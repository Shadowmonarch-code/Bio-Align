   133→function parseCSVData(csvText: string): { genotypes: string[], environments: string[], dataMatrix: number[][] } | null {
   134→  const lines = csvText.trim().split('\n').map(line => line.trim()).filter(line => line.length > 0)
   135→  
   136→  if (lines.length < 2) return null
   137→  
   138→  const headers = lines[0].split(',').map(h => h.trim())
   139→  
   140→  // First column is genotype names, rest are environments
   141→  const environments = headers.slice(1)
   142→  const genotypes: string[] = []
   143→  const dataMatrix: number[][] = []
   144→  
   145→  for (let i = 1; i < lines.length; i++) {
   146→    const values = lines[i].split(',').map(v => v.trim())
   147→    
   148→    if (values.length !== headers.length) continue
   149→    
   150→    const genotypeName = values[0]
   151→    const yields = values.slice(1).map(v => parseFloat(v))
   152→    
   153→    // Validate all values are numbers
   154→    if (yields.some(y => isNaN(y))) continue
   155→    
   156→    genotypes.push(genotypeName)
   157→    dataMatrix.push(yields)
   158→  }
   159→  
   160→  if (genotypes.length === 0 || environments.length === 0) return null
   161→  
   162→  return { genotypes, environments, dataMatrix }
   163→}

   169→function matrixTranspose(matrix: number[][]): number[][] {
   170→  if (matrix.length === 0) return []
   171→  return matrix[0].map((_, colIdx) => matrix.map(row => row[colIdx]))
   172→}
   173→
   174→function matrixMultiply(A: number[][], B: number[][]): number[][] {
   175→  const rowsA = A.length
   176→  const colsA = A[0]?.length || 0
   177→  const colsB = B[0]?.length || 0
   178→  
   179→  const result: number[][] = []
   180→  for (let i = 0; i < rowsA; i++) {
   181→    result[i] = []
   182→    for (let j = 0; j < colsB; j++) {
   183→      let sum = 0
   184→      for (let k = 0; k < colsA; k++) {
   185→        sum += A[i][k] * B[k][j]
   186→      }
   187→      result[i][j] = sum
   188→    }
   189→  }
   190→  return result
   191→}
   192→
   193→function copyMatrix(mat: number[][]): number[][] {
   194→  return mat.map(row => [...row])
   195→}
   196→
   197→function matrixMultiplyVector(matrix: number[][], vec: number[]): number[] {
   198→  return matrix.map(row => 
   199→    row.reduce((sum, val, i) => sum + val * (vec[i] || 0), 0)
   200→  )
   201→}

   203→/**
   204→ * Compute SVD using power iteration method for AMMI analysis
   205→ */
   206→function computeSVD(matrix: number[][]): { U: number[][]; S: number[]; Vt: number[][] } {
   207→  const m = matrix.length
   208→  const n = matrix[0]?.length || 0
   209→  
   210→  if (m === 0 || n === 0) {
   211→    return { U: [], S: [], Vt: [] }
   212→  }
   213→  
   214→  const AtA = matrixMultiply(matrixTranspose(matrix), matrix)
   215→  const size = Math.min(m, n)
   216→  
   217→  const eigenvalues: number[] = []
   218→  const eigenvectors: number[][] = []
   219→  
   220→  let workingMatrix = copyMatrix(AtA)
   221→  
   222→  for (let iter = 0; iter < size; iter++) {
   223→    let v = new Array(n).fill(0)
   224→    v[iter < n ? iter : 0] = 1
   225→    
   226→    for (let powerIter = 0; powerIter < 100; powerIter++) {
   227→      let Av = matrixMultiplyVector(workingMatrix, v)
   228→      
   229→      // Orthogonalize against previous eigenvectors
   230→      for (const prevVec of eigenvectors) {
   231→        const dot = v.reduce((sum, vi, i) => sum + vi * prevVec[i], 0)
   232→        Av = Av.map((val, i) => val - dot * prevVec[i])
   233→      }
   234→      
   235→      const norm = Math.sqrt(Av.reduce((sum, x) => sum + x * x, 0))
   236→      if (norm < 1e-10) break
   237→      
   238→      v = Av.map(x => x / norm)
   239→    }
   240→    
   241→    const Av = matrixMultiplyVector(workingMatrix, v)
   242→    const eigenvalue = v.reduce((sum, vi, i) => sum + vi * Av[i], 0)
   243→    
   244→    if (eigenvalue > 1e-10) {
   245→      eigenvalues.push(Math.max(0, eigenvalue))
   246→      eigenvectors.push([...v])
   247→      
   248→      for (let i = 0; i < n; i++) {
   249→        for (let j = 0; j < n; j++) {
   250→          workingMatrix[i][j] -= eigenvalue * v[i] * v[j]
   251→        }
   252→      }
   253→    } else {
   254→      break
   255→    }
   256→  }
   257→  
   258→  const S = eigenvalues.map(ev => Math.sqrt(ev))
   259→  const V = eigenvectors.length > 0 ? matrixTranspose(eigenvectors) : []
   260→  const Vt = matrixTranspose(V)
   261→  
   262→  const U: number[][] = []
   263→  for (let i = 0; i < m; i++) {
   264→    U[i] = []
   265→    for (let j = 0; j < S.length; j++) {
   266→      if (S[j] > 1e-10) {
   267→        let sum = 0
   268→        for (let k = 0; k < n; k++) {
   269→          sum += matrix[i][k] * (V[k]?.[j] || 0)
   270→        }
   271→        U[i][j] = sum / S[j]
   272→      } else {
   273→        U[i][j] = 0
   274→      }
   275→    }
   276→  }
   277→  
   278→  return { U, S, Vt }
   279→}

   281→// ============================================================================
   282→// G×E ANALYSIS FUNCTIONS
   283→// ============================================================================
   284→
   285→/**
   286→ * Perform comprehensive Genotype × Environment Interaction Analysis
   287→ * Implements multiple stability analysis methods:
   288→ * - Finlay-Wilkinson (1963) Joint Regression Analysis
   289→ * - Eberhart-Russell (1966) Stability Parameters
   290→ * - Shukla (1972) Stability Variance
   291→ * - Wricke (1962) Ecovalence
   292→ * - AMMI Stability Value (ASV)
   293→ */
   294→function performGXEAnalysis(
   295→  genotypeNames: string[],
   296→  environmentNames: string[],
   297→  dataMatrix: number[][]
   298→): GEXEResult {
   299→  const g = genotypeNames.length   // Number of genotypes
   300→  const e = environmentNames.length // Number of environments
   301→  const N = g * e                  // Total observations
   302→  
   303→  // ========================================================================
   304→  // BASIC STATISTICS
   305→  // ========================================================================
   306→  
   307→  // Calculate grand mean
   308→  const allValues = dataMatrix.flat()
   309→  const grandMean = mean(allValues)
   310→  
   311→  // Calculate genotype means (row means)
   312→  const genotypeMeans = dataMatrix.map(row => mean(row))
   313→  
   314→  // Calculate environment means (column means)
   315→  const environmentMeans: number[] = []
   316→  for (let j = 0; j < e; j++) {
   317→    const colValues = dataMatrix.map(row => row[j])
   318→    environmentMeans.push(mean(colValues))
   319→  }
   320→  
   321→  // ========================================================================
   322→  // TWO-WAY ANOVA FOR G×E INTERACTION
   323→  // ========================================================================
   324→  
   325→  // Sum of Squares calculations
   326→  // Total SS
   327→  const totalSS = sumOfSquares(allValues) - (Math.pow(allValues.reduce((a, b) => a + b, 0), 2) / N)
   328→  
   329→  // Genotype SS
   330→  let genotypeSS = 0
   331→  for (let i = 0; i < g; i++) {
   332→    genotypeSS += e * Math.pow(genotypeMeans[i] - grandMean, 2)
   333→  }
   334→  
   335→  // Environment SS
   336→  let environmentSS = 0
   337→  for (let j = 0; j < e; j++) {
   338→    environmentSS += g * Math.pow(environmentMeans[j] - grandMean, 2)
   339→  }
   340→  
   341→  // G×E Interaction SS (calculated as residual after main effects)
   342→  let geSS = 0
   343→  for (let i = 0; i < g; i++) {
   344→    for (let j = 0; j < e; j++) {
   345→      const observed = dataMatrix[i][j]
   346→      const expected = grandMean + (genotypeMeans[i] - grandMean) + (environmentMeans[j] - grandMean)
   347→      geSS += Math.pow(observed - expected, 2)
   348→    }
   349→  }
   350→  
   351→  // Degrees of freedom
   352→  const dfGenotype = g - 1
   353→  const dfEnvironment = e - 1
   354→  const dfGE = (g - 1) * (e - 1)
   355→  const dfTotal = N - 1
   356→  
   357→  // Mean Squares
   358→  const MSgenotype = genotypeSS / dfGenotype
   359→  const MSenvironment = environmentSS / dfEnvironment
   360→  const MSge = geSS / dfGE
   361→  
   362→  // F-values (using GE MS as error term for this combined model)
   363→  const Fgenotype = MSgenotype / MSge
   364→  const Fenvironment = MSenvironment / MSge
   365→  
   366→  // P-values
   367→  const pValueGenotype = 1 - fDistributionCDF(Fgenotype, dfGenotype, dfGE)
   368→  const pValueEnvironment = 1 - fDistributionCDF(Fenvironment, dfEnvironment, dfGE)
   369→  
   370→  function getSignificance(p: number): string {
   371→    if (p < 0.001) return '***'
   372→    if (p < 0.01) return '**'
   373→    if (p < 0.05) return '*'
   374→    if (p < 0.1) return '.'
   375→    return 'ns'
   376→  }
   377→  
   378→  const anova: ANOVARow[] = [
   379→    {
   380→      source: 'Genotype',
   381→      df: dfGenotype,
   382→      ss: genotypeSS,
   383→      ms: MSgenotype,
   384→      f: Fgenotype,
   385→      pValue: pValueGenotype,
   386→      significance: getSignificance(pValueGenotype)
   387→    },
   388→    {
   389→      source: 'Environment',
   390→      df: dfEnvironment,
   391→      ss: environmentSS,
   392→      ms: MSenvironment,
   393→      f: Fenvironment,
   394→      pValue: pValueEnvironment,
   395→      significance: getSignificance(pValueEnvironment)
   396→    },
   397→    {
   398→      source: 'G × E Interaction',
   399→      df: dfGE,
   400→      ss: geSS,
   401→      ms: MSge,
   402→      f: NaN,  // No F-test for interaction without error term
   403→      pValue: NaN,
   404→      significance: '-'
   405→    },
   406→    {
   407→      source: 'Total',
   408→      df: dfTotal,
   409→      ss: totalSS,
   410→      ms: NaN,
   411→      f: NaN,
   412→      pValue: NaN,
   413→      significance: '-'
   414→    }
   415→  ]
   416→  
   417→  // ========================================================================
   418→  // ENVIRONMENTAL INDICES
   419→  // ========================================================================
   420→  
   421→  // Environmental Index (I_j) = EnvMean_j - GrandMean
   422→  const envIndices: EnvironmentIndex[] = environmentNames.map((name, idx) => ({
   423→    environment: name,
   424→    index: environmentMeans[idx] - grandMean,
   425→    meanYield: environmentMeans[idx],
   426→    rank: 1  // Will be updated below
   427→  }))
   428→  
   429→  // Rank by environmental index (descending)
   430→  envIndices.sort((a, b) => b.index - a.index)
   431→  envIndices.forEach((env, idx) => env.rank = idx + 1)
   432→  
   433→  // Restore original order
   434→  envIndices.sort((a, b) => 
   435→    environmentNames.indexOf(a.environment) - environmentNames.indexOf(b.environment)
   436→  )
   437→  
   438→  // ========================================================================
   439→  // FINLAY-WILKINSON REGRESSION ANALYSIS (1963)
   440→  // Regress each genotype's yield on environmental index
   441→  // ========================================================================
   442→  
   443→  const finlayWilkinsonResults: Array<{
   444→    b: number         // Regression coefficient
   445→    s2d: number       // Deviation from regression (S²d)
   446→    rSquared: number  // R² value
   447→    intercept: number
   448→  }> = []
   449→  
   450→  for (let i = 0; i < g; i++) {
   451→    // X = environmental indices, Y = genotype yields across environments
   452→    const xValues = environmentMeans.map(em => em - grandMean)  // Environmental indices
   453→    const yValues = dataMatrix[i]  // Genotype i yields in each environment
   454→    
   455→    const regression = simpleLinearRegression(xValues, yValues)
   456→    
   457→    // Calculate deviation from regression (S²d)
   458→    // S²d = Σ(Y_ij - Ŷ_ij)² / (e - 2)
   459→    let devSumSq = 0
   460→    for (let j = 0; j < e; j++) {
   461→      const predicted = regression.coefficients[0] + regression.coefficients[1] * xValues[j]
   462→      devSumSq += Math.pow(yValues[j] - predicted, 2)
   463→    }
   464→    const s2d = devSumSq / Math.max(1, e - 2)
   465→    
   466→    finlayWilkinsonResults.push({
   467→      b: regression.coefficients[1],  // Slope = regression coefficient
   468→      s2d: s2d,
   469→      rSquared: regression.rSquared,
   470→      intercept: regression.coefficients[0]
   471→    })
   472→  }
   473→  
   474→  // ========================================================================
   475→  // EBERHART-RUSSELL STABILITY ANALYSIS (1966)
   476→  // Uses same regression but adds interpretation framework
   477→  // Ideal: b_i ≈ 1 and S²d_i not significantly different from 0
   478→  // ========================================================================
   479→  
   480→  // Pooled deviation MS (error estimate)
   481→  const pooledDevMS = finlayWilkinsonResults.reduce((sum, r) => sum + r.s2d, 0) / g
   482→  const dfDeviation = g * (e - 2)  // Total df for deviations
   483→  
   484→  function getERClassification(b: number, s2d: number): string {
   485→    const bCloseToOne = Math.abs(b - 1) < 0.2
   486→    const s2dSmall = s2d < pooledDevMS * 2  // Less than 2x pooled MS
   487→    
   488→    if (bCloseToOne && s2dSmall) return 'Average stability, predictable'
   489→    if (b < 0.8 && s2dSmall) return 'High stability, poor env specific'
   490→    if (b > 1.2 && s2dSmall) return 'Low stability, responsive to good env'
   491→    if (bCloseToOne && !s2dSmall) return 'Average response, unpredictable'
   492→    if (b < 0.8 && !s2dSmall) return 'Below average, unstable'
   493→    return 'Above average, unstable'
   494→  }
   495→  
   496→  // ========================================================================
   497→  // SHUKLA'S STABILITY VARIANCE (1972)
   498→  // σ²_i = [(n-1)/(g-1)(n-2)] * ΣΣ(Y_ij - Ȳ_i. - Ȳ_.j + Ȳ..)²
   499→  // Lower values indicate more stable genotypes
   500→  // ========================================================================
   501→  
   502→  const shuklaFactor = ((e - 1) / ((g - 1) * (e - 2)))
   503→  const shukaVariances: number[] = []
   504→  
   505→  for (let i = 0; i < g; i++) {
   506→    let shuklaSum = 0
   507→    for (let j = 0; j < e; j++) {
   508→      const cellMean = dataMatrix[i][j]
   509→      const genoMean = genotypeMeans[i]
   510→      const envMean = environmentMeans[j]
   511→      // Interaction effect for this cell
   512→      const interactionEffect = cellMean - genoMean - envMean + grandMean
   513→      shuklaSum += Math.pow(interactionEffect, 2)
   514→    }
   515→    shukaVariances.push(shuklaFactor * shuklaSum)
   516→  }
   517→  
   518→  // ========================================================================
   519→  // WRICKE'S ECOVALENCE (1962)
   520→  // W_i = Σ_j(Ȳ_ij - Ȳ_i. - Ȳ_.j + Ȳ..)²
   521→  // Contribution of each genotype to G×E sum of squares
   522→  // ========================================================================
   523→  
   524→  const ecovalences: number[] = []
   525→  
   526→  for (let i = 0; i < g; i++) {
   527→    let wSum = 0
   528→    for (let j = 0; j < e; j++) {
   529→      const cellMean = dataMatrix[i][j]
   530→      const genoMean = genotypeMeans[i]
   531→      const envMean = environmentMeans[j]
   532→      const interactionEffect = cellMean - genoMean - envMean + grandMean
   533→      wSum += Math.pow(interactionEffect, 2)
   534→    }
   535→    ecovalences.push(wSum)
   536→  }
   537→  
   538→  // Calculate percentage contribution to G×E SS
   539→  const totalEcovalence = ecovalences.reduce((a, b) => a + b, 0)
   540→  const ecovalencePercents = ecovalences.map(w => (w / totalEcovalence) * 100)
   541→  
   542→  // ========================================================================
   543→  // AMMI STABILITY VALUE (ASV)
   544→  // ASV = √[(SS_IPCA1/SS_IPCA2 * IPCA1_score)² + (IPCA2_score)²]
   545→  // Lower ASV = more stable
   546→  // ========================================================================
   547→  
   548→  // Calculate interaction effects matrix (residuals after removing main effects)
   549→  const interactionMatrix: number[][] = []
   550→  for (let i = 0; i < g; i++) {
   551→    interactionMatrix[i] = []
   552→    for (let j = 0; j < e; j++) {
   553→      const cellMean = dataMatrix[i][j]
   554→      const genoEffect = genotypeMeans[i] - grandMean
   555→      const envEffect = environmentMeans[j] - grandMean
   556→      interactionMatrix[i][j] = cellMean - grandMean - genoEffect - envEffect
   557→    }
   558→  }
   559→  
   560→  // Perform SVD on interaction matrix
   561→  const { U, S, Vt } = computeSVD(interactionMatrix)
   562→  
   563→  // Calculate ASV for each genotype
   564→  const asvValues: number[] = []
   565→  const ssIPCA1 = S.length > 0 ? Math.pow(S[0], 2) : 1
   566→  const ssIPCA2 = S.length > 1 ? Math.pow(S[1], 2) : 1
   567→  const weightFactor = ssIPCA1 / Math.max(ssIPCA2, 1)
   568→  
   569→  for (let i = 0; i < g; i++) {
   570→    const ipca1Score = U[i]?.[0] || 0
   571→    const ipca2Score = U[i]?.[1] || 0
   572→    const asv = Math.sqrt(
   573→      Math.pow(weightFactor * ipca1Score, 2) + Math.pow(ipca2Score, 2)
   574→    )
   575→    asvValues.push(asv)
   576→  }
   577→  
   578→  // ========================================================================
   579→  // COMPILE STABILITY RESULTS
   580→  // ========================================================================
   581→  
   582→  const stabilityResults: StabilityResult[] = genotypeNames.map((name, i) => {
   583→    const fw = finlayWilkinsonResults[i]
   584→    
   585→    // F-test for S²d significance (Eberhart-Russell)
   586→    const s2dFValue = fw.s2d / Math.max(pooledDevMS, 0.001)
   587→    const s2dPValue = 1 - fDistributionCDF(s2dFValue, e - 2, dfDeviation)
   588→    
   589→    // Determine overall stability class based on composite score
   590→    // Combine multiple metrics into a single ranking score
   591→    const bDeviation = Math.abs(fw.b - 1)  // Deviation from ideal b=1
   592→    const normalizedS2d = fw.s2d / Math.max(...finlayWilkinsonResults.map(r => r.s2d), 1)
   593→    const normalizedShukla = shukaVariances[i] / Math.max(...shukaVariances, 1)
   594→    const normalizedWricke = ecovalences[i] / Math.max(...ecovalences, 1)
   595→    const normalizedASV = asvValues[i] / Math.max(...asvValues, 1)
   596→    
   597→    // Composite instability score (higher = more unstable)
   598→    const instabilityScore = (
   599→      bDeviation * 0.2 +
   600→      normalizedS2d * 0.2 +
   601→      normalizedShukla * 0.2 +
   602→      normalizedWricke * 0.2 +
   603→      normalizedASV * 0.2
   604→    )
   605→    
   606→    let stabilityClass: StabilityResult['stabilityClass']
   607→    if (instabilityScore < 0.15) stabilityClass = 'Very Stable'
   608→    else if (instabilityScore < 0.35) stabilityClass = 'Stable'
   609→    else if (instabilityScore < 0.55) stabilityClass = 'Moderate'
   610→    else if (instabilityScore < 0.75) stabilityClass = 'Unstable'
   611→    else stabilityClass = 'Very Unstable'
   612→    
   613→    return {
   614→      genotype: name,
   615→      meanYield: genotypeMeans[i],
   616→      regressionCoeff: fw.b,
   617→      deviationMS: fw.s2d,
   618→      rSquared: fw.rSquared,
   619→      erClassification: getERClassification(fw.b, fw.s2d),
   620→      s2dFValue: s2dFValue,
   621→      s2dSignificance: getSignificance(s2dPValue),
   622→      shuklaVariance: shukaVariances[i],
   623→      ecovalence: ecovalences[i],
   624→      ecovalencePercent: ecovalencePercents[i],
   625→      asv: asvValues[i],
   626→      overallRank: 0,  // Will be calculated below
   627→      stabilityClass
   628→    }
   629→  })
   630→  
   631→  // Sort by composite stability and assign ranks
   632→  stabilityResults.sort((a, b) => {
   633→    // Primary sort by stability class (Very Stable first)
   634→    const classOrder = { 'Very Stable': 0, 'Stable': 1, 'Moderate': 2, 'Unstable': 3, 'Very Unstable': 4 }
   635→    const classDiff = classOrder[a.stabilityClass] - classOrder[b.stabilityClass]
   636→    if (classDiff !== 0) return classDiff
   637→    // Secondary sort by ASV (lower is better)
   638→    return a.asv - b.asv
   639→  })
   640→  
   641→  stabilityResults.forEach((r, idx) => r.overallRank = idx + 1)
   642→  
   643→  // Find most stable and most responsive genotypes
   644→  const sortedByStability = [...stabilityResults].sort((a, b) => a.asv - b.asv)
   645→  const sortedByResponse = [...stabilityResults].sort((a, b) => b.regressionCoeff - a.regressionCoeff)
   646→  
   647→  return {
   648→    genotypes: genotypeNames,
   649→    environments: environmentNames,
   650→    dataMatrix,
   651→    grandMean,
   652→    genotypeMeans,
   653→    environmentMeans,
   654→    anova,
   655→    environmentalIndices: envIndices,
   656→    stabilityResults,
   657→    summary: {
   658→      totalGenotypes: g,
   659→      totalEnvironments: e,
   660→      geSignificant: pValueGenotype < 0.05 || pValueEnvironment < 0.05,
   661→      mostStableGenotype: sortedByStability[0]?.genotype || '',
   662→      mostResponsiveGenotype: sortedByResponse[0]?.genotype || ''
   663→    }
   664→  }
   665→}

   671→function exportResultsToCSV(result: GEXEResult): string {
   672→  const lines: string[] = []
   673→  
   674→  // Header
   675→  lines.push('G×E Interaction Analysis Results')
   676→  lines.push('')
   677→  
   678→  // ANOVA Table
   679→  lines.push('ANOVA TABLE')
   680→  lines.push('Source,DF,SS,MS,F-value,P-value,Sig')
   681→  result.anova.forEach(row => {
   682→    lines.push(`${row.source},${row.df},${row.ss.toFixed(2)},${row.ms.toFixed(2)},${isNaN(row.f) ? '-' : row.f.toFixed(2)},${isNaN(row.pValue) ? '-' : row.pValue.toFixed(4)},${row.significance}`)
   683→  })
   684→  lines.push('')
   685→  
   686→  // Environmental Indices
   687→  lines.push('ENVIRONMENTAL INDICES')
   688→  lines.push('Environment,Environmental Index,Mean Yield,Rank')
   689→  result.environmentalIndices.forEach(env => {
   690→    lines.push(`${env.environment},${env.index.toFixed(2)},${env.meanYield.toFixed(2)},${env.rank}`)
   691→  })
   692→  lines.push('')
   693→  
   694→  // Stability Parameters
   695→  lines.push('STABILITY PARAMETERS')
   696→  lines.push('Rank,Genotype,Mean Yield,b (Reg Coef),S²d,R²,Shukla σ²,Wricke W,ASV,Stability Class')
   697→  result.stabilityResults.sort((a, b) => a.overallRank - b.overallRank)
   698→  result.stabilityResults.forEach(r => {
   699→    lines.push(`${r.overallRank},${r.genotype},${r.meanYield.toFixed(2)},${r.regressionCoeff.toFixed(3)},${r.deviationMS.toFixed(2)},${r.rSquared.toFixed(4)},${r.shuklaVariance.toFixed(2)},${r.ecovalence.toFixed(2)},${r.asv.toFixed(3)},${r.stabilityClass}`)
   700→  })
   701→  
   702→  return lines.join('\n')
   703→}
