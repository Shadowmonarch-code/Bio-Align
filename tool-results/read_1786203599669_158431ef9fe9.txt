   201→/**
   202→ * Calculates the natural logarithm of the gamma function using Lanczos approximation.
   203→ * Used internally for distribution calculations.
   204→ * @param x - Input value
   205→ * @returns ln(Gamma(x))
   206→ */
   207→function lnGamma(x: number): number {
   208→  if (x < 0.5) {
   209→    return Math.log(Math.PI / Math.sin(Math.PI * x)) - lnGamma(1 - x);
   210→  }
   211→  
   212→  x -= 1;
   213→  const g = 7;
   214→  const coef = [
   215→    0.99999999999980993,
   216→    676.5203681218851,
   217→    -1259.1392167224028,
   218→    771.32342877765313,
   219→    -176.61502916214059,
   220→    12.507343278686905,
   221→    -0.13857109526572012,
   222→    9.9843695780195716e-6,
   223→    1.5056327351493116e-7
   224→  ];
   225→  
   226→  let x0 = coef[0];
   227→  for (let i = 1; i < g + 2; i++) {
   228→    x0 += coef[i] / (x + i);
   229→  }
   230→  
   231→  const t = x + g + 0.5;
   232→  return 0.5 * Math.log(2 * Math.PI) + (x + 0.5) * Math.log(t) - t + Math.log(x0);
   233→}

   235→/**
   236→ * Regularized incomplete beta function I_x(a, b)
   237→ * Uses continued fraction representation (Lentz's method)
   238→ * @param x - Value between 0 and 1
   239→ * @param a - First shape parameter
   240→ * @param b - Second shape parameter
   241→ * @returns I_x(a, b)
   242→ */
   243→function regularizedIncompleteBeta(x: number, a: number, b: number): number {
   244→  if (x === 0) return 0;
   245→  if (x === 1) return 1;
   246→  
   247→  const logBetaFn = lnGamma(a) + lnGamma(b) - lnGamma(a + b);
   248→  
   249→  // Use symmetry transformation if x > (a+1)/(a+b+2)
   250→  const front = Math.exp(a * Math.log(x) + b * Math.log(1 - x) - logBetaFn);
   251→  
   252→  if (x < (a + 1) / (a + b + 2)) {
   253→    return front * betaContinuedFraction(x, a, b) / a;
   254→  } else {
   255→    return 1 - front * betaContinuedFraction(1 - x, b, a) / b;
   256→  }
   257→}
   258→
   259→/**
   260→ * Continued fraction for incomplete beta function using Lentz's method
   261→ */
   262→function betaContinuedFraction(x: number, a: number, b: number): number {
   263→  const maxIterations = 200;
   264→  const epsilon = 3e-14;
   265→  
   266→  const qab = a + b;
   267→  const qap = a + 1;
   268→  const qam = a - 1;
   269→  
   270→  let c = 1;
   271→  let d = 1 - qab * x / qap;
   272→  if (Math.abs(d) < epsilon) d = epsilon;
   273→  d = 1 / d;
   274→  let h = d;
   275→  
   276→  for (let m = 1; m <= maxIterations; m++) {
   277→    let m2 = 2 * m;
   278→    
   279→    // Even step
   280→    let aa = m * (b - m) * x / ((qam + m2) * (a + m2));
   281→    d = 1 + aa * d;
   282→    if (Math.abs(d) < epsilon) d = epsilon;
   283→    c = 1 + aa / c;
   284→    if (Math.abs(c) < epsilon) c = epsilon;
   285→    d = 1 / d;
   286→    h *= d * c;
   287→    
   288→    // Odd step
   289→    aa = -(a + m) * (qab + m) * x / ((a + m2) * (qap + m2));
   290→    d = 1 + aa * d;
   291→    if (Math.abs(d) < epsilon) d = epsilon;
   292→    c = 1 + aa / c;
   293→    if (Math.abs(c) < epsilon) c = epsilon;
   294→    d = 1 / d;
   295→    const del = d * c;
   296→    h *= del;
   297→    
   298→    if (Math.abs(del - 1) < epsilon) break;
   299→  }
   300→  
   301→  return h;
   302→}

   304→/**
   305→ * Cumulative Distribution Function for F-distribution
   306→ * P(F ≤ f) where F ~ F(df1, df2)
   307→ * Uses relationship with Beta distribution: P(F ≤ f) = I_{df1*f/(df1*f+df2)}(df1/2, df2/2)
   308→ * 
   309→ * @param f - F-statistic value
   310→ * @param df1 - Numerator degrees of freedom
   311→ * @param df2 - Denominator degrees of freedom
   312→ * @returns Cumulative probability
   313→ */
   314→export function fDistributionCDF(f: number, df1: number, df2: number): number {
   315→  if (f <= 0) return 0;
   316→  if (!isFinite(f)) return 1;
   317→  
   318→  const x = df1 * f / (df1 * f + df2);
   319→  return regularizedIncompleteBeta(x, df1 / 2, df2 / 2);
   320→}
   321→
   322→/**
   323→ * Cumulative Distribution Function for t-distribution
   324→ * P(T ≤ t) where T ~ t(df)
   325→ * Uses relationship with Beta distribution
   326→ * 
   327→ * @param t - t-statistic value
   328→ * @param df - Degrees of freedom
   329→ * @returns Cumulative probability
   330→ */
   331→export function tDistributionCDF(t: number, df: number): number {
   332→  if (t === 0) return 0.5;
   333→  
   334→  const x = df / (df + t * t);
   335→  const ib = regularizedIncompleteBeta(x, df / 2, 0.5);
   336→  
   337→  return t > 0 ? 1 - 0.5 * ib : 0.5 * ib;
   338→}

   340→/**
   341→ * Standard Normal Cumulative Distribution Function
   342→ * Uses Abramowitz and Stegun approximation (error function approximation)
   343→ * 
   344→ * @param z - Z-score
   345→ * @returns P(Z ≤ z)
   346→ */
   347→export function normalCDF(z: number): number {
   348→  // Approximation with maximum error < 7.5e-8
   349→  const a1 = 0.254829592;
   350→  const a2 = -0.284496736;
   351→  const a3 = 1.421413741;
   352→  const a4 = -1.453152027;
   353→  const a5 = 1.061405429;
   354→  const p = 0.3275911;
   355→  
   356→  const sign = z < 0 ? -1 : 1;
   357→  z = Math.abs(z) / Math.sqrt(2);
   358→  
   359→  const t = 1.0 / (1.0 + p * z);
   360→  const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-z * z);
   361→  
   362→  return 0.5 * (1.0 + sign * y);
   363→}

   365→/**
   366→ * Cumulative Distribution Function for Chi-Square distribution
   367→ * Uses relationship with Gamma distribution
   368→ * 
   369→ * @param x - Chi-square value
   370→ * @param df - Degrees of freedom
   371→ * @returns P(X² ≤ x)
   372→ */
   373→export function chiSquareCDF(x: number, df: number): number {
   374→  if (x <= 0) return 0;
   375→  return regularizedIncompleteBeta(x / (x + df), df / 2, 0.5);
   376→}

   382→/**
   383→ * Calculate arithmetic mean of an array
   384→ * Formula: μ = Σxᵢ / n
   385→ * 
   386→ * @param arr - Array of numbers
   387→ * @returns Mean value
   388→ */
   389→export function mean(arr: number[]): number {
   390→  if (!arr || arr.length === 0) throw new Error('Cannot calculate mean of empty array');
   391→  return sum(arr) / arr.length;
   392→}
   393→
   394→/**
   395→ * Calculate sum of array elements
   396→ * Formula: Σxᵢ
   397→ * 
   398→ * @param arr - Array of numbers
   399→ * @returns Sum value
   400→ */
   401→export function sum(arr: number[]): number {
   402→  return arr.reduce((acc, val) => acc + val, 0);
   403→}
   404→
   405→/**
   406→ * Calculate sum of squares
   407→ * Formula: Σxᵢ²
   408→ * 
   409→ * @param arr - Array of numbers
   410→ * @returns Sum of squares
   411→ */
   412→export function sumOfSquares(arr: number[]): number {
   413→  return arr.reduce((acc, val) => acc + val * val, 0);
   414→}

   416→/**
   417→ * Calculate variance of an array
   418→ * Population variance: σ² = Σ(xᵢ - μ)² / N
   419→ * Sample variance: s² = Σ(xᵢ - μ)² / (n-1)
   420→ * 
   421→ * @param arr - Array of numbers
   422→ * @param population - If true, use population formula (divide by N); otherwise sample (divide by n-1)
   423→ * @returns Variance
   424→ */
   425→export function variance(arr: number[], population: boolean = false): number {
   426→  if (!arr || arr.length === 0) throw new Error('Cannot calculate variance of empty array');
   427→  if (arr.length === 1) return population ? 0 : NaN;
   428→  
   429→  const mu = mean(arr);
   430→  const squaredDiffs = arr.map(x => (x - mu) ** 2);
   431→  const divisor = population ? arr.length : arr.length - 1;
   432→  return sum(squaredDiffs) / divisor;
   433→}
   434→
   435→/**
   436→ * Calculate standard deviation
   437→ * σ = √σ² or s = √s²
   438→ * 
   439→ * @param arr - Array of numbers
   440→ * @param population - If true, use population SD; otherwise sample SD
   441→ * @returns Standard deviation
   442→ */
   443→export function stddev(arr: number[], population: boolean = false): number {
   444→  return Math.sqrt(variance(arr, population));
   445→}

   447→/**
   448→ * Calculate median of an array
   449→ * 
   450→ * @param arr - Array of numbers
   451→ * @returns Median value
   452→ */
   453→export function median(arr: number[]): number {
   454→  if (!arr || arr.length === 0) throw new Error('Cannot calculate median of empty array');
   455→  
   456→  const sorted = [...arr].sort((a, b) => a - b);
   457→  const mid = Math.floor(sorted.length / 2);
   458→  
   459→  if (sorted.length % 2 === 0) {
   460→    return (sorted[mid - 1] + sorted[mid]) / 2;
   461→  }
   462→  return sorted[mid];
   463→}

   465→/**
   466→ * Calculate covariance between two arrays
   467→ * Cov(X,Y) = Σ((xᵢ - μₓ)(yᵢ - μᵧ)) / (n-1)
   468→ * 
   469→ * @param x - First array
   470→ * @param y - Second array
   471→ * @returns Covariance value
   472→ */
   473→export function covariance(x: number[], y: number[]): number {
   474→  if (x.length !== y.length) throw new Error('Arrays must have equal length');
   475→  if (x.length === 0) throw new Error('Cannot calculate covariance of empty arrays');
   476→  
   477→  const meanX = mean(x);
   478→  const meanY = mean(y);
   479→  let covSum = 0;
   480→  
   481→  for (let i = 0; i < x.length; i++) {
   482→    covSum += (x[i] - meanX) * (y[i] - meanY);
   483→  }
   484→  
   485→  return covSum / (x.length - 1);
   486→}

   498→/**
   499→ * Calculate critical F-value using inverse CDF approximation
   500→ * Uses Newton-Raphson method on the F-distribution CDF
   501→ * 
   502→ * @param alpha - Significance level (e.g., 0.05)
   503→ * @param df1 - Numerator degrees of freedom
   504→ * @param df2 - Denominator degrees of freedom
   505→ * @returns Critical F-value
   506→ */
   507→export function criticalF(alpha: number, df1: number, df2: number): number {
   508→  // Initial guess using Wilson-Hilferty approximation for chi-square
   509→  // F ≈ χ²₁/df1 / (χ²₂/df2), approximated
   510→  
   511→  let x = 1; // Starting guess
   512→  const target = 1 - alpha;
   513→  
   514→  // Newton-Raphson iteration
   515→  for (let iter = 0; iter < 100; iter++) {
   516→    const cdf = fDistributionCDF(x, df1, df2);
   517→    const diff = cdf - target;
   518→    
   519→    if (Math.abs(diff) < 1e-10) break;
   520→    
   521→    // Numerical derivative
   522→    const eps = 1e-7;
   523→    const deriv = (fDistributionCDF(x + eps, df1, df2) - cdf) / eps;
   524→    
   525→    if (Math.abs(deriv) < 1e-15) break;
   526→    
   527→    x -= diff / deriv;
   528→    x = Math.max(0.001, x); // Keep positive
   529→  }
   530→  
   531→  return x;
   532→}

   534→/**
   535→ * Calculate critical t-value using inverse CDF approximation
   536→ * 
   537→ * @param alpha - Significance level (two-tailed: 0.05 gives 95% CI)
   538→ * @param df - Degrees of freedom
   539→ * @returns Critical t-value
   540→ */
   541→export function criticalT(alpha: number, df: number): number {
   542→  // For large df, approximate with normal
   543→  if (df > 100) {
   544→    return normalCDF(1 - alpha / 2) > 0.5 ? 
   545→      findZForProbability(1 - alpha / 2) : 
   546→      -findZForProbability(alpha / 2);
   547→  }
   548→  
   549→  let t = 1; // Starting guess
   550→  const target = 1 - alpha / 2; // Two-tailed
   551→  
   552→  for (let iter = 0; iter < 100; iter++) {
   553→    const cdf = tDistributionCDF(t, df);
   554→    const diff = cdf - target;
   555→    
   556→    if (Math.abs(diff) < 1e-10) break;
   557→    
   558→    const eps = 1e-7;
   559→    const deriv = (tDistributionCDF(t + eps, df) - cdf) / eps;
   560→    
   561→    if (Math.abs(deriv) < 1e-15) break;
   562→    
   563→    t -= diff / deriv;
   564→  }
   565→  
   566→  return t;
   567→}

   569→/**
   570→ * Find z-score for given probability (inverse normal CDF)
   571→ */
   572→function findZForProbability(p: number): number {
   573→  // Rational approximation by Peter Acklam
   574→  if (p <= 0 || p >= 1) throw new Error('Probability must be between 0 and 1');
   575→  
   576→  const a = [
   577→    -3.969683028665376e+01,
   578→    2.209460984245205e+02,
   579→    -2.759285104469687e+02,
   580→    1.383577518672690e+02,
   581→    -3.066479806614716e+01,
   582→    2.506628277459239e+00
   583→  ];
   584→  
   585→  const b = [
   586→    -5.447608881703079e+01,
   587→    1.615858368580409e+02,
   588→    -1.556989798598866e+02,
   589→    6.680131188771972e+01,
   590→    -1.328068155288572e+01
   591→  ];
   592→  
   593→  const c = [
   594→    -7.784894002430293e-03,
   595→    -3.223964580411365e-01,
   596→    -2.400758277161838e+00,
   597→    -2.549732539343734e+00,
   598→    4.374664141464968e+00,
   599→    2.938163982698783e+00
   600→  ];
   601→  
   602→  const d = [
   603→    7.784695709041462e-03,
   604→    3.224671290700398e-01,
   605→    2.445134137142996e+00,
   606→    3.754408661907416e+00
   607→  ];
   608→  
   609→  const pLow = 0.02425;
   610→  const pHigh = 1 - pLow;
   611→  
   612→  let q, r: number;
   613→  
   614→  if (p < pLow) {
   615→    q = Math.sqrt(-2 * Math.log(p));
   616→    return (((((c[0]*q+c[1])*q+c[2])*q+c[3])*q+c[4])*q+c[5]) /
   617→           ((((d[0]*q+d[1])*q+d[2])*q+d[3])*q+1);
   618→  } else if (p <= pHigh) {
   619→    q = p - 0.5;
   620→    r = q * q;
   621→    return (((((a[0]*r+a[1])*r+a[2])*r+a[3])*r+a[4])*r+a[5]) * q /
   622→           (((((b[0]*r+b[1])*r+b[2])*r+b[3])*r+b[4])*r+1);
   623→  } else {
   624→    q = Math.sqrt(-2 * Math.log(1 - p));
   625→    return -(((((c[0]*q+c[1])*q+c[2])*q+c[3])*q+c[4])*q+c[5]) /
   626→            ((((d[0]*q+d[1])*q+d[2])*q+d[3])*q+1);
   627→  }
   628→}
   629→
   630→/**
   631→ * Calculate p-value from F-statistic
   632→ * p = P(F ≥ f) = 1 - CDF(f)
   633→ * 
   634→ * @param fStat - F-statistic value
   635→ * @param df1 - Numerator degrees of freedom
   636→ * @param df2 - Denominator degrees of freedom
   637→ * @returns Two-tailed p-value
   638→ */
   639→export function pValueFromF(fStat: number, df1: number, df2: number): number {
   640→  if (fStat < 0) return 1;
   641→  return 1 - fDistributionCDF(fStat, df1, df2);
   642→}
   643→
   644→/**
   645→ * Calculate p-value from t-statistic
   646→ * p = 2 × P(|T| ≥ |t|) (two-tailed)
   647→ * 
   648→ * @param tStat - t-statistic value
   649→ * @param df - Degrees of freedom
   650→ * @returns Two-tailed p-value
   651→ */
   652→export function pValueFromT(tStat: number, df: number): number {
   653→  const absT = Math.abs(tStat);
   654→  return 2 * (1 - tDistributionCDF(absT, df));
   655→}
   656→
   657→/**
   658→ * Get significance stars based on p-value
   659→ */
   660→function getSignificance(p: number): string {
   661→  if (p < 0.001) return '***';
   662→  if (p < 0.01) return '**';
   663→  if (p < 0.05) return '*';
   664→  if (p < 0.1) return '.';
   665→  return '';
   666→}

   668→// ============================================================================
   669→// ANOVA ENGINE
   670→// ============================================================================
   671→
   672→/**
   673→ * Perform One-Way Analysis of Variance (ANOVA)
   674→ * 
   675→ * Tests whether means of three or more groups are significantly different.
   676→ * 
   677→ * **Formulas:**
   678→ * - Total SS: SST = ΣΣ(yᵢⱼ - ȳ..)²
   679→ * - Treatment SS: SSt = Σnᵢ(ȳᵢ. - ȳ..)²
   680→ * - Error SS: SSE = SST - SSt
   681→ * - MS = SS / df
   682→ * - F = MStreatment / MSerror
   683→ * 
   684→ * @param groups - Array of arrays, each inner array is a treatment group
   685→ * @returns Complete ANOVA results including F-test and significance
   686→ * 
   687→ * @example
   688→ * ```typescript
   689→ * const data = [[10, 12, 14], [15, 17, 19], [8, 9, 11]];
   690→ * const result = oneWayANOVA(data);
   691→ * console.log(result.f); // F-statistic
   692→ * console.log(result.pValue[0]); // p-value for treatment effect
   693→ * ```
   694→ */
   695→export function oneWayANOVA(groups: number[][]): ANOVAResult {
   696→  // Validate input
   697→  if (!groups || groups.length < 2) {
   698→    throw new Error('ANOVA requires at least 2 groups');
   699→  }
   700→  
   701→  const validGroups = groups.filter(g => g && g.length > 0);
   702→  if (validGroups.length < 2) {
   703→    throw new Error('ANOVA requires at least 2 non-empty groups');
   704→  }
   705→  
   706→  const k = validGroups.length; // Number of groups
   707→  const ni = validGroups.map(g => g.length); // Observations per group
   708→  const N = sum(ni); // Total observations
   709→  
   710→  // Calculate group means and grand mean
   711→  const groupMeans = validGroups.map(g => mean(g));
   712→  const grandMean = mean(validGroups.flat());
   713→  
   714→  // Sum of Squares calculations
   715→  // Total SS: ΣΣ(yᵢⱼ - ȳ..)²
   716→  let totalSS = 0;
   717→  for (const group of validGroups) {
   718→    for (const val of group) {
   719→      totalSS += (val - grandMean) ** 2;
   720→    }
   721→  }
   722→  
   723→  // Treatment (Between Groups) SS: Σnᵢ(ȳᵢ. - ȳ..)²
   724→  let treatmentSS = 0;
   725→  for (let i = 0; i < k; i++) {
   726→    treatmentSS += ni[i] * (groupMeans[i] - grandMean) ** 2;
   727→  }
   728→  
   729→  // Error (Within Groups) SS: SST - SSt
   730→  const errorSS = totalSS - treatmentSS;
   731→  
   732→  // Degrees of Freedom
   733→  const treatmentDF = k - 1;
   734→  const errorDF = N - k;
   735→  const totalDF = N - 1;
   736→  
   737→  // Mean Squares
   738→  const treatmentMS = treatmentSS / treatmentDF;
   739→  const errorMS = errorSS / errorDF;
   740→  
   741→  // F-statistic
   742→  const fStatistic = treatmentMS / errorMS;
   743→  
   744→  // P-value
   745→  const pVal = pValueFromF(fStatistic, treatmentDF, errorDF);
   746→  
   747→  // Critical F at α = 0.05
   748→  const fCrit = criticalF(0.05, treatmentDF, errorDF);
   749→  
   750→  return {
   751→    source: ['Treatment', 'Error', 'Total'],
   752→    df: [treatmentDF, errorDF, totalDF],
   753→    ss: [treatmentSS, errorSS, totalSS],
   754→    ms: [treatmentMS, errorMS, null as unknown as number], // No MS for total
   755→    f: [fStatistic, null as unknown as number, null as unknown as number],
   756→    pValue: [pVal, null as unknown as number, null as unknown as number],
   757→    significance: [getSignificance(pVal), '', ''],
   758→    totalObservations: N,
   759→    grandMean,
   760→    fCritical: fCrit
   761→  };
   762→}
   763→
   764→/**
   765→ * Perform Two-Way Analysis of Variance (with replication)
   766→ * 
   767→ * Tests effects of two factors and their interaction on response variable.
   768→ * 
   769→ * **Formulas:**
   770→ * - Factor A SS: SSA = nb × Σ(ȳᵢ.. - ȳ...)²
   771→ * - Factor B SS: SSB = na × Σ(ȳ.ⱼ. - ȳ...)²
   772→ * - Interaction SS: SSAB = n × ΣΣ(ȳᵢⱼ. - ȳᵢ.. - ȸ.ⱼ. + ȳ...)²
   773→ * - Error SS: SSE = ΣΣΣ(yᵢⱼₖ - ȳᵢⱼ.)²
   774→ * 
   775→ * @param data - 3D array [factorA][factorB][replicates] or flattened with indices
   776→ * @param factorA - Labels for factor A levels
   777→ * @param factorB - Labels for factor B levels
   778→ * @returns Complete Two-Way ANOVA results
   779→ */
   780→export function twoWayANOVA(data: number[][], factorA: string[], factorB: string[]): TwoWayANOVAResult {
   781→  // For simplicity, assume data is organized as rows = factorA, columns = factorB
   782→  // Each cell contains one observation (no replication version first)
   783→  
   784→  const aLevels = factorA.length;
   785→  const bLevels = factorB.length;
   786→  
   787→  if (data.length !== aLevels) {
   788→    throw new Error('Data rows must match factorA length');
   789→  }
   790→  
   791→  // Flatten all values
   792→  const allValues: number[] = [];
   793→  const cellMeans: number[][] = [];
   794→  const rowSums: number[] = [];
   795→  const colSums: number[] = [];
   796→  
   797→  let grandTotal = 0;
   798→  let totalCount = 0;
   799→  
   800→  for (let i = 0; i < aLevels; i++) {
   801→    rowSums[i] = 0;
   802→    cellMeans[i] = [];
   803→    for (let j = 0; j < bLevels; j++) {
   804→      const val = data[i][j];
   805→      allValues.push(val);
   806→      cellMeans[i][j] = val;
   807→      rowSums[i] += val;
   808→      grandTotal += val;
   809→      totalCount++;
   810→      
   811→      if (colSums[j] === undefined) colSums[j] = 0;
   812→      colSums[j] += val;
   813→    }
   814→  }
   815→  
   816→  const grandMean = grandTotal / totalCount;
   817→  
   818→  // Row means (Factor A)
   819→  const rowMeans = rowSums.map(s => s / bLevels);
   820→  
   821→  // Column means (Factor B)
   822→  const colMeans = colSums.map(s => s / aLevels);
   823→  
   824→  // Calculate sums of squares
   825→  // Total SS
   826→  let totalSS = 0;
   827→  for (const val of allValues) {
   828→    totalSS += (val - grandMean) ** 2;
   829→  }
   830→  
   831→  // Factor A SS (Rows)
   832→  let ssA = 0;
   833→  for (let i = 0; i < aLevels; i++) {
   834→    ssA += bLevels * (rowMeans[i] - grandMean) ** 2;
   835→  }
   836→  
   837→  // Factor B SS (Columns)
   838→  let ssB = 0;
   839→  for (let j = 0; j < bLevels; j++) {
   840→    ssB += aLevels * (colMeans[j] - grandMean) ** 2;
   841→  }
   842→  
   843→  // For no-replication model, we can't separate interaction from error
   844→  // So we combine them into residual/error term
   845→  const ssError = totalSS - ssA - ssB;
   846→  
   847→  // Degrees of freedom
   848→  const dfA = aLevels - 1;
   849→  const dfB = bLevels - 1;
   850→  const dfError = (aLevels - 1) * (bLevels - 1);
   851→  const dfTotal = totalCount - 1;
   852→  
   853→  // Mean Squares
   854→  const msA = ssA / dfA;
   855→  const msB = ssB / dfB;
   856→  const msError = ssError / dfError;
   857→  
   858→  // F-statistics
   859→  const fA = msA / msError;
   860→  const fB = msB / msError;
   861→  
   862→  // P-values
   863→  const pA = pValueFromF(fA, dfA, dfError);
   864→  const pB = pValueFromF(fB, dfB, dfError);
   865→  
   866→  return {
   867→    source: ['Factor A', 'Factor B', 'Error', 'Total'],
   868→    df: [dfA, dfB, dfError, dfTotal],
   869→    ss: [ssA, ssB, ssError, totalSS],
   870→    ms: [msA, msB, msError, null as unknown as number],
   871→    f: [fA, fB, null as unknown as number, null as unknown as number],
   872→    pValue: [pA, pB, null as unknown as number, null as unknown as number],
   873→    significance: [getSignificance(pA), getSignificance(pB), '', ''],
   874→    totalObservations: totalCount,
   875→    grandMean,
   876→    factorA: { ss: ssA, df: dfA, ms: msA, f: fA, pValue: pA },
   877→    factorB: { ss: ssB, df: dfB, ms: msB, f: fB, pValue: pB },
   878→    interaction: { ss: 0, df: 0, ms: 0, f: 0, pValue: 1 }, // Not available without replication
   879→    error: { ss: ssError, df: dfError, ms: msError }
   880→  };
   881→}
   882→
   883→/**
   884→ * Analyze Randomized Complete Block Design (RCBD)
   885→ * 
   886→ * RCBD is used when experimental units can be grouped into homogeneous blocks.
   887→ * This reduces experimental error by accounting for block-to-block variation.
   888→ * 
   889→ * **Formulas:**
   890→ * - Treatment SS: SSt = b × Σ(ȳₜ. - ȳ..)²
   891→ * - Block SS: SSbl = t × Σ(ȳ.b - ȳ..)²
   892→ * - Error SS: SSE = SST - SSt - SSbl
   893→ * - CV = (√MSE / Grand Mean) × 100
   894→ * - LSD = t(α/2, dfE) × √(2 × MSE / b)
   895→ * 
   896→ * @param treatments - 2D array [treatments][replications/blocks]
   897→ * @param blocks - Names/labels for each block
   898→ * @returns Complete RCBD analysis results
   899→ * 
   900→ * @example
   901→ * ```typescript
   902→ * // 3 treatments, each measured in 4 blocks
   903→ * const data = [[25, 28, 30, 27], [35, 38, 33, 36], [20, 22, 19, 23]];
   904→ * const result = analyzeRCBD(data, ['Block1', 'Block2', 'Block3', 'Block4']);
   905→ * ```
   906→ */
   907→export function analyzeRCBD(treatments: number[][], blocks: string[]): RBCDResult {
   908→  if (!treatments || treatments.length < 2) {
   909→    throw new Error('RCBD requires at least 2 treatments');
   910→  }
   911→  
   912→  const t = treatments.length; // Number of treatments
   913→  const b = blocks.length; // Number of blocks
   914→  
   915→  // Verify all treatments have same number of replications
   916→  for (let i = 0; i < t; i++) {
   917→    if (!treatments[i] || treatments[i].length !== b) {
   918→      throw new Error(`Treatment ${i} must have exactly ${b} observations (one per block)`);
   919→    }
   920→  }
   921→  
   922→  const N = t * b; // Total observations
   923→  
   924→  // Calculate means
   925→  const treatmentMeans = treatments.map(trt => mean(trt));
   926→  const blockMeans: number[] = [];
   927→  
   928→  for (let j = 0; j < b; j++) {
   929→    const blockVals = treatments.map(trt => trt[j]);
   930→    blockMeans[j] = mean(blockVals);
   931→  }
   932→  
   933→  const grandMean = mean(treatments.flat());
   934→  
   935→  // Sum of Squares
   936→  // Total SS
   937→  let totalSS = 0;
   938→  for (const trt of treatments) {
   939→    for (const val of trt) {
   940→      totalSS += (val - grandMean) ** 2;
   941→    }
   942→  }
   943→  
   944→  // Treatment SS
   945→  let treatmentSS = 0;
   946→  for (let i = 0; i < t; i++) {
   947→    treatmentSS += b * (treatmentMeans[i] - grandMean) ** 2;
   948→  }
   949→  
   950→  // Block SS
   951→  let blockSS = 0;
   952→  for (let j = 0; j < b; j++) {
   953→    blockSS += t * (blockMeans[j] - grandMean) ** 2;
   954→  }
   955→  
   956→  // Error SS
   957→  const errorSS = totalSS - treatmentSS - blockSS;
   958→  
   959→  // Degrees of Freedom
   960→  const treatmentDF = t - 1;
   961→  const blockDF = b - 1;
   962→  const errorDF = (t - 1) * (b - 1);
   963→  const totalDF = N - 1;
   964→  
   965→  // Mean Squares
   966→  const treatmentMS = treatmentSS / treatmentDF;
   967→  const blockMS = blockSS / blockDF;
   968→  const errorMS = errorSS / errorDF;
   969→  
   970→  // F-statistics
   971→  const fTreatment = treatmentMS / errorMS;
   972→  const fBlock = blockMS / errorMS;
   973→  
   974→  // P-values
   975→  const pTreatment = pValueFromF(fTreatment, treatmentDF, errorDF);
   976→  const pBlock = pValueFromF(fBlock, blockDF, errorDF);
   977→  
   978→  // Coefficient of Variation (%)
   979→  const cv = (Math.sqrt(errorMS) / grandMean) * 100;
   980→  
   981→  // Least Significant Difference at α = 0.05
   982→  const tCritical = criticalT(0.05, errorDF);
   983→  const lsd = tCritical * Math.sqrt(2 * errorMS / b);
   984→  
   985→  return {
   986→    source: ['Treatments', 'Blocks', 'Error', 'Total'],
   987→    df: [treatmentDF, blockDF, errorDF, totalDF],
   988→    ss: [treatmentSS, blockSS, errorSS, totalSS],
   989→    ms: [treatmentMS, blockMS, errorMS, null as unknown as number],
   990→    f: [fTreatment, fBlock, null as unknown as number, null as unknown as number],
   991→    pValue: [pTreatment, pBlock, null as unknown as number, null as unknown as number],
   992→    significance: [getSignificance(pTreatment), getSignificance(pBlock), '', ''],
   993→    totalObservations: N,
   994→    grandMean,
   995→    treatments: { ss: treatmentSS, df: treatmentDF, ms: treatmentMS, f: fTreatment, pValue: pTreatment },
   996→    blocks: { ss: blockSS, df: blockDF, ms: blockMS, f: fBlock, pValue: pBlock },
   997→    error: { ss: errorSS, df: errorDF, ms: errorMS },
   998→    cv,
   999→    treatmentMeans,
  1000→    blockMeans,
  1001→    lsd
  1002→  };
  1003→}
  1004→
  1005→/**
  1006→ * Analyze Completely Randomized Design (CRD)
  1007→ * 
  1008→ * CRD is the simplest experimental design where treatments are assigned randomly.
  1009→ * All experimental units are assumed homogeneous.
  1010→ * 
  1011→ * **Formulas:**
  1012→ * - Treatment SS: SSt = Σnᵢ(ȳᵢ. - ȳ..)²
  1013→ * - Error SS: SSE = SST - SSt
  1014→ * - CV = (√MSE / Grand Mean) × 100
  1015→ * - SE = √(MSE / nᵢ)
  1016→ * - LSD = t(α/2, dfE) × √(2 × MSE / n̄)
  1017→ * 
  1018→ * @param treatments - Array of arrays, each inner array is a treatment's replicates
  1019→ * @returns Complete CRD analysis results
  1020→ * 
  1021→ * @example
  1022→ * ```typescript
  1023→ * // 3 treatments with varying replication
  1024→ * const data = [[25, 28, 30, 27], [35, 38, 33], [20, 22, 19, 23, 21]];
  1025→ * const result = analyzeCRD(data);
  1026→ * ```
  1027→ */
  1028→export function analyzeCRD(treatments: number[][]): CRDResult {
  1029→  // Use one-way ANOVA as base
  1030→  const anovaResult = oneWayANOVA(treatments);
  1031→  
  1032→  const t = treatments.length;
  1033→  const ni = treatments.map(trt => trt.length);
  1034→  const N = sum(ni);
  1035→  
  1036→  const treatmentMeans = treatments.map(trt => mean(trt));
  1037→  const errorMS = anovaResult.ms[1];
  1038→  const errorDF = anovaResult.df[1];
  1039→  
  1040→  // Coefficient of Variation (%)
  1041→  const cv = (Math.sqrt(errorMS) / anovaResult.grandMean) * 100;
  1042→  
  1043→  // Standard Errors for each treatment mean
  1044→  const standardErrors = treatments.map(trt => Math.sqrt(errorMS / trt.length));
  1045→  
  1046→  // Average number of replicates for LSD calculation
  1047→  const avgN = N / t;
  1048→  const harmonicN = t / sum(ni.map(n => 1 / n)); // Harmonic mean for unequal reps
  1049→  
  1050→  // Least Significant Difference at α = 0.05
  1051→  const tCritical = criticalT(0.05, errorDF);
  1052→  const lsd = tCritical * Math.sqrt(2 * errorMS / harmonicN);
  1053→  
  1054→  return {
  1055→    ...anovaResult,
  1056→    cv,
  1057→    treatmentMeans,
  1058→    standardErrors,
  1059→    lsd
  1060→  };
  1061→}
  1062→
  1063→// ============================================================================
  1064→// CORRELATION ANALYSIS
  1065→// ============================================================================
  1066→
  1067→/**
  1068→ * Calculate Pearson Product-Moment Correlation Coefficient matrix
  1069→ * 
  1070→ * Formula: r = Cov(X,Y) / (sX × sY) = Σ[(xᵢ-x̄)(yᵢ-ȳ)] / √[Σ(xᵢ-x̄)² × Σ(yᵢ-ȳ)²]
  1071→ * Range: [-1, 1] where ±1 is perfect linear relationship
  1072→ * 
  1073→ * @param data - 2D array where each column is a variable, each row is an observation
  1074→ * @param variableNames - Optional names for variables
  1075→ * @returns Correlation matrix with p-values and sample sizes
  1076→ */
  1077→export function pearsonCorrelation(data: number[][], variableNames?: string[]): CorrelationResult {
  1078→  if (!data || data.length === 0) {
  1079→    throw new Error('Data cannot be empty');
  1080→  }
  1081→  
  1082→  const nVars = data[0].length;
  1083→  const nObs = data.length;
  1084→  
  1085→  // Transpose to work with variables as arrays
  1086→  const vars: number[][] = [];
  1087→  for (let j = 0; j < nVars; j++) {
  1088→    vars[j] = data.map(row => row[j]);
  1089→  }
  1090→  
  1091→  // Check for missing values
  1092→  for (let j = 0; j < nVars; j++) {
  1093→    if (vars[j].some(v => v === undefined || v === null || isNaN(v))) {
  1094→      throw new Error(`Variable ${j} contains missing values`);
  1095→    }
  1096→  }
  1097→  
  1098→  // Generate default names if not provided
  1099→  const names = variableNames || vars.map((_, i) => `Var${i + 1}`);
  1100→  
  1101→  // Calculate correlation matrix
  1102→  const corrMatrix: number[][] = [];
  1103→  const pMatrix: number[][] = [];
  1104→  const nMatrix: number[][] = [];
  1105→  
  1106→  for (let i = 0; i < nVars; i++) {
  1107→    corrMatrix[i] = [];
  1108→    pMatrix[i] = [];
  1109→    nMatrix[i] = [];
  1110→    
  1111→    for (let j = 0; j < nVars; j++) {
  1112→      if (i === j) {
  1113→        corrMatrix[i][j] = 1;
  1114→        pMatrix[i][j] = 0;
  1115→        nMatrix[i][j] = nObs;
  1116→        continue;
  1117→      }
  1118→      
  1119→      const x = vars[i];
  1120→      const y = vars[j];
  1121→      const n = x.length;
  1122→      
  1123→      const meanX = mean(x);
  1124→      const meanY = mean(y);
  1125→      
  1126→      let sumXY = 0, sumX2 = 0, sumY2 = 0;
  1127→      for (let k = 0; k < n; k++) {
  1128→        const dx = x[k] - meanX;
  1129→        const dy = y[k] - meanY;
  1130→        sumXY += dx * dy;
  1131→        sumX2 += dx * dx;
  1132→        sumY2 += dy * dy;
  1133→      }
  1134→      
  1135→      const denom = Math.sqrt(sumX2 * sumY2);
  1136→      const r = denom === 0 ? 0 : sumXY / denom;
  1137→      
  1138→      corrMatrix[i][j] = r;
  1139→      nMatrix[i][j] = n;
  1140→      
  1141→      // Calculate p-value using t-distribution
  1142→      // t = r × √[(n-2)/(1-r²)], df = n-2
  1143→      if (n > 2 && Math.abs(r) < 1) {
  1144→        const tStat = r * Math.sqrt((n - 2) / (1 - r * r));
  1145→        pMatrix[i][j] = pValueFromT(tStat, n - 2);
  1146→      } else {
  1147→        pMatrix[i][j] = 1;
  1148→      }
  1149→    }
  1150→  }
  1151→  
  1152→  return {
  1153→    matrix: corrMatrix,
  1154→    pValues: pMatrix,
  1155→    n: nMatrix,
  1156→    variables: names
  1157→  };
  1158→}
  1159→
  1160→/**
  1161→ * Calculate Spearman Rank Correlation Coefficient matrix
  1162→ * 
  1163→ * Non-parametric measure of rank correlation.
  1164→ * Formula: ρ = 1 - (6 × Σdᵢ²) / (n × (n² - 1))
  1165→ * where dᵢ is difference in ranks for observation i
  1166→ * 
  1167→ * @param data - 2D array where each column is a variable
  1168→ * @param variableNames - Optional names for variables
  1169→ * @returns Spearman correlation matrix with p-values
  1170→ */
  1171→export function spearmanCorrelation(data: number[][], variableNames?: string[]): CorrelationResult {
  1172→  if (!data || data.length === 0) {
  1173→    throw new Error('Data cannot be empty');
  1174→  }
  1175→  
  1176→  const nVars = data[0].length;
  1177→  
  1178→  // Rank each variable
  1179→  const rankData: number[][] = [];
  1180→  for (let j = 0; j < nVars; j++) {
  1181→    const col = data.map(row => row[j]);
  1182→    rankData[j] = calculateRanks(col);
  1183→  }
  1184→  
  1185→  // Transpose ranked data for pearsonCorrelation
  1186→  const transposed: number[][] = [];
  1187→  for (let i = 0; i < data.length; i++) {
  1188→    transposed[i] = [];
  1189→    for (let j = 0; j < nVars; j++) {
  1190→      transposed[i][j] = rankData[j][i];
  1191→    }
  1192→  }
  1193→  
  1194→  return pearsonCorrelation(transposed, variableNames);
  1195→}
  1196→
  1197→/**
  1198→ * Calculate ranks for an array, handling ties with average rank method
  1199→ */
  1200→function calculateRanks(arr: number[]): number[] {
  1201→  const indexed = arr.map((val, idx) => ({ val, idx }));
  1202→  indexed.sort((a, b) => a.val - b.val);
  1203→  
  1204→  const ranks = new Array(arr.length);
  1205→  let i = 0;
  1206→  
  1207→  while (i < arr.length) {
  1208→    let j = i;
  1209→    while (j < arr.length && indexed[j].val === indexed[i].val) {
  1210→      j++;
  1211→    }
  1212→    // Average rank for ties
  1213→    const avgRank = (i + j - 1) / 2 + 1;
  1214→    for (let k = i; k < j; k++) {
  1215→      ranks[indexed[k].idx] = avgRank;
  1216→    }
  1217→    i = j;
  1218→  }
  1219→  
  1220→  return ranks;
  1221→}

  1227→/**
  1228→ * Perform Simple Linear Regression (y = β₀ + β₁x + ε)
  1229→ * 
  1230→ * Uses Ordinary Least Squares (OLS) estimation:
  1231→ * - β₁ = Σ[(xᵢ-x̄)(yᵢ-ȳ)] / Σ(xᵢ-x̄)²
  1232→ * - β₀ = ȳ - β₁x̄
  1233→ * 
  1234→ * @param x - Independent variable values
  1235→ * @param y - Dependent variable values
  1236→ * @returns Complete regression analysis results
  1237→ */
  1238→export function simpleLinearRegression(x: number[], y: number[]): RegressionResult {
  1239→  if (x.length !== y.length) {
  1240→    throw new Error('X and Y must have equal length');
  1241→  }
  1242→  if (x.length < 3) {
  1243→    throw new Error('Need at least 3 observations for regression');
  1244→  }
  1245→  
  1246→  const n = x.length;
  1247→  const meanX = mean(x);
  1248→  const meanY = mean(y);
  1249→  
  1250→  // Calculate slope (β₁) and intercept (β₀)
  1251→  let sumXY = 0, sumXX = 0;
  1252→  for (let i = 0; i < n; i++) {
  1253→    sumXY += (x[i] - meanX) * (y[i] - meanY);
  1254→    sumXX += (x[i] - meanX) ** 2;
  1255→  }
  1256→  
  1257→  const beta1 = sumXY / sumXX;
  1258→  const beta0 = meanY - beta1 * meanX;
  1259→  
  1260→  // Calculate predictions and residuals
  1261→  const predicted = x.map(xi => beta0 + beta1 * xi);
  1262→  const residuals = y.map((yi, i) => yi - predicted[i]);
  1263→  
  1264→  // Sum of Squares
  1265→  const ssReg = sum(predicted.map(p => (p - meanY) ** 2)); // Regression SS
  1266→  const ssRes = sum(residuals.map(r => r ** 2)); // Residual SS
  1267→  const ssTot = sum(y.map(val => (val - meanY) ** 2)); // Total SS
  1268→  
  1269→  // R-squared
  1270→  const rSquared = ssReg / ssTot;
  1271→  
  1272→  // Adjusted R-squared: 1 - (1-R²)(n-1)/(n-p-1)
  1273→  const adjustedRSquared = 1 - (1 - rSquared) * (n - 1) / (n - 2);
  1274→  
  1275→  // Mean Square Error
  1276→  const mse = ssRes / (n - 2);
  1277→  
  1278→  // Standard errors of coefficients
  1279→  const seBeta1 = Math.sqrt(mse / sumXX);
  1280→  const seBeta0 = Math.sqrt(mse * (1 / n + meanX ** 2 / sumXX));
  1281→  
  1282→  // F-statistic: MSR/MSE where MSR = SSR/1
  1283→  const fStatistic = (ssReg / 1) / mse;
  1284→  
  1285→  // P-value for F-test
  1286→  const pValue = pValueFromF(fStatistic, 1, n - 2);
  1287→  
  1288→  // Confidence intervals for regression line (95% CI)
  1289→  const tCrit = criticalT(0.05, n - 2);
  1290→  const confidenceIntervals: [number, number][] = predicted.map((p, i) => {
  1291→    const sePred = Math.sqrt(mse * (1 / n + (x[i] - meanX) ** 2 / sumXX));
  1292→    return [p - tCrit * sePred, p + tCrit * sePred];
  1293→  });
  1294→  
  1295→  // Standardized coefficients (beta weights)
  1296→  const stdX = stddev(x);
  1297→  const stdY = stddev(y);
  1298→  const standardizedCoefficients = [beta0 * (stdX / stdY) + meanY/stdY - beta0/stdY, beta1 * stdX / stdY];
  1299→  
  1300→  return {
  1301→    equation: `y = ${beta0.toFixed(4)} + ${beta1.toFixed(4)}x`,
  1302→    coefficients: [beta0, beta1],
  1303→    stdErrors: [seBeta0, seBeta1],
  1304→    rSquared,
  1305→    adjustedRSquared,
  1306→    fStatistic,
  1307→    pValue,
  1308→    residuals,
  1309→    predicted,
  1310→    confidenceIntervals,
  1311→    standardizedCoefficients
  1312→  };
  1313→}
  1314→
  1315→/**
  1316→ * Perform Multiple Linear Regression (y = Xβ + ε)
  1317→ * 
  1318→ * Uses matrix approach with OLS: β = (X'X)⁻¹X'y
  1319→ * Implements QR decomposition for numerical stability
  1320→ * 
  1321→ * @param X - Matrix of independent variables (including intercept column if needed)
  1322→ * @param y - Dependent variable vector
  1323→ * @returns Complete multiple regression results
  1324→ */
  1325→export function multipleRegression(X: number[][], y: number[]): RegressionResult {
  1326→  const n = X.length; // Number of observations
  1327→  const p = X[0].length; // Number of predictors (including intercept)
  1328→  
  1329→  if (n !== y.length) {
  1330→    throw new Error('X rows must match y length');
  1331→  }
  1332→  if (n < p + 1) {
  1333→    throw new Error(`Need at least ${p + 1} observations for ${p} predictors`);
  1334→  }
  1335→  
  1336→  // Add intercept column if not present (check if first column is all 1s)
  1337→  const hasIntercept = X.every(row => row[0] === 1);
  1338→  const Xmat = hasIntercept ? [...X.map(row => [...row])] : X.map(row => [1, ...row]);
  1339→  const numPredictors = Xmat[0].length;
  1340→  
  1341→  // Matrix operations for OLS: β = (X'X)⁻¹X'y
  1342→  const Xt = transpose(Xmat);
  1343→  const XtX = matMult(Xt, Xmat);
  1344→  const Xty = matVecMult(Xt, y);
  1345→  
  1346→  // Solve using Gaussian elimination with partial pivoting
  1347→  const coefficients = solveLinearSystem(XtX, Xty);
  1348→  
  1349→  // Calculate predicted values and residuals
  1350→  const predicted = Xmat.map(row => dotProduct(row, coefficients));
  1351→  const residuals = y.map((yi, i) => yi - predicted[i]);
  1352→  
  1353→  // Statistics
  1354→  const meanY = mean(y);
  1355→  const ssTot = sum(y.map(val => (val - meanY) ** 2));
  1356→  const ssRes = sum(residuals.map(r => r ** 2));
  1357→  const ssReg = ssTot - ssRes;
  1358→  
  1359→  const rSquared = ssReg / ssTot;
  1360→  const adjRSquared = 1 - (1 - rSquared) * (n - 1) / (n - numPredictors);
  1361→  
  1362→  // MSE and F-statistic
  1363→  const mse = ssRes / (n - numPredictors);
  1364→  const msReg = ssReg / (numPredictors - 1);
  1365→  const fStatistic = msReg / mse;
  1366→  const pValue = pValueFromF(fStatistic, numPredictors - 1, n - numPredictors);
  1367→  
  1368→  // Standard errors of coefficients: sqrt(diag(MSE × (X'X)⁻¹))
  1369→  const XtXinv = invertMatrix(XtX);
  1370→  const varCovar = XtXinv.map(row => row.map(val => val * mse));
  1371→  const stdErrors = varCovar.map(row => Math.sqrt(row[varCovar.indexOf(row)]));
  1372→  
  1373→  // Confidence intervals for coefficients
  1374→  const tCrit = criticalT(0.05, n - numPredictors);
  1375→  const confIntervals: [number, number][] = coefficients.map((coef, i) => {
  1376→    return [coef - tCrit * stdErrors[i], coef + tCrit * stdErrors[i]];
  1377→  });
  1378→  
  1379→  // Standardized coefficients
  1380→  const stdDevs = transpose(Xmat).map(col => stddev(col));
  1381→  const standardizedCoefficients = coefficients.map((coef, i) => 
  1382→    coef * stdDevs[i] / stddev(y)
  1383→  );
  1384→  
  1385→  // Format equation
  1386→  const terms = coefficients.slice(1).map((coef, i) => 
  1387→    `${coef >= 0 ? '+' : ''}${coef.toFixed(4)}×X${i + 1}`
  1388→  );
  1389→  const equation = `y = ${coefficients[0].toFixed(4)} ${terms.join(' ')}`;
  1390→  
  1391→  return {
  1392→    equation,
  1393→    coefficients,
  1394→    stdErrors,
  1395→    rSquared,
  1396→    adjustedRSquared: adjRSquared,
  1397→    fStatistic,
  1398→    pValue,
  1399→    residuals,
  1400→    predicted,
  1401→    confidenceIntervals: confIntervals,
  1402→    standardizedCoefficients
  1403→  };
  1404→}

  1410→/** Transpose a matrix */
  1411→function transpose(matrix: number[][]): number[][] {
  1412→  return matrix[0].map((_, colIdx) => matrix.map(row => row[colIdx]));
  1413→}
  1414→
  1415→/** Matrix multiplication */
  1416→function matMult(A: number[][], B: number[][]): number[][] {
  1417→  const rows = A.length;
  1418→  const cols = B[0].length;
  1419→  const inner = B.length;
  1420→  
  1421→  const result: number[][] = [];
  1422→  for (let i = 0; i < rows; i++) {
  1423→    result[i] = [];
  1424→    for (let j = 0; j < cols; j++) {
  1425→      let sum = 0;
  1426→      for (let k = 0; k < inner; k++) {
  1427→        sum += A[i][k] * B[k][j];
  1428→      }
  1429→      result[i][j] = sum;
  1430→    }
  1431→  }
  1432→  return result;
  1433→}
  1434→
  1435→/** Matrix-vector multiplication */
  1436→function matVecMult(matrix: number[][], vec: number[]): number[] {
  1437→  return matrix.map(row => dotProduct(row, vec));
  1438→}
  1439→
  1440→/** Dot product of two vectors */
  1441→function dotProduct(a: number[], b: number[]): number {
  1442→  return a.reduce((sum, ai, i) => sum + ai * b[i], 0);
  1443→}
  1444→
  1445→/** Solve linear system Ax = b using Gaussian elimination with partial pivoting */
  1446→function solveLinearSystem(A: number[][], b: number[]): number[] {
  1447→  const n = A.length;
  1448→  
  1449→  // Create augmented matrix
  1450→  const aug = A.map((row, i) => [...row, b[i]]);
  1451→  
  1452→  // Forward elimination with partial pivoting
  1453→  for (let col = 0; col < n; col++) {
  1454→    // Find pivot
  1455→    let maxRow = col;
  1456→    for (let row = col + 1; row < n; row++) {
  1457→      if (Math.abs(aug[row][col]) > Math.abs(aug[maxRow][col])) {
  1458→        maxRow = row;
  1459→      }
  1460→    }
  1461→    
  1462→    // Swap rows
  1463→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
  1464→    
  1465→    // Check for singularity
  1466→    if (Math.abs(aug[col][col]) < 1e-10) {
  1467→      throw new Error('Matrix is singular or nearly singular');
  1468→    }
  1469→    
  1470→    // Eliminate below
  1471→    for (let row = col + 1; row < n; row++) {
  1472→      const factor = aug[row][col] / aug[col][col];
  1473→      for (let j = col; j <= n; j++) {
  1474→        aug[row][j] -= factor * aug[col][j];
  1475→      }
  1476→    }
  1477→  }
  1478→  
  1479→  // Back substitution
  1480→  const x = new Array(n).fill(0);
  1481→  for (let i = n - 1; i >= 0; i--) {
  1482→    x[i] = aug[i][n];
  1483→    for (let j = i + 1; j < n; j++) {
  1484→      x[i] -= aug[i][j] * x[j];
  1485→    }
  1486→    x[i] /= aug[i][i];
  1487→  }
  1488→  
  1489→  return x;
  1490→}
  1491→
  1492→/** Invert a matrix using Gauss-Jordan elimination */
  1493→function invertMatrix(matrix: number[][]): number[][] {
  1494→  const n = matrix.length;
  1495→  
  1496→  // Augment with identity matrix
  1497→  const aug = matrix.map((row, i) => {
  1498→    const identityRow = new Array(n).fill(0);
  1499→    identityRow[i] = 1;
  1500→    return [...row, ...identityRow];
  1501→  });
  1502→  
  1503→  // Forward elimination
  1504→  for (let col = 0; col < n; col++) {
  1505→    // Partial pivoting
  1506→    let maxRow = col;
  1507→    for (let row = col + 1; row < n; row++) {
  1508→      if (Math.abs(aug[row][col]) > Math.abs(aug[maxRow][col])) {
  1509→        maxRow = row;
  1510→      }
  1511→    }
  1512→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
  1513→    
  1514→    // Scale pivot row
  1515→    const pivot = aug[col][col];
  1516→    if (Math.abs(pivot) < 1e-10) {
  1517→      throw new Error('Matrix is singular');
  1518→    }
  1519→    for (let j = 0; j < 2 * n; j++) {
  1520→      aug[col][j] /= pivot;
  1521→    }
  1522→    
  1523→    // Eliminate other rows
  1524→    for (let row = 0; row < n; row++) {
  1525→      if (row !== col) {
  1526→        const factor = aug[row][col];
  1527→        for (let j = 0; j < 2 * n; j++) {
  1528→          aug[row][j] -= factor * aug[col][j];
  1529→        }
  1530→      }
  1531→    }
  1532→  }
  1533→  
  1534→  // Extract inverse
  1535→  return aug.map(row => row.slice(n));
  1536→}

  1538→// ============================================================================
  1539→// GENETIC PARAMETERS (PLANT BREEDING)
  1540→// ============================================================================
  1541→
  1542→/**
  1543→ * Calculate Genetic Parameters for plant breeding analysis
  1544→ * 
  1545→ * Based on methodology from:
  1546→ * - Falconer & Mackay (1996) Introduction to Quantitative Genetics
  1547→ * - Singh & Chaudhary (1985) Biometrical Methods in Quantitative Genetics Analysis
  1548→ * 
  1549→ * **Key Formulas:**
  1550→ * - Phenotypic Variance (σ²P) = σ²g + σ²e
  1551→ * - Genotypic Variance (σ²g) = (MSg - MSe) / r
  1552→ * - Environmental Variance (σ²e) = MSe
  1553→ * - Broad-sense Heritability (H²) = σ²g / σ²P
  1554→ * - GCV = (σg / μ) × 100
  1555→ * - PCV = (σp / μ) × 100
  1556→ * - GA = K × H × σp (where K = selection intensity)
  1557→ * - GA% = (GA / μ) × 100
  1558→ * 
  1559→ * @param genotypeData - 2D array [genotypes x replications]
  1560→ * @param selectionIntensity - Selection intensity K (default 2.06 for 5% selection)
  1561→ * @returns Complete genetic parameters with interpretation
  1562→ * 
  1563→ * @example
  1564→ * ```typescript
  1565→ * // 10 genotypes, 3 replications each
  1566→ * const data = [
  1567→ *   [45, 48, 47],  // Genotype 1
  1568→ *   [52, 55, 53],  // Genotype 2
  1569→ *   // ... more genotypes
  1570→ * ];
  1571→ * const params = calculateGeneticParameters(data);
  1572→ * console.log(params.broadSenseHeritability); // e.g., 0.85 (85%)
  1573→ * ```
  1574→ */
  1575→export function calculateGeneticParameters(
  1576→  genotypeData: number[][],
  1577→  selectionIntensity: number = 2.06
  1578→): GeneticParameters {
  1579→  if (!genotypeData || genotypeData.length < 2) {
  1580→    throw new Error('Need at least 2 genotypes for genetic parameter estimation');
  1581→  }
  1582→  
  1583→  const g = genotypeData.length; // Number of genotypes
  1584→  const r = genotypeData[0].length; // Number of replications
  1585→  
  1586→  // Verify all genotypes have same replications
  1587→  for (let i = 0; i < g; i++) {
  1588→    if (!genotypeData[i] || genotypeData[i].length !== r) {
  1589→      throw new Error(`Genotype ${i} must have exactly ${r} replications`);
  1590→    }
  1591→  }
  1592→  
  1593→  // Perform CRD ANOVA to get variance components
  1594→  const crdResult = analyzeCRD(genotypeData);
  1595→  
  1596→  const msGenotype = crdResult.ms[0]; // MS for genotypes (treatments)
  1597→  const msError = crdResult.ms[1]; // MS for error
  1598→  
  1599→  // Variance Components
  1600→  // σ²g = (MSg - MSe) / r
  1601→  const genotypicVariance = Math.max(0, (msGenotype - msError) / r);
  1602→  const environmentalVariance = msError;
  1603→  const phenotypicVariance = genotypicVariance + environmentalVariance;
  1604→  
  1605→  // Trait statistics
  1606→  const allValues = genotypeData.flat();
  1607→  const traitMean = mean(allValues);
  1608→  const traitSD = stddev(allValues);
  1609→  const genotypicSD = Math.sqrt(genotypicVariance);
  1610→  const phenotypicSD = Math.sqrt(phenotypicVariance);
  1611→  
  1612→  // Coefficients of Variation
  1613→  // GCV = (σg / μ) × 100
  1614→  const gcv = traitMean !== 0 ? (genotypicSD / Math.abs(traitMean)) * 100 : 0;
  1615→  // PCV = (σp / μ) × 100
  1616→  const pcv = traitMean !== 0 ? (phenotypicSD / Math.abs(traitMean)) * 100 : 0;
  1617→  
  1618→  // Broad-sense Heritability
  1619→  // H² = σ²g / σ²P
  1620→  const broadSenseHeritability = phenotypicVariance > 0 ? genotypicVariance / phenotypicVariance : 0;
  1621→  
  1622→  // Standard Error of Heritability (approximate)
  1623→  // SE(H²) ≈ √[(4(1-H²)²(1+(r-1)H²)²) / (r(r-1)(g-1)(1+H²)²)]
  1624→  const heritabilitySE = Math.sqrt(
  1625→    (4 * Math.pow(1 - broadSenseHeritability, 2) * Math.pow(1 + (r - 1) * broadSenseHeritability, 2)) /
  1626→    (r * (r - 1) * (g - 1) * Math.pow(1 + broadSenseHeritability, 2))
  1627→  );
  1628→  
  1629→  // Genetic Advance
  1630→  // GA = K × H × σp
  1631→  const geneticAdvance = selectionIntensity * Math.sqrt(broadSenseHeritability) * phenotypicSD;
  1632→  // GA%
  1633→  const geneticAdvancePercent = traitMean !== 0 ? (geneticAdvance / Math.abs(traitMean)) * 100 : 0;
  1634→  
  1635→  // Interpretation based on Johnson et al. (1955) criteria
  1636→  let interpretation = '';
  1637→  
  1638→  // Heritability interpretation
  1639→  if (broadSenseHeritability >= 0.80) {
  1640→    interpretation += 'Very high heritability (>80%). ';
  1641→  } else if (broadSenseHeritability >= 0.60) {
  1642→    interpretation += 'High heritability (60-80%). ';
  1643→  } else if (broadSenseHeritability >= 0.40) {
  1644→    interpretation += 'Moderate heritability (40-60%). ';
  1645→  } else if (broadSenseHeritability >= 0.20) {
  1646→    interpretation += 'Low heritability (20-40%). ';
  1647→  } else {
  1648→    interpretation += 'Very low heritability (<20%). ';
  1649→  }
  1650→  
  1651→  // Genetic Advance interpretation
  1652→  if (geneticAdvancePercent >= 20) {
  1653→    interpretation += 'Very high genetic advance expected. Selection will be highly effective.';
  1654→  } else if (geneticAdvancePercent >= 10) {
  1655→    interpretation += 'High genetic advance expected. Selection should be reasonably effective.';
  1656→  } else if (geneticAdvancePercent >= 5) {
  1657→    interpretation += 'Moderate genetic advance expected. Some progress possible through selection.';
  1658→  } else {
  1659→    interpretation += 'Low genetic advance expected. Consider alternative breeding approaches.';
  1660→  }
  1661→  
  1662→  return {
  1663→    phenotypicVariance,
  1664→    genotypicVariance,
  1665→    environmentalVariance,
  1666→    gcv,
  1667→    pcv,
  1668→    broadSenseHeritability,
  1669→    heritabilitySE,
  1670→    geneticAdvance,
  1671→    geneticAdvancePercent,
  1672→    genotypeCount: g,
  1673→    replicationCount: r,
  1674→    traitMean,
  1675→    traitSD,
  1676→    interpretation
  1677→  };
  1678→}

  1682→// ============================================================================
  1683→
  1684→/**
  1685→ * Calculate Selection Index for multi-trait selection
  1686→ * 
  1687→ * The selection index combines information from multiple traits weighted
  1688→ * by their economic importance and genetic parameters.
  1689→ * 
  1690→ * **Methods:**
  1691→ * - Smith (1936): I = Pb⁻¹g where P=phenotypic var-cov, g=genetic-economic vector
  1692→ * - Base Index: Simple weighted sum with base trait values
  1693→ * - Desired Gains: Weights adjusted to achieve target gains
  1694→ * 
  1695→ * @param traitData - 2D array [genotypes x traits]
  1696→ * @param weights - Relative importance weights for each trait
  1697→ * @param options - Additional options (names, economic weights, etc.)
  1698→ * @returns Selection index values with rankings
  1699→ * 
  1700→ * @example
  1701→ * ```typescript
  1702→ * const traitData = [
  1703→ *   [45, 120, 8.5],  // Genotype 1: yield, height, protein
  1704→ *   [52, 135, 7.8],  // Genotype 2
  1705→ *   // ...
  1706→ * ];
  1707→ * const weights = [0.5, 0.2, 0.3];  // Yield most important
  1708→ * const result = calculateSelectionIndex(traitData, weights);
  1709→ * ```
  1710→ */
  1711→export function calculateSelectionIndex(
  1712→  traitData: number[][],
  1713→  weights: number[],
  1714→  options: {
  1715→    genotypeNames?: string[];
  1716→    traitNames?: string[];
  1717→    economicWeights?: number[];
  1718→    desiredGains?: number[];
  1719→    baseMeans?: number[];
  1720→    method?: 'smith' | 'base' | 'desired';
  1721→  } = {}
  1722→): SelectionIndexResult {
  1723→  const {
  1724→    genotypeNames = [],
  1725→    traitNames = [],
  1726→    economicWeights = weights,
  1727→    desiredGains = [],
  1728→    baseMeans = [],
  1729→    method = 'base'
  1730→  } = options;
  1731→  
  1732→  const nGenotypes = traitData.length;
  1733→  const nTraits = traitData[0]?.length || 0;
  1734→  
  1735→  if (nTraits === 0) {
  1736→    throw new Error('No traits provided');
  1737→  }
  1738→  if (weights.length !== nTraits) {
  1739→    throw new Error('Weights length must match number of traits');
  1740→  }
  1741→  
  1742→  // Default genotype names
  1743→  const genotypes = genotypeNames.length > 0 
  1744→    ? genotypeNames 
  1745→    : traitData.map((_, i) => `G${i + 1}`);
  1746→  
  1747→  const traits = traitNames.length > 0
  1748→    ? traitNames
  1749→    : traitData[0].map((_, i) => `Trait${i + 1}`);
  1750→  
  1751→  // Normalize weights to sum to 1
  1752→  const totalWeight = sum(weights);
  1753→  const normWeights = weights.map(w => w / totalWeight);
  1754→  
  1755→  // Calculate index values based on method
  1756→  let indexValues: number[];
  1757→  
  1758→  switch (method) {
  1759→    case 'smith': {
  1760→      // Smith's selection index: I = Pb⁻¹g
  1761→      // Simplified version using phenotypic correlations
  1762→      const traitMeans = traitData[0].map((_, j) => 
  1763→        mean(traitData.map(row => row[j]))
  1764→      );
  1765→      
  1766→      // Standardize traits
  1767→      const stdTraits = traitData.map(row =>
  1768→        row.map((val, j) => (val - traitMeans[j]) / stddev(traitData.map(r => r[j])))
  1769→      );
  1770→      
  1771→      // Apply economic weights
  1772→      indexValues = stdTraits.map(row =>
  1773→        dotProduct(row, economicWeights.map((w, i) => w * normWeights[i]))
  1774→      );
  1775→      break;
  1776→    }
  1777→    
  1778→    case 'desired': {
  1779→      // Desired gains index
  1780→      const traitMeans = traitData[0].map((_, j) => 
  1781→        mean(traitData.map(row => row[j]))
  1782→      );
  1783→      
  1784→      if (desiredGains.length === nTraits) {
  1785→        // Weight by ratio of observed to desired gain potential
  1786→        indexValues = traitData.map(row =>
  1787→          row.reduce((sum, val, j) => {
  1788→            const deviation = (val - traitMeans[j]) / (traitMeans[j] || 1);
  1789→            return sum + deviation * normWeights[j] * desiredGains[j];
  1790→          }, 0)
  1791→        );
  1792→      } else {
  1793→        // Fall back to base method
  1794→        indexValues = traitData.map(row => dotProduct(row, normWeights));
  1795→      }
  1796→      break;
  1797→    }
  1798→    
  1799→    case 'base':
  1800→    default: {
  1801→      // Base index: simple weighted sum
  1802→      // Optionally adjust by base means
  1803→      if (baseMeans.length === nTraits) {
  1804→        indexValues = traitData.map(row =>
  1805→          row.reduce((sum, val, j) => {
  1806→            const relativePerformance = val / (baseMeans[j] || 1);
  1807→            return sum + relativePerformance * normWeights[j];
  1808→          }, 0)
  1809→        );
  1810→      } else {
  1811→        indexValues = traitData.map(row => dotProduct(row, normWeights));
  1812→      }
  1813→      break;
  1814→    }
  1815→  }
  1816→  
  1817→  // Rank genotypes (highest index = rank 1)
  1818→  const ranked = indexValues
  1819→    .map((val, idx) => ({ val, idx }))
  1820→    .sort((a, b) => b.val - a.val);
  1821→  
  1822→  const ranks = new Array(nGenotypes).fill(0);
  1823→  ranked.forEach((item, rank) => {
  1824→    ranks[item.idx] = rank + 1;
  1825→  });
  1826→  
  1827→  // Select top 20% (or at least 1) genotypes
  1828→  const nSelected = Math.max(1, Math.ceil(nGenotypes * 0.2));
  1829→  const selectedGenotypes = ranked.slice(0, nSelected).map(item => genotypes[item.idx]);
  1830→  
  1831→  // Build criteria info
  1832→  const criteria = traits.map((trait, i) => ({
  1833→    trait,
  1834→    weight: normWeights[i],
  1835→    desiredGain: desiredGains[i]
  1836→  }));
  1837→  
  1838→  return {
  1839→    genotypes,
  1840→    traitValues: traitData,
  1841→    weights: normWeights,
  1842→    indexValues,
  1843→    ranks,
  1844→    selectedGenotypes,
  1845→    criteria
  1846→  };
  1847→}

  1853→/**
  1854→ * Perform Path Coefficient Analysis
  1855→ * 
  1856→ * Path analysis decomposes correlation coefficients into direct and indirect
  1857→ * effects, helping understand causal relationships among variables.
  1858→ * 
  1859→ * **Method:** Wright's path analysis using standardized partial regression
  1860→ * 
  1861→ * **Formulas:**
  1862→ * - Direct effects: Solution to R × P = r where R is correlation matrix
  1863→ *   among independent variables, r is their correlations with dependent
  1864→ * - Indirect effects: rij × pj (correlation × path coefficient)
  1865→ * - Residual effect: √(1 - R²) where R² = Σ(pi × riy)
  1866→ * 
  1867→ * @param correlationMatrix - Full correlation matrix including dependent variable
  1868→ * @param dependentIndex - Column/row index of dependent variable
  1869→ * @param variableNames - Optional names for variables
  1870→ * @returns Path analysis results with direct and indirect effects
  1871→ * 
  1872→ * @example
  1873→ * ```typescript
  1874→ * // Correlation matrix: [yield, height, biomass, nodes]
  1875→ * const corrMat = [
  1876→ *   [1.0, 0.6, 0.8, 0.4],  // yield
  1877→ *   [0.6, 1.0, 0.5, 0.7],  // height
  1878→ *   [0.8, 0.5, 1.0, 0.3],  // biomass
  1879→ *   [0.4, 0.7, 0.3, 1.0],  // nodes
  1880→ * ];
  1881→ * const result = pathAnalysis(corrMat, 0); // yield is dependent
  1882→ * console.log(result.directEffects); // Path coefficients
  1883→ * ```
  1884→ */
  1885→export function pathAnalysis(
  1886→  correlationMatrix: number[][],
  1887→  dependentIndex: number,
  1888→  variableNames?: string[]
  1889→): PathAnalysisResult {
  1890→  const n = correlationMatrix.length;
  1891→  
  1892→  if (dependentIndex < 0 || dependentIndex >= n) {
  1893→    throw new Error('Dependent index out of range');
  1894→  }
  1895→  
  1896→  // Variable names
  1897→  const names = variableNames || correlationMatrix.map((_, i) => `X${i + 1}`);
  1898→  const depName = names[dependentIndex];
  1899→  const indepNames = names.filter((_, i) => i !== dependentIndex);
  1900→  
  1901→  // Extract submatrices
  1902→  // Independent variable correlations (R matrix)
  1903→  const R: number[][] = [];
  1904→  const r: number[] = []; // Correlations with dependent variable
  1905→  
  1906→  for (let i = 0; i < n; i++) {
  1907→    if (i === dependentIndex) continue;
  1908→    
  1909→    const rowR: number[] = [];
  1910→    for (let j = 0; j < n; j++) {
  1911→      if (j === dependentIndex) continue;
  1912→      rowR.push(correlationMatrix[i][j]);
  1913→    }
  1914→    R.push(rowR);
  1915→    r.push(correlationMatrix[i][dependentIndex]);
  1916→  }
  1917→  
  1918→  const nIndep = R.length;
  1919→  
  1920→  // Solve for direct effects: R × P = r
  1921→  // Using matrix inversion: P = R⁻¹ × r
  1922→  let directEffects: number[];
  1923→  
  1924→  try {
  1925→    const Rinv = invertMatrix(R);
  1926→    directEffects = matVecMult(Rinv, r);
  1927→  } catch (e) {
  1928→    // If matrix is singular, use pseudo-inverse approximation
  1929→    directEffects = new Array(nIndep).fill(1 / nIndep);
  1930→  }
  1931→  
  1932→  // Calculate indirect effects
  1933→  // Indirect effect of Xj on Y through Xi = rij × pi
  1934→  const indirectEffects: number[][] = [];
  1935→  
  1936→  for (let i = 0; i < nIndep; i++) {
  1937→    indirectEffects[i] = [];
  1938→    for (let j = 0; j < nIndep; j++) {
  1939→      if (i === j) {
  1940→        indirectEffects[i][j] = 0; // No self indirect effect
  1941→      } else {
  1942→        // Indirect of Xi through Xj = corr(Xi,Xj) × pj
  1943→        indirectEffects[i][j] = R[i][j] * directEffects[j];
  1944→      }
  1945→    }
  1946→  }
  1947→  
  1948→  // Coefficient of determination (R²)
  1949→  // R² = Σ(pi × riy)
  1950→  let rSquared = 0;
  1951→  for (let i = 0; i < nIndep; i++) {
  1952→    rSquared += directEffects[i] * r[i];
  1953→  }
  1954→  
  1955→  // Residual effect
  1956→  const residualEffect = Math.sqrt(Math.max(0, 1 - rSquared));
  1957→  
  1958→  // Build full path coefficient matrix
  1959→  const pathCoefficients: number[][] = [];
  1960→  for (let i = 0; i < nIndep; i++) {
  1961→    pathCoefficients[i] = [...indirectEffects[i]];
  1962→    pathCoefficients[i][i] = directEffects[i]; // Diagonal = direct effects
  1963→  }
  1964→  
  1965→  return {
  1966→    directEffects,
  1967→    indirectEffects,
  1968→    residualEffect,
  1969→    correlationMatrix,
  1970→    pathCoefficients,
  1971→    rSquared,
  1972→    dependentVariable: depName,
  1973→    independentVariables: indepNames
  1974→  };
  1975→}
