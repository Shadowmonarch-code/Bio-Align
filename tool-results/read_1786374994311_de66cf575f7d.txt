   142→function matrixTranspose(matrix: number[][]): number[][] {
   143→  if (matrix.length === 0) return []
   144→  return matrix[0].map((_, colIdx) => matrix.map(row => row[colIdx]))
   145→}
   146→
   147→function matrixMultiply(A: number[][], B: number[][]): number[][] {
   148→  const rowsA = A.length
   149→  const colsA = A[0]?.length || 0
   150→  const colsB = B[0]?.length || 0
   151→  
   152→  const result: number[][] = []
   153→  for (let i = 0; i < rowsA; i++) {
   154→    result[i] = []
   155→    for (let j = 0; j < colsB; j++) {
   156→      let sum = 0
   157→      for (let k = 0; k < colsA; k++) {
   158→        sum += A[i][k] * B[k][j]
   159→      }
   160→      result[i][j] = sum
   161→    }
   162→  }
   163→  return result
   164→}

   166→function copyMatrix(mat: number[][]): number[][] {
   167→  return mat.map(row => [...row])
   168→}
   169→
   170→function matrixMultiplyVector(matrix: number[][], vec: number[]): number[] {
   171→  return matrix.map(row => 
   172→    row.reduce((sum, val, i) => sum + val * (vec[i] || 0), 0)
   173→  )
   174→}

   176→/**
   177→ * Compute Singular Value Decomposition using power iteration with deflation
   178→ * Returns U, singular values array, and V transpose
   179→ */
   180→function computeSVD(
   181→  matrix: number[][]
   182→): { U: number[][]; S: number[]; Vt: number[][] } {
   183→  const m = matrix.length
   184→  const n = matrix[0]?.length || 0
   185→  
   186→  if (m === 0 || n === 0) {
   187→    return { U: [], S: [], Vt: [] }
   188→  }
   189→  
   190→  // Work with A^T * A for numerical stability
   191→  const AtA = matrixMultiply(matrixTranspose(matrix), matrix)
   192→  const size = Math.min(m, n)
   193→  
   194→  const eigenvalues: number[] = []
   195→  const eigenvectors: number[][] = []
   196→  
   197→  let workingMatrix = copyMatrix(AtA)
   198→  
   199→  for (let iter = 0; iter < size; iter++) {
   200→    let v = new Array(n).fill(0)
   201→    v[iter < n ? iter : 0] = 1
   202→    
   203→    for (let powerIter = 0; powerIter < 100; powerIter++) {
   204→      const Av = eigenvectors.length > 0 && iter > 0 
   205→        ? deflateVector(matrixMultiply(workingMatrix, v), eigenvectors, eigenvalues)
   206→        : matrixMultiplyVector(workingMatrix, v)
   207→      
   208→      const norm = Math.sqrt(Av.reduce((sum, x) => sum + x * x, 0))
   209→      if (norm < 1e-10) break
   210→      
   211→      v = Av.map(x => x / norm)
   212→    }
   213→    
   214→    const Av = matrixMultiplyVector(workingMatrix, v)
   215→    const eigenvalue = v.reduce((sum, vi, i) => sum + vi * Av[i], 0)
   216→    
   217→    if (eigenvalue > 1e-10) {
   218→      eigenvalues.push(Math.max(0, eigenvalue))
   219→      eigenvectors.push(v)
   220→      
   221→      // Deflate
   222→      for (let i = 0; i < n; i++) {
   223→        for (let j = 0; j < n; j++) {
   224→          workingMatrix[i][j] -= eigenvalue * v[i] * v[j]
   225→        }
   226→      }
   227→    } else {
   228→      break
   229→    }
   230→  }
   231→  
   232→  const S = eigenvalues.map(ev => Math.sqrt(ev))
   233→  const V = eigenvectors.length > 0 ? matrixTranspose(eigenvectors) : []
   234→  const Vt = matrixTranspose(V)
   235→  
   236→  // Compute U = A * V * Σ^(-1)
   237→  const U: number[][] = []
   238→  for (let i = 0; i < m; i++) {
   239→    U[i] = []
   240→    for (let j = 0; j < S.length; j++) {
   241→      if (S[j] > 1e-10) {
   242→        let sum = 0
   243→        for (let k = 0; k < n; k++) {
   244→          sum += matrix[i][k] * (V[k]?.[j] || 0)
   245→        }
   246→        U[i][j] = sum / S[j]
   247→      } else {
   248→        U[i][j] = 0
   249→      }
   250→    }
   251→  }
   252→  
   253→  return { U, S, Vt }
   254→}
   255→
   256→function deflateVector(
   257→  v: number[], 
   258→  eigenvectors: number[][], 
   259→  eigenvalues: number[]
   260→): number[] {
   261→  let result = [...v]
   262→  for (let k = 0; k < eigenvectors.length; k++) {
   263→    const dot = v.reduce((sum, vi, i) => sum + vi * eigenvectors[k][i], 0)
   264→    for (let i = 0; i < v.length; i++) {
   265→      result[i] -= dot * eigenvalues[k] * eigenvectors[k][i]
   266→    }
   267→  }
   268→  return result
   269→}

   271→/**
   272→ * Calculate angle between two vectors in radians
   273→ */
   274→function angleBetween(v1: [number, number], v2: [number, number]): number {
   275→  const dot = v1[0] * v2[0] + v1[1] * v2[1]
   276→  const norm1 = Math.sqrt(v1[0] ** 2 + v1[1] ** 2)
   277→  const norm2 = Math.sqrt(v2[0] ** 2 + v2[1] ** 2)
   278→  if (norm1 < 1e-10 || norm2 < 1e-10) return 0
   279→  return Math.acos(Math.max(-1, Math.min(1, dot / (norm1 * norm2))))
   280→}
   281→
   282→/**
   283→ * Calculate cross product (for determining sector)
   284→ */
   285→function crossProduct(v1: [number, number], v2: [number, number]): number {
   286→  return v1[0] * v2[1] - v1[1] * v2[0]
   287→}

   292→
   293→/**
   294→ * Perform complete GGE Biplot Analysis based on Yan & Kang (2003) methodology
   295→ * 
   296→ * Key concept: GGE Biplot = "Genotype main effect + Genotype × Environment interaction"
   297→ * This is different from AMMI which analyzes only the G×E interaction.
   298→ * 
   299→ * Steps:
   300→ * 1. Center data by environment means (remove environment main effect)
   301→ * 2. SVD of the centered (G+GE) matrix
   302→ * 3. Calculate coordinates for genotypes and environments
   303→ * 4. Determine which-won-where pattern
   304→ */
   305→function performGGEBiplotAnalysis(
   306→  genotypeNames: string[],
   307→  environmentNames: string[],
   308→  dataMatrix: number[][],
   309→  scalingMethod: 'symmetric' | 'genotype' | 'environment' = 'symmetric'
   310→): GGEBiplotResult {
   311→  const g = genotypeNames.length
   312→  const e = environmentNames.length
   313→  const N = g * e
   314→  
   315→  // Step 1: Calculate environment means and center the data
   316→  const environmentMeans: number[] = []
   317→  const centeredMatrix: number[][] = []
   318→  
   319→  for (let j = 0; j < e; j++) {
   320→    let envSum = 0
   321→    for (let i = 0; i < g; i++) {
   322→      envSum += dataMatrix[i][j]
   323→    }
   324→    environmentMeans.push(envSum / g)
   325→  }
   326→  
   327→  // Center by environment means: Y_ij - Ȳ_j
   328→  // This removes environment main effect, leaving G + GE
   329→  let totalCenteredSS = 0
   330→  for (let i = 0; i < g; i++) {
   331→    centeredMatrix[i] = []
   332→    for (let j = 0; j < e; j++) {
   333→      const centered = dataMatrix[i][j] - environmentMeans[j]
   334→      centeredMatrix[i][j] = centered
   335→      totalCenteredSS += centered ** 2
   336→    }
   337→  }
   338→  
   339→  // Step 2: Total SS and Environment SS for reference
   340→  const grandMean = dataMatrix.flat().reduce((a, b) => a + b, 0) / N
   341→  let totalSS = 0
   342→  for (let i = 0; i < g; i++) {
   343→    for (let j = 0; j < e; j++) {
   344→      totalSS += (dataMatrix[i][j] - grandMean) ** 2
   345→    }
   346→  }
   347→  
   348→  let environmentSS = 0
   349→  for (let j = 0; j < e; j++) {
   350→    environmentSS += g * (environmentMeans[j] - grandMean) ** 2
   351→  }
   352→  
   353→  const ggeSS = totalCenteredSS
   354→  
   355→  // Step 3: SVD of centered (G+GE) matrix
   356→  const { U, S, Vt } = computeSVD(centeredMatrix)
   357→  
   358→  // Step 4: Calculate variance explained
   359→  const numPCs = Math.min(S.length, Math.min(g, e))
   360→  let cumulativeVariance = 0
   361→  const pcAxes: PCAxis[] = []
   362→  
   363→  for (let k = 0; k < numPCs; k++) {
   364→    const eigenvalue = S[k] ** 2
   365→    const variancePercent = (eigenvalue / ggeSS) * 100
   366→    cumulativeVariance += variancePercent
   367→    
   368→    pcAxes.push({
   369→      axis: k + 1,
   370→      singularValue: S[k],
   371→      eigenvalue,
   372→      variancePercent,
   373→      cumulativePercent: cumulativeVariance
   374→    })
   375→  }
   376→  
   377→  // Step 5: Calculate coordinates based on scaling method
   378→  // Symmetric scaling: sqrt(lambda) for both
   379→  // Genotype-focused: full lambda to genotype
   380→  // Environment-focused: full lambda to environment
   381→  
   382→  let genotypeScaling: number[]
   383→  let environmentScaling: number[]
   384→  
   385→  if (scalingMethod === 'symmetric') {
   386→    const sqrtS = S.map(s => Math.sqrt(s))
   387→    genotypeScaling = sqrtS
   388→    environmentScaling = sqrtS
   389→  } else if (scalingMethod === 'genotype') {
   390→    genotypeScaling = S.map(s => s)
   391→    environmentScaling = S.map(s => 1)
   392→  } else {
   393→    genotypeScaling = S.map(s => 1)
   394→    environmentScaling = S.map(s => s)
   395→  }
   396→  
   397→  // Step 6: Calculate genotype coordinates
   398→  const genotypeMeans: number[] = []
   399→  for (let i = 0; i < g; i++) {
   400→    let sum = 0
   401→    for (let j = 0; j < e; j++) {
   402→      sum += dataMatrix[i][j]
   403→    }
   404→    genotypeMeans.push(sum / e)
   405→  }
   406→  
   407→  const genotypeCoords: GenotypeCoordinate[] = []
   408→  for (let i = 0; i < g; i++) {
   409→    const pc1 = (U[i]?.[0] || 0) * (genotypeScaling[0] || 0)
   410→    const pc2 = (U[i]?.[1] || 0) * (genotypeScaling[1] || 0)
   411→    const pc3 = (U[i]?.[2] || 0) * (genotypeScaling[2] || 0)
   412→    
   413→    // Stability measure: distance from origin in PC space
   414→    // In GGE biplot, stability is inversely related to this distance
   415→    const stabilityMeasure = Math.sqrt(pc1 ** 2 + pc2 ** 2)
   416→    
   417→    // Mean yield rank (higher is better)
   418→    const meanYield = genotypeMeans[i]
   419→    
   420→    // Ideal genotype distance: distance from "ideal" point
   421→    // The ideal genotype has highest mean AND perfect stability
   422→    // We'll calculate this after determining the ideal point
   423→    
   424→    let category: string
   425→    if (stabilityMeasure < 50) category = 'Very Stable'
   426→    else if (stabilityMeasure < 100) category = 'Stable'
   427→    else if (stabilityMeasure < 200) category = 'Moderate'
   428→    else if (stabilityMeasure < 350) category = 'Unstable'
   429→    else category = 'Very Unstable'
   430→    
   431→    genotypeCoords.push({
   432→      genotype: genotypeNames[i],
   433→      pc1,
   434→      pc2,
   435→      pc3,
   436→      meanYield,
   437→      stabilityMeasure,
   438→      stabilityRank: 0,
   439→      meanRank: 0,
   440→      idealDistance: 0,
   441→      category
   442→    })
   443→  }
   444→  
   445→  // Rank genotypes by mean yield and stability
   446→  const sortedByMean = [...genotypeCoords].sort((a, b) => b.meanYield - a.meanYield)
   447→  sortedByMean.forEach((g, idx) => {
   448→    const original = genotypeCoords.find(gc => gc.genotype === g.genotype)
   449→    if (original) original.meanRank = idx + 1
   450→  })
   451→  
   452→  const sortedByStability = [...genotypeCoords].sort((a, b) => a.stabilityMeasure - b.stabilityMeasure)
   453→  sortedByStability.forEach((g, idx) => {
   454→    const original = genotypeCoords.find(gc => gc.genotype === g.genotype)
   455→    if (original) original.stabilityRank = idx + 1
   456→  })
   457→  
   458→  // Step 7: Calculate environment coordinates
   459→  const environmentCoords: EnvironmentCoordinate[] = []
   460→  for (let j = 0; j < e; j++) {
   461→    const pc1 = (Vt[0]?.[j] || 0) * (environmentScaling[0] || 0)
   462→    const pc2 = (Vt[1]?.[j] || 0) * (environmentScaling[1] || 0)
   463→    const pc3 = (Vt[2]?.[j] || 0) * (environmentScaling[2] || 0)
   464→    
   465→    // Vector length = discriminating power
   466→    const vectorLength = Math.sqrt(pc1 ** 2 + pc2 ** 2)
   467→    
   468→    let discriminatingPower: string
   469→    if (vectorLength > 150) discriminatingPower = 'High'
   470→    else if (vectorLength > 80) discriminatingPower = 'Moderate'
   471→    else discriminatingPower = 'Low'
   472→    
   473→    // Representativeness: correlation with average environment
   474→    // Average environment has coordinates at average of all env coordinates
   475→    // For now, use proximity to origin as proxy (will refine after all coords calculated)
   476→    
   477→    environmentCoords.push({
   478→      environment: environmentNames[j],
   479→      pc1,
   480→      pc2,
   481→      pc3,
   482→      meanYield: environmentMeans[j],
   483→      vectorLength,
   484→      discriminatingPower,
   485→      representativeness: 0,
   486→      idealEnvDistance: 0
   487→    })
   488→  }
   489→  
   490→  // Calculate representativeness (angle with average environment axis)
   491→  const avgEnvPC1 = environmentCoords.reduce((sum, e) => sum + e.pc1, 0) / e
   492→  const avgEnvPC2 = environmentCoords.reduce((sum, e) => sum + e.pc2, 0) / e
   493→  const avgEnvVector: [number, number] = [avgEnvPC1, avgEnvPC2]
   494→  const avgEnvNorm = Math.sqrt(avgEnvPC1 ** 2 + avgEnvPC2 ** 2)
   495→  
   496→  for (const env of environmentCoords) {
   497→    const envVector: [number, number] = [env.pc1, env.pc2]
   498→    const cosAngle = avgEnvNorm > 1e-10 
   499→      ? (envVector[0] * avgEnvVector[0] + envVector[1] * avgEnvVector[1]) / (Math.sqrt(envVector[0] ** 2 + envVector[1] ** 2) * avgEnvNorm)
   500→      : 1
   501→    env.representativeness = Math.max(-1, Math.min(1, cosAngle))
   502→  }
   503→  
   504→  // Step 8: Determine ideal genotype point
   505→  // Ideal genotype: one with highest mean performance AND absolute stability
   506→  // Located at the intersection of "ideal" position - longest vector in positive PC1 direction
   507→  // that also has minimum PC2 (stability)
   508→  
   509→  // Find the genotype with maximum PC1 (mean performance proxy in symmetric scaling)
   510→  const maxPC1 = Math.max(...genotypeCoords.map(g => g.pc1))
   511→  const minAbsPC2AmongTop = Math.min(
   512→    ...genotypeCoords
   513→      .filter(g => g.pc1 > maxPC1 * 0.8)
   514→      .map(g => Math.abs(g.pc2))
   515→  )
   516→  
   517→  // Ideal genotype is at the "arrow head" position
   518→  const idealGenotype = {
   519→    pc1: maxPC1 * 1.15,  // Extend slightly beyond best genotype
   520→    pc2: 0               // Perfect stability
   521→  }
   522→  
   523→  // Calculate distances to ideal genotype
   524→  for (const geno of genotypeCoords) {
   525→    const dx = geno.pc1 - idealGenotype.pc1
   526→    const dy = geno.pc2 - idealGenotype.pc2
   527→    geno.idealDistance = Math.sqrt(dx ** 2 + dy ** 2)
   528→  }
   529→  
   530→  // Step 9: Determine ideal environment point
   531→  // Ideal environment: high discriminating power + high representativeness
   532→  // Located in direction of average environment with long vector
   533→  const idealEnvironment = {
   534→    pc1: avgEnvPC1 * 1.3,
   535→    pc2: avgEnvPC2 * 1.3
   536→  }
   537→  
   538→  // Calculate distances to ideal environment
   539→  for (const env of environmentCoords) {
   540→    const dx = env.pc1 - idealEnvironment.pc1
   541→    const dy = env.pc2 - idealEnvironment.pc2
   542→    env.idealEnvDistance = Math.sqrt(dx ** 2 + dy ** 2)
   543→  }
   544→  
   545→  // Step 10: Which-Won-Where analysis
   546→  // Draw convex hull around genotypes, determine sectors, assign environments
   547→  const whichWonWhere = performWhichWonWhere(genotypeCoords, environmentCoords)
   548→  
   549→  return {
   550→    genotypes: genotypeNames,
   551→    environments: environmentNames,
   552→    dataMatrix,
   553→    centeredMatrix,
   554→    environmentMeans,
   555→    singularValues: S,
   556→    pcAxes,
   557→    genotypeCoords,
   558→    environmentCoords,
   559→    whichWonWhere,
   560→    idealGenotype,
   561→    idealEnvironment,
   562→    anova: {
   563→      totalSS,
   564→      environmentSS,
   565→      ggeSS,
   566→      ggePercent: (ggeSS / totalSS) * 100
   567→    },
   568→    scalingMethod
   569→  }
   570→}

   572→/**
   573→ * Perform Which-Won-Where analysis using polygon view approach
   574→ * Based on Yan et al. (2000) methodology
   575→ */
   576→function performWhichWonWhere(
   577→  genotypeCoords: GenotypeCoordinate[],
   578→  environmentCoords: EnvironmentCoordinate[]
   579→): WhichWonWhereEntry[] {
   580→  if (genotypeCoords.length < 3) {
   581→    return [{
   582→      sector: 1,
   583→      winningGenotype: genotypeCoords[0]?.genotype || '',
   584→      environments: environmentCoords.map(e => e.environment),
   585→      vertexGenotypes: genotypeCoords.map(g => g.genotype)
   586→    }]
   587→  }
   588→  
   589→  // Find convex hull vertices (genotypes on the outer boundary)
   590→  const points = genotypeCoords.map(g => ({ x: g.pc1, y: g.pc2, name: g.genotype }))
   591→  const hullIndices = convexHull(points)
   592→  const vertexGenotypes = hullIndices.map(i => genotypeCoords[i].genotype)
   593→  
   594→  // Determine sectors from the origin through each vertex
   595→  // Each sector is defined by perpendicular bisectors of adjacent hull edges
   596→  const sectors: { startAngle: number; endAngle: number; vertex: string }[] = []
   597→  
   598→  for (let i = 0; i < hullIndices.length; i++) {
   599→    const currIdx = hullIndices[i]
   600→    const nextIdx = hullIndices[(i + 1) % hullIndices.length]
   601→    
   602→    const currPoint = points[currIdx]
   603→    const nextPoint = points[nextIdx]
   604→    
   605→    // Angle to current vertex
   606→    const angleCurr = Math.atan2(currPoint.y, currPoint.x)
   607→    const angleNext = Math.atan2(nextPoint.y, nextPoint.x)
   608→    
   609→    // Sector midpoint (direction to winning area)
   610→    let startAngle: number
   611→    let endAngle: number
   612→    
   613→    if (i === 0) {
   614→      // First sector: from previous vertex's bisector to current bisector
   615→      const prevIdx = hullIndices[hullIndices.length - 1]
   616→      const prevPoint = points[prevIdx]
   617→      const anglePrev = Math.atan2(prevPoint.y, prevPoint.x)
   618→      
   619→      startAngle = (anglePrev + angleCurr) / 2
   620→      if (angleCurr > anglePrev) startAngle += Math.PI
   621→      
   622→      endAngle = (angleCurr + angleNext) / 2
   623→      if (angleNext < angleCurr) endAngle += Math.PI
   624→    } else {
   625→      const prevIdx = hullIndices[i - 1]
   626→      const prevPoint = points[prevIdx]
   627→      const anglePrev = Math.atan2(prevPoint.y, prevPoint.x)
   628→      
   629→      startAngle = (anglePrev + angleCurr) / 2
   630→      endAngle = (angleCurr + angleNext) / 2
   631→    }
   632→    
   633→    sectors.push({
   634→      startAngle,
   635→      endAngle,
   636→      vertex: genotypeCoords[currIdx].genotype
   637→    })
   638→  }
   639→  
   640→  // Assign each environment to a sector
   641→  const results: Map<string, { winner: string; sector: number; vertices: string[] }> = new Map()
   642→  
   643→  for (const env of environmentCoords) {
   644→    const envAngle = Math.atan2(env.pc2, env.pc1)
   645→    
   646→    let assignedSector = 0
   647→    let minAngleDiff = Infinity
   648→    
   649→    for (let s = 0; s < sectors.length; s++) {
   650→      const sector = sectors[s]
   651→      
   652→      // Normalize angles for comparison
   653→      let angleDiff = Math.abs(envAngle - (sector.startAngle + sector.endAngle) / 2)
   654→      if (angleDiff > Math.PI) angleDiff = 2 * Math.PI - angleDiff
   655→      
   656→      if (angleDiff < minAngleDiff) {
   657→        minAngleDiff = angleDiff
   658→        assignedSector = s
   659→      }
   660→    }
   661→    
   662→    results.set(env.environment, {
   663→      winner: sectors[assignedSector]?.vertex || genotypeCoords[0].genotype,
   664→      sector: assignedSector + 1,
   665→      vertices: vertexGenotypes
   666→    })
   667→  }
   668→  
   669→  // Group by winning genotype
   670→  const winnerMap = new Map<string, string[]>()
   671→  for (const [, value] of results) {
   672→    if (!winnerMap.has(value.winner)) {
   673→      winnerMap.set(value.winner, [])
   674→    }
   675→  }
   676→  
   677→  for (const [envName, value] of results) {
   678→    winnerMap.get(value.winner)?.push(envName)
   679→  }
   680→  
   681→  // Build final result
   682→  const whichWonWhere: WhichWonWhereEntry[] = []
   683→  let sectorNum = 1
   684→  
   685→  for (const [winner, envs] of winnerMap) {
   686→    whichWonWhere.push({
   687→      sector: sectorNum++,
   688→      winningGenotype: winner,
   689→      environments: envs.sort(),
   690→      vertexGenotypes: vertexGenotypes
   691→    })
   692→  }
   693→  
   694→  // Sort by sector number
   695→  whichWonWhere.sort((a, b) => a.sector - b.sector)
   696→  
   697→  // If no sectors determined, put everything in one
   698→  if (whichWonWhere.length === 0) {
   699→    whichWonWhere.push({
   700→      sector: 1,
   701→      winningGenotype: genotypeCoords.sort((a, b) => b.meanYield - a.meanYield)[0]?.genotype || '',
   702→      environments: environmentCoords.map(e => e.environment),
   703→      vertexGenotypes: vertexGenotypes
   704→    })
   705→  }
   706→  
   707→  return whichWonWhere
   708→}

   710→/**
   711→ * Compute convex hull using Graham scan algorithm
   712→ * Returns indices of points forming the convex hull
   713→ */
   714→function convexHull(points: { x: number; y: number; name: string }[]): number[] {
   715→  if (points.length <= 2) return points.map((_, i) => i)
   716→  
   717→  // Find lowest point (by y, then by x)
   718→  let lowest = 0
   719→  for (let i = 1; i < points.length; i++) {
   720→    if (points[i].y < points[lowest].y || 
   721→        (points[i].y === points[lowest].y && points[i].x < points[lowest].x)) {
   722→      lowest = i
   723→    }
   724→  }
   725→  
   726→  // Sort by polar angle from lowest point
   727→  const start = points[lowest]
   728→  const sorted = points
   729→    .map((p, i) => ({ ...p, originalIndex: i }))
   730→    .filter((_, i) => i !== lowest)
   731→    .sort((a, b) => {
   732→      const angleA = Math.atan2(a.y - start.y, a.x - start.x)
   733→      const angleB = Math.atan2(b.y - start.y, b.x - start.x)
   734→      return angleA - angleB
   735→    })
   736→  
   737→  // Build hull
   738→  const hull: number[] = [lowest]
   739→  
   740→  for (const p of sorted) {
   741→    while (hull.length > 1) {
   742→      const top = points[hull[hull.length - 1]]
   743→      const nextToTop = points[hull[hull.length - 2]]
   744→      
   745→      // Cross product to check orientation
   746→      const cross = (top.x - nextToTop.x) * (p.y - top.y) - (top.y - nextToTop.y) * (p.x - top.x)
   747→      
   748→      if (cross <= 0) {
   749→        hull.pop()
   750→      } else {
   751→        break
   752→      }
   753→    }
   754→    hull.push(p.originalIndex)
   755→  }
   756→  
   757→  return hull
   758→}
