   236→function StatusIcon({ status }: { status: ToolStatus }) {
   237→  switch (status) {
   238→    case "available":
   239→      return (
   240→        <Tooltip>
   241→          <TooltipTrigger asChild>
   242→            <CheckCircle2 className="size-4 text-emerald-500" />
   243→          </TooltipTrigger>
   244→          <TooltipContent>Available</TooltipContent>
   245→        </Tooltip>
   246→      )
   247→    case "coming-soon":
   248→      return (
   249→        <Tooltip>
   250→          <TooltipTrigger asChild>
   251→            <Clock className="size-4 text-amber-500" />
   252→          </TooltipTrigger>
   253→          <TooltipContent>Coming Soon</TooltipContent>
   254→        </Tooltip>
   255→      )

   257→      return (
   258→        <Tooltip>

   330→// ==================== CATEGORY CARD COMPONENT ====================
   331→function CategoryCard({
   332→  category,
   333→  isExpanded,
   334→  onToggle,
   335→  onClick,
   336→}: {
   337→  category: ToolCategory
   338→  isExpanded: boolean
   339→  onToggle: () => void
   340→  onClick: () => void
   341→}) {
   342→  const availableTools = category.tools?.filter(
   343→    (t) => t.status === "available"
   344→  ).length || 0
   345→
   346→  return (
   347→    <motion.div
   348→      layout
   349→      initial={{ opacity: 0, y: 20 }}
   350→      animate={{ opacity: 1, y: 0 }}
   351→      exit={{ opacity: 0, y: -20 }}
   352→      transition={{ duration: 0.3 }}
   353→    >
   354→      <Card
   355→        className={cn(
   356→          "group glass-card cursor-pointer transition-all duration-300 hover:shadow-lg hover:border-primary/30 hover:-translate-y-1",
   357→          isExpanded && "ring-2 ring-primary/20 border-primary/30"
   358→        )}
   359→        onClick={() => {
   360→          onToggle()
   361→          if (!isExpanded) onClick()
   362→        }}
   363→      >
   364→        <CardContent className="p-6">
   365→          <div className="flex items-start gap-4">
   366→            {/* Icon */}
   367→            <motion.div
   368→              whileHover={{ scale: 1.05, rotate: 5 }}
   369→              className={cn(
   370→                "size-12 rounded-xl flex items-center justify-center shrink-0",
   371→                "bg-gradient-to-br shadow-lg"
   372→              )}
   373→              style={{
   374→                background: `linear-gradient(135deg, ${category.color}20, ${category.color}40)`,
   375→                color: category.color,
   376→              }}
   377→            >
   378→              <DynamicIcon name={category.icon} className="size-6" />
   379→            </motion.div>
   380→
   381→            {/* Content */}
   382→            <div className="flex-1 min-w-0">
   383→              <div className="flex items-center gap-2 mb-1">
   384→                <CardTitle className="text-lg font-semibold group-hover:text-primary transition-colors">
   385→                  {category.name}
   386→                </CardTitle>
   387→                <motion.div
   388→                  animate={{ rotate: isExpanded ? 180 : 0 }}
   389→                  transition={{ duration: 0.2 }}
   390→                >
   391→                  <ChevronDown className="size-4 text-muted-foreground" />
   392→                </motion.div>
   393→              </div>
   394→              <CardDescription className="text-sm line-clamp-2 mb-3">
   395→                {category.description}
   396→              </CardDescription>
   397→              <div className="flex items-center gap-2 flex-wrap">
   398→                <Badge
   399→                  variant="secondary"
   400→                  className="text-xs font-medium"
   401→                  style={{
   402→                    backgroundColor: `${category.color}15`,
   403→                    color: category.color,
   404→                  borderColor: `${category.color}30`,
   405→                  }}
   406→                >
   407→                  {category.tools?.length || 0} tools
   408→                </Badge>
   409→                <Badge
   410→                  variant="outline"
   411→                  className="text-xs text-emerald-600 border-emerald-300 dark:border-emerald-700"
   412→                >
   413→                  {availableTools} available
   414→                </Badge>
   415→              </div>
   416→            </div>
   417→
   418→            {/* Action Button */}
   419→            <Button
   420→              variant="ghost"
   421→              size="icon"
   422→              className="shrink-0 opacity-0 group-hover:opacity-100 transition-opacity"
   423→              onClick={(e) => {
   424→                e.stopPropagation()
   425→                onToggle()
   426→              }}
   427→            >
   428→              <ExternalLink className="size-4" />
   429→            </Button>
   430→          </div>
   431→        </CardContent>
   432→      </Card>
   433→    </motion.div>
   434→  )
   435→}

   437→// ==================== TOOL CARD COMPONENT ====================

   439→  tool,
   440→  viewMode,
   441→  isFavorite,
   442→  onToggleFavorite,
   443→  onLaunch,
