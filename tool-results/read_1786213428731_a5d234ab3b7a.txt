     1→'use client'
     2→
     3→import React, { useState, useCallback } from 'react'
     4→import { motion } from 'framer-motion'
     5→import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card'
     6→import { Button } from '@/components/ui/button'
     7→import { Input } from '@/components/ui/input'
     8→import { Label } from '@/components/ui/label'
     9→import { Textarea } from '@/components/ui/textarea'
    10→import { Badge } from '@/components/ui/badge'
    11→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    12→import { 
    13→  Microscope, 
    14→  Upload, 
    15→  FileText, 
    16→  Download,
    17→  Info,
    18→  Dna,
    19→  AlertCircle,
    20→  CheckCircle2,
    21→  XCircle
    22→} from 'lucide-react'
    23→import { analyzePopulationGenetics, PopulationGeneticsResult } from '@/lib/statistics-engine'
    24→
    25→interface PopulationGeneticsProps {
    26→  className?: string
    27→}
    28→
    29→// Sample genotype data (diploid genotypes)
    30→const SAMPLE_DATA = `AA,Aa,aa,AA,Aa,Aa,aa,aa,AA,Aa,AA,aa,Aa,AA,Aa,aa,aa,AA,Aa,Aa,aa,AA,Aa,aa`
    31→
    32→export default function PopulationGeneticsAnalyzer({ className }: PopulationGeneticsProps) {
    33→  const [rawData, setRawData] = useState('')
    34→  const [parsedGenotypes, setParsedGenotypes] = useState<string[]>([])
    35→  const [result, setResult] = useState<PopulationGeneticsResult | null>(null)
    36→  const [error, setError] = useState<string | null>(null)
    37→  const [isLoading, setIsLoading] = useState(false)
    38→
    39→  // Parse genotype data
    40→  const parseData = useCallback((text: string): boolean => {
    41→    try {
    42→      // Support multiple formats: comma-separated, space-separated, or newline-separated
    43→      let genotypes: string[] = []
    44→      
    45→      if (text.includes('\n')) {
    46→        // Newline separated - each line is a genotype or multiple
    47→        text.trim().split('\n').forEach(line => {
    48→          const items = line.split(/[,\s]+/).filter(g => g.trim())
    49→          genotypes.push(...items.map(g => g.trim().toUpperCase()))
    50→        })
    51→      } else if (text.includes(',')) {
    52→        genotypes = text.split(',').map(g => g.trim().toUpperCase()).filter(g => g)
    53→      } else {
    54→        genotypes = text.split(/\s+/).map(g => g.trim().toUpperCase()).filter(g => g)
    55→      }
    56→      
    57→      if (genotypes.length < 4) throw new Error('Need at least 4 genotypes for meaningful analysis')
    58→      
    59→      // Validate format - should be like AA, Aa, aa, AB, etc.
    60→      for (const gt of genotypes) {
    61→        const clean = gt.replace(/[^A-Za-z]/g, '')
    62→        if (clean.length < 2 || clean.length > 2) {
    63→          console.warn(`Unusual genotype format: ${gt}, attempting to continue`)
    64→        }
    65→      }
    66→      
    67→      setParsedGenotypes(genotypes)
    68→      setError(null)
    69→      return true
    70→    } catch (err) {
    71→      setError(err instanceof Error ? err.message : 'Failed to parse data')
    72→      return false
    73→    }
    74→  }, [])
    75→
    76→  // Run analysis
    77→  const handleAnalyze = useCallback(() => {
    78→    if (parsedGenotypes.length === 0) {
    79→      if (!parseData(rawData)) return
    80→    }
    81→    
    82→    runAnalysis(parsedGenotypes)
    83→  }, [parsedGenotypes, rawData, parseData])
    84→
    85→  const runAnalysis = (genotypes: string[]) => {
    86→    setIsLoading(true)
    87→    setError(null)
    88→    
    89→    setTimeout(() => {
    90→      try {
    91→        const popResult: PopulationGeneticsResult = analyzePopulationGenetics(genotypes)
    92→        setResult(popResult)
    93→      } catch (err) {
    94→        setError(err instanceof Error ? err.message : 'Population genetics analysis failed')
    95→      }
    96→      setIsLoading(false)
    97→    }, 300)
    98→  }
    99→
   100→  // Load sample data
   101→  const loadSampleData = () => {
   102→    setRawData(SAMPLE_DATA)
   103→    parseData(SAMPLE_DATA)
   104→  }
   105→
   106→  // Handle file upload
   107→  const handleFileUpload = (event: React.ChangeEvent<HTMLInputElement>) => {
   108→    const file = event.target.files?.[0]
   109→    if (!file) return
   110→
   111→    const reader = new FileReader()
   112→    reader.onload = (e) => {
   113→      const text = e.target?.result as string
   114→      setRawData(text)
   115→      parseData(text)
   116→    }
   117→    reader.readAsText(file)
   118→  }
   119→
   120→  // Export results
   121→  const exportResults = () => {
   122→    if (!result) return
   123→    
   124→    const rows = [
   125→      ['Population Genetics Analysis Results'],
   126→      [''],
   127→      ['Allele Frequencies'],
   128→      ...Object.entries(result.alleleFrequencies).map(([allele, freq]) => [allele, freq.toFixed(4)]),
   129→      [''],
   130→      ['Genotype Frequencies'],
   131→      ...Object.entries(result.genotypeFrequencies).map(([gt, freq]) => [gt, freq.toFixed(4)]),
   132→      [''],
   133→      ['Diversity Measures', 'Value', 'Interpretation'],
   134→      ['Expected Heterozygosity (He)', result.expectedHeterozygosity.toFixed(4), 'Gene diversity'],
   135→      ['Observed Heterozygosity (Ho)', result.observedHeterozygosity.toFixed(4), 'Actual heterozygotes proportion'],
   136→      ['Fixation Index (F)', result.fixationIndex.toFixed(4), result.fixationIndex > 0 ? 'Deficit' : 'Excess'],
   137→      ['Genetic Diversity', result.geneticDiversity.toFixed(4), '-'],
   138→      [''],
   139→      ['Hardy-Weinberg Equilibrium Test'],
   140→      ['Chi-square statistic', result.hardyWeinberg.chiSquare.toFixed(4)],
   141→      ['Degrees of freedom', result.hardyWeinberg.df.toString()],
   142→      ['In HWE?', result.hardyWeinberg.equilibrium ? 'Yes' : 'No']
   143→    ]
   144→    
   145→    const csv = rows.map(r => r.join(',')).join('\n')
   146→    const blob = new Blob([csv], { type: 'text/csv' })
   147→    const url = URL.createObjectURL(blob)
   148→    const a = document.createElement('a')
   149→    a.href = url
   150→    a.download = 'population_genetics.csv'
   151→    a.click()
   152→    URL.revokeObjectURL(url)
   153→  }
   154→
   155→  // Render pie chart for allele frequencies
   156→  const renderAllelePieChart = (): React.ReactNode => {
   157→    if (!result) return null
   158→    
   159→    const alleles = Object.entries(result.alleleFrequencies)
   160→    const total = alleles.reduce((sum, [, f]) => sum + f, 0)
   161→    
   162→    let currentAngle = -90 // Start from top
   163→    const radius = 80
   164→    const centerX = 100
   165→    const centerY = 100
   166→    
   167→    const colors = ['#22c55e', '#3b82f6', '#f59e0b', '#ef4444', '#8b5cf6', '#ec4899']
   168→    
   169→    const paths = alleles.map(([allele, freq], i) => {
   170→      const angleSpan = (freq / total) * 360
   171→      const startAngle = currentAngle
   172→      const endAngle = currentAngle + angleSpan
   173→      
   174→      const x1 = centerX + radius * Math.cos((startAngle * Math.PI) / 180)
   175→      const y1 = centerY + radius * Math.sin((startAngle * Math.PI) / 180)
   176→      const x2 = centerX + radius * Math.cos((endAngle * Math.PI) / 180)
   177→      const y2 = centerY + radius * Math.sin((endAngle * Math.PI) / 180)
   178→      
   179→      const largeArc = angleSpan > 180 ? 1 : 0
   180→      
   181→      const pathD = [
   182→        `M ${centerX} ${centerY}`,
   183→        `L ${x1} ${y1}`,
   184→        `A ${radius} ${radius} 0 ${largeArc} 1 ${x2} ${y2}`,
   185→        'Z'
   186→      ].join(' ')
   187→      
   188→      // Label position
   189→      const midAngle = startAngle + angleSpan / 2
   190→      const labelRadius = radius * 0.7
   191→      const labelX = centerX + labelRadius * Math.cos((midAngle * Math.PI) / 180)
   192→      const labelY = centerY + labelRadius * Math.sin((midAngle * Math.PI) / 180)
   193→      
   194→      currentAngle = endAngle
   195→      
   196→      return (
   197→        <g key={allele}>
   198→          <path d={pathD} fill={colors[i % colors.length]} stroke="white" strokeWidth="2" />
   199→          {(freq / total) > 0.05 && (
   200→            <>
   201→              <text x={labelX} y={labelY} textAnchor="middle" dominantBaseline="middle" fontSize="12" fontWeight="bold" fill="white">
   202→                {allele}
   203→              </text>
   204→              <text x={labelX} y={labelY + 14} textAnchor="middle" dominantBaseline="middle" fontSize="10" fill="white">
   205→                {(freq * 100).toFixed(1)}%
   206→              </text>
   207→            </>
   208→          )}
   209→        </g>
   210→      )
   211→    })
   212→    
   213→    return (
   214→      <svg viewBox="0 0 200 200" className="w-48 h-48 mx-auto">
   215→        {paths}
   216→      </svg>
   217→    )
   218→  }
   219→
   220→  // Render heterozygosity bar chart
   221→  const renderHeterozygosityChart = (): React.ReactNode => {
   222→    if (!result) return null
   223→    
   224→    const { expectedHeterozygosity, observedHeterozygosity } = result
   225→    const maxVal = Math.max(expectedHeterozygosity, observedHeterozygosity, 0.5)
   226→    const width = 250
   227→    const height = 120
   228→    const barHeight = 30
   229→    const padding = 20
   230→    
   231→    return (
   232→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full max-w-sm mx-auto">
   233→        {/* Expected Heterozygosity */}
   234→        <text x={padding} y={25} fontSize="11" fill="#374151" fontWeight="medium">Expected (He)</text>
   235→        <rect 
   236→          x={padding + 90} 
   237→          y={10} 
   238→          width={(expectedHeterozygosity / maxVal) * (width - padding - 100)} 
   239→          height={barHeight - 5} 
   240→          rx="4"
   241→          fill="#22c55e"
   242→        />
   243→        <text x={padding + 95 + (expectedHeterozygosity / maxVal) * (width - padding - 100)} y={30} fontSize="11" fill="#22c55e" fontWeight="bold">
   244→          {expectedHeterozygosity?.toFixed(3)}
   245→        </text>
   246→        
   247→        {/* Observed Heterozygosity */}
   248→        <text x={padding} y={70} fontSize="11" fill="#374151" fontWeight="medium">Observed (Ho)</text>
   249→        <rect 
   250→          x={padding + 90} 
   251→          y={55} 
   252→          width={(observedHeterozygosity / maxVal) * (width - padding - 100)} 
   253→          height={barHeight - 5} 
   254→          rx="4"
   255→          fill="#3b82f6"
   256→        />
   257→        <text x={padding + 95 + (observedHeterozygosity / maxVal) * (width - padding - 100)} y={75} fontSize="11" fill="#3b82f6" fontWeight="bold">
   258→          {observedHeterozygosity.toFixed(3)}
   259→        </text>
   260→        
   261→        {/* Fixation Index */}
   262→        <text x={padding} y={110} fontSize="11" fill="#374151" fontWeight="medium">Fixation (F)</text>
   263→        <rect 
   264→          x={padding + 90} 
   265→          y={95} 
   266→          width={(Math.abs(result.fixationIndex) / 1) * (width - padding - 100)} 
   267→          height={barHeight - 5} 
   268→          rx="4"
   269→          fill={result.fixationIndex > 0 ? '#ef4444' : '#f59e0b'}
   270→        />
   271→        <text x={padding + 95 + (Math.abs(result.fixationIndex) / 1) * (width - padding - 100)} y={110} fontSize="11" fill={result.fixationIndex > 0 ? '#ef4444' : '#f59e0b'} fontWeight="bold">
   272→          {result.fixationIndex.toFixed(3)}
   273→        </text>
   274→      </svg>
   275→    )
   276→  }
   277→
   278→  return (
   279→    <div className={`space-y-6 ${className || ''}`}>
   280→      {/* Input Section */}
   281→      <Card>
   282→        <CardHeader>
   283→          <CardTitle className="flex items-center gap-2">
   284→            <Microscope className="h-5 w-5 text-primary" />
   285→            Population Genetics Analyzer
   286→          </CardTitle>
   287→          <CardDescription>
   288→            Analyze allele frequencies, test Hardy-Weinberg equilibrium, and compute diversity indices
   289→          </CardDescription>
   290→        </CardHeader>
   291→        <CardContent>
   292→          <Tabs defaultValue="paste" className="w-full">
   293→            <TabsList className="grid w-full grid-cols-2">
   294→              <TabsTrigger value="paste" className="gap-2">
   295→                <FileText className="h-4 w-4" />
   296→                Paste Data
   297→              </TabsTrigger>
   298→              <TabsTrigger value="upload" className="gap-2">
   299→                <Upload className="h-4 w-4" />
   300→                Upload File
   301→              </TabsTrigger>
   302→            </TabsList>
   303→
   304→            <TabsContent value="paste" className="space-y-4 mt-4">
   305→              <div>
   306→                <Label htmlFor="data-input">Enter diploid genotypes</Label>
   307→                <Textarea
   308→                  id="data-input"
   309→                  placeholder={`Enter genotypes in any of these formats:
   310→• Comma-separated: AA, Aa, aa, AA, Aa...
   311→• Space-separated: AA Aa aa AA Aa...
   312→• One per line:
   313→  AA
   314→  Aa
   315→  aa
   316→  
   317→Supported formats: AA, Aa, aa, AB, etc.`}
   318→                  value={rawData}
   319→                  onChange={(e) => setRawData(e.target.value)}
   320→                  className="mt-2 font-mono text-sm min-h-[140px]"
   321→                />
   322→              </div>
   323→
   324→              <Button onClick={loadSampleData} variant="outline" size="sm">
   325→                Load Sample Data (n=24)
   326→              </Button>
   327→
   328→              {parsedGenotypes.length > 0 && !result && (
   329→                <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }}
   330→                  className="mt-4 p-3 bg-muted/30 rounded-lg">
   331→                  <p className="text-sm font-medium mb-2">Parsed Genotypes ({parsedGenotypes.length}):</p>
   332→                  <div className="flex flex-wrap gap-1 max-h-20 overflow-y-auto">
   333→                    {parsedGenotypes.slice(0, 30).map((gt, i) => (
   334→                      <Badge key={i} variant="secondary" className="font-mono text-xs">{gt}</Badge>
   335→                    ))}
   336→                    {parsedGenotypes.length > 30 && (
   337→                      <Badge variant="outline" className="text-xs">+{parsedGenotypes.length - 30} more</Badge>
   338→                    )}
   339→                  </div>
   340→                </motion.div>
   341→              )}
   342→            </TabsContent>
   343→
   344→            <TabsContent value="upload" className="space-y-4 mt-4">
   345→              <div className="border-2 border-dashed border-muted-foreground/25 rounded-lg p-6 text-center">
   346→                <Upload className="mx-auto h-12 w-12 text-muted-foreground mb-4" />
   347→                <p className="text-sm text-muted-foreground mb-4">
   348→                  Upload a file with genotype data (one per line or comma/space separated)
   349→                </p>
   350→                <Input type="file" accept=".csv,.txt,.gen" onChange={handleFileUpload} className="max-w-xs mx-auto" />
   351→              </div>
   352→            </TabsContent>
   353→          </Tabs>
   354→
   355→          <div className="mt-6 flex justify-end">
   356→            <Button onClick={handleAnalyze} disabled={!rawData.trim()} className="gap-2">
   357→              <Dna className="h-4 w-4" />
   358→              Analyze Population
   359→            </Button>
   360→          </div>
   361→
   362→          {error && (
   363→            <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }}
   364→              className="mt-4 p-3 bg-destructive/10 border border-destructive/20 rounded-lg flex items-start gap-2">
   365→              <AlertCircle className="h-4 w-4 text-destructive mt-0.5 shrink-0" />
   366→              <span className="text-sm text-destructive">{error}</span>
   367→            </motion.div>
   368→          )}
   369→        </CardContent>
   370→      </Card>
   371→
   372→      {/* Results Section */}
   373→      {result && (
   374→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="space-y-6">
   375→          {/* Summary Cards */}
   376→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
   377→            <Card className="bg-gradient-to-br from-green-50 to-green-100 dark:from-green-950 dark:to-green-900">
   378→              <CardContent className="pt-4">
   379→                <p className="text-xs text-green-600 dark:text-green-400">Expected Het. (He)</p>
   380→                <p className="text-2xl font-bold text-green-900 dark:text-green-100">{result.expectedHeterozygosity.toFixed(4)}</p>
   381→                <p className="text-xs text-muted-foreground mt-1">Gene diversity</p>
   382→              </CardContent>
   383→            </Card>
   384→            
   385→            <Card className="bg-gradient-to-br from-blue-50 to-blue-100 dark:from-blue-950 dark:to-blue-900">
   386→              <CardContent className="pt-4">
   387→                <p className="text-xs text-blue-600 dark:text-blue-400">Observed Het. (Ho)</p>
   388→                <p className="text-2xl font-bold text-blue-900 dark:text-blue-100">{result.observedHeterozygosity.toFixed(4)}</p>
   389→                <p className="text-xs text-muted-foreground mt-1">Actual heterozygotes</p>
   390→              </CardContent>
   391→            </Card>
   392→            
   393→            <Card className="bg-gradient-to-br from-purple-50 to-purple-100 dark:from-purple-950 dark:to-purple-900">
   394→              <CardContent className="pt-4">
   395→                <p className="text-xs text-purple-600 dark:text-purple-400">Fixation Index (F)</p>
   396→                <p className="text-2xl font-bold text-purple-900 dark:text-purple-100">{result.fixationIndex.toFixed(4)}</p>
   397→                <Badge variant={Math.abs(result.fixationIndex) < 0.05 ? "default" : "secondary"} className="mt-1">
   398→                  {Math.abs(result.fixationIndex) < 0.05 ? 'Near equilibrium' : 'Deviation'}
   399→                </Badge>
   400→              </CardContent>
   401→            </Card>
   402→            
   403→            <Card className={`bg-gradient-to-br ${
   404→              result.hardyWeinberg.equilibrium 
   405→                ? 'from-emerald-50 to-emerald-100 dark:from-emerald-950 dark:to-emerald-900'
   406→                : 'from-red-50 to-red-100 dark:from-red-950 dark:to-red-900'
   407→            }`}>
   408→              <CardContent className="pt-4">
   409→                <p className={`text-xs ${
   410→                  result.hardyWeinberg.equilibrium 
   411→                    ? 'text-emerald-600 dark:text-emerald-400'
   412→                    : 'text-red-600 dark:text-red-400'
   413→                }`}>
   414→                  Hardy-Weinberg
   415→                </p>
   416→                <div className="flex items-center gap-2">
   417→                  <p className="text-xl font-bold">
   418→                    {result.hardyWeinberg.equilibrium ? 'In Equilibrium' : 'Deviates'}
   419→                  </p>
   420→                  {result.hardyWeinberg.equilibrium 
   421→                    ? <CheckCircle2 className="h-5 w-5 text-emerald-500" />
   422→                    : <XCircle className="h-5 w-5 text-red-500" />
   423→                  }
   424→                </div>
   425→              </CardContent>
   426→            </Card>
   427→          </div>
   428→
   429→          <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   430→            {/* Allele Frequencies */}
   431→            <Card>
   432→              <CardHeader>
   433→                <CardTitle className="text-lg">Allele Frequencies</CardTitle>
   434→                <CardDescription>Frequency of each allele in the population</CardDescription>
   435→              </CardHeader>
   436→              <CardContent>
   437→                <div className="flex flex-col md:flex-row items-center gap-6">
   438→                  {renderAllelePieChart()}
   439→                  
   440→                  <div className="flex-1 space-y-2">
   441→                    {Object.entries(result.alleleFrequencies).sort((a, b) => b[1] - a[1]).map(([allele, freq], i) => (
   442→                      <div key={allele} className="flex items-center gap-3">
   443→                        <span className="font-mono font-bold w-8">{allele}</span>
   444→                        <div className="flex-1 h-6 bg-gray-100 dark:bg-gray-800 rounded overflow-hidden">
   445→                          <div 
   446→                            className="h-full bg-gradient-to-r from-primary to-primary/60 rounded transition-all"
   447→                            style={{ width: `${freq * 100}%` }}
   448→                          ></div>
   449→                        </div>
   450→                        <span className="font-mono text-sm w-14 text-right">{(freq * 100).toFixed(2)}%</span>
   451→                      </div>
   452→                    ))}
   453→                  </div>
   454→                </div>
   455→              </CardContent>
   456→            </Card>
   457→
   458→            {/* Genotype Frequencies */}
   459→            <Card>
   460→              <CardHeader>
   461→                <CardTitle className="text-lg">Genotype Frequencies</CardTitle>
   462→                <CardDescription>Observed frequency of each genotype</CardDescription>
   463→              </CardHeader>
   464→              <CardContent>
   465→                <div className="space-y-3">
   466→                  {Object.entries(result.genotypeFrequencies)
   467→                    .sort((a, b) => b[1] - a[1])
   468→                    .map(([gt, freq]) => {
   469→                      const count = Math.round(freq * parsedGenotypes.length)
   470→                      return (
   471→                        <div key={gt} className="flex items-center gap-3">
   472→                          <Badge variant="outline" className="font-mono w-12 justify-center">{gt}</Badge>
   473→                          <div className="flex-1 h-7 bg-gray-100 dark:bg-gray-800 rounded overflow-hidden">
   474→                            <div 
   475→                              className={`h-full rounded transition-all ${
   476→                                gt === gt[0] + gt[0].toLowerCase() || gt === gt.toUpperCase() 
   477→                                  ? 'bg-blue-500' 
   478→                                  : gt[0] !== gt[1]
   479→                                    ? 'bg-green-500'
   480→                                    : 'bg-orange-500'
   481→                              }`}
   482→                              style={{ width: `${freq * 100}%` }}
   483→                            ></div>
   484→                          </div>
   485→                          <div className="text-right">
   486→                            <span className="font-mono text-sm">{(freq * 100).toFixed(1)}%</span>
   487→                            <span className="text-xs text-muted-foreground ml-1">(n={count})</span>
   488→                          </div>
   489→                        </div>
   490→                      )
   491→                    })}
   492→                  
   493→                  <div className="mt-4 pt-3 border-t text-xs text-muted-foreground space-y-1">
   494→                    <div className="flex items-center gap-2">
   495→                      <span className="w-3 h-3 rounded bg-blue-500 inline-block"></span>
   496→                      Homozygous dominant
   497→                    </div>
   498→                    <div className="flex items-center gap-2">
   499→                      <span className="w-3 h-3 rounded bg-green-500 inline-block"></span>
   500→                      Heterozygous
