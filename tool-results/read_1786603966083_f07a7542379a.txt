   128→  const [isScrolled, setIsScrolled] = React.useState(false);
   129→  const [isMobileOpen, setIsMobileOpen] = React.useState(false);
   130→  const [activeSection, setActiveSection] = React.useState("");
   131→  const [mounted, setMounted] = React.useState(false);
   132→  const [showAuthModal, setShowAuthModal] = React.useState(false);
   133→  const [authMode, setAuthMode] = React.useState<"signin" | "signup">("signin");
   134→  
   135→  // Search state
   136→  const [showSearch, setShowSearch] = React.useState(false);
   137→  const [searchQuery, setSearchQuery] = React.useState("");
   138→  const [searchResults, setSearchResults] = React.useState<typeof databaseOptions>([]);
   139→  
   140→  // Form state
   141→  const [authEmail, setAuthEmail] = React.useState("");
   142→  const [authPassword, setAuthPassword] = React.useState("");
   143→  const [authName, setAuthName] = React.useState("");
   144→  const [isLoading, setIsLoading] = React.useState(false);
   145→  const [authError, setAuthError] = React.useState("");
   146→  
   147→  // Contact modal
   148→  const [showContactModal, setShowContactModal] = React.useState(false);
   149→  
   150→  const { theme, setTheme } = useTheme();
   151→  const { data: session, status } = useSession();
   152→
   153→  // Handle mount state for theme
   154→  React.useEffect(() => {
   155→    setMounted(true);
   156→  }, []);
   157→

   160→    const handleScroll = () => {
   161→      setIsScrolled(window.scrollY > 20);

   164→      const sections = navLinks.map((link) => link.href.substring(1));
   165→      for (const section of sections.reverse()) {
   166→        const element = document.getElementById(section);
   167→        if (element) {
   168→          const rect = element.getBoundingClientRect();
   169→          if (rect.top <= 150) {
   170→            setActiveSection(section);
   171→            break;
   172→          }
   173→        }
   174→      }
   175→    };
   176→
   177→    window.addEventListener("scroll", handleScroll, { passive: true });
   178→    return () => window.removeEventListener("scroll", handleScroll);
   179→  }, []);
   180→
   181→  // Toggle theme
   182→  const toggleTheme = () => {
   183→    setTheme(theme === "dark" ? "light" : "dark");
   184→  };

   187→  const handleNavClick = (href: string, action?: string) => {
   188→    setIsMobileOpen(false);
   189→    
   190→    if (action && onNavigate) {
   191→      onNavigate(action);
   192→      window.scrollTo({ top: 0 });
   193→      return;
   194→    }
   195→    
   196→    // For landing page sections that exist
   197→    const elementId = href.substring(1);
   198→    const element = document.getElementById(elementId);
   199→    if (element) {
   200→      element.scrollIntoView({ behavior: "smooth" });
   201→    } else if (onNavigate) {
   202→      // Navigate to view if section doesn't exist on current page
   203→      onNavigate(elementId);
   204→      window.scrollTo({ top: 0 });
   205→    }
   206→  };
   207→
   208→  // Handle Get Started button click
   209→  const handleGetStarted = () => {
   210→    if (onNavigate) {
   211→      onNavigate('dashboard');
   212→      window.scrollTo({ top: 0 });
   213→    }
   214→  };

   216→  // Handle Sign In button click
   217→  const handleSignIn = () => {
   218→    setShowAuthModal(true);
   219→    setAuthMode('signin');
   220→    setAuthError("");
   221→  };
   222→
   223→  // Handle Sign Up button click  
   224→  const handleSignUp = () => {
   225→    setShowAuthModal(true);
   226→    setAuthMode('signup');
   227→    setAuthError("");
   228→  };
   229→
   230→  // Handle sign out with proper cleanup
   231→  const handleSignOut = async () => {
   232→    try {
   233→      setIsLoading(true)
   234→      // Sign out from NextAuth
   235→      await signOut({ redirect: false })
   236→      // Force page refresh to clear any cached state
   237→      window.location.href = '/'
   238→    } catch (error) {
   239→      console.error('Sign out error:', error)
   240→      // Fallback: force redirect anyway
   241→      window.location.href = '/'
   242→    }
   243→  };
   244→
   245→  // Reset form when modal opens
   246→  React.useEffect(() => {
   247→    if (showAuthModal) {
   248→      setAuthEmail("");
   249→      setAuthPassword("");
   250→      setAuthName("");
   251→      setAuthError("");

   254→

   257→    e.preventDefault();
   258→    setAuthError("");
   259→    setIsLoading(true);
   260→
   261→    try {
   262→      if (authMode === 'signup') {
   263→        // Registration
   264→        const response = await fetch('/api/auth/register', {
   265→          method: 'POST',
   266→          headers: { 'Content-Type': 'application/json' },
   267→          body: JSON.stringify({
   268→            name: authName,
   269→            email: authEmail,
   270→            password: authPassword,
   271→          }),
   272→        });
   273→
   274→        const data = await response.json();
   275→
   276→        if (!response.ok) {
   277→          throw new Error(data.error || 'Registration failed');
   278→        }
   279→
   280→        // After successful registration, sign in
   281→        const result = await signIn('credentials', {
   282→          email: authEmail,
   283→          password: authPassword,
   284→          redirect: false,
   285→        });
   286→
   287→        if (result?.error) {
   288→          throw new Error(result.error);
   289→        }

   293→          email: authEmail,
   294→          password: authPassword,
   295→          redirect: false,
   296→        });
   297→
   298→        if (result?.error) {
   299→          throw new Error('Invalid email or password');
   300→        }
   301→      }
   302→
   303→      // Success - close modal and navigate to dashboard
   304→      setShowAuthModal(false);
   305→      if (onNavigate) {
   306→        onNavigate('dashboard');
   307→        window.scrollTo({ top: 0 });
   308→      }
   309→    } catch (error: any) {
   310→      setAuthError(error.message || 'An error occurred');

   312→      setIsLoading(false);
   313→    }
   314→  };
