   127→export default function SequenceAnalysisTool() {
   128→  // State
   129→  const [activeTool, setActiveTool] = useState<ToolId>('pairwise');
   130→  const [sequence1, setSequence1] = useState('');
   131→  const [sequence2, setSequence2] = useState('');
   132→  const [isAnalyzing, setIsAnalyzing] = useState(false);
   133→  const [progress, setProgress] = useState(0);
   134→  const [result, setResult] = useState<AnalysisResult | null>(null);
   135→  const [showAdvanced, setShowAdvanced] = useState(false);
   136→  const [copied, setCopied] = useState(false);
   137→  
   138→  // Tool-specific options state
   139→  const [alignmentOptions, setAlignmentOptions] = useState({
   140→    algorithm: 'global' as 'global' | 'local',
   141→    matchScore: 1,
   142→    mismatchPenalty: -1,
   143→    gapPenalty: -1,
   144→    matrix: 'BLOSUM62',
   145→  });
   146→
   147→  const [blastOptions, setBlastOptions] = useState({
   148→    program: 'blastn',
   149→    database: 'nr',
   150→    eValue: '0.001',
   151→    maxTargets: 10,
   152→    wordSize: 11,
   153→    lowComplexityFilter: true,
   154→  });
   155→
   156→  const [orfOptions, setOrfOptions] = useState({
   157→    geneticCode: 1,
   158→    minLength: 30,
   159→    showAllFrames: true,
   160→    startCodons: ['ATG'],
   161→  });
   162→
   163→  const [translationOptions, setTranslationOptions] = useState({
   164→    frame: 1,
   165→    geneticCode: 1,
   166→  });
   167→
   168→  const [motifPattern, setMotifPattern] = useState('');
   169→  const [patternSearchOptions, setPatternSearchOptions] = useState({
   170→    caseSensitive: false,
   171→    allowMismatch: false,
   172→    maxMismatches: 0,
   173→  });
   174→
   175→  // Get sequence stats for display
   176→  const seq1Stats = useMemo(() => 
   177→    sequence1 ? getSequenceStats(sequence1) : null, [sequence1]
   178→  );
   179→
   180→  const seq2Stats = useMemo(() => 
   181→    sequence2 ? getSequenceStats(sequence2) : null, [sequence2]
   182→  );
   183→
   184→  // Load example sequences
   185→  const loadExampleDNA = useCallback(() => {
   186→    setSequence1(EXAMPLE_SEQUENCES.dna);
   187→  }, []);
   188→
   189→  const loadExampleProtein = useCallback(() => {
   190→    setSequence1(EXAMPLE_SEQUENCES.protein);
   191→  }, []);
   192→
   193→  // Handle file upload
   194→  const handleFileUpload = useCallback((event: React.ChangeEvent<HTMLInputElement>) => {
   195→    const file = event.target.files?.[0];
   196→    if (file) {
   197→      const reader = new FileReader();
   198→      reader.onload = (e) => {
   199→        const content = e.target?.result as string;
   200→        if (content.startsWith('>')) {
   201→          const parsed = parseFASTA(content);
   202→          if (parsed.length > 0) {
   203→            setSequence1(parsed[0].sequence);
   204→            if (parsed.length > 1) {
   205→              setSequence2(parsed[1].sequence);
   206→            }
   207→          }
   208→        } else {
   209→          setSequence1(content);
   210→        }
   211→      };
   212→      reader.readAsText(file);
   213→    }
   214→  }, []);
   215→
   216→  // Run analysis based on active tool
   217→  const runAnalysis = useCallback(async () => {
   218→    setIsAnalyzing(true);
   219→    setProgress(0);
   220→    setResult(null);
   221→
   222→    // Simulate progress
   223→    const progressInterval = setInterval(() => {
   224→      setProgress(prev => Math.min(prev + Math.random() * 15, 90));
   225→    }, 200);
   226→
   227→    // Simulate processing delay
   228→    await new Promise(resolve => setTimeout(resolve, 1000 + Math.random() * 1500));
   229→
   230→    let analysisResult: AnalysisResult;
   231→
   232→    try {
   233→      switch (activeTool) {
   234→        case 'pairwise': {
   235→          const alignFn = alignmentOptions.algorithm === 'global' ? globalAlignment : localAlignment;
   236→          const alignmentResult = alignFn(sequence1, sequence2, {
   237→            matchScore: alignmentOptions.matchScore,
   238→            mismatchPenalty: alignmentOptions.mismatchPenalty,
   239→            gapPenalty: alignmentOptions.gapPenalty,
   240→          });
   241→          analysisResult = { type: 'pairwise', data: alignmentResult, timestamp: new Date() };
   242→          break;
   243→        }
   244→
   245→        case 'blast': {
   246→          const blastResult = mockBLASTSearch(sequence1, {
   247→            database: blastOptions.database,
   248→            maxTargets: blastOptions.maxTargets,
   249→            program: blastOptions.program,
   250→          });
   251→          analysisResult = { type: 'blast', data: blastResult, timestamp: new Date() };
   252→          break;
   253→        }
   254→
   255→        case 'orf': {
   256→          const orfs = findORFs(sequence1, {
   257→            minLength: orfOptions.minLength,
   258→            geneticCode: orfOptions.geneticCode,
   259→            allFrames: orfOptions.showAllFrames,
   260→            startCodons: orfOptions.startCodons,
   261→          });
   262→          analysisResult = { type: 'orf', data: orfs, timestamp: new Date() };
   263→          break;
   264→        }
   265→
   266→        case 'reverse-complement': {
   267→          const rcResult = rc(sequence1);
   268→          analysisResult = { type: 'reverse-complement', data: rcResult, timestamp: new Date() };
   269→          break;
   270→        }
   271→
   272→        case 'translation': {
   273→          const translationResult = translate(
   274→            sequence1,
   275→            translationOptions.frame,
   276→            translationOptions.geneticCode
   277→          );
   278→          analysisResult = { type: 'translation', data: translationResult, timestamp: new Date() };
   279→          break;
   280→        }
   281→
   282→        case 'gc-content': {
   283→          const gcResult = getSequenceStats(sequence1);
   284→          analysisResult = { type: 'gc-content', data: gcResult, timestamp: new Date() };
   285→          break;
   286→        }
   287→
   288→        case 'motif-search':
   289→        case 'pattern-search': {
   290→          const motifResults = searchMotif(sequence1, motifPattern || 'ATG', patternSearchOptions);
   291→          analysisResult = { type: activeTool, data: motifResults, timestamp: new Date() };
   292→          break;
   293→        }
   294→
   295→        case 'restriction-map': {
   296→          const restrictionSites = findRestrictionSites(sequence1);
   297→          analysisResult = { type: 'restriction-map', data: restrictionSites, timestamp: new Date() };
   298→          break;
   299→        }
   300→
   301→        default:
   302→          throw new Error('Unknown tool');
   303→      }
   304→
   305→      clearInterval(progressInterval);
   306→      setProgress(100);
   307→      
   308→      setTimeout(() => {
   309→        setResult(analysisResult);
   310→        setIsAnalyzing(false);
   311→        setProgress(0);
   312→      }, 300);
   313→
   314→    } catch (error) {
   315→      clearInterval(progressInterval);
   316→      setIsAnalyzing(false);
   317→      console.error('Analysis error:', error);
   318→    }
   319→  }, [activeTool, sequence1, sequence2, alignmentOptions, blastOptions, orfOptions, translationOptions, motifPattern, patternSearchOptions]);
   320→
   321→  // Copy to clipboard
   322→  const copyToClipboard = useCallback((text: string) => {
   323→    navigator.clipboard.writeText(text);
   324→    setCopied(true);
   325→    setTimeout(() => setCopied(false), 2000);
   326→  }, []);
   327→
   328→  // Download result
   329→  const downloadResult = useCallback((format: 'json' | 'csv' | 'txt') => {
   330→    if (!result) return;
   331→
   332→    let content = '';
   333→    let filename = '';
   334→    let mimeType = '';
   335→
   336→    switch (format) {
   337→      case 'json':
   338→        content = JSON.stringify(result.data, null, 2);
   339→        filename = `bioalign-${result.type}-result.json`;
   340→        mimeType = 'application/json';
   341→        break;
   342→      case 'csv':
   343→        content = convertToCSV(result.data);
   344→        filename = `bioalign-${result.type}-result.csv`;
   345→        mimeType = 'text/csv';
   346→        break;
   347→      case 'txt':
   348→        content = formatPlainText(result);
   349→        filename = `bioalign-${result.type}-result.txt`;
   350→        mimeType = 'text/plain';
   351→        break;
   352→    }
   353→
   354→    const blob = new Blob([content], { type: mimeType });
   355→    const url = URL.createObjectURL(blob);
   356→    const a = document.createElement('a');
   357→    a.href = url;
   358→    a.download = filename;
   359→    a.click();
   360→    URL.revokeObjectURL(url);
   361→  }, [result]);
   362→
   363→  // Highlight nucleotides in sequence
   364→  const highlightSequence = (seq: string): React.ReactNode[] => {
   365→    return seq.split('').map((char, i) => (
   366→      <span key={i} className={NUCLEOTIDE_COLORS[char.toUpperCase()] || ''}>
   367→        {char}
   368→      </span>
   369→    ));
   370→  };
   371→
   372→  // Render tool-specific options
   373→  const renderToolOptions = () => {
   374→    switch (activeTool) {
   375→      case 'pairwise':
   376→        return (
   377→          <div className="space-y-4">
   378→            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   379→              <div>
   380→                <Label htmlFor="algorithm">Algorithm</Label>
   381→                <Select value={alignmentOptions.algorithm} onValueChange={(v) => 
   382→                  setAlignmentOptions(prev => ({ ...prev, algorithm: v as 'global' | 'local' }))
   383→                }>
   384→                  <SelectTrigger><SelectValue /></SelectTrigger>
   385→                  <SelectContent>
   386→                    <SelectItem value="global">Global (Needleman-Wunsch)</SelectItem>
   387→                    <SelectItem value="local">Local (Smith-Waterman)</SelectItem>
   388→                  </SelectContent>
   389→                </Select>
   390→              </div>
   391→              <div>
   392→                <Label htmlFor="matrix">Substitution Matrix</Label>
   393→                <Select value={alignmentOptions.matrix} onValueChange={(v) =>
   394→                  setAlignmentOptions(prev => ({ ...prev, matrix: v }))
   395→                }>
   396→                  <SelectTrigger><SelectValue /></SelectTrigger>
   397→                  <SelectContent>
   398→                    {Object.keys(SUBSTITUTION_MATRICES).map(m => (
   399→                      <SelectItem key={m} value={m}>{m}</SelectItem>
   400→                    ))}
   401→                  </SelectContent>
   402→                </Select>
   403→              </div>
   404→            </div>
   405→            
   406→            <Collapsible open={showAdvanced} onOpenChange={setShowAdvanced}>
   407→              <CollapsibleTrigger asChild>
   408→                <Button variant="ghost" size="sm" className="w-full justify-between">
   409→                  <span className="flex items-center gap-2">
   410→                    <Settings2 className="w-4 h-4" />
   411→                    Advanced Scoring Options
   412→                  </span>
   413→                  {showAdvanced ? <ChevronUp className="w-4 h-4" /> : <ChevronDown className="w-4 h-4" />}
   414→                </Button>
   415→              </CollapsibleTrigger>
   416→              <CollapsibleContent className="space-y-4 pt-4">
   417→                <div className="grid grid-cols-3 gap-4">
   418→                  <div>
   419→                    <Label>Match Score</Label>
   420→                    <Input
   421→                      type="number"
   422→                      value={alignmentOptions.matchScore}
   423→                      onChange={(e) => setAlignmentOptions(prev => ({ ...prev, matchScore: Number(e.target.value) }))}
   424→                    />
   425→                  </div>
   426→                  <div>
   427→                    <Label>Mismatch Penalty</Label>
   428→                    <Input
   429→                      type="number"
   430→                      value={alignmentOptions.mismatchPenalty}
   431→                      onChange={(e) => setAlignmentOptions(prev => ({ ...prev, mismatchPenalty: Number(e.target.value) }))}
   432→                    />
   433→                  </div>
   434→                  <div>
   435→                    <Label>Gap Penalty</Label>
   436→                    <Input
   437→                      type="number"
   438→                      value={alignmentOptions.gapPenalty}
   439→                      onChange={(e) => setAlignmentOptions(prev => ({ ...prev, gapPenalty: Number(e.target.value) }))}
   440→                    />
   441→                  </div>
   442→                </div>
   443→              </CollapsibleContent>
   444→            </Collapsible>
   445→
   446→            {/* Second sequence input */}
   447→            <div>
   448→              <Label htmlFor="seq2">Second Sequence</Label>
   449→              <Textarea
   450→                id="seq2"
   451→                placeholder="Enter second sequence..."
   452→                value={sequence2}
   453→                onChange={(e) => setSequence2(e.target.value)}
   454→                rows={4}
   455→                className="font-mono text-sm"
   456→              />
   457→              {seq2Stats && (
   458→                <div className="flex flex-wrap gap-2 mt-2">
   459→                  <Badge variant="secondary">{seq2Stats.length} bp</Badge>
   460→                  <Badge variant="secondary">{seq2Stats.type}</Badge>
   461→                  <Badge variant="secondary">GC: {seq2Stats.gcContent.toFixed(1)}%</Badge>
   462→                </div>
   463→              )}
   464→            </div>
   465→          </div>
   466→        );
   467→
   468→      case 'blast':
   469→        return (
   470→          <div className="space-y-4">
   471→            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   472→              <div>
   473→                <Label>Program</Label>
   474→                <Select value={blastOptions.program} onValueChange={(v) =>
   475→                  setBlastOptions(prev => ({ ...prev, program: v }))
   476→                }>
   477→                  <SelectTrigger><SelectValue /></SelectTrigger>
   478→                  <SelectContent>
   479→                    <SelectItem value="blastn">BLASTN (Nucleotide)</SelectItem>
   480→                    <SelectItem value="blastp">BLASTP (Protein)</SelectItem>
   481→                    <SelectItem value="blastx">BLASTX (Translated)</SelectItem>
   482→                    <SelectItem value="tblastn">TBLASTN</SelectItem>
   483→                    <SelectItem value="tblastx">TBLASTX</SelectItem>
   484→                  </SelectContent>
   485→                </Select>
   486→              </div>
   487→              <div>
   488→                <Label>Database</Label>
   489→                <Select value={blastOptions.database} onValueChange={(v) =>
   490→                  setBlastOptions(prev => ({ ...prev, database: v }))
   491→                }>
   492→                  <SelectTrigger><SelectValue /></SelectTrigger>
   493→                  <SelectContent>
   494→                    <SelectItem value="nr">NR (Non-redundant)</SelectItem>
   495→                    <SelectItem value="nt">NT (Nucleotide)</SelectItem>
   496→                    <SelectItem value="pdb">PDB</SelectItem>
   497→                    <SelectItem value="swissprot">SwissProt</SelectItem>
   498→                    <SelectItem value="refseq">RefSeq</SelectItem>
   499→                  </SelectContent>
   500→                </Select>
   501→              </div>
   502→            </div>
   503→            
   504→            <Collapsible open={showAdvanced} onOpenChange={setShowAdvanced}>
   505→              <CollapsibleTrigger asChild>
   506→                <Button variant="ghost" size="sm" className="w-full justify-between">
   507→                  <span className="flex items-center gap-2">
   508→                    <Settings2 className="w-4 h-4" />
   509→                    Advanced Parameters
   510→                  </span>
   511→                  {showAdvanced ? <ChevronUp className="w-4 h-4" /> : <ChevronDown className="w-4 h-4" />}
   512→                </Button>
   513→              </CollapsibleTrigger>
   514→              <CollapsibleContent className="space-y-4 pt-4">
   515→                <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
   516→                  <div>
   517→                    <Label>E-value Threshold</Label>
   518→                    <Input
   519→                      value={blastOptions.eValue}
   520→                      onChange={(e) => setBlastOptions(prev => ({ ...prev, eValue: e.target.value }))}
   521→                    />
   522→                  </div>
   523→                  <div>
   524→                    <Label>Max Targets</Label>
   525→                    <Input
   526→                      type="number"
   527→                      min={1}
   528→                      max={100}
   529→                      value={blastOptions.maxTargets}
   530→                      onChange={(e) => setBlastOptions(prev => ({ ...prev, maxTargets: Number(e.target.value) }))}
   531→                    />
   532→                  </div>
   533→                  <div>
   534→                    <Label>Word Size</Label>
   535→                    <Input
   536→                      type="number"
   537→                      value={blastOptions.wordSize}
   538→                      onChange={(e) => setBlastOptions(prev => ({ ...prev, wordSize: Number(e.target.value) }))}
   539→                    />
   540→                  </div>
   541→                </div>
   542→                <div className="flex items-center space-x-2">
   543→                  <Switch
   544→                    checked={blastOptions.lowComplexityFilter}
   545→                    onCheckedChange={(checked) => setBlastOptions(prev => ({ ...prev, lowComplexityFilter: checked }))}
   546→                  />
   547→                  <Label>Low complexity filter</Label>
   548→                </div>
   549→              </CollapsibleContent>
   550→            </Collapsible>
   551→          </div>
   552→        );
   553→
   554→      case 'orf':
   555→        return (
   556→          <div className="space-y-4">
   557→            <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
   558→              <div>
   559→                <Label>Genetic Code</Label>
   560→                <Select value={String(orfOptions.geneticCode)} onValueChange={(v) =>
   561→                  setOrfOptions(prev => ({ ...prev, geneticCode: Number(v) }))
   562→                }>
   563→                  <SelectTrigger><SelectValue /></SelectTrigger>
   564→                  <SelectContent>
   565→                    <SelectItem value="1">Standard (1)</SelectItem>
   566→                    <SelectItem value="2">Vertebrate Mitochondrial (2)</SelectItem>
   567→                    <SelectItem value="4">Invertebrate Mitochondrial (4)</SelectItem>
   568→                    <SelectItem value="11">Bacterial (11)</SelectItem>
   569→                  </SelectContent>
   570→                </Select>
   571→              </div>
   572→              <div>
   573→                <Label>Min ORF Length (bp)</Label>
   574→                <Input
   575→                  type="number"
   576→                  min={10}

   581→              <div className="flex items-end pb-1">

   586→                  />
   587→                  <Label>Show all 6 frames</Label>
   588→                </div>

  1285→function convertToCSV(data: unknown): string {
  1286→  if (typeof data === 'string') return data;
  1287→  if (Array.isArray(data)) {
  1288→    if (data.length === 0) return '';
  1289→    const headers = Object.keys(data[0]);
  1290→    const rows = data.map(item => 
  1291→      headers.map(h => JSON.stringify((item as Record<string, unknown>)[h] ?? '')).join(',')
  1292→    );
  1293→    return [headers.join(','), ...rows].join('\n');
  1294→  }
  1295→  if (typeof data === 'object' && data !== null) {
  1296→    const entries = Object.entries(data as Record<string, unknown>);
  1297→    return entries.map(([k, v]) => `${k},${JSON.stringify(v)}`).join('\n');
  1298→  }
  1299→  return String(data);
  1300→}

  1304→  
  1305→  if (typeof result.data === 'string') {

  1311→      Object.entries(item as Record<string, unknown>).forEach(([k, v]) => {
  1312→        lines.push(`${k}: ${JSON.stringify(v)}`);
  1313→      });
  1314→    });
  1315→  } else if (typeof result.data === 'object' && result.data !== null) {
  1316→    Object.entries(result.data as Record<string, unknown>).forEach(([k, v]) => {
  1317→      lines.push(`${k}: ${JSON.stringify(v)}`);
  1318→    });
  1319→  }
  1320→  
  1321→  return lines.join('\n');
