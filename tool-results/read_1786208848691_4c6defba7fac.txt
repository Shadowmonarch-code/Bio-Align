   108→/**
   109→ * Extracts view name and optional toolId from an href
   110→ * Maps URL paths to internal view names for navigation
   111→ */
   112→function getNavigationTarget(href: string): { view: string; toolId?: string } {
   113→  // Tool items - map to analysis view with toolId
   114→  if (href.startsWith("/dashboard/tools/")) {
   115→    const toolId = href.replace("/dashboard/tools/", "")
   116→    return { view: "analysis", toolId }
   117→  }
   118→
   119→  // Main navigation items
   120→  switch (href) {
   121→    case "/dashboard":
   122→      return { view: "dashboard" }
   123→    case "/analyze-data":
   124→      return { view: "analyze-data" }
   125→    case "/dashboard/workspaces":
   126→      return { view: "workspaces" }
   127→    case "/dashboard/databases":
   128→      return { view: "databases" }
   129→    case "/docs":
   130→      return { view: "documentation" }
   131→    case "/dashboard/settings":
   132→      return { view: "settings" }
   133→    default:
   134→      // Fallback: use the path as view name
   135→      return { view: href.replace(/^\//, "") }
   136→  }
   137→}

   271→interface SidebarContentProps {
   272→  collapsed: boolean
   273→  activeItem: string
   274→  openSections: Record<string, boolean>
   275→  setOpenSections: React.Dispatch<React.SetStateAction<Record<string, boolean>>>
   276→  onNavigate?: (view: string, toolId?: string) => void
   277→}
   278→
   279→function SidebarContent({
   280→  collapsed,
   281→  activeItem,
   282→  openSections,
   283→  setOpenSections,
   284→  onNavigate,
   285→}: SidebarContentProps) {
   286→  const { 
   287→    isAuthenticated,
   288→    userDisplayName, 
   289→    userEmail, 
   290→    userInitials,
   291→  } = useUser()
   292→
   293→  // Handle navigation click
   294→  const handleNavClick = React.useCallback(
   295→    (href: string) => {
   296→      if (onNavigate) {
   297→        const { view, toolId } = getNavigationTarget(href)
   298→        onNavigate(view, toolId)
   299→      }
   300→    },
   301→    [onNavigate]
   302→  )
   303→
   304→  const toggleSection = (sectionId: string) => {
   305→    setOpenSections((prev) => ({
   306→      ...prev,
   307→      [sectionId]: !prev[sectionId],
   308→    }))
   309→  }
   310→
   311→  const renderNavItem = (item: NavItem, isSubItem = false, sectionColor?: string) => {
   312→    const isActive = item.href === activeItem || item.isActive
   313→    
   314→    if (collapsed && !isSubItem) {
   315→      return (
   316→        <Tooltip key={item.title}>
   317→          <TooltipTrigger asChild>
   318→            <Button
   319→              variant="ghost"
   320→              size={isSubItem ? "sm" : "default"}
   321→              className={cn(
   322→                "w-full justify-center relative",
   323→                isSubItem ? "h-8 px-2" : "h-10",
   324→                isActive && sectionColor 
   325→                  ? `bg-[${sectionColor}]/10 hover:bg-[${sectionColor}]/15` 
   326→                  : isActive && "bg-biored/10 text-biored hover:bg-biored/15 hover:text-biored",
   327→                !isActive && "hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   328→              )}
   329→              onClick={() => handleNavClick(item.href)}
   330→            >
   331→              <item.icon className={cn(
   332→                "size-4 shrink-0", 
   333→                isActive && (sectionColor ? `text-[${sectionColor}]` : "text-biored")
   334→              )} />
   335→            </Button>
   336→          </TooltipTrigger>
   337→          <TooltipContent side="right">
   338→            {item.title}
   339→          </TooltipContent>
   340→        </Tooltip>
   341→      )
   342→    }
   343→
   344→    return (
   345→      <Button
   346→        key={item.title}
   347→        variant="ghost"
   348→        size={isSubItem ? "sm" : "default"}
   349→        className={cn(
   350→          "w-full justify-start gap-3 relative",
   351→          isSubItem ? "h-8 pl-7 text-sm" : "h-10 px-3",
   352→          isActive && sectionColor
   353→            ? `bg-[${sectionColor}]/10 text-[${sectionColor}] hover:bg-[${sectionColor}]/15 font-medium`
   354→            : isActive && "bg-biored/10 text-biored hover:bg-biored/15 hover:text-biored font-medium",
   355→          !isActive && "hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   356→        )}
   357→        onClick={() => handleNavClick(item.href)}
   358→      >
   359→        {isActive && (
   360→          <motion.div
   361→            layoutId={`activeIndicator-${item.title}`}
   362→            className="absolute left-0 top-1/2 -translate-y-1/2 w-[3px] h-6 rounded-r-full"
   363→            style={{ backgroundColor: sectionColor || 'var(--biored)' }}
   364→            transition={{ type: "spring", stiffness: 350, damping: 30 }}
   365→          />
   366→        )}
   367→        <item.icon className={cn(
   368→          "size-4 shrink-0", 
   369→          isActive && (sectionColor || "text-biored")
   370→        )} />
   371→        {!collapsed && (
   372→          <motion.span
   373→            initial={{ opacity: 0 }}
   374→            animate={{ opacity: 1 }}
   375→            exit={{ opacity: 0 }}
   376→            transition={{ duration: 0.2 }}
   377→          >
   378→            {item.title}
   379→          </motion.span>
   380→        )}
   381→      </Button>
   382→    )
   383→  }
   384→
   385→  const renderCollapsibleSection = (section: SectionDefinition) => {
   386→    const isOpen = openSections[section.id] ?? false
   387→    const hasNestedGroups = section.nestedGroups && Object.keys(section.nestedGroups).length > 0
   388→
   389→    if (collapsed) {
   390→      return (
   391→        <div key={section.id} className="space-y-1">
   392→          <Tooltip>
   393→            <TooltipTrigger asChild>
   394→              <Button
   395→                variant="ghost"
   396→                size="default"
   397→                className="w-full justify-center h-10 hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   398→                onClick={() => toggleSection(section.id)}
   399→              >
   400→                <section.icon className="size-4" style={{ color: section.color }} />
   401→              </Button>
   402→            </TooltipTrigger>
   403→            <TooltipContent side="right">
   404→              <span className="flex items-center gap-2">
   405→                <span>{section.emoji}</span>
   406→                <span>{section.title}</span>
   407→              </span>
   408→            </TooltipContent>
   409→          </Tooltip>
   410→        </div>
   411→      )
   412→    }
   413→
   414→    return (
   415→      <Collapsible key={section.id} open={isOpen} onOpenChange={() => toggleSection(section.id)}>
   416→        <CollapsibleTrigger asChild>
   417→          <Button
   418→            variant="ghost"
   419→            size="default"
   420→            className="w-full justify-between h-10 px-3 hover:bg-sidebar-accent hover:text-sidebar-accent-foreground group"
   421→          >
   422→            <div className="flex items-center gap-3">
   423→              <section.icon 
   424→                className="size-4 shrink-0 transition-colors group-hover:opacity-80" 
   425→                style={{ color: section.color }} 
   426→              />
   427→              <span className="font-medium">{section.emoji} {section.title}</span>
   428→            </div>
   429→            <motion.div
   430→              animate={{ rotate: isOpen ? 90 : 0 }}
   431→              transition={{ duration: 0.2 }}
   432→            >
   433→              <ChevronRight className="size-4 opacity-50" />
   434→            </motion.div>
   435→          </Button>
   436→        </CollapsibleTrigger>
   437→        <CollapsibleContent>
   438→          <motion.div
   439→            initial={{ height: 0, opacity: 0 }}
   440→            animate={{ height: "auto", opacity: 1 }}
   441→            exit={{ height: 0, opacity: 0 }}
   442→            transition={{ duration: 0.25, ease: "easeInOut" }}
   443→            className="overflow-hidden"
   444→          >
   445→            <div className="mt-1 space-y-0.5 pl-2">
   446→              {section.items.map((item) => {
   447→                // Check if this item has a nested group
   448→                const nestedKey = Object.keys(section.nestedGroups || {}).find(
   449→                  key => item.href.includes(key.toLowerCase().replace(/\s+/g, "-"))
   450→                )
   451→                
   452→                if (nestedKey && section.nestedGroups) {
   453→                  const nestedItems = section.nestedGroups[nestedKey]
   454→                  const isNestedOpen = openSections[`${section.id}-${nestedKey}`] ?? false
   455→                  
   456→                  return (
   457→                    <Collapsible key={item.title} open={isNestedOpen} onOpenChange={() => 
   458→                      setOpenSections(prev => ({ ...prev, [`${section.id}-${nestedKey}`]: !isNestedOpen }))
   459→                    }>
   460→                      <CollapsibleTrigger asChild>
   461→                        <Button
   462→                          variant="ghost"
   463→                          size="sm"
   464→                          className="w-full justify-between h-8 pl-6 pr-3 text-sm hover:bg-sidebar-accent/50"
   465→                        >
   466→                          <div className="flex items-center gap-2">
   467→                            <item.icon className="size-3.5" style={{ color: section.color }} />
   468→                            <span>{item.title}</span>
   469→                          </div>
   470→                          <motion.div
   471→                            animate={{ rotate: isNestedOpen ? 90 : 0 }}
   472→                            transition={{ duration: 0.15 }}
   473→                          >
   474→                            <ChevronRight className="size-3 opacity-50" />
   475→                          </motion.div>
   476→                        </Button>
   477→                      </CollapsibleTrigger>
   478→                      <CollapsibleContent>
   479→                        <motion.div
   480→                          initial={{ height: 0, opacity: 0 }}
   481→                          animate={{ height: "auto", opacity: 1 }}
   482→                          exit={{ height: 0, opacity: 0 }}
   483→                          transition={{ duration: 0.2 }}
   484→                          className="overflow-hidden"
   485→                        >
   486→                          <div className="py-1 space-y-0.5 pl-4">
   487→                            {nestedItems.map((nestedItem) => 
   488→                              renderNavItem(nestedItem, true, section.color)
   489→                            )}
   490→                          </div>
   491→                        </motion.div>
   492→                      </CollapsibleContent>
   493→                    </Collapsible>
   494→                  )
   495→                }
   496→                
   497→                return renderNavItem(item, true, section.color)
   498→              })}
   499→            </div>
   500→          </motion.div>
   501→        </CollapsibleContent>
   502→      </Collapsible>
   503→    )
   504→  }
   505→
   506→  return (
   507→    <div className="flex flex-col h-full">
   508→      {/* Logo Section */}
   509→      <div className={cn(
   510→        "flex items-center h-16 border-b border-sidebar-border px-4",
   511→        collapsed ? "justify-center" : "gap-3"
   512→      )}>
   513→        <div className="flex items-center justify-center w-9 h-9 rounded-lg bg-gradient-to-br from-biored to-biored-dark shadow-lg shadow-biored/25">
   514→          <Dna className="size-5 text-white" />
   515→        </div>
   516→        <AnimatePresence mode="wait">
   517→          {!collapsed && (
   518→            <motion.div
   519→              initial={{ opacity: 0, width: 0 }}
   520→              animate={{ opacity: 1, width: "auto" }}
   521→              exit={{ opacity: 0, width: 0 }}
   522→              transition={{ duration: 0.2, ease: "easeInOut" }}
   523→              className="overflow-hidden whitespace-nowrap"
   524→            >
   525→              <span className="text-lg font-bold tracking-tight gradient-text">BioAlign</span>
   526→            </motion.div>
   527→          )}
   528→        </AnimatePresence>
   529→      </div>
   530→
   531→      {/* Navigation */}
   532→      <ScrollArea className="flex-1 py-4">
   533→        <div className="px-3 space-y-4">
   534→          {/* Main Navigation */}
   535→          <div className="space-y-1">
   536→            {!collapsed && (
   537→              <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground/60 px-3 mb-2">
   538→                Main
   539→              </p>
   540→            )}
   541→            {mainNavItems.map((item) => renderNavItem(item))}
   542→          </div>
   543→
   544→          {/* Separator after main */}
   545→          <Separator className="bg-sidebar-border/50" />
   546→
   547→          {/* Collapsible Sections */}
   548→          {collapsibleSections.map((section) => (
   549→            <div key={section.id} className="space-y-1">
   550→              {renderCollapsibleSection(section)}
   551→            </div>
   552→          ))}
   553→
   554→          {/* Separator before other */}
   555→          <Separator className="bg-sidebar-border/50" />
   556→
   557→          {/* AI Assistant - Special standalone item */}
   558→          <div className="space-y-1">
   559→            {renderNavItem(
   560→              { 
   561→                title: "AI Assistant", 
   562→                href: "/ai-assistant", 
   563→                icon: Brain 
   564→              }, 
   565→              false, 
   566→              "#A855F7"
   567→            )}
   568→          </div>
   569→
   570→          {/* Other Navigation */}
   571→          <div className="space-y-1">
   572→            {!collapsed && (
   573→              <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground/60 px-3 mb-2 mt-4">
   574→                System
   575→              </p>
   576→            )}
   577→            {otherNavItems.map((item) => renderNavItem(item))}
   578→          </div>
   579→        </div>
   580→      </ScrollArea>
   581→
   582→      {/* User Profile Section - Only show when authenticated */}
   583→      <Separator className="bg-sidebar-border" />
   584→      {isAuthenticated ? (
   585→        <div className={cn(
   586→          "flex items-center h-16 px-4 border-t border-sidebar-border",

   614→      ) : (

   730→export function useSidebarState() {
   731→  const [collapsed, setCollapsed] = React.useState(false)
   732→  const [mobileOpen, setMobileOpen] = React.useState(false)
   733→  const isMobile = useIsMobile()
   734→
   735→  const toggle = React.useCallback(() => {
   736→    if (isMobile) {
   737→      setMobileOpen((prev) => !prev)

   739→      setCollapsed((prev) => !prev)
   740→    }
   741→  }, [isMobile])
   742→

   744→    collapsed,
   745→    setCollapsed,
   746→    mobileOpen,
   747→    setMobileOpen,
   748→    isMobile,
   749→    toggle,
   750→  }
