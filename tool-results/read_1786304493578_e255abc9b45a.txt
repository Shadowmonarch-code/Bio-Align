   108→   108→/**
   109→   109→ * Extracts view name and optional toolId from an href
   110→   110→ * Maps URL paths to internal view names for navigation
   111→   111→ */
   112→   112→function getNavigationTarget(href: string): { view: string; toolId?: string } {
   113→   113→  // Tool items - map to analysis view with toolId
   114→   114→  if (href.startsWith("/dashboard/tools/")) {
   115→   115→    const toolId = href.replace("/dashboard/tools/", "")
   116→   116→    return { view: "analysis", toolId }
   117→   117→  }
   118→   118→
   119→   119→  // Plant Breeding section - all map to plant-breeding view
   120→   120→  if (href.startsWith("/plant-breeding")) {
   121→   121→    return { view: "plant-breeding" }
   122→   122→  }
   123→   123→
   124→   124→  // Bioinformatics section - map to analysis view with toolId
   125→   125→  if (href.startsWith("/bioinformatics")) {
   126→   126→    const toolId = href.replace("/bioinformatics/", "")
   127→   127→    return { view: "analysis", toolId }
   128→   128→  }
   129→   129→
   130→   130→  // Thesis Studio section - map to thesis view
   131→   131→  if (href.startsWith("/thesis")) {
   132→   132→    return { view: "thesis" }
   133→   133→  }
   134→   134→
   135→   135→  // Visualization section - could map to a visualization view or tools
   136→   136→  if (href.startsWith("/visualization")) {
   137→   137→    const toolId = href.replace("/visualization/", "")
   138→   138→    return { view: "analysis", toolId: `viz-${toolId}` }
   139→   139→  }
   140→   140→
   141→   141→  // Main navigation items
   142→   142→  switch (href) {
   143→   143→    case "/dashboard":
   144→   144→      return { view: "dashboard" }
   145→   145→    case "/analyze-data":
   146→   146→      return { view: "analyze-data" }
   147→   147→    case "/dashboard/workspaces":
   148→   148→    case "/dashboard/projects":
   149→   149→      return { view: "workspaces" }
   150→   150→    case "/dashboard/databases":
   151→   151→      return { view: "databases" }
   152→   152→    case "/docs":
   153→   153→      return { view: "documentation" }
   154→   154→    case "/dashboard/settings":
   155→   155→      return { view: "settings" }
   156→   156→    case "/dashboard/workflows":
   157→   157→      return { view: "workspaces" }
   158→   158→    default:
   159→   159→      // Fallback: try to extract meaningful view name
   160→   160→      const path = href.replace(/^\//, "")
   161→   161→      // Map common patterns
   162→   162→      if (path.startsWith("plant-breeding")) return { view: "plant-breeding" }
   163→   163→      if (path.startsWith("bioinformatics")) return { view: "analysis", toolId: path.replace("bioinformatics/", "") }
   164→   164→      if (path.startsWith("thesis")) return { view: "thesis" }
   165→   165→      return { view: "dashboard" }
   166→   166→  }
   167→   167→}

   308→   308→
   309→   309→function SidebarContent({
   310→   310→  collapsed,
   311→   311→  activeItem,
   312→   312→  openSections,
   313→   313→  setOpenSections,
   314→   314→  onNavigate,
   315→   315→}: SidebarContentProps) {
   316→   316→  const { 
   317→   317→    isAuthenticated,
   318→   318→    userDisplayName, 
   319→   319→    userEmail, 
   320→   320→    userInitials,
   321→   321→  } = useUser()
   322→   322→
   323→   323→  // Handle navigation click
   324→   324→  const handleNavClick = React.useCallback(
   325→   325→    (href: string) => {
   326→   326→      if (onNavigate) {
   327→   327→        const { view, toolId } = getNavigationTarget(href)
   328→   328→        onNavigate(view, toolId)
   329→   329→      }
   330→   330→    },
   331→   331→    [onNavigate]
   332→   332→  )
   333→   333→
   334→   334→  const toggleSection = (sectionId: string) => {
   335→   335→    setOpenSections((prev) => ({
   336→   336→      ...prev,
   337→   337→      [sectionId]: !prev[sectionId],
   338→   338→    }))
   339→   339→  }
   340→   340→
   341→   341→  const renderNavItem = (item: NavItem, isSubItem = false, sectionColor?: string) => {
   342→   342→    const isActive = item.href === activeItem || item.isActive
   343→   343→    
   344→   344→    if (collapsed && !isSubItem) {
   345→   345→      return (
   346→   346→        <Tooltip key={item.title}>
   347→   347→          <TooltipTrigger asChild>
   348→   348→            <Button
   349→   349→              variant="ghost"
   350→   350→              size={isSubItem ? "sm" : "default"}
   351→   351→              className={cn(
   352→   352→                "w-full justify-center relative",
   353→   353→                isSubItem ? "h-8 px-2" : "h-10",
   354→   354→                isActive && sectionColor 
   355→   355→                  ? `bg-[${sectionColor}]/10 hover:bg-[${sectionColor}]/15` 
   356→   356→                  : isActive && "bg-biored/10 text-biored hover:bg-biored/15 hover:text-biored",
   357→   357→                !isActive && "hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   358→   358→              )}
   359→   359→              onClick={() => handleNavClick(item.href)}
   360→   360→            >
   361→   361→              <item.icon className={cn(
   362→   362→                "size-4 shrink-0", 
   363→   363→                isActive && (sectionColor ? `text-[${sectionColor}]` : "text-biored")
   364→   364→              )} />
   365→   365→            </Button>
   366→   366→          </TooltipTrigger>
   367→   367→          <TooltipContent side="right">
   368→   368→            {item.title}
   369→   369→          </TooltipContent>
   370→   370→        </Tooltip>
   371→   371→      )
   372→   372→    }
   373→   373→
   374→   374→    return (
   375→   375→      <Button
   376→   376→        key={item.title}
   377→   377→        variant="ghost"
   378→   378→        size={isSubItem ? "sm" : "default"}
   379→   379→        className={cn(
   380→   380→          "w-full justify-start gap-3 relative",
   381→   381→          isSubItem ? "h-8 pl-7 text-sm" : "h-10 px-3",
   382→   382→          isActive && sectionColor
   383→   383→            ? `bg-[${sectionColor}]/10 text-[${sectionColor}] hover:bg-[${sectionColor}]/15 font-medium`

   390→   390→          <motion.div
   391→   391→            layoutId={`activeIndicator-${item.title}`}

   394→   394→            transition={{ type: "spring", stiffness: 350, damping: 30 }}

   396→   396→        )}
   397→   397→        <item.icon className={cn(
   398→   398→          "size-4 shrink-0", 
   399→   399→          isActive && (sectionColor || "text-biored")
   400→   400→        )} />
   401→   401→        {!collapsed && (
   402→   402→          <motion.span
   403→   403→            initial={{ opacity: 0 }}
   404→   404→            animate={{ opacity: 1 }}
   405→   405→            exit={{ opacity: 0 }}
   406→   406→            transition={{ duration: 0.2 }}
   407→   407→          >
   408→   408→            {item.title}
   409→   409→          </motion.span>
   410→   410→        )}

   414→   414→
   415→   415→  const renderCollapsibleSection = (section: SectionDefinition) => {
   416→   416→    const isOpen = openSections[section.id] ?? false
   417→   417→    const hasNestedGroups = section.nestedGroups && Object.keys(section.nestedGroups).length > 0

   419→   419→    if (collapsed) {
   420→   420→      return (
   421→   421→        <div key={section.id} className="space-y-1">
   422→   422→          <Tooltip>
   423→   423→            <TooltipTrigger asChild>
   424→   424→              <Button
   425→   425→                variant="ghost"
   426→   426→                size="default"
   427→   427→                className="w-full justify-center h-10 hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"

   451→   451→            size="default"

   455→   455→              <span style={{ color: section.color }} className="size-4 shrink-0 flex items-center justify-center transition-colors group-hover:opacity-80">

   469→   469→        </CollapsibleTrigger>
   470→   470→        <CollapsibleContent>
   471→   471→          <motion.div
   472→   472→            initial={{ height: 0, opacity: 0 }}
   473→   473→            animate={{ height: "auto", opacity: 1 }}
   474→   474→            exit={{ height: 0, opacity: 0 }}
   475→   475→            transition={{ duration: 0.25, ease: "easeInOut" }}

   477→   477→          >

   479→   479→              {section.items.map((item) => {
