   108→/**
   109→ * Extracts view name and optional toolId from an href
   110→ * Maps URL paths to internal view names for navigation
   111→ */
   112→function getNavigationTarget(href: string): { view: string; toolId?: string } {
   113→  // Tool items - map to analysis view with toolId
   114→  if (href.startsWith("/dashboard/tools/")) {
   115→    const toolId = href.replace("/dashboard/tools/", "")
   116→    return { view: "analysis", toolId }
   117→  }
   118→
   119→  // Main navigation items
   120→  switch (href) {
   121→    case "/dashboard":
   122→      return { view: "dashboard" }
   123→    case "/analyze-data":
   124→      return { view: "analyze-data" }
   125→    case "/dashboard/workspaces":
   126→      return { view: "workspaces" }
   127→    case "/dashboard/databases":
   128→      return { view: "databases" }
   129→    case "/docs":
   130→      return { view: "documentation" }
   131→    case "/dashboard/settings":
   132→      return { view: "settings" }
   133→    default:
   134→      // Fallback: use the path as view name
   135→      return { view: href.replace(/^\//, "") }
   136→  }
   137→}

   271→interface SidebarContentProps {
   272→  collapsed: boolean
   273→  activeItem: string
   274→  openSections: Record<string, boolean>
   275→  setOpenSections: React.Dispatch<React.SetStateAction<Record<string, boolean>>>
   276→  onNavigate?: (view: string, toolId?: string) => void
   277→}
   278→
   279→function SidebarContent({
   280→  collapsed,
   281→  activeItem,
   282→  openSections,
   283→  setOpenSections,
   284→  onNavigate,
   285→}: SidebarContentProps) {
   286→  const { 
   287→    isAuthenticated,
   288→    userDisplayName, 
   289→    userEmail, 
   290→    userInitials,
   291→  } = useUser()
   292→
   293→  // Handle navigation click
   294→  const handleNavClick = React.useCallback(
   295→    (href: string) => {
   296→      if (onNavigate) {
   297→        const { view, toolId } = getNavigationTarget(href)
   298→        onNavigate(view, toolId)
   299→      }
   300→    },
   301→    [onNavigate]
   302→  )
   303→
   304→  const toggleSection = (sectionId: string) => {
   305→    setOpenSections((prev) => ({
   306→      ...prev,
   307→      [sectionId]: !prev[sectionId],
   308→    }))
   309→  }
   310→
   311→  const renderNavItem = (item: NavItem, isSubItem = false, sectionColor?: string) => {
   312→    const isActive = item.href === activeItem || item.isActive
   313→    
   314→    if (collapsed && !isSubItem) {
   315→      return (
   316→        <Tooltip key={item.title}>
   317→          <TooltipTrigger asChild>
   318→            <Button
   319→              variant="ghost"
   320→              size={isSubItem ? "sm" : "default"}
   321→              className={cn(
   322→                "w-full justify-center relative",
   323→                isSubItem ? "h-8 px-2" : "h-10",
   324→                isActive && sectionColor 
   325→                  ? `bg-[${sectionColor}]/10 hover:bg-[${sectionColor}]/15` 
   326→                  : isActive && "bg-biored/10 text-biored hover:bg-biored/15 hover:text-biored",
   327→                !isActive && "hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   328→              )}
   329→              onClick={() => handleNavClick(item.href)}
   330→            >
   331→              <item.icon className={cn(
   332→                "size-4 shrink-0", 
   333→                isActive && (sectionColor ? `text-[${sectionColor}]` : "text-biored")
   334→              )} />
   335→            </Button>
   336→          </TooltipTrigger>
   337→          <TooltipContent side="right">
   338→            {item.title}
   339→          </TooltipContent>
   340→        </Tooltip>
   341→      )
   342→    }
   343→
   344→    return (
   345→      <Button
   346→        key={item.title}
   347→        variant="ghost"
   348→        size={isSubItem ? "sm" : "default"}
   349→        className={cn(
   350→          "w-full justify-start gap-3 relative",
   351→          isSubItem ? "h-8 pl-7 text-sm" : "h-10 px-3",
   352→          isActive && sectionColor
   353→            ? `bg-[${sectionColor}]/10 text-[${sectionColor}] hover:bg-[${sectionColor}]/15 font-medium`
   354→            : isActive && "bg-biored/10 text-biored hover:bg-biored/15 hover:text-biored font-medium",
   355→          !isActive && "hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   356→        )}
   357→        onClick={() => handleNavClick(item.href)}
   358→      >
   359→        {isActive && (
   360→          <motion.div
   361→            layoutId={`activeIndicator-${item.title}`}
   362→            className="absolute left-0 top-1/2 -translate-y-1/2 w-[3px] h-6 rounded-r-full"
   363→            style={{ backgroundColor: sectionColor || 'var(--biored)' }}
   364→            transition={{ type: "spring", stiffness: 350, damping: 30 }}
   365→          />
   366→        )}
   367→        <item.icon className={cn(
   368→          "size-4 shrink-0", 
   369→          isActive && (sectionColor || "text-biored")
   370→        )} />
   371→        {!collapsed && (
   372→          <motion.span
   373→            initial={{ opacity: 0 }}
   374→            animate={{ opacity: 1 }}
   375→            exit={{ opacity: 0 }}
   376→            transition={{ duration: 0.2 }}
   377→          >
   378→            {item.title}
   379→          </motion.span>
   380→        )}
   381→      </Button>
   382→    )
   383→  }
   384→
   385→  const renderCollapsibleSection = (section: SectionDefinition) => {
   386→    const isOpen = openSections[section.id] ?? false
   387→    const hasNestedGroups = section.nestedGroups && Object.keys(section.nestedGroups).length > 0
   388→
   389→    if (collapsed) {
   390→      return (
   391→        <div key={section.id} className="space-y-1">
   392→          <Tooltip>
   393→            <TooltipTrigger asChild>
   394→              <Button
   395→                variant="ghost"
   396→                size="default"
   397→                className="w-full justify-center h-10 hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   398→                onClick={() => toggleSection(section.id)}
   399→              >
   400→                <span style={{ color: section.color }} className="size-4 flex items-center justify-center">
   401→                  <section.icon className="size-4" />
   402→                </span>
   403→              </Button>
   404→            </TooltipTrigger>
   405→            <TooltipContent side="right">
   406→              <span className="flex items-center gap-2">
   407→                <span>{section.emoji}</span>
   408→                <span>{section.title}</span>
   409→              </span>
   410→            </TooltipContent>
   411→          </Tooltip>
   412→        </div>
   413→      )
   414→    }
   415→
   416→    return (
   417→      <Collapsible key={section.id} open={isOpen} onOpenChange={() => toggleSection(section.id)}>
   418→        <CollapsibleTrigger asChild>
   419→          <Button
   420→            variant="ghost"
   421→            size="default"
   422→            className="w-full justify-between h-10 px-3 hover:bg-sidebar-accent hover:text-sidebar-accent-foreground group"
   423→          >
   424→            <div className="flex items-center gap-3">
   425→              <span style={{ color: section.color }} className="size-4 shrink-0 flex items-center justify-center transition-colors group-hover:opacity-80">
   426→                <section.icon 
   427→                  className="size-4" 
   428→                />
   429→              </span>
   430→              <span className="font-medium">{section.emoji} {section.title}</span>
   431→            </div>
   432→            <motion.div
   433→              animate={{ rotate: isOpen ? 90 : 0 }}
   434→              transition={{ duration: 0.2 }}
   435→            >
   436→              <ChevronRight className="size-4 opacity-50" />
   437→            </motion.div>
   438→          </Button>
   439→        </CollapsibleTrigger>
   440→        <CollapsibleContent>
   441→          <motion.div
   442→            initial={{ height: 0, opacity: 0 }}
   443→            animate={{ height: "auto", opacity: 1 }}
   444→            exit={{ height: 0, opacity: 0 }}
   445→            transition={{ duration: 0.25, ease: "easeInOut" }}
   446→            className="overflow-hidden"
   447→          >
   448→            <div className="mt-1 space-y-0.5 pl-2">
   449→              {section.items.map((item) => {
   450→                // Check if this item has a nested group
   451→                const nestedKey = Object.keys(section.nestedGroups || {}).find(
   452→                  key => item.href.includes(key.toLowerCase().replace(/\s+/g, "-"))
   453→                )
   454→                
   455→                if (nestedKey && section.nestedGroups) {
   456→                  const nestedItems = section.nestedGroups[nestedKey]
   457→                  const isNestedOpen = openSections[`${section.id}-${nestedKey}`] ?? false
   458→                  
   459→                  return (
   460→                    <Collapsible key={item.title} open={isNestedOpen} onOpenChange={() => 
   461→                      setOpenSections(prev => ({ ...prev, [`${section.id}-${nestedKey}`]: !isNestedOpen }))
   462→                    }>
   463→                      <CollapsibleTrigger asChild>
   464→                        <Button
   465→                          variant="ghost"
   466→                          size="sm"
   467→                          className="w-full justify-between h-8 pl-6 pr-3 text-sm hover:bg-sidebar-accent/50"
   468→                        >
   469→                          <div className="flex items-center gap-2">
   470→                            <span style={{ color: section.color }} className="size-3.5 flex items-center justify-center">
   471→                              <item.icon className="size-3.5" />
   472→                            </span>
   473→                            <span>{item.title}</span>
   474→                          </div>
   475→                          <motion.div
   476→                            animate={{ rotate: isNestedOpen ? 90 : 0 }}
   477→                            transition={{ duration: 0.15 }}
   478→                          >
   479→                            <ChevronRight className="size-3 opacity-50" />
   480→                          </motion.div>
   481→                        </Button>
   482→                      </CollapsibleTrigger>
   483→                      <CollapsibleContent>
   484→                        <motion.div
   485→                          initial={{ height: 0, opacity: 0 }}
   486→                          animate={{ height: "auto", opacity: 1 }}
   487→                          exit={{ height: 0, opacity: 0 }}
   488→                          transition={{ duration: 0.2 }}
   489→                          className="overflow-hidden"
   490→                        >
   491→                          <div className="py-1 space-y-0.5 pl-4">
   492→                            {nestedItems.map((nestedItem) => 
   493→                              renderNavItem(nestedItem, true, section.color)
   494→                            )}
   495→                          </div>
   496→                        </motion.div>
   497→                      </CollapsibleContent>
   498→                    </Collapsible>
   499→                  )
   500→                }
   501→                
   502→                return renderNavItem(item, true, section.color)
   503→              })}
   504→            </div>
   505→          </motion.div>
   506→        </CollapsibleContent>
   507→      </Collapsible>
   508→    )
   509→  }
   510→
   511→  return (
   512→    <div className="flex flex-col h-full">
   513→      {/* Logo Section */}
   514→      <div className={cn(
   515→        "flex items-center h-16 border-b border-sidebar-border px-4",
   516→        collapsed ? "justify-center" : "gap-3"
   517→      )}>
   518→        <div className="flex items-center justify-center w-9 h-9 rounded-lg bg-gradient-to-br from-biored to-biored-dark shadow-lg shadow-biored/25">
   519→          <Dna className="size-5 text-white" />
   520→        </div>
   521→        <AnimatePresence mode="wait">
   522→          {!collapsed && (
   523→            <motion.div
   524→              initial={{ opacity: 0, width: 0 }}
   525→              animate={{ opacity: 1, width: "auto" }}
   526→              exit={{ opacity: 0, width: 0 }}
   527→              transition={{ duration: 0.2, ease: "easeInOut" }}
   528→              className="overflow-hidden whitespace-nowrap"
   529→            >
   530→              <span className="text-lg font-bold tracking-tight gradient-text">BioAlign</span>
   531→            </motion.div>
   532→          )}
   533→        </AnimatePresence>
   534→      </div>
   535→
   536→      {/* Navigation */}
   537→      <ScrollArea className="flex-1 py-4">
   538→        <div className="px-3 space-y-4">
   539→          {/* Main Navigation */}
   540→          <div className="space-y-1">
   541→            {!collapsed && (
   542→              <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground/60 px-3 mb-2">
   543→                Main
   544→              </p>
   545→            )}
   546→            {mainNavItems.map((item) => renderNavItem(item))}
   547→          </div>
   548→
   549→          {/* Separator after main */}
   550→          <Separator className="bg-sidebar-border/50" />
   551→
   552→          {/* Collapsible Sections */}
   553→          {collapsibleSections.map((section) => (
   554→            <div key={section.id} className="space-y-1">
   555→              {renderCollapsibleSection(section)}
   556→            </div>
   557→          ))}
   558→
   559→          {/* Separator before other */}
   560→          <Separator className="bg-sidebar-border/50" />
   561→
   562→          {/* AI Assistant - Special standalone item */}
   563→          <div className="space-y-1">
   564→            {renderNavItem(
   565→              { 
   566→                title: "AI Assistant", 
   567→                href: "/ai-assistant", 
   568→                icon: Brain 
   569→              }, 
   570→              false, 
   571→              "#A855F7"
   572→            )}
   573→          </div>
   574→
   575→          {/* Other Navigation */}
   576→          <div className="space-y-1">
   577→            {!collapsed && (

   579→                System

   585→      </ScrollArea>

   587→      {/* User Profile Section - Only show when authenticated */}

   589→      {isAuthenticated ? (

   608→              >

   613→                  </span>
   614→                </div>
   615→              </motion.div>
   616→            )}
   617→          </AnimatePresence>
   618→        </div>
   619→      ) : (
   620→        <div className={cn(
   621→          "flex items-center h-16 px-4 border-t border-sidebar-border justify-center",
   622→        )}>
   623→          <span className="text-xs text-muted-foreground">Not signed in</span>
   624→        </div>
   625→      )}
   626→    </div>
   627→  )

   735→export function useSidebarState() {
   736→  const [collapsed, setCollapsed] = React.useState(false)
   737→  const [mobileOpen, setMobileOpen] = React.useState(false)
   738→  const isMobile = useIsMobile()
   739→

   741→    if (isMobile) {
   742→      setMobileOpen((prev) => !prev)

   744→      setCollapsed((prev) => !prev)
   745→    }
   746→  }, [isMobile])
   747→

   749→    collapsed,
   750→    setCollapsed,
   751→    mobileOpen,
   752→    setMobileOpen,
   753→    isMobile,
   754→    toggle,
   755→  }
