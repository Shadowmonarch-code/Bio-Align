   127→export default function Navbar({ onNavigate }: NavbarProps) {
   128→  const [isScrolled, setIsScrolled] = React.useState(false);
   129→  const [isMobileOpen, setIsMobileOpen] = React.useState(false);
   130→  const [activeSection, setActiveSection] = React.useState("");
   131→  const [mounted, setMounted] = React.useState(false);
   132→  const [showAuthModal, setShowAuthModal] = React.useState(false);
   133→  const [authMode, setAuthMode] = React.useState<"signin" | "signup">("signin");
   134→  
   135→  // Search state
   136→  const [showSearch, setShowSearch] = React.useState(false);
   137→  const [searchQuery, setSearchQuery] = React.useState("");
   138→  const [searchResults, setSearchResults] = React.useState<typeof databaseOptions>([]);
   139→  
   140→  // Form state
   141→  const [authEmail, setAuthEmail] = React.useState("");
   142→  const [authPassword, setAuthPassword] = React.useState("");
   143→  const [authName, setAuthName] = React.useState("");
   144→  const [isLoading, setIsLoading] = React.useState(false);
   145→  const [authError, setAuthError] = React.useState("");
   146→  
   147→  // Contact modal
   148→  const [showContactModal, setShowContactModal] = React.useState(false);
   149→  
   150→  const { theme, setTheme } = useTheme();
   151→  const { data: session, status } = useSession();
   152→
   153→  // Handle mount state for theme
   154→  React.useEffect(() => {
   155→    setMounted(true);
   156→  }, []);
   157→
   158→  // Scroll detection for glassmorphism effect
   159→  React.useEffect(() => {
   160→    const handleScroll = () => {
   161→      setIsScrolled(window.scrollY > 20);
   162→      
   163→      // Detect active section
   164→      const sections = navLinks.map((link) => link.href.substring(1));
   165→      for (const section of sections.reverse()) {
   166→        const element = document.getElementById(section);
   167→        if (element) {
   168→          const rect = element.getBoundingClientRect();
   169→          if (rect.top <= 150) {
   170→            setActiveSection(section);
   171→            break;
   172→          }
   173→        }
   174→      }
   175→    };
   176→
   177→    window.addEventListener("scroll", handleScroll, { passive: true });
   178→    return () => window.removeEventListener("scroll", handleScroll);
   179→  }, []);
   180→
   181→  // Toggle theme
   182→  const toggleTheme = () => {
   183→    setTheme(theme === "dark" ? "light" : "dark");
   184→  };
   185→
   186→  // Handle navigation click - ALL links are now functional
   187→  const handleNavClick = (href: string, action?: string) => {
   188→    setIsMobileOpen(false);
   189→    
   190→    if (action && onNavigate) {
   191→      onNavigate(action);
   192→      window.scrollTo({ top: 0 });
   193→      return;
   194→    }
   195→    
   196→    // For landing page sections that exist
   197→    const elementId = href.substring(1);
   198→    const element = document.getElementById(elementId);
   199→    if (element) {
   200→      element.scrollIntoView({ behavior: "smooth" });
   201→    } else if (onNavigate) {
   202→      // Navigate to view if section doesn't exist on current page
   203→      onNavigate(elementId);
   204→      window.scrollTo({ top: 0 });
   205→    }
   206→  };
   207→
   208→  // Handle Get Started button click
   209→  const handleGetStarted = () => {
   210→    if (onNavigate) {
   211→      onNavigate('dashboard');
   212→      window.scrollTo({ top: 0 });
   213→    }
   214→  };
   215→
   216→  // Handle Sign In button click
   217→  const handleSignIn = () => {
   218→    setShowAuthModal(true);
   219→    setAuthMode('signin');
   220→    setAuthError("");
   221→  };
   222→
   223→  // Handle Sign Up button click  
   224→  const handleSignUp = () => {
   225→    setShowAuthModal(true);
   226→    setAuthMode('signup');
   227→    setAuthError("");
   228→  };
   229→
   230→  // Handle sign out
   231→  const handleSignOut = async () => {
   232→    await signOut({ callbackUrl: '/' });
   233→  };
   234→
   235→  // Reset form when modal opens
   236→  React.useEffect(() => {
   237→    if (showAuthModal) {
   238→      setAuthEmail("");
   239→      setAuthPassword("");
   240→      setAuthName("");
   241→      setAuthError("");
   242→    }
   243→  }, [showAuthModal, authMode]);
   244→
   245→  // Handle form submission with real authentication
   246→  const handleAuthSubmit = async (e: React.FormEvent) => {
   247→    e.preventDefault();
   248→    setAuthError("");
   249→    setIsLoading(true);
   250→
   251→    try {
   252→      if (authMode === 'signup') {
   253→        // Registration
   254→        const response = await fetch('/api/auth/register', {
   255→          method: 'POST',
   256→          headers: { 'Content-Type': 'application/json' },
   257→          body: JSON.stringify({
   258→            name: authName,
   259→            email: authEmail,
   260→            password: authPassword,
   261→          }),
   262→        });
   263→
   264→        const data = await response.json();
   265→
   266→        if (!response.ok) {
   267→          throw new Error(data.error || 'Registration failed');
   268→        }
   269→
   270→        // After successful registration, sign in
   271→        const result = await signIn('credentials', {
   272→          email: authEmail,
   273→          password: authPassword,
   274→          redirect: false,
   275→        });
   276→
   277→        if (result?.error) {
   278→          throw new Error(result.error);
   279→        }
   280→      } else {
   281→        // Sign in
   282→        const result = await signIn('credentials', {
   283→          email: authEmail,
   284→          password: authPassword,
   285→          redirect: false,
   286→        });
   287→
   288→        if (result?.error) {
   289→          throw new Error('Invalid email or password');
   290→        }
   291→      }
   292→
   293→      // Success - close modal and navigate to dashboard
   294→      setShowAuthModal(false);
   295→      if (onNavigate) {
   296→        onNavigate('dashboard');
   297→        window.scrollTo({ top: 0 });
   298→      }
   299→    } catch (error: any) {
   300→      setAuthError(error.message || 'An error occurred');
   301→    } finally {
   302→      setIsLoading(false);
   303→    }
   304→  };
   305→
   306→  // Handle database search
   307→  const handleDatabaseSearch = (query: string) => {
   308→    setSearchQuery(query);
   309→    if (query.trim().length > 0) {
   310→      const filtered = databaseOptions.filter(db => 
   311→        db.name.toLowerCase().includes(query.toLowerCase()) ||
   312→        db.description.toLowerCase().includes(query.toLowerCase())
   313→      );
   314→      setSearchResults(filtered);
   315→    } else {
   316→      setSearchResults([]);
   317→    }
   318→  };
   319→
   320→  // Open external database link
   321→  const openDatabaseLink = (url: string, dbName: string) => {
   322→    window.open(url, '_blank', 'noopener,noreferrer');
   323→  };
   324→
   325→  // User initials for avatar
   326→  const getUserInitials = () => {
   327→    if (session?.user?.name) {
   328→      return session.user.name.split(' ').map(n => n[0]).join('').toUpperCase().slice(0, 2);
   329→    }
   330→    if (session?.user?.email) {
   331→      return session.user.email[0].toUpperCase();
   332→    }
   333→    return 'U';
   334→  };
   335→
   336→  return (
   337→    <>
   338→      <motion.header
   339→        initial={{ y: -100, opacity: 0 }}
   340→        animate={{ y: 0, opacity: 1 }}
   341→        transition={{ duration: 0.5 }}
   342→        className={cn(
   343→          "fixed top-0 left-0 right-0 z-50 w-full transition-all duration-300",
   344→          isScrolled
   345→            ? "glass shadow-sm border-b border-border/50"
   346→            : "bg-transparent"
   347→        )}
   348→      >
   349→        <nav
   350→          className={cn(
   351→            "mx-auto flex h-16 max-w-7xl items-center justify-between px-4 sm:px-6 lg:px-8",
   352→            "transition-all duration-300"
   353→          )}
   354→        >
   355→          {/* Logo */}
   356→          <motion.div className="flex items-center gap-2">
   357→            <a
   358→              href="#"
   359→              className="group flex items-center gap-2.5 no-underline"
   360→              onClick={(e) => {
   361→                e.preventDefault();
   362→                if (onNavigate) {
   363→                  onNavigate('landing');
   364→                  window.scrollTo({ top: 0 });
   365→                } else {
   366→                  window.scrollTo({ top: 0, behavior: "smooth" });
   367→                }
   368→              }}
   369→            >
   370→              <div className="relative flex items-center justify-center">
   371→                <div className="absolute inset-0 rounded-lg bg-biored/20 blur-lg group-hover:bg-biored/30 transition-colors duration-300" />
   372→                <Dna className="relative size-8 text-biored transition-transform duration-300 group-hover:scale-110 group-hover:rotate-12" />
   373→              </div>
   374→              <span className="text-xl font-bold tracking-tight">
   375→                <span className="text-foreground">Bio</span>
   376→                <span className="text-biored">Align</span>
   377→              </span>
   378→            </a>
   379→          </motion.div>
   380→
   381→          {/* Desktop Navigation Links */}
   382→          <div className="hidden lg:flex items-center gap-1">
   383→            {navLinks.map((link) => (
   384→              <a
   385→                key={link.name}
   386→                href={link.href}
   387→                onClick={(e) => {
   388→                  e.preventDefault();
   389→                  handleNavClick(link.href, link.action);
   390→                }}
   391→                className={cn(
   392→                  "relative px-3 py-2 text-sm font-medium rounded-md transition-all duration-200 cursor-pointer",
   393→                  "hover:text-foreground hover:bg-accent/10",
   394→                  activeSection === link.href.substring(1) || 
   395→                  (link.href.substring(1) === 'coffee' && activeSection === 'pricing')
   396→                    ? "text-biored font-semibold"
   397→                    : "text-muted-foreground"
   398→                )}
   399→              >
   400→                <span className="relative z-10">{link.name}</span>
   401→                {/* Active indicator */}
   402→                {(activeSection === link.href.substring(1) || 
   403→                  (link.href.substring(1) === 'coffee' && activeSection === 'pricing')) && (
   404→                  <motion.div
   405→                    layoutId="activeNav"
   406→                    className="absolute inset-0 rounded-md bg-biored/10 border border-biored/20"
   407→                    transition={{
   408→                      type: "spring",
   409→                      stiffness: 380,
   410→                      damping: 30,
   411→                    }}
   412→                  />
   413→                )}
   414→              </a>

   416→          </div>

   418→          {/* Right Side Actions */}
   419→          <div className="flex items-center gap-2 sm:gap-3">
   420→            {/* Global Search Button */}
   421→            <Button

   423→              size="icon"

   425→              className="relative size-9 rounded-full hover:bg-accent/10 cursor-pointer"

   429→            </Button>
   430→
   431→            {/* Theme Toggle */}
   432→            {mounted && (
   433→              <Button
   434→                variant="ghost"
   435→                size="icon"
   436→                onClick={toggleTheme}
   437→                className="relative size-9 rounded-full overflow-hidden hover:bg-accent/10 cursor-pointer"
   438→                aria-label="Toggle theme"
   439→              >
   440→                <AnimatePresence mode="wait" initial={false}>
   441→                  {theme === "dark" ? (
   442→                    <motion.div
   443→                      key="sun"
   444→                      initial={{ rotate: -90, scale: 0, opacity: 0 }}
   445→                      animate={{ rotate: 0, scale: 1, opacity: 1 }}
   446→                      exit={{ rotate: 90, scale: 0, opacity: 0 }}
   447→                      transition={{ duration: 0.2 }}
   448→                      className="absolute"
   449→                    >
   450→                      <Sun className="size-4 text-yellow-400" />
   451→                    </motion.div>
   452→                  ) : (
   453→                    <motion.div
   454→                      key="moon"
   455→                      initial={{ rotate: 90, scale: 0, opacity: 0 }}
   456→                      animate={{ rotate: 0, scale: 1, opacity: 1 }}
   457→                      exit={{ rotate: -90, scale: 0, opacity: 0 }}
   458→                      transition={{ duration: 0.2 }}
   459→                      className="absolute"
   460→                    >
   461→                      <Moon className="size-4 text-slate-700" />
   462→                    </motion.div>
   463→                  )}
   464→                </AnimatePresence>

   468→            {/* Contact Button */}
   469→            <Button
   470→              variant="ghost"
   471→              size="icon"
   472→              onClick={() => setShowContactModal(true)}

   475→            >

   477→            </Button>
   478→
   479→            {/* Auth Section */}
   480→            {status === "loading" ? (
   481→              <div className="flex items-center gap-2">
   482→                <Loader2 className="size-4 animate-spin text-muted-foreground" />
   483→              </div>
   484→            ) : session?.user ? (
   485→              /* Logged In State */
   486→              <div className="flex items-center gap-2">
   487→                {/* User Menu Button */}
   488→                <Button
   489→                  variant="ghost"
   490→                  size="sm"
   491→                  onClick={() => onNavigate?.("dashboard")}
   492→                  className="hidden sm:flex items-center gap-2 text-muted-foreground hover:text-foreground hover:bg-accent/10 cursor-pointer"
   493→                >
   494→                  <div className="size-7 rounded-full bg-biored/10 flex items-center justify-center">
   495→                    <span className="text-xs font-medium text-biored">{getUserInitials()}</span>
   496→                  </div>
   497→                  <span className="text-sm font-medium">{session.user.name || 'User'}</span>
   498→                </Button>
   499→                
   500→                {/* Sign Out Button */}
   501→                <Button
   502→                  variant="ghost"
   503→                  size="sm"
   504→                  onClick={handleSignOut}
   505→                  className="hidden sm:inline-flex text-muted-foreground hover:text-destructive hover:bg-destructive/10 cursor-pointer gap-2"
   506→                >
   507→                  <LogOut className="size-4" />
   508→                  <span className="hidden md:inline">Sign Out</span>
   509→                </Button>
   510→
   511→                {/* Mobile Sign Out */}
   512→                <Button
   513→                  variant="ghost"
   514→                  size="icon"
   515→                  onClick={handleSignOut}
   516→                  className="sm:hidden size-9 hover:bg-accent/10 cursor-pointer"
   517→                >
   518→                  <LogOut className="size-4" />
   519→                </Button>
   520→              </div>
   521→            ) : (
   522→              /* Logged Out State */
   523→              <>
   524→                {/* Sign In Button - Desktop */}
   525→                <Button
   526→                  variant="ghost"
   527→                  size="sm"
   528→                  onClick={handleSignIn}
   529→                  className="hidden sm:inline-flex text-muted-foreground hover:text-foreground hover:bg-accent/10 cursor-pointer gap-2"
   530→                >
   531→                  <LogIn className="size-4" />
   532→                  Sign In
   533→                </Button>
   534→
   535→                {/* Get Started CTA Button */}
   536→                <Button
   537→                  size="sm"
   538→                  onClick={handleGetStarted}
   539→                  className="hidden sm:inline-flex bg-biored hover:bg-biored-dark text-white shadow-lg shadow-biored/25 hover:shadow-biored/40 transition-all duration-300 group cursor-pointer"
   540→                >

   550→                variant="ghost"
   551→                size="icon"
   552→                className="lg:hidden size-9 hover:bg-accent/10 cursor-pointer"
   553→                onClick={() => setIsMobileOpen(true)}
   554→              >
   555→                <Menu className="size-5" />
   556→              </Button>
   557→              <SheetContent side="right" className="w-[300px] sm:w-[350px] p-0 bg-background/95 backdrop-blur-xl">
   558→                <div className="flex flex-col h-full">

   561→                    <div className="flex items-center justify-between">

   563→                        <Dna className="size-6 text-biored" />
   564→                        <span className="text-xl font-bold">
   565→                          <span className="text-foreground">Bio</span>
   566→                          <span className="text-biored">Align</span>
   567→                        </span>
   568→                      </SheetTitle>
   569→                    </div>
   570→                    <SheetDescription className="sr-only">
   571→                      Navigation menu for BioAlign platform
   572→                    </SheetDescription>
   573→                  </SheetHeader>
   574→
   575→                  <Separator className="bg-border/50" />
   576→
   577→                  {/* Auth Status in Mobile Menu */}
   578→                  <div className="px-4 py-3 border-b border-border/50">
   579→                    {session?.user ? (
   580→                      <div className="flex items-center justify-between">
   581→                        <div className="flex items-center gap-3">
   582→                          <div className="size-9 rounded-full bg-biored/10 flex items-center justify-center">
   583→                            <User className="size-4 text-biored" />
   584→                          </div>
   585→                          <div>
   586→                            <p className="text-sm font-medium">{session.user.name || 'User'}</p>
   587→                            <p className="text-xs text-muted-foreground truncate max-w-[150px]">
   588→                              {session.user.email}
   589→                            </p>
   590→                          </div>
   591→                        </div>
   592→                        <Button

   595→                          onClick={handleSignOut}

   597→                        >

   601→                      </div>
   602→                    ) : (

   607→                          className="flex-1 cursor-pointer"
   608→                          onClick={() => {

   612→                        >
