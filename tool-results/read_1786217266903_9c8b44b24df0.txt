    99→function SearchDialog({ onNavigate }: SearchDialogProps) {
   100→  const [open, setOpen] = React.useState(false)
   101→  const [searchQuery, setSearchQuery] = React.useState("")
   102→  const inputRef = React.useRef<HTMLInputElement>(null)
   103→
   104→  // Filter tools based on search query
   105→  const filteredTools = React.useMemo(() => {
   106→    if (!searchQuery.trim()) return []
   107→    const query = searchQuery.toLowerCase()
   108→    return searchableTools.filter(tool => 
   109→      tool.name.toLowerCase().includes(query) || 
   110→      tool.category.toLowerCase().includes(query) ||
   111→      tool.id.toLowerCase().includes(query)
   112→    )
   113→  }, [searchQuery])
   114→
   115→  // Filter databases based on search query
   116→  const filteredDatabases = React.useMemo(() => {
   117→    if (!searchQuery.trim()) return []
   118→    const query = searchQuery.toLowerCase()
   119→    return databaseLinks.filter(db => 
   120→      db.name.toLowerCase().includes(query)
   121→    )
   122→  }, [searchQuery])
   123→
   124→  React.useEffect(() => {
   125→    const handleKeyDown = (e: KeyboardEvent) => {
   126→      if ((e.metaKey || e.ctrlKey) && e.key === "k") {
   127→        e.preventDefault()
   128→        setOpen((prev) => !prev)
   129→      }
   130→      // Close on Escape
   131→      if (e.key === "Escape" && open) {
   132→        e.preventDefault()
   133→        setOpen(false)
   134→      }
   135→    }
   136→
   137→    document.addEventListener("keydown", handleKeyDown)
   138→    return () => document.removeEventListener("keydown", handleKeyDown)
   139→  }, [open])
   140→
   141→  // Focus input when dialog opens
   142→  React.useEffect(() => {
   143→    if (open && inputRef.current) {
   144→      setTimeout(() => inputRef.current?.focus(), 100)
   145→    }
   146→    // Reset search when closing
   147→    if (!open) {
   148→      setSearchQuery("")
   149→    }
   150→  }, [open])
   151→
   152→  const handleToolClick = (toolId: string) => {
   153→    setOpen(false)
   154→    if (onNavigate) {
   155→      onNavigate('analysis', toolId)
   156→    }
   157→  }
   158→
   159→  const handleDatabaseClick = (url: string) => {
   160→    window.open(url, '_blank', 'noopener,noreferrer')

   162→  }

   165→  const showInitial = !searchQuery.trim()

   175→          >
   176→            <Search className="size-4" />

   178→            <span className="inline-flex lg:hidden">Search</span>
   179→            <kbd className="pointer-events-none absolute right-1.5 top-1/2 hidden h-5 -translate-y-1/2 select-none items-center gap-1 rounded border bg-muted px-1.5 font-mono text-[10px] font-medium opacity-100 sm:flex">
   180→              <Command className="size-3" />K
   181→            </kbd>
   182→          </Button>
   183→        </TooltipTrigger>
   184→        <TooltipContent>
   185→          <p>Search <kbd className="rounded border bg-muted px-1 text-xs">⌘K</kbd></p>
   186→        </TooltipContent>
   187→      </Tooltip>
   188→
   189→      {/* Search Overlay */}
   190→      <AnimatePresence>
   191→        {open && (
   192→          <>
   193→            <motion.div
   194→              initial={{ opacity: 0 }}
   195→              animate={{ opacity: 1 }}
   196→              exit={{ opacity: 0 }}
   197→              className="fixed inset-0 z-50 bg-black/40 backdrop-blur-sm"
   198→              onClick={() => setOpen(false)}
   199→            />
   200→            <motion.div
   201→              initial={{ opacity: 0, scale: 0.95, y: -20 }}
   202→              animate={{ opacity: 1, scale: 1, y: 0 }}
   203→              exit={{ opacity: 0, scale: 0.95, y: -20 }}
   204→              transition={{ duration: 0.2, ease: "easeOut" }}

   349→// Theme Toggle Component
   350→function ThemeToggle() {
   351→  const { theme, setTheme } = useTheme()
   352→  const [mounted, setMounted] = React.useState(false)
   353→
   354→  React.useEffect(() => {
   355→    setMounted(true)
   356→  }, [])
   357→
   358→  if (!mounted) {
   359→    return (
   360→      <Button variant="ghost" size="icon" className="size-9">
   361→        <Sun className="size-4" />
   362→      </Button>
   363→    )
   364→  }
   365→
   366→  return (
   367→    <Tooltip>
   368→      <TooltipTrigger asChild>
   369→        <Button
   370→          variant="ghost"
   371→          size="icon"
   372→          className="size-9"
   373→          onClick={() => setTheme(theme === "dark" ? "light" : "dark")}
   374→        >
   375→          <AnimatePresence mode="wait">
   376→            {theme === "dark" ? (
   377→              <motion.div
   378→                key="moon"
   379→                initial={{ rotate: -90, opacity: 0 }}
   380→                animate={{ rotate: 0, opacity: 1 }}
   381→                exit={{ rotate: 90, opacity: 0 }}
   382→                transition={{ duration: 0.2 }}
   383→              >
   384→                <Moon className="size-4" />
   385→              </motion.div>
   386→            ) : (
   387→              <motion.div
   388→                key="sun"
   389→                initial={{ rotate: 90, opacity: 0 }}
   390→                animate={{ rotate: 0, opacity: 1 }}
   391→                exit={{ rotate: -90, opacity: 0 }}
   392→                transition={{ duration: 0.2 }}
   393→              >
   394→                <Sun className="size-4" />
   395→              </motion.div>
   396→            )}
   397→          </AnimatePresence>
   398→          <span className="sr-only">Toggle theme</span>
   399→        </Button>
   400→      </TooltipTrigger>
   401→      <TooltipContent>
   402→        Switch to {theme === "dark" ? "light" : "dark"} mode
   403→      </TooltipContent>
   404→    </Tooltip>
   405→  )
   406→}

   434→// User Menu Component with proper authentication
   435→function UserMenu({ onNavigate }: { onNavigate?: (view: string) => void }) {
   436→  const router = useRouter()
   437→  const { 
   438→    isAuthenticated, 
   439→    userDisplayName, 
   440→    userEmail, 
   441→    userInitials, 
   442→    userProfile,
   443→    status 
   444→  } = useUser()
   445→  const [isSigningOut, setIsSigningOut] = React.useState(false)
   446→
   447→  // Handle sign out with proper cleanup
   448→  const handleSignOut = async () => {
   449→    setIsSigningOut(true)
   450→    try {
   451→      // Sign out and redirect to home page
   452→      await signOut({ 
   453→        redirect: false 
   454→      })
   455→      // Clear any local state and redirect
   456→      router.push('/')
   457→      router.refresh()
   458→    } catch (error) {
   459→      console.error('Sign out error:', error)
   460→      // Force redirect even on error
   461→      window.location.href = '/'
   462→    } finally {
   463→      setIsSigningOut(false)
   464→    }
   465→  }
   466→
   467→  // Show loading state while checking auth
   468→  if (status === 'loading') {
   469→    return (
   470→      <Button variant="ghost" className="relative h-9 gap-2 pl-2 pr-3" disabled>
   471→        <div className="size-7 rounded-full bg-muted animate-pulse" />
   472→        <span className="hidden md:inline-flex text-sm">Loading...</span>
   473→      </Button>
   474→    )
   475→  }
   476→
   477→  // Not authenticated - show nothing or login prompt
   478→  if (!isAuthenticated) {
   479→    return null
   480→  }
   481→
   482→  return (
   483→    <DropdownMenu>
   484→      <DropdownMenuTrigger asChild>
   485→        <Button variant="ghost" className="relative h-9 gap-2 pl-2 pr-3 cursor-pointer">
   486→          <Avatar className="size-7 ring-2 ring-biored/20">
   487→            <AvatarImage src={userProfile?.image || undefined} alt={userDisplayName} />
   488→            <AvatarFallback className="bg-biored/10 text-biored text-xs font-medium">
   489→              {userInitials}
   490→            </AvatarFallback>
   491→          </Avatar>
   492→          <span className="hidden md:inline-flex text-sm font-medium max-w-[100px] truncate">
   493→            {userDisplayName.split(' ')[0]}
   494→          </span>
   495→          <ChevronRight className="size-3.5 rotate-90 text-muted-foreground hidden md:block" />
   496→        </Button>
   497→      </DropdownMenuTrigger>
   498→      <DropdownMenuContent align="end" className="w-56">
   499→        <DropdownMenuLabel className="font-normal">
   500→          <div className="flex flex-col space-y-1">
   501→            <div className="flex items-center gap-2">
   502→              <p className="text-sm font-medium leading-none">{userDisplayName}</p>
   503→              {userProfile?.role === 'admin' && (
   504→                <Badge variant="default" className="text-[10px] px-1 py-0 bg-biored">Admin</Badge>
   505→              )}
   506→            </div>
   507→            <p className="text-xs leading-none text-muted-foreground truncate max-w-[200px]">
   508→              {userEmail}
   509→            </p>
   510→            {userProfile?.institution && (
   511→              <p className="text-xs leading-none text-muted-foreground mt-1">
   512→                🏛️ {userProfile.institution}
   513→              </p>
   514→            )}
   515→          </div>
   516→        </DropdownMenuLabel>

   604→// Main Dashboard Layout Component
   605→export default function DashboardLayout({
   606→  children,
   607→  title,
   608→  subtitle,
   609→  actions,
   610→  breadcrumbs = [],
   611→  onSidebarNavigate,
   612→}: DashboardLayoutProps) {
   613→  const sidebarState = useSidebarState()
   614→
   615→  // Default breadcrumbs if not provided
   616→  const defaultBreadcrumbs = title
   617→    ? [{ label: "Dashboard", href: "/dashboard" }, { label: title }]
   618→    : []
   619→
   620→  const displayBreadcrumbs = breadcrumbs.length > 0 ? breadcrumbs : defaultBreadcrumbs
   621→
   622→  return (
   623→    <TooltipProvider delayDuration={0}>
   624→      <div className="min-h-screen bg-background">
   625→        {/* Desktop Sidebar */}
   626→        <div className="hidden md:block">
   627→          <DashboardSidebar
   628→            collapsed={sidebarState.collapsed}
   629→            onToggle={sidebarState.toggle}
   630→            onNavigate={onSidebarNavigate}
   631→          />
   632→        </div>
   633→
   634→        {/* Mobile Sidebar */}
   635→        <MobileSidebar
   636→          open={sidebarState.mobileOpen}
   637→          onClose={() => sidebarState.setMobileOpen(false)}
   638→          onNavigate={onSidebarNavigate}
   639→        />
   640→
   641→        {/* Main Content Area */}
   642→        <motion.main
   643→          animate={{
   644→            marginLeft: sidebarState.isMobile ? 0 : sidebarState.collapsed ? 64 : 280,
   645→          }}
   646→          transition={{ type: "spring", stiffness: 300, damping: 30 }}
   647→          className="min-h-screen flex flex-col"
   648→        >
   649→          {/* Header */}
   650→          <Header
   651→            onMobileMenuClick={sidebarState.toggle}
   652→            title={title}
   653→            subtitle={subtitle}
   654→            onNavigate={onSidebarNavigate}
   655→            actions={actions}
   656→            breadcrumbs={displayBreadcrumbs}
   657→            onSearchNavigate={onSidebarNavigate}
   658→          />
   659→
   660→          {/* Page Title & Breadcrumbs Section */}
   661→          {(title || displayBreadcrumbs.length > 0) && (
   662→            <div className="border-b bg-background/50 backdrop-blur-sm px-4 sm:px-6 lg:px-8 py-4">
   663→              {/* Breadcrumbs */}
   664→              {displayBreadcrumbs.length > 0 && (
   665→                <Breadcrumb className="mb-2">
   666→                  <BreadcrumbList>
   667→                    {displayBreadcrumbs.map((crumb, index) => (
   668→                      <React.Fragment key={crumb.label}>
   669→                        <BreadcrumbItem>
   670→                          {index === displayBreadcrumbs.length - 1 || (!crumb.href && !('onClick' in crumb)) ? (
   671→                            <BreadcrumbPage>{crumb.label}</BreadcrumbPage>
   672→                          ) : 'onClick' in crumb && crumb.onClick ? (
   673→                            <button onClick={(e) => { e.preventDefault(); (crumb.onClick as () => void)(); }} className="flex items-center gap-1.5 hover:text-biored transition-colors">
   674→                              {crumb.label}
   675→                            </button>
   676→                          ) : (
   677→                            <BreadcrumbLink href={crumb.href || '#'}>{crumb.label}</BreadcrumbLink>
   678→                          )}
   679→                        </BreadcrumbItem>
   680→                        {index < displayBreadcrumbs.length - 1 && (
   681→                          <BreadcrumbSeparator />
   682→                        )}
   683→                      </React.Fragment>
   684→                    ))}
   685→                  </BreadcrumbList>
   686→                </Breadcrumb>
   687→              )}
   688→
   689→              {/* Title & Actions Row */}
   690→              <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   691→                <div>
   692→                  {title && (
   693→                    <motion.h1
   694→                      initial={{ opacity: 0, y: -10 }}
   695→                      animate={{ opacity: 1, y: 0 }}
   696→                      className="text-2xl font-bold tracking-tight"
   697→                    >
   698→                      {title}
   699→                    </motion.h1>
   700→                  )}
   701→                  {subtitle && (
   702→                    <motion.p
   703→                      initial={{ opacity: 0 }}
   704→                      animate={{ opacity: 1 }}
   705→                      transition={{ delay: 0.1 }}
   706→                      className="text-sm text-muted-foreground mt-1"
   707→                    >
   708→                      {subtitle}
   709→                    </motion.p>
   710→                  )}
   711→                </div>
   712→                {actions && (
   713→                  <motion.div
   714→                    initial={{ opacity: 0, x: 20 }}
   715→                    animate={{ opacity: 1, x: 0 }}
   716→                    transition={{ delay: 0.15 }}
   717→                    className="flex items-center gap-2"
   718→                  >
   719→                    {actions}
   720→                  </motion.div>
   721→                )}
   722→              </div>
   723→            </div>
   724→          )}
   725→
   726→          {/* Main Content */}
   727→          <ScrollArea className="flex-1">
   728→            <div className="p-4 sm:p-6 lg:p-8">
   729→              <motion.div
   730→                initial={{ opacity: 0, y: 20 }}
   731→                animate={{ opacity: 1, y: 0 }}
   732→                transition={{ duration: 0.3, ease: "easeOut" }}
   733→              >
   734→                {children}
   735→              </motion.div>
   736→            </div>
   737→          </ScrollArea>
   738→        </motion.main>
   739→      </div>
   740→    </TooltipProvider>
   741→  )
   742→}
