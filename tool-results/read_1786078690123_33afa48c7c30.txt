   520→interface ToolsCatalogProps {
   521→  onToolSelect?: (toolId: string) => void
   522→}
   523→
   524→export default function ToolsCatalog({ onToolSelect }: ToolsCatalogProps) {
   525→  // State
   526→  const [isLoading, setIsLoading] = React.useState(true)
   527→  const [searchQuery, setSearchQuery] = React.useState("")
   528→  const [viewMode, setViewMode] = React.useState<ViewMode>("grid")
   529→  const [expandedCategories, setExpandedCategories] = React.useState<
   530→    Set<string>
   531→  >(new Set())
   532→  const [favorites, setFavorites] = React.useState<Set<string>>(new Set())
   533→  const [activeTab, setActiveTab] = React.useState("all")
   534→  const [statusFilter, setStatusFilter] = React.useState<ToolStatus | "all">(
   535→    "all"
   536→  )
   537→
   538→  // Simulate loading
   539→  React.useEffect(() => {
   540→    const timer = setTimeout(() => setIsLoading(false), 800)
   541→    return () => clearTimeout(timer)
   542→  }, [])
   543→
   544→  // Load favorites from localStorage
   545→  React.useEffect(() => {
   546→    try {
   547→      const saved = localStorage.getItem("bioalign-favorites")
   548→      if (saved) {
   549→        setFavorites(new Set(JSON.parse(saved)))
   550→      }
   551→    } catch (error) {
   552→      console.error("Failed to load favorites:", error)
   553→    }
   554→  }, [])
   555→
   556→  // Save favorites to localStorage
   557→  const saveFavorites = React.useCallback((newFavs: Set<string>) => {
   558→    setFavorites(newFavs)
   559→    try {
   560→      localStorage.setItem(
   561→        "bioalign-favorites",
   562→        JSON.stringify(Array.from(newFavs))
   563→      )
   564→    } catch (error) {
   565→      console.error("Failed to save favorites:", error)
   566→    }
   567→  }, [])
   568→
   569→  // Toggle favorite
   570→  const toggleFavorite = React.useCallback(
   571→    (toolId: string) => {
   572→      const newFavs = new Set(favorites)
   573→      if (newFavs.has(toolId)) {
   574→        newFavs.delete(toolId)
   575→      } else {
   576→        newFavs.add(toolId)
   577→      }
   578→      saveFavorites(newFavs)
   579→    },
   580→    [favorites, saveFavorites]
   581→  )
   582→
   583→  // Toggle category expansion
   584→  const toggleCategory = React.useCallback((categoryId: string) => {
   585→    setExpandedCategories((prev) => {
   586→      const next = new Set(prev)
   587→      if (next.has(categoryId)) {
   588→        next.delete(categoryId)
   589→      } else {
   590→        next.add(categoryId)
   591→      }
   592→      return next
   593→    })
   594→  }, [])
   595→
   596→  // Get filtered tools
   597→  const getFilteredTools = React.useCallback((): BioTool[] => {
   598→    let tools: BioTool[]
   599→
   600→    switch (activeTab) {
   601→      case "recent":
   602→        tools = recentlyUsedTools
   603→        break
   604→      case "favorites":
   605→        tools = getAllTools().filter((t) => favorites.has(t.id))
   606→        break
   607→      case "popular":
   608→        tools = popularTools
   609→        break
   610→      default:
   611→        tools = getAllTools()
   612→    }
   613→
   614→    // Apply search filter
   615→    if (searchQuery.trim()) {
   616→      tools = searchTools(searchQuery.trim())
   617→    }
   618→
   619→    // Apply status filter
   620→    if (statusFilter !== "all") {
   621→      tools = tools.filter((t) => t.status === statusFilter)
   622→    }
   623→
   624→    // Apply tab-specific filtering
   625→    if (activeTab === "favorites") {
   626→      tools = tools.filter((t) => favorites.has(t.id))
   627→    }
   628→
   629→    return tools
   630→  }, [searchQuery, activeTab, statusFilter, favorites])
   631→
   632→  // Get filtered categories
   633→  const getFilteredCategories = React.useCallback((): ToolCategory[] => {
   634→    if (!searchQuery.trim() && statusFilter === "all") {
   635→      return toolCategories
   636→    }
   637→
   638→    return toolCategories
   639→      .map((cat) => ({
   640→        ...cat,
   641→        tools: cat.tools.filter((tool) => {
   642→          const matchesSearch =
   643→            !searchQuery.trim() ||
   644→            tool.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
   645→            tool.description
   646→              .toLowerCase()
   647→              .includes(searchQuery.toLowerCase())
   648→          const matchesStatus =
   649→            statusFilter === "all" || tool.status === statusFilter
