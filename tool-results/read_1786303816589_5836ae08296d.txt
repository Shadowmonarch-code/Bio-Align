   130→
   131→export default function SequenceAnalysisTool({ selectedTool }: SequenceAnalysisToolProps) {
   132→  // State
   133→  // Map tool IDs from sidebar to internal tool IDs
   134→  const getInitialTool = (): ToolId => {
   135→    if (!selectedTool) return 'pairwise';
   136→    const toolMap: Record<string, ToolId> = {
   137→      'sequence': 'pairwise',
   138→      'blast': 'blast',
   139→      'orf': 'orf',
   140→      'reverse-complement': 'reverse-complement',
   141→      'translate': 'translation',
   142→      'gc-content': 'gc-content',
   143→      'motif-search': 'motif-search',
   144→      'pattern-search': 'pattern-search',
   145→      'restriction-map': 'restriction-map',
   146→    };
   147→    return toolMap[selectedTool] || 'pairwise';
   148→  };
   149→  
   150→  const [activeTool, setActiveTool] = useState<ToolId>(getInitialTool);
   151→  const [sequence1, setSequence1] = useState('');
   152→  const [sequence2, setSequence2] = useState('');
   153→  const [isAnalyzing, setIsAnalyzing] = useState(false);
   154→  const [progress, setProgress] = useState(0);
   155→  const [result, setResult] = useState<AnalysisResult | null>(null);
   156→  const [showAdvanced, setShowAdvanced] = useState(false);
   157→  const [copied, setCopied] = useState(false);
   158→  
   159→  // Tool-specific options state
   160→  const [alignmentOptions, setAlignmentOptions] = useState({
   161→    algorithm: 'global' as 'global' | 'local',
   162→    matchScore: 1,
   163→    mismatchPenalty: -1,
   164→    gapPenalty: -1,
   165→    matrix: 'BLOSUM62',
   166→  });
   167→
   168→  const [blastOptions, setBlastOptions] = useState({
   169→    program: 'blastn',
   170→    database: 'nr',
   171→    eValue: '0.001',
   172→    maxTargets: 10,
   173→    wordSize: 11,
   174→    lowComplexityFilter: true,
   175→  });
   176→
   177→  const [orfOptions, setOrfOptions] = useState({
   178→    geneticCode: 1,
   179→    minLength: 30,
   180→    showAllFrames: true,
   181→    startCodons: ['ATG'],
   182→  });
   183→
   184→  const [translationOptions, setTranslationOptions] = useState({
   185→    frame: 1,
   186→    geneticCode: 1,
   187→  });
   188→
   189→  const [motifPattern, setMotifPattern] = useState('');
   190→  const [patternSearchOptions, setPatternSearchOptions] = useState({
   191→    caseSensitive: false,
   192→    allowMismatch: false,
   193→    maxMismatches: 0,
   194→  });
   195→
   196→  // Get sequence stats for display
   197→  const seq1Stats = useMemo(() => 
   198→    sequence1 ? getSequenceStats(sequence1) : null, [sequence1]
   199→  );
   200→
   201→  const seq2Stats = useMemo(() => 
   202→    sequence2 ? getSequenceStats(sequence2) : null, [sequence2]
   203→  );
   204→
   205→  // Load example sequences
   206→  const loadExampleDNA = useCallback(() => {
   207→    setSequence1(EXAMPLE_SEQUENCES.dna);
   208→  }, []);
   209→
   210→  const loadExampleProtein = useCallback(() => {
   211→    setSequence1(EXAMPLE_SEQUENCES.protein);
   212→  }, []);
   213→
   214→  // Handle file upload
   215→  const handleFileUpload = useCallback((event: React.ChangeEvent<HTMLInputElement>) => {
   216→    const file = event.target.files?.[0];
   217→    if (file) {
   218→      const reader = new FileReader();
   219→      reader.onload = (e) => {
   220→        const content = e.target?.result as string;
   221→        if (content.startsWith('>')) {
   222→          const parsed = parseFASTA(content);
   223→          if (parsed.length > 0) {
   224→            setSequence1(parsed[0].sequence);
   225→            if (parsed.length > 1) {
   226→              setSequence2(parsed[1].sequence);
   227→            }
   228→          }
   229→        } else {
   230→          setSequence1(content);
   231→        }
   232→      };
   233→      reader.readAsText(file);
   234→    }
   235→  }, []);
   236→
   237→  // Run analysis based on active tool
   238→  const runAnalysis = useCallback(async () => {
   239→    setIsAnalyzing(true);
   240→    setProgress(0);
   241→    setResult(null);
   242→
   243→    // Simulate progress
   244→    const progressInterval = setInterval(() => {
   245→      setProgress(prev => Math.min(prev + Math.random() * 15, 90));
   246→    }, 200);
   247→
   248→    // Simulate processing delay
   249→    await new Promise(resolve => setTimeout(resolve, 1000 + Math.random() * 1500));
   250→
   251→    let analysisResult: AnalysisResult;
   252→
   253→    try {
   254→      switch (activeTool) {
   255→        case 'pairwise': {
   256→          const alignFn = alignmentOptions.algorithm === 'global' ? globalAlignment : localAlignment;
   257→          const alignmentResult = alignFn(sequence1, sequence2, {
   258→            matchScore: alignmentOptions.matchScore,
   259→            mismatchPenalty: alignmentOptions.mismatchPenalty,
   260→            gapPenalty: alignmentOptions.gapPenalty,
   261→          });
   262→          analysisResult = { type: 'pairwise', data: alignmentResult, timestamp: new Date() };
   263→          break;
   264→        }
   265→
   266→        case 'blast': {
   267→          const blastResult = mockBLASTSearch(sequence1, {
   268→            database: blastOptions.database,
   269→            maxTargets: blastOptions.maxTargets,
   270→            program: blastOptions.program,
   271→          });
   272→          analysisResult = { type: 'blast', data: blastResult, timestamp: new Date() };
   273→          break;
   274→        }
   275→
   276→        case 'orf': {
   277→          const orfs = findORFs(sequence1, {
   278→            minLength: orfOptions.minLength,
   279→            geneticCode: orfOptions.geneticCode,
   280→            allFrames: orfOptions.showAllFrames,
   281→            startCodons: orfOptions.startCodons,
   282→          });
   283→          analysisResult = { type: 'orf', data: orfs, timestamp: new Date() };
   284→          break;
   285→        }
   286→
   287→        case 'reverse-complement': {
   288→          const rcResult = rc(sequence1);
   289→          analysisResult = { type: 'reverse-complement', data: rcResult, timestamp: new Date() };
   290→          break;
   291→        }
   292→
   293→        case 'translation': {
   294→          const translationResult = translate(
   295→            sequence1,
   296→            translationOptions.frame,
   297→            translationOptions.geneticCode
   298→          );
   299→          analysisResult = { type: 'translation', data: translationResult, timestamp: new Date() };
   300→          break;
   301→        }
   302→
   303→        case 'gc-content': {
   304→          const gcResult = getSequenceStats(sequence1);
   305→          analysisResult = { type: 'gc-content', data: gcResult, timestamp: new Date() };
   306→          break;
   307→        }
   308→
   309→        case 'motif-search':
   310→        case 'pattern-search': {
   311→          const motifResults = searchMotif(sequence1, motifPattern || 'ATG', patternSearchOptions);
   312→          analysisResult = { type: activeTool, data: motifResults, timestamp: new Date() };
   313→          break;
   314→        }
   315→
   316→        case 'restriction-map': {
   317→          const restrictionSites = findRestrictionSites(sequence1);
   318→          analysisResult = { type: 'restriction-map', data: restrictionSites, timestamp: new Date() };
   319→          break;
   320→        }
   321→
   322→        default:
   323→          throw new Error('Unknown tool');
   324→      }
   325→
   326→      clearInterval(progressInterval);
   327→      setProgress(100);
   328→      
   329→      setTimeout(() => {
   330→        setResult(analysisResult);
   331→        setIsAnalyzing(false);
   332→        setProgress(0);
   333→      }, 300);
   334→
   335→    } catch (error) {
   336→      clearInterval(progressInterval);
   337→      setIsAnalyzing(false);
   338→      console.error('Analysis error:', error);
   339→    }
   340→  }, [activeTool, sequence1, sequence2, alignmentOptions, blastOptions, orfOptions, translationOptions, motifPattern, patternSearchOptions]);
   341→
   342→  // Copy to clipboard
   343→  const copyToClipboard = useCallback((text: string) => {
   344→    navigator.clipboard.writeText(text);
   345→    setCopied(true);
   346→    setTimeout(() => setCopied(false), 2000);
   347→  }, []);
   348→
   349→  // Download result
   350→  const downloadResult = useCallback((format: 'json' | 'csv' | 'txt') => {
   351→    if (!result) return;
   352→
   353→    let content = '';
   354→    let filename = '';
   355→    let mimeType = '';
   356→
   357→    switch (format) {
   358→      case 'json':
   359→        content = JSON.stringify(result.data, null, 2);
   360→        filename = `bioalign-${result.type}-result.json`;
   361→        mimeType = 'application/json';
   362→        break;
   363→      case 'csv':
   364→        content = convertToCSV(result.data);
   365→        filename = `bioalign-${result.type}-result.csv`;
   366→        mimeType = 'text/csv';
   367→        break;
   368→      case 'txt':
   369→        content = formatPlainText(result);
   370→        filename = `bioalign-${result.type}-result.txt`;
   371→        mimeType = 'text/plain';
   372→        break;
   373→    }
   374→
   375→    const blob = new Blob([content], { type: mimeType });
   376→    const url = URL.createObjectURL(blob);
   377→    const a = document.createElement('a');
   378→    a.href = url;
   379→    a.download = filename;
   380→    a.click();
   381→    URL.revokeObjectURL(url);
   382→  }, [result]);
   383→
   384→  // Highlight nucleotides in sequence
   385→  const highlightSequence = (seq: string): React.ReactNode[] => {
   386→    return seq.split('').map((char, i) => (
   387→      <span key={i} className={NUCLEOTIDE_COLORS[char.toUpperCase()] || ''}>
   388→        {char}
   389→      </span>
   390→    ));
   391→  };
   392→
   393→  // Render tool-specific options
   394→  const renderToolOptions = () => {
   395→    switch (activeTool) {
   396→      case 'pairwise':
   397→        return (
   398→          <div className="space-y-4">
   399→            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   400→              <div>
   401→                <Label htmlFor="algorithm">Algorithm</Label>
   402→                <Select value={alignmentOptions.algorithm} onValueChange={(v) => 
   403→                  setAlignmentOptions(prev => ({ ...prev, algorithm: v as 'global' | 'local' }))
   404→                }>
   405→                  <SelectTrigger><SelectValue /></SelectTrigger>
   406→                  <SelectContent>
   407→                    <SelectItem value="global">Global (Needleman-Wunsch)</SelectItem>
   408→                    <SelectItem value="local">Local (Smith-Waterman)</SelectItem>
   409→                  </SelectContent>
   410→                </Select>
   411→              </div>
   412→              <div>
   413→                <Label htmlFor="matrix">Substitution Matrix</Label>
   414→                <Select value={alignmentOptions.matrix} onValueChange={(v) =>
   415→                  setAlignmentOptions(prev => ({ ...prev, matrix: v }))
   416→                }>
   417→                  <SelectTrigger><SelectValue /></SelectTrigger>
   418→                  <SelectContent>
   419→                    {Object.keys(SUBSTITUTION_MATRICES).map(m => (
   420→                      <SelectItem key={m} value={m}>{m}</SelectItem>
   421→                    ))}
   422→                  </SelectContent>
   423→                </Select>
   424→              </div>
   425→            </div>
   426→            
   427→            <Collapsible open={showAdvanced} onOpenChange={setShowAdvanced}>
   428→              <CollapsibleTrigger asChild>
   429→                <Button variant="ghost" size="sm" className="w-full justify-between">
   430→                  <span className="flex items-center gap-2">
   431→                    <Settings2 className="w-4 h-4" />
   432→                    Advanced Scoring Options
   433→                  </span>
   434→                  {showAdvanced ? <ChevronUp className="w-4 h-4" /> : <ChevronDown className="w-4 h-4" />}
   435→                </Button>
   436→              </CollapsibleTrigger>
   437→              <CollapsibleContent className="space-y-4 pt-4">
   438→                <div className="grid grid-cols-3 gap-4">
   439→                  <div>
   440→                    <Label>Match Score</Label>
   441→                    <Input
   442→                      type="number"
   443→                      value={alignmentOptions.matchScore}
   444→                      onChange={(e) => setAlignmentOptions(prev => ({ ...prev, matchScore: Number(e.target.value) }))}
   445→                    />
   446→                  </div>
   447→                  <div>
   448→                    <Label>Mismatch Penalty</Label>
   449→                    <Input
   450→                      type="number"
   451→                      value={alignmentOptions.mismatchPenalty}
   452→                      onChange={(e) => setAlignmentOptions(prev => ({ ...prev, mismatchPenalty: Number(e.target.value) }))}
   453→                    />
   454→                  </div>
   455→                  <div>
   456→                    <Label>Gap Penalty</Label>
   457→                    <Input
   458→                      type="number"
   459→                      value={alignmentOptions.gapPenalty}
   460→                      onChange={(e) => setAlignmentOptions(prev => ({ ...prev, gapPenalty: Number(e.target.value) }))}
   461→                    />
   462→                  </div>
   463→                </div>
   464→              </CollapsibleContent>
   465→            </Collapsible>
   466→
   467→            {/* Second sequence input */}
   468→            <div>
   469→              <Label htmlFor="seq2">Second Sequence</Label>
   470→              <Textarea
   471→                id="seq2"
   472→                placeholder="Enter second sequence..."
   473→                value={sequence2}
   474→                onChange={(e) => setSequence2(e.target.value)}
   475→                rows={4}
   476→                className="font-mono text-sm"
   477→              />
   478→              {seq2Stats && (
   479→                <div className="flex flex-wrap gap-2 mt-2">
   480→                  <Badge variant="secondary">{seq2Stats.length} bp</Badge>
   481→                  <Badge variant="secondary">{seq2Stats.type}</Badge>
   482→                  <Badge variant="secondary">GC: {seq2Stats.gcContent.toFixed(1)}%</Badge>
   483→                </div>
   484→              )}
   485→            </div>
   486→          </div>
   487→        );
   488→
   489→      case 'blast':
   490→        return (
   491→          <div className="space-y-4">
   492→            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   493→              <div>
   494→                <Label>Program</Label>
   495→                <Select value={blastOptions.program} onValueChange={(v) =>
   496→                  setBlastOptions(prev => ({ ...prev, program: v }))
   497→                }>
   498→                  <SelectTrigger><SelectValue /></SelectTrigger>
   499→                  <SelectContent>
   500→                    <SelectItem value="blastn">BLASTN (Nucleotide)</SelectItem>
   501→                    <SelectItem value="blastp">BLASTP (Protein)</SelectItem>
   502→                    <SelectItem value="blastx">BLASTX (Translated)</SelectItem>
   503→                    <SelectItem value="tblastn">TBLASTN</SelectItem>
   504→                    <SelectItem value="tblastx">TBLASTX</SelectItem>
   505→                  </SelectContent>
   506→                </Select>
   507→              </div>
   508→              <div>
   509→                <Label>Database</Label>
   510→                <Select value={blastOptions.database} onValueChange={(v) =>
   511→                  setBlastOptions(prev => ({ ...prev, database: v }))
   512→                }>
   513→                  <SelectTrigger><SelectValue /></SelectTrigger>
   514→                  <SelectContent>
   515→                    <SelectItem value="nr">NR (Non-redundant)</SelectItem>
   516→                    <SelectItem value="nt">NT (Nucleotide)</SelectItem>
   517→                    <SelectItem value="pdb">PDB</SelectItem>
   518→                    <SelectItem value="swissprot">SwissProt</SelectItem>
   519→                    <SelectItem value="refseq">RefSeq</SelectItem>
   520→                  </SelectContent>
   521→                </Select>
   522→              </div>
   523→            </div>
   524→            
   525→            <Collapsible open={showAdvanced} onOpenChange={setShowAdvanced}>
   526→              <CollapsibleTrigger asChild>
   527→                <Button variant="ghost" size="sm" className="w-full justify-between">
   528→                  <span className="flex items-center gap-2">
   529→                    <Settings2 className="w-4 h-4" />
   530→                    Advanced Parameters
   531→                  </span>
   532→                  {showAdvanced ? <ChevronUp className="w-4 h-4" /> : <ChevronDown className="w-4 h-4" />}
   533→                </Button>
   534→              </CollapsibleTrigger>
   535→              <CollapsibleContent className="space-y-4 pt-4">
   536→                <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
   537→                  <div>
   538→                    <Label>E-value Threshold</Label>
   539→                    <Input
   540→                      value={blastOptions.eValue}
   541→                      onChange={(e) => setBlastOptions(prev => ({ ...prev, eValue: e.target.value }))}
   542→                    />
   543→                  </div>
   544→                  <div>
   545→                    <Label>Max Targets</Label>
   546→                    <Input
   547→                      type="number"
   548→                      min={1}
   549→                      max={100}
   550→                      value={blastOptions.maxTargets}
   551→                      onChange={(e) => setBlastOptions(prev => ({ ...prev, maxTargets: Number(e.target.value) }))}
   552→                    />
   553→                  </div>
   554→                  <div>
   555→                    <Label>Word Size</Label>
   556→                    <Input
   557→                      type="number"
   558→                      value={blastOptions.wordSize}
   559→                      onChange={(e) => setBlastOptions(prev => ({ ...prev, wordSize: Number(e.target.value) }))}
   560→                    />
   561→                  </div>
   562→                </div>
   563→                <div className="flex items-center space-x-2">
   564→                  <Switch
   565→                    checked={blastOptions.lowComplexityFilter}
   566→                    onCheckedChange={(checked) => setBlastOptions(prev => ({ ...prev, lowComplexityFilter: checked }))}
   567→                  />
   568→                  <Label>Low complexity filter</Label>
   569→                </div>
   570→              </CollapsibleContent>
   571→            </Collapsible>
   572→          </div>
   573→        );
   574→
   575→      case 'orf':
   576→        return (
   577→          <div className="space-y-4">
   578→            <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
   579→              <div>
   580→                <Label>Genetic Code</Label>
   581→                <Select value={String(orfOptions.geneticCode)} onValueChange={(v) =>
   582→                  setOrfOptions(prev => ({ ...prev, geneticCode: Number(v) }))
   583→                }>
   584→                  <SelectTrigger><SelectValue /></SelectTrigger>
   585→                  <SelectContent>
   586→                    <SelectItem value="1">Standard (1)</SelectItem>
   587→                    <SelectItem value="2">Vertebrate Mitochondrial (2)</SelectItem>
   588→                    <SelectItem value="4">Invertebrate Mitochondrial (4)</SelectItem>

  1306→function convertToCSV(data: unknown): string {
  1307→  if (typeof data === 'string') return data;
  1308→  if (Array.isArray(data)) {
  1309→    if (data.length === 0) return '';
  1310→    const headers = Object.keys(data[0]);
  1311→    const rows = data.map(item => 
  1312→      headers.map(h => JSON.stringify((item as Record<string, unknown>)[h] ?? '')).join(',')
  1313→    );
  1314→    return [headers.join(','), ...rows].join('\n');
  1315→  }
  1316→  if (typeof data === 'object' && data !== null) {
  1317→    const entries = Object.entries(data as Record<string, unknown>);
  1318→    return entries.map(([k, v]) => `${k},${JSON.stringify(v)}`).join('\n');
  1319→  }
  1320→  return String(data);
  1321→}

  1323→function formatPlainText(result: AnalysisResult): string {
  1324→  const lines = [`BioAlign ${result.type.toUpperCase()} Results`, `Generated: ${result.timestamp.toISOString()}`, ''];
  1325→  
  1326→  if (typeof result.data === 'string') {
  1327→    lines.push(result.data);
  1328→  } else if (Array.isArray(result.data)) {
  1329→    lines.push(`Total entries: ${result.data.length}`);
  1330→    result.data.forEach((item, idx) => {
  1331→      lines.push(`\n--- Entry ${idx + 1} ---`);
  1332→      Object.entries(item as Record<string, unknown>).forEach(([k, v]) => {
  1333→        lines.push(`${k}: ${JSON.stringify(v)}`);
  1334→      });
  1335→    });
  1336→  } else if (typeof result.data === 'object' && result.data !== null) {
  1337→    Object.entries(result.data as Record<string, unknown>).forEach(([k, v]) => {
  1338→      lines.push(`${k}: ${JSON.stringify(v)}`);
  1339→    });
  1340→  }
  1341→  
  1342→  return lines.join('\n');
  1343→}
