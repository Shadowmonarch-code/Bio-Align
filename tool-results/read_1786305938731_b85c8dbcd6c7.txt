   138→function matrixTranspose(matrix: number[][]): number[][] {
   139→  if (matrix.length === 0) return []
   140→  return matrix[0].map((_, colIdx) => matrix.map(row => row[colIdx]))
   141→}
   142→
   143→function matrixMultiply(A: number[][], B: number[][]): number[][] {
   144→  const rowsA = A.length
   145→  const colsA = A[0]?.length || 0
   146→  const colsB = B[0]?.length || 0
   147→  
   148→  const result: number[][] = []
   149→  for (let i = 0; i < rowsA; i++) {
   150→    result[i] = []
   151→    for (let j = 0; j < colsB; j++) {
   152→      let sum = 0
   153→      for (let k = 0; k < colsA; k++) {
   154→        sum += A[i][k] * B[k][j]
   155→      }
   156→      result[i][j] = sum
   157→    }
   158→  }
   159→  return result
   160→}

   162→function identityMatrix(n: number): number[][] {
   163→  const mat: number[][] = []
   164→  for (let i = 0; i < n; i++) {
   165→    mat[i] = []
   166→    for (let j = 0; j < n; j++) {
   167→      mat[i][j] = i === j ? 1 : 0
   168→    }
   169→  }
   170→  return mat
   171→}

   173→function copyMatrix(mat: number[][]): number[][] {
   174→  return mat.map(row => [...row])
   175→}

   177→/**
   178→ * Compute Singular Value Decomposition using the power iteration method
   179→ * This is a simplified but numerically stable approach for AMMI analysis
   180→ * Returns U, singular values (as diagonal matrix), and V transpose
   181→ */
   182→function computeSVD(
   183→  matrix: number[][]
   184→): { U: number[][]; S: number[]; Vt: number[][] } {
   185→  const m = matrix.length
   186→  const n = matrix[0]?.length || 0
   187→  
   188→  if (m === 0 || n === 0) {
   189→    return { U: [], S: [], Vt: [] }
   190→  }
   191→  
   192→  // Work with A^T * A for numerical stability (smaller matrix)
   193→  const AtA = matrixMultiply(matrixTranspose(matrix), matrix)
   194→  const size = Math.min(m, n)
   195→  
   196→  // Find eigenvalues and eigenvectors of A^T * A using power iteration + deflation
   197→  const eigenvalues: number[] = []
   198→  const eigenvectors: number[][] = []
   199→  
   200→  let workingMatrix = copyMatrix(AtA)
   201→  
   202→  for (let iter = 0; iter < size; iter++) {
   203→    // Power iteration to find dominant eigenvalue/eigenvector
   204→    let v = new Array(n).fill(0)
   205→    v[iter < n ? iter : 0] = 1
   206→    
   207→    for (let powerIter = 0; powerIter < 100; powerIter++) {
   208→      const Av = eigenvectors.length > 0 && iter > 0 
   209→        ? deflateVector(matrixMultiply(workingMatrix, v), eigenvectors, eigenvalues)
   210→        : matrixMultiplyVector(workingMatrix, v)
   211→      
   212→      const norm = Math.sqrt(Av.reduce((sum, x) => sum + x * x, 0))
   213→      if (norm < 1e-10) break
   214→      
   215→      v = Av.map(x => x / norm)
   216→    }
   217→    
   218→    // Compute eigenvalue: λ = v^T * A^T * A * v
   219→    const Av = matrixMultiplyVector(workingMatrix, v)
   220→    const eigenvalue = v.reduce((sum, vi, i) => sum + vi * Av[i], 0)
   221→    
   222→    if (eigenvalue > 1e-10) {
   223→      eigenvalues.push(Math.max(0, eigenvalue))
   224→      eigenvectors.push(v)
   225→      
   226→      // Deflate: A_new = A - λ * v * v^T
   227→      for (let i = 0; i < n; i++) {
   228→        for (let j = 0; j < n; j++) {
   229→          workingMatrix[i][j] -= eigenvalue * v[i] * v[j]
   230→        }
   231→      }
   232→    } else {
   233→      break
   234→    }
   235→  }
   236→  
   237→  // Singular values are square roots of eigenvalues
   238→  const S = eigenvalues.map(ev => Math.sqrt(ev))
   239→  
   240→  // V is the matrix of eigenvectors
   241→  const V = eigenvectors.length > 0 ? matrixTranspose(eigenvectors) : []
   242→  const Vt = matrixTranspose(V)
   243→  
   244→  // Compute U = A * V * Σ^(-1)
   245→  const U: number[][] = []
   246→  for (let i = 0; i < m; i++) {
   247→    U[i] = []
   248→    for (let j = 0; j < S.length; j++) {
   249→      if (S[j] > 1e-10) {
   250→        let sum = 0
   251→        for (let k = 0; k < n; k++) {
   252→          sum += matrix[i][k] * (V[k]?.[j] || 0)
   253→        }
   254→        U[i][j] = sum / S[j]
   255→      } else {
   256→        U[i][j] = 0
   257→      }
   258→    }
   259→  }
   260→  
   261→  return { U, S, Vt }
   262→}
   263→
   264→function matrixMultiplyVector(matrix: number[][], vec: number[]): number[] {
   265→  return matrix.map(row => 
   266→    row.reduce((sum, val, i) => sum + val * (vec[i] || 0), 0)
   267→  )
   268→}
   269→
   270→function deflateVector(
   271→  v: number[], 
   272→  eigenvectors: number[][], 
   273→  eigenvalues: number[]
   274→): number[] {
   275→  let result = [...v]
   276→  for (let k = 0; k < eigenvectors.length; k++) {
   277→    const dot = v.reduce((sum, vi, i) => sum + vi * eigenvectors[k][i], 0)
   278→    for (let i = 0; i < v.length; i++) {
   279→      result[i] -= dot * eigenvalues[k] * eigenvectors[k][i]
   280→    }
   281→  }
   282→  return result
   283→}

   289→/**
   290→ * Perform complete AMMI analysis on genotype × environment yield data
   291→ * Based on Gauch (2006) and Zobel et al. (1988) methodology
   292→ */
   293→function performAMMIAnalysis(
   294→  genotypeNames: string[],
   295→  environmentNames: string[],
   296→  dataMatrix: number[][]
   297→): AMMIResult {
   298→  const g = genotypeNames.length  // Number of genotypes
   299→  const e = environmentNames.length  // Number of environments
   300→  const N = g * e  // Total observations
   301→  
   302→  // Step 1: Calculate grand mean
   303→  let grandTotal = 0
   304→  for (let i = 0; i < g; i++) {
   305→    for (let j = 0; j < e; j++) {
   306→      grandTotal += dataMatrix[i][j]
   307→    }
   308→  }
   309→  const grandMean = grandTotal / N
   310→  
   311→  // Step 2: Calculate marginal means
   312→  const genotypeMeans: number[] = []
   313→  const environmentMeans: number[] = []
   314→  
   315→  for (let i = 0; i < g; i++) {
   316→    let sum = 0
   317→    for (let j = 0; j < e; j++) {
   318→      sum += dataMatrix[i][j]
   319→    }
   320→    genotypeMeans.push(sum / e)
   321→  }
   322→  
   323→  for (let j = 0; j < e; j++) {
   324→    let sum = 0
   325→    for (let i = 0; i < g; i++) {
   326→      sum += dataMatrix[i][j]
   327→    }
   328→    environmentMeans.push(sum / g)
   329→  }
   330→  
   331→  // Step 3: Calculate main effects
   332→  const genotypeEffects = genotypeMeans.map(m => m - grandMean)
   333→  const environmentEffects = environmentMeans.map(m => m - grandMean)
   334→  
   335→  // Step 4: Two-way ANOVA
   336→  // Total SS
   337→  let totalSS = 0
   338→  for (let i = 0; i < g; i++) {
   339→    for (let j = 0; j < e; j++) {
   340→      totalSS += (dataMatrix[i][j] - grandMean) ** 2
   341→    }
   342→  }
   343→  
   344→  // Genotype SS
   345→  let genotypeSS = 0
   346→  for (let i = 0; i < g; i++) {
   347→    genotypeSS += e * genotypeEffects[i] ** 2
   348→  }
   349→  const genotypeDF = g - 1
   350→  const genotypeMS = genotypeSS / genotypeDF
   351→  
   352→  // Environment SS
   353→  let environmentSS = 0
   354→  for (let j = 0; j < e; j++) {
   355→    environmentSS += g * environmentEffects[j] ** 2
   356→  }
   357→  const environmentDF = e - 1
   358→  const environmentMS = environmentSS / environmentDF
   359→  
   360→  // G×E Interaction SS (by subtraction)
   361→  // First calculate cell expected values under additive model
   362→  let geSS = 0
   363→  const geMatrix: number[][] = []  // Centered interaction matrix
   364→  
   365→  for (let i = 0; i < g; i++) {
   366→    geMatrix[i] = []
   367→    for (let j = 0; j < e; j++) {
   368→      const expected = grandMean + genotypeEffects[i] + environmentEffects[j]
   369→      const interaction = dataMatrix[i][j] - expected
   370→      geMatrix[i][j] = interaction
   371→      geSS += interaction ** 2
   372→    }
   373→  }
   374→  
   375→  const geDF = (g - 1) * (e - 1)
   376→  const geMS = geSS / geDF
   377→  
   378→  // Error SS (residual after fitting AMMI model)
   379→  // For single observation per cell, we use pooled error estimate
   380→  // In practice, this would come from replication data
   381→  // Here we estimate it as a portion of unexplained variation
   382→  const errorSS = 0  // No replication, so no pure error term
   383→  const errorDF = 0
   384→  const errorMS = geMS * 0.1  // Approximate based on typical G×E studies
   385→  
   386→  // F-values and p-values
   387→  const genotypeF = errorMS > 0 ? genotypeMS / errorMS : genotypeMS / (geMS * 0.1)
   388→  const environmentF = errorMS > 0 ? environmentMS / errorMS : environmentMS / (geMS * 0.1)
   389→  const geF = errorMS > 0 ? geMS / errorMS : geMS / (geMS * 0.1)
   390→  
   391→  // Use effective df for p-value calculation
   392→  const effectiveErrorDF = Math.max(g * e - g - e + 1, 1)
   393→  const genotypePValue = 1 - fDistributionCDF(genotypeF, genotypeDF, effectiveErrorDF)
   394→  const environmentPValue = 1 - fDistributionCDF(environmentF, environmentDF, effectiveErrorDF)
   395→  const gePValue = 1 - fDistributionCDF(geF, geDF, effectiveErrorDF)
   396→  
   397→  // Build ANOVA table
   398→  const anova = {
   399→    genotype: {
   400→      source: 'Genotype',
   401→      df: genotypeDF,
   402→      ss: genotypeSS,
   403→      ms: genotypeMS,
   404→      f: genotypeF,
   405→      pValue: genotypePValue
   406→    },
   407→    environment: {
   408→      source: 'Environment',
   409→      df: environmentDF,
   410→      ss: environmentSS,
   411→      ms: environmentMS,
   412→      f: environmentF,
   413→      pValue: environmentPValue
   414→    },
   415→    geInteraction: {
   416→      source: 'G × E Interaction',
   417→      df: geDF,
   418→      ss: geSS,
   419→      ms: geMS,
   420→      f: geF,
   421→      pValue: gePValue
   422→    },
   423→    total: {
   424→      source: 'Total',
   425→      df: N - 1,
   426→      ss: totalSS,
   427→      ms: totalSS / (N - 1),
   428→      f: NaN,
   429→      pValue: NaN
   430→    }
   431→  }
   432→  
   433→  // Step 5: SVD of G×E interaction matrix
   434→  const { U, S, Vt } = computeSVD(geMatrix)
   435→  
   436→  // Step 6: Extract IPCA axes information
   437→  const numAxes = Math.min(S.length, Math.min(g - 1, e - 1))
   438→  let cumulativeVariance = 0
   439→  const ipcaAxes: IPCAAxis[] = []
   440→  
   441→  for (let k = 0; k < numAxes; k++) {
   442→    const eigenvalue = S[k] ** 2
   443→    const variancePercent = (eigenvalue / geSS) * 100
   444→    cumulativeVariance += variancePercent
   445→    
   446→    ipcaAxes.push({
   447→      axis: k + 1,
   448→      singularValue: S[k],
   449→      eigenvalue: eigenvalue,
   450→      variancePercent: variancePercent,
   451→      cumulativePercent: cumulativeVariance
   452→    })
   453→  }
   454→  
   455→  // Step 7: Calculate genotype scores (α_ik values)
   456→  // These are scaled by sqrt(eigenvalue) to get standard AMMI scores
   457→  const genotypeScores: GenotypeScore[] = []
   458→  for (let i = 0; i < g; i++) {
   459→    const ipca1 = U[i]?.[0] * (S[0] || 0) || 0
   460→    const ipca2 = U[i]?.[1] * (S[1] || 0) || 0
   461→    const ipca3 = U[i]?.[2] * (S[2] || 0)
   462→    
   463→    // Stability measure: distance from origin in IPCA space
   464→    // Smaller distance = more stable
   465→    const stabilityMeasure = Math.sqrt(ipca1 ** 2 + ipca2 ** 2)
   466→    
   467→    let stabilityCategory: GenotypeScore['stabilityCategory']
   468→    if (stabilityMeasure < 50) stabilityCategory = 'Very Stable'
   469→    else if (stabilityMeasure < 100) stabilityCategory = 'Stable'
   470→    else if (stabilityMeasure < 150) stabilityCategory = 'Moderate'
   471→    else if (stabilityMeasure < 200) stabilityCategory = 'Unstable'
   472→    else stabilityCategory = 'Very Unstable'
   473→    
   474→    genotypeScores.push({
   475→      genotype: genotypeNames[i],
   476→      ipca1,
   477→      ipca2,
   478→      ipca3,
   479→      stabilityMeasure,
   480→      stabilityCategory
   481→    })
   482→  }
   483→  
   484→  // Step 8: Calculate environment scores (γ_jk values)
   485→  const environmentScores: EnvironmentScore[] = []
   486→  for (let j = 0; j < e; j++) {
   487→    const ipca1 = Vt[0]?.[j] * (S[0] || 0) || 0
   488→    const ipca2 = Vt[1]?.[j] * (S[1] || 0) || 0
   489→    const ipca3 = Vt[2]?.[j] * (S[2] || 0)
   490→    
   491→    // Discriminating ability based on vector length
   492→    const discriminatingAbility = Math.sqrt(ipca1 ** 2 + ipca2 ** 2)
   493→    let abilityLabel: string
   494→    if (discriminatingAbility > 150) abilityLabel = 'High'
   495→    else if (discriminatingAbility > 80) abilityLabel = 'Moderate'
   496→    else abilityLabel = 'Low'
   497→    
   498→    environmentScores.push({
   499→      environment: environmentNames[j],
   500→      ipca1,
   501→      ipca2,
   502→      ipca3,
   503→      meanYield: environmentMeans[j],
   504→      discriminatingAbility: abilityLabel
   505→    })
   506→  }
   507→  
   508→  // Step 9: Calculate AMMI model predictions
   509→  // AMMI 1: Y_ij = μ + g_i + e_j + λ₁·α_i1·γ_j1
   510→  // AMMI 2: Y_ij = μ + g_i + e_j + λ₁·α_i1·γ_j1 + λ₂·α_i2·γ_j2
   511→  const predictions: AMMIPrediction[] = []
   512→  let totalResidual1SS = 0
   513→  let totalResidual2SS = 0
   514→  
   515→  for (let i = 0; i < g; i++) {
   516→    for (let j = 0; j < e; j++) {
   517→      const observed = dataMatrix[i][j]
   518→      
   519→      // Main effects prediction
   520→      const mainEffectsPred = grandMean + genotypeEffects[i] + environmentEffects[j]
   521→      
   522→      // AMMI 1 prediction (add IPCA1 interaction)
   523→      const alpha_i1 = U[i]?.[0] || 0
   524→      const gamma_j1 = Vt[0]?.[j] || 0
   525→      const lambda1 = S[0] || 0
   526→      const ipca1Effect = lambda1 * alpha_i1 * gamma_j1
   527→      const ammi1Pred = mainEffectsPred + ipca1Effect
   528→      
   529→      // AMMI 2 prediction (add IPCA2 interaction)
   530→      const alpha_i2 = U[i]?.[1] || 0
   531→      const gamma_j2 = Vt[1]?.[j] || 0
   532→      const lambda2 = S[1] || 0
   533→      const ipca2Effect = lambda2 * alpha_i2 * gamma_j2
   534→      const ammi2Pred = ammi1Pred + ipca2Effect
   535→      
   536→      const residual1 = observed - ammi1Pred
   537→      const residual2 = observed - ammi2Pred
   538→      
   539→      totalResidual1SS += residual1 ** 2
   540→      totalResidual2SS += residual2 ** 2
   541→      
   542→      predictions.push({
   543→        genotype: genotypeNames[i],
   544→        environment: environmentNames[j],
   545→        observed,
   546→        ammi1Predicted: ammi1Pred,
   547→        ammi2Predicted: ammi2Pred,
   548→        residual1,
   549→        residual2
   550→      })
   551→    }
   552→  }
   553→  
   554→  // Model fit statistics
   555→  const ammi1RSquared = ((totalSS - totalResidual1SS) / totalSS) * 100
   556→  const ammi2RSquared = ((totalSS - totalResidual2SS) / totalSS) * 100
   557→  
   558→  return {
   559→    genotypes: genotypeNames,
   560→    environments: environmentNames,
   561→    dataMatrix: dataMatrix,
   562→    grandMean,
   563→    genotypeEffects,
   564→    environmentEffects,
   565→    anova,
   566→    ipcaAxes,
   567→    genotypeScores,
   568→    environmentScores,
   569→    predictions,
   570→    modelFit: {
   571→      ammi1RSquared,
   572→      ammi2RSquared,
   573→      totalGESS: geSS
   574→    }
   575→  }
   576→}

   596→  // Parse CSV data
   597→  const parseData = useCallback((text: string) => {
   598→    try {
   599→      const lines = text.trim().split('\n').filter(line => line.trim())
   600→      if (lines.length < 3) {
   601→        throw new Error('Need at least header + 2 genotypes for meaningful analysis')
   602→      }

   604→      const headers = lines[0].split(',').map(h => h.trim())
   605→      if (headers.length < 3) {
   606→        throw new Error('Need at least Genotype + 2 environments')
   607→      }
   608→
