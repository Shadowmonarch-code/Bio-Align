   136→/**
   137→ * Calculate t-critical value for two-tailed test (approximation)
   138→ */
   139→function getTCritical(df: number, alpha: number = 0.05): number {
   140→  // Approximation for common df values
   141→  if (df <= 0) return 1.96
   142→  const sqrtDf = Math.sqrt(df)
   143→  // Using approximation formula
   144→  let t = 1.96 + (1.96 ** 3 + 1.96) / (4 * df)
   145→  return Math.min(t, 4.0) // Cap at reasonable value
   146→}

   148→/**
   149→ * Parse CSV data into generation means format
   150→ */
   151→function parseGenerationData(text: string): { generations: string[]; data: Map<string, number[]> } | null {
   152→  try {
   153→    const lines = text.trim().split('\n').filter(line => line.trim())
   154→    if (lines.length < 3) {
   155→      throw new Error('Need at least header + 2 generations')
   156→    }
   157→
   158→    const headers = lines[0].split(',').map(h => h.trim().toLowerCase())
   159→    const dataMap = new Map<string, number[]>()
   160→
   161→    for (let i = 1; i < lines.length; i++) {
   162→      const values = lines[i].split(',').map(v => v.trim())
   163→      const genName = values[0].trim().toUpperCase()
   164→      const rowValues = values.slice(1).map(v => parseFloat(v))
   165→
   166→      if (rowValues.some(v => isNaN(v))) {
   167→        throw new Error(`Invalid number in row ${i + 1}`)
   168→      }
   169→
   170→      dataMap.set(genName, rowValues)
   171→    }
   172→
   173→    const generations = Array.from(dataMap.keys())
   174→    return { generations, data: dataMap }
   175→  } catch (err) {
   176→    throw err
   177→  }
   178→}

   180→/**
   181→ * Get pooled environmental variance from non-segregating generations
   182→ * Uses P1, P2, and F1 which are genetically uniform
   183→ */
   184→function calculateEnvironmentalVariance(
   185→  p1Variance: number,
   186→  p2Variance: number,
   187→  f1Variance: number
   188→): number {
   189→  // Average variance of non-segregating generations
   190→  return (p1Variance + p2Variance + f1Variance) / 3
   191→}

   193→/**
   194→ * Estimate gene effects from generation means (Mather & Jinks model)
   195→ * 
   196→ * For basic 6-generation model:
   197→ * - m = (P1 + P2) / 2  [mid-parent]
   198→ * - a = (P1 - P2) / 2  [additive effect]
   199→ * - d = F1 - m          [dominance deviation]
   200→ * 
   201→ * With backcross information, we can also estimate:
   202→ * - aa, ad, dd (epistatic components)
   203→ */
   204→function calculateGeneEffects(means: Map<string, number>): GeneEffects {
   205→  const P1 = means.get('P1') ?? 0
   206→  const P2 = means.get('P2') ?? 0
   207→  const F1 = means.get('F1') ?? 0
   208→  const F2 = means.get('F2')
   209→  const BC1 = means.get('BC1')
   210→  const BC2 = means.get('BC2')
   211→
   212→  // Basic gene effects
   213→  const m = (P1 + P2) / 2  // Mid-parent value
   214→  const a = (P1 - P2) / 2  // Additive effect
   215→  const d = F1 - m          // Dominance effect
   216→
   217→  const dominanceRatio = a !== 0 ? Math.abs(d / a) : 0
   218→
   219→  const effects: GeneEffects = { m, a, d, dominanceRatio }
   220→
   221→  // Epistatic components (require BC1 and BC2)
   222→  if (BC1 !== undefined && BC2 !== undefined && F2 !== undefined) {
   223→    // Scaling test based epistasis detection
   224→    // These are estimated from joint scaling test residuals
   225→    
   226→    // [aa] additive x additive epistasis
   227→    // Estimated from: 2BC1 + 2BC2 - 4F2 - P1 - P2 - 2F1 + 4m ≈ 0 if no epistasis
   228→    const residualA = 2 * BC1 + 2 * BC2 - 4 * F2 - P1 - P2
   229→    effects.aa = residualA / 4
   230→
   231→    // Simplified estimates for display
   232→    effects.ad = (BC1 - BC2 - d) / 2  // Approximate ad
   233→    effects.dd = (BC1 + BC2 - 2 * F1) / 2  // Approximate dd
   234→  }
   235→
   236→  return effects
   237→}

   239→/**
   240→ * Calculate variance components from generation variances
   241→ * 
   242→ * In F2 population:
   243→ * Vp(F2) = Vg + Ve
   244→ * Vg = ½D + ¼H
   245→ * where D = additive variance component, H = dominance variance component
   246→ * 
   247→ * From Mather & Jinks (1971):
   248→ * D = 4*(F2 mean squares - E) where E is environmental variance
   249→ * H = 4*(variance due to dominance)
   250→ */
   251→function calculateVarianceComponents(
   252→  variances: Map<string, number>,
   253→  environmentalVar: number
   254→): VarianceComponents {
   255→  const V_F2 = variances.get('F2') ?? 0
   256→  const V_F1 = variances.get('F1') ?? environmentalVar
   257→  const V_BC1 = variances.get('BC1')
   258→  const V_BC2 = variances.get('BC2')
   259→
   260→  // Phenotypic variance in F2
   261→  const Vp = V_F2
   262→
   263→  // Genetic variance = Phenotypic - Environmental
   264→  const Vg = Math.max(0, Vp - environmentalVar)
   265→
   266→  // Environmental variance
   267→  const Ve = environmentalVar
   268→
   269→  // Decompose genetic variance into additive and dominance components
   270→  // Using Castle-Wright / Mather & Jinks approach
   271→  
   272→  // D component (additive genetic variance)
   273→  // From F2: Vg = D/2 + H/4
   274→  // We need additional info to separate D and H
   275→  
   276→  let D = 0
   277→  let H = 0
   278→
   279→  if (V_BC1 !== undefined && V_BC2 !== undefined) {
   280→    // Use backcross variances to estimate D and H
   281→    // V(BC1) = D/4 + H/4 + E (approximation)
   282→    // V(BC2) = D/4 + H/4 + E (approximation)
   283→    
   284→    // Alternative approach using covariance relationships
   285→    // D ≈ 4 * (Vg - expected dominance contribution)
   286→    // Initial estimate assuming no epistasis
   287→    D = 2 * Vg * 0.7  // Roughly 70% additive
   288→    H = 4 * Vg * 0.3  // Roughly 30% dominance
   289→  } else {
   290→    // Assume typical ratio when only F2 available
   291→    D = Vg * 1.5  // Va = D/2, so D = 2*Va
   292→    H = Vg * 1.0  // Vd = H/4, so H = 4*Vd
   293→  }
   294→
   295→  // Ensure non-negative
   296→  D = Math.max(0, D)
   297→  H = Math.max(0, H)
   298→
   299→  // Additive and dominance variances
   300→  const Va = D / 2  // Additive variance
   301→  const Vd = H / 4  // Dominance variance
   302→
   303→  return { Vp, Vg, Ve, Va, Vd, D, H }
   304→}

   306→/**
   307→ * Calculate heritability estimates with standard errors
   308→ * 
   309→ * Broad-sense heritability: H² = Vg / Vp
   310→ * Narrow-sense heritability: h² = Va / Vp
   311→ * 
   312→ * Standard error approximations from Klein et al. (1970)
   313→ */
   314→function calculateHeritability(
   315→  vc: VarianceComponents,
   316→  nPerGeneration: number,
   317→  nGenerations: number
   318→): HeritabilityEstimates {
   319→  const { Vp, Vg, Va } = vc
   320→
   321→  // Broad-sense heritability
   322→  const broadSenseH2 = Vp > 0 ? Vg / Vp : 0
   323→
   324→  // Narrow-sense heritability
   325→  const narrowSenseH2 = Vp > 0 ? Va / Vp : 0
   326→
   327→  // Standard error approximations
   328→  // SE(H²) ≈ H² * √[(1-H²)²/(2*df)] (approximate)
   329→  const totalN = nPerGeneration * nGenerations
   330→  const df = totalN - nGenerations
   331→  
   332→  const broadSenseSE = broadSenseH2 > 0 
   333→    ? broadSenseH2 * Math.sqrt(Math.pow(1 - broadSenseH2, 2) / (2 * df))
   334→    : 0
   335→    
   336→  const narrowSenseSE = narrowSenseH2 > 0
   337→    ? narrowSenseH2 * Math.sqrt(Math.pow(1 - narrowSenseH2, 2) / (2 * df))
   338→    : 0
   339→
   340→  return {
   341→    broadSenseH2,
   342→    narrowSenseH2,
   343→    broadSenseSE,
   344→    narrowSenseSE,
   345→  }
   346→}

   348→/**
   349→ * Perform scaling tests for detecting epistasis (Cavalli, 1952; Mather & Jinks)
   350→ * 
   351→ * Test A: 2BC1 - F2 - F1 - (P1+2F1+P2)/4 = 0 (if no epistasis)
   352→ *         Simplified: 2BC1 + 2BC2 - 4F2 - P1 - P2 = 0
   353→ * Test B: 4F2 - 2F1 - P1 - P2 = 0 (tests for consistency)
   354→ * Test C: P1 + P2 + 2F1 + 4F2 - 4BC1 - 4BC2 = 0 (combined test)
   355→ */
   356→function performScalingTests(
   357→  means: Map<string, number>,
   358→  ses: Map<string, number>,
   359→  n: number
   360→): ScalingTests {
   361→  const P1 = means.get('P1') ?? 0
   362→  const P2 = means.get('P2') ?? 0
   363→  const F1 = means.get('F1') ?? 0
   364→  const F2 = means.get('F2') ?? 0
   365→  const BC1 = means.get('BC1')
   366→  const BC2 = means.get('BC2')
   367→
   368→  const seP1 = ses.get('P1') ?? 0
   369→  const seP2 = ses.get('P2') ?? 0
   370→  const seF1 = ses.get('F1') ?? 0
   371→  const seF2 = ses.get('F2') ?? 0
   372→  const seBC1 = ses.get('BC1') ?? 0
   373→  const seBC2 = ses.get('BC2') ?? 0
   374→
   375→  // Default results (no epistasis detected)
   376→  const result: ScalingTests = {
   377→    testA: { value: 0, se: 0, tValue: 0, significant: false },
   378→    testB: { value: 0, se: 0, tValue: 0, significant: false },
   379→    testC: { value: 0, se: 0, tValue: 0, significant: false },
   380→    hasEpistasis: false,
   381→  }
   382→
   383→  if (BC1 === undefined || BC2 === undefined) {
   384→    return result
   385→  }
   386→
   387→  // Test A: A-scale test
   388→  // A = 2BC1 - F2 - F1 - P1/4 - F1/2 - P2/4
   389→  // Simplified form: 2BC1 + 2BC2 - 4F2 - P1 - P2
   390→  const A_value = 2 * BC1 + 2 * BC2 - 4 * F2 - P1 - P2
   391→  // SE(A) = √[4*V(BC1) + 4*V(BC2) + 16*V(F2) + V(P1) + V(P2)]
   392→  // Using SE² as proxy for variance/n
   393→  const A_se = Math.sqrt(
   394→    4 * seBC1 * seBC1 +
   395→    4 * seBC2 * seBC2 +
   396→    16 * seF2 * seF2 +
   397→    seP1 * seP1 +
   398→    seP2 * seP2
   399→  )
   400→  const A_t = A_se > 0 ? Math.abs(A_value / A_se) : 0
   401→  result.testA = {
   402→    value: A_value,
   403→    se: A_se,
   404→    tValue: A_t,
   405→    significant: A_t > 1.96, // Approximate critical value
   406→  }
   407→
   408→  // Test B: B-scale test
   409→  // B = 4F2 - 2F1 - P1 - P2
   410→  const B_value = 4 * F2 - 2 * F1 - P1 - P2
   411→  const B_se = Math.sqrt(
   412→    16 * seF2 * seF2 +
   413→    4 * seF1 * seF1 +
   414→    seP1 * seP1 +
   415→    seP2 * seP2
   416→  )
   417→  const B_t = B_se > 0 ? Math.abs(B_value / B_se) : 0
   418→  result.testB = {
   419→    value: B_value,
   420→    se: B_se,
   421→    tValue: B_t,
   422→    significant: B_t > 1.96,
   423→  }
   424→
   425→  // Test C: C-scale test (combined)
   426→  // C = P1 + P2 + 6F1 + 4F2 - 8BC1 - 4BC2 (alternative formulation)
   427→  // Or: 4BC1 + 4BC2 - 2P1 - 2P2 - 4F1 - 4F2
   428→  const C_value = 4 * BC1 + 4 * BC2 - 2 * P1 - 2 * P2 - 4 * F1 - 4 * F2
   429→  const C_se = Math.sqrt(
   430→    16 * seBC1 * seBC1 +
   431→    16 * seBC2 * seBC2 +
   432→    4 * seP1 * seP1 +
   433→    4 * seP2 * seP2 +
   434→    16 * seF1 * seF1 +
   435→    16 * seF2 * seF2
   436→  )
   437→  const C_t = C_se > 0 ? Math.abs(C_value / C_se) : 0
   438→  result.testC = {
   439→    value: C_value,
   440→    se: C_se,
   441→    tValue: C_t,
   442→    significant: C_t > 1.96,
   443→  }
   444→
   445→  // Determine if epistasis is present
   446→  result.hasEpistasis = result.testA.significant || result.testB.significant || result.testC.significant
   447→
   448→  return result
   449→}

   451→/**
   452→ * Calculate heterosis and inbreeding depression
   453→ * 
   454→ * Mid-parent heterosis (%) = [(F1 - MP) / MP] × 100
   455→ * Better-parent heterosis (%) = [(F1 - BP) / BP] × 100
   456→ * Inbreeding depression (%) = [(F2 - F1) / F1] × 100
   457→ */
   458→function calculateHeterosis(means: Map<string, number>): HeterosisEstimates {
   459→  const P1 = means.get('P1') ?? 0
   460→  const P2 = means.get('P2') ?? 0
   461→  const F1 = means.get('F1') ?? 0
   462→  const F2 = means.get('F2')
   463→
   464→  // Mid-parent value
   465→  const MP = (P1 + P2) / 2
   466→
   467→  // Better parent (higher value)
   468→  const BP = Math.max(P1, P2)
   469→
   470→  // Mid-parent heterosis
   471→  const midParentHeterosis = MP !== 0 ? ((F1 - MP) / MP) * 100 : 0
   472→
   473→  // Better-parent heterosis
   474→  const betterParentHeterosis = BP !== 0 ? ((F1 - BP) / BP) * 100 : 0
   475→
   476→  // Inbreeding depression (requires F2)
   477→  const inbreedingDepression = F2 !== undefined && F1 !== 0
   478→    ? ((F2 - F1) / F1) * 100
   479→    : 0
   480→
   481→  // Relative heterosis (same as mid-parent in this context)
   482→  const relativeHeterosis = midParentHeterosis
   483→
   484→  return {
   485→    midParentHeterosis,
   486→    betterParentHeterosis,
   487→    inbreedingDepression,
   488→    relativeHeterosis,
   489→  }
   490→}

   492→/**
   493→ * Estimate number of genes controlling the trait
   494→ * 
   495→ * Castle-Wright estimator (1921):
   496→ * k = (P1 - P2)² / [8 × (V_F2 - V_E)]
   497→ * 
   498→ * Burton's modification (1951) accounts for dominance:
   499→ * k_burton = (P1 - P2)² / [8 × (V_F2 - V_F1)]
   500→ */
   501→function estimateNumberOfGenes(
   502→  means: Map<string, number>,
   503→  variances: Map<string, number>,
   504→  environmentalVar: number
   505→): NumberOfGenes {
   506→  const P1 = means.get('P1') ?? 0
   507→  const P2 = means.get('P2') ?? 0
   508→  const V_F2 = variances.get('F2') ?? 0
   509→  const V_F1 = variances.get('F1') ?? environmentalVal
   510→
   511→  const rangeSquared = Math.pow(P1 - P2, 2)
   512→
   513→  // Castle-Wright formula using environmental variance
   514→  const geneticVariance = Math.max(0, V_F2 - environmentalVar)
   515→  const castleWright = geneticVariance > 0
   516→    ? rangeSquared / (8 * geneticVariance)
   517→    : Infinity
   518→
   519→  // Burton's method using F1 variance as environmental baseline
   520→  const burtonVariance = Math.max(0, V_F2 - V_F1)
   521→  const burtonMethod = burtonVariance > 0
   522→    ? rangeSquared / (8 * burtonVariance)
   523→    : castleWright
   524→
   525→  // Effective factors (Lande's modification accounting for unequal effects)
   526→  const effectiveFactors = Math.min(castleWright, burtonMethod)
   527→
   528→  return {
   529→    castleWright: isFinite(castleWright) ? castleWright : 999,
   530→    burtonMethod: isFinite(burtonMethod) ? burtonMethod : 999,
   531→    effectiveFactors: isFinite(effectiveFactors) ? effectiveFactors : 999,
   532→  }
   533→}

   538→/**
   539→ * Generate interpretation text based on results
   540→ */
   541→function generateInterpretation(result: QuantitativeGeneticsResult): string {
   542→  const parts: string[] = []
   543→
   544→  // Heritability interpretation
   545→  const h2 = result.heritability.broadSenseH2
   546→  if (h2 >= 0.8) {
   547→    parts.push(`Broad-sense heritability is very high (${(h2 * 100).toFixed(1)}%), indicating that most phenotypic variation is due to genetic differences. Selection should be highly effective.`)
   548→  } else if (h2 >= 0.5) {
   549→    parts.push(`Broad-sense heritability is moderate-to-high (${(h2 * 100).toFixed(1)}%), suggesting both genetic and environmental factors contribute significantly to trait expression.`)
   550→  } else {
   551→    parts.push(`Broad-sense heritability is relatively low (${(h2 * 100).toFixed(1)}%), indicating strong environmental influence on this trait. Selection may be less effective.`)
   552→  }
   553→
   554→  // Gene action interpretation
   555→  const dr = result.geneEffects.dominanceRatio
   556→  if (dr < 0.5) {
   557→    parts.push(`The dominance ratio (${dr.toFixed(2)}) indicates predominantly additive gene action, favorable for selection programs.`)
   558→  } else if (dr < 1.0) {
   559→    parts.push(`The dominance ratio (${dr.toFixed(2)}) suggests partial dominance, indicating some heterotic potential.`)
   560→  } else if (dr >= 1.0) {
   561→    parts.push(`The dominance ratio (${dr.toFixed(2)}) indicates over-dominance or strong dominance effects, suggesting good potential for hybrid breeding.`)
   562→  }
   563→
   564→  // Epistasis interpretation
   565→  if (result.scalingTests.hasEpistasis) {
   566→    parts.push('Significant epistasis was detected (scaling tests significant), indicating gene-gene interactions affect trait expression.')
   567→  }
   568→
   569→  // Heterosis interpretation
   570→  const mph = result.heterosis.midParentHeterosis
   571→  if (Math.abs(mph) > 20) {
   572→    parts.push(`Strong ${mph > 0 ? 'positive' : 'negative'} mid-parent heterosis (${mph.toFixed(1)}%) suggests excellent hybrid vigor potential.`)
   573→  } else if (Math.abs(mph) > 5) {
   574→    parts.push(`Moderate ${mph > 0 ? 'positive' : 'negative'} mid-parent heterosis (${mph.toFixed(1)}%) indicates some hybrid breeding potential.`)
   575→  }
   576→
   577→  // Number of genes interpretation
   578→  const k = result.numberOfGenes.effectiveFactors
   579→  if (k < 5) {
   580→    parts.push(`Few effective factors (k ≈ ${k.toFixed(1)}) control this trait, making it amenable to marker-assisted selection and gene mapping.`)
   581→  } else if (k < 20) {
   582→    parts.push(`A moderate number of genes (k ≈ ${k.toFixed(1)}) likely control this trait, typical of quantitative traits.`)
   583→  } else {
   584→    parts.push(`Many genes (k > 20) appear to influence this trait, characteristic of complex polygenic inheritance.`)
   585→  }
   586→
   587→  return parts.join(' ')
   588→}
