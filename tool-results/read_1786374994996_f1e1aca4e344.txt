   164→  // Parse genotypic data
   165→  const parseData = useCallback((text: string): boolean => {
   166→    try {
   167→      setError(null)
   168→      const lines = text.trim().split('\n').filter(line => line.trim())

   173→      }

   204→        samples,

   222→  // Calculate allele frequencies for a single marker
   223→  const calculateAlleleFrequencies = useCallback(
   224→    (genotypes: string[]): { freq: Record<string, number>; ho: number } => {
   225→      const alleleCounts: Record<string, number> = {}
   226→      let heteroCount = 0
   227→      const n = genotypes.length
   228→
   229→      for (const geno of genotypes) {
   230→        if (geno.length !== 2) continue
   231→        
   232→        const a1 = geno[0]
   233→        const a2 = geno[1]
   234→        
   235→        alleleCounts[a1] = (alleleCounts[a1] || 0) + 1
   236→        alleleCounts[a2] = (alleleCounts[a2] || 0) + 1
   237→        
   238→        if (a1 !== a2) {
   239→          heteroCount++
   240→        }
   241→      }
   242→
   243→      const totalAlleles = 2 * n
   244→      const freq: Record<string, number> = {}
   245→      
   246→      for (const [allele, count] of Object.entries(alleleCounts)) {
   247→        freq[allele] = count / totalAlleles
   248→      }
   249→
   250→      return { freq, ho: heteroCount / n }
   251→    },
   252→    []
   253→  )
   254→
   255→  // Calculate PIC (Polymorphism Information Content) - Botstein et al. (1980)
   256→  const calculatePIC = useCallback((freq: Record<string, number>): number => {
   257→    const alleles = Object.keys(freq)
   258→    let pic = 1
   259→    
   260→    // PIC = 1 - Σp_i² - 2ΣΣ(p_i * p_j)² for i < j
   261→    for (let i = 0; i < alleles.length; i++) {
   262→      const pi = freq[alleles[i]]
   263→      pic -= pi * pi
   264→      
   265→      for (let j = i + 1; j < alleles.length; j++) {
   266→        const pj = freq[alleles[j]]
   267→        pic -= 2 * pi * pi * pj * pj
   268→      }
   269→    }
   270→    
   271→    return Math.max(0, pic)
   272→  }, [])
   273→
   274→  // Calculate expected heterozygosity (gene diversity) - Nei (1973)
   275→  const calculateExpectedHeterozygosity = useCallback((freq: Record<string, number>): number => {
   276→    let he = 1
   277→    for (const p of Object.values(freq)) {
   278→      he -= p * p
   279→    }
   280→    return Math.max(0, he)
   281→  }, [])
   282→
   283→  // Calculate MAF (Minor Allele Frequency)
   284→  const calculateMAF = useCallback((freq: Record<string, number>): number => {
   285→    const frequencies = Object.values(freq)
   286→    return Math.min(...frequencies)
   287→  }, [])
   288→
   289→  // Calculate Linkage Disequilibrium - Hill & Weir (1988)
   290→  const calculateLD = useCallback(
   291→    (
   292→      genotypes1: string[],
   293→      genotypes2: string[],
   294→      alpha: number = 0.05
   295→    ): LDResult => {
   296→      const n = genotypes1.length
   297→      
   298→      // Get alleles for each marker
   299→      const alleles1 = new Set<string>()
   300→      const alleles2 = new Set<string>()
   301→      
   302→      for (const g of genotypes1) {
   303→        alleles1.add(g[0])
   304→        alleles1.add(g[1])
   305→      }
   306→      for (const g of genotypes2) {
   307→        alleles2.add(g[0])
   308→        alleles2.add(g[1])
   309→      }
   310→
   311→      const aArr = Array.from(alleles1).sort()
   312→      const bArr = Array.from(alleles2).sort()
   313→      
   314→      // Use first allele as reference for biallelic case
   315→      const A = aArr[0] || 'A'
   316→      const B = bArr[0] || 'A'
   317→
   318→      // Count haplotype-like frequencies from genotypes
   319→      // For diploid data, we estimate gamete frequencies using EM or simple counting
   320→      let nAB = 0, nAb = 0, naB = 0, nab = 0
   321→      
   322→      for (let i = 0; i < n; i++) {
   323→        const g1 = genotypes1[i]
   324→        const g2 = genotypes2[i]
   325→        
   326→        // Count allele occurrences
   327→        const g1hasA = g1.includes(A)
   328→        const g2hasB = g2.includes(B)
   329→        
   330→        // Simplified: count based on presence/absence
   331→        if (g1hasA && g2hasB) nAB++
   332→        if (g1hasA && !g2hasB) nAb++
   333→        if (!g1hasA && g2hasB) naB++
   334→        if (!g1hasA && !g2hasB) nab++
   335→      }
   336→
   337→      // Allele frequencies
   338→      const pA = (nAB + nAb) / (2 * n)
   339→      const pa = 1 - pA
   340→      const pB = (nAB + naB) / (2 * n)
   341→      const pb = 1 - pB
   342→
   343→      // Expected frequency of AB under independence
   344→      const expectedAB = pA * pB
   345→      const observedAB = nAB / n
   346→      
   347→      // D coefficient
   348→      const D = observedAB - expectedAB
   349→      
   350→      // r² = D² / (pA * pa * pB * pb)
   351→      const denom = pA * pa * pB * pb
   352→      const r2 = denom > 0 ? (D * D) / denom : 0
   353→      
   354→      // D' (standardized disequilibrium coefficient)
   355→      // D' = D / D_max where D_max = min(pA*pb, pa*pB) if D > 0
   356→      //                   D_max = min(pA*pB, pa*pb) if D < 0
   357→      let dMax: number
   358→      if (D > 0) {
   359→        dMax = Math.min(pA * pb, pa * pB)
   360→      } else {
   361→        dMax = Math.min(pA * pB, pa * pb)
   362→      }
   363→      const dPrime = dMax > 0 ? Math.abs(D) / dMax : 0
   364→
   365→      // Chi-square test for LD significance
   366→      // Contingency table: 2x2 (presence/absence of each allele)
   367→      const contingencyTable = [
   368→        [nAB, nAb],
   369→        [naB, nab]
   370→      ]
   371→      
   372→      const chiSqResult = chiSquareTest(contingencyTable)
   373→
   374→      return {
   375→        marker1: '',
   376→        marker2: '',
   377→        r2: Math.min(1, r2),
   378→        dPrime: Math.min(1, dPrime),
   379→        dValue: D,
   380→        chiSquare: chiSqResult.chiSquare,
   381→        pValue: chiSqResult.pValue,
   382→        significant: chiSqResult.pValue < alpha
   383→      }
   384→    },
   385→    []
   386→  )
   387→
   388→  // Single Marker Analysis (SMA) for QTL mapping
   389→  const calculateQTLSMA = useCallback(
   390→    (
   391→      genotypes: string[],
   392→      traitValues: number[],
   393→      alpha: number = 0.05
   394→    ): QTLResult => {
   395→      // Group trait values by genotype
   396→      const groups: Record<string, number[]> = {}
   397→      
   398→      for (let i = 0; i < genotypes.length; i++) {
   399→        const geno = genotypes[i]
   400→        if (!groups[geno]) {
   401→          groups[geno] = []
   402→        }
   403→        groups[geno].push(traitValues[i])
   404→      }
   405→
   406→      const groupKeys = Object.keys(groups)
   407→      const k = groupKeys.length
   408→      const n = traitValues.length
   409→
   410→      if (k < 2) {
   411→        return {
   412→          marker: '',
   413→          fStatistic: 0,
   414→          pValue: 1,
   415→          rSquared: 0,
   416→          lodScore: 0,
   417→          effect: 0,
   418→          significant: false
   419→        }
   420→      }
   421→
   422→      // One-way ANOVA
   423→      const grandMean = mean(traitValues)
   424→      
   425→      // Sum of squares between groups
   426→      let ssb = 0
   427→      const groupMeans: number[] = []
   428→      
   429→      for (const key of groupKeys) {
   430→        const groupData = groups[key]
   431→        const groupMean = mean(groupData)
   432→        groupMeans.push(groupMean)
   433→        ssb += groupData.length * (groupMean - grandMean) ** 2
   434→      }
   435→      
   436→      // Sum of squares within groups (error)
   437→      let ssw = 0
   438→      for (const key of groupKeys) {
   439→        const groupData = groups[key]
   440→        const groupMean = mean(groupData)
   441→        for (const val of groupData) {
   442→          ssw += (val - groupMean) ** 2
   443→        }
   444→      }
   445→
   446→      // Degrees of freedom
   447→      const dfBetween = k - 1
   448→      const dfWithin = n - k
   449→
   450→      // Mean squares
   451→      const msb = ssb / dfBetween
   452→      const msw = ssw / dfWithin
   453→
   454→      // F-statistic
   455→      const fStat = msw > 0 ? msb / msw : 0
   456→
   457→      // R-squared (proportion of variance explained)
   458→      const ssTotal = ssb + ssw
   459→      const rSquared = ssTotal > 0 ? ssb / ssTotal : 0
   460→
   461→      // Approximate p-value using F-distribution
   462→      // Using chi-square approximation for simplicity
   463→      const pValueApprox = 1 - chiSquareCDF(fStat * dfBetween, dfBetween)
   464→
   465→      // LOD score approximation: LOD = (n/2) * log10(1 - R²) for null vs alternative
   466→      // Or LOD ≈ (n * R²) / (2 * ln(10) * (1-R²))
   467→      const lodScore = rSquared < 1 
   468→        ? (n * rSquared) / (2 * Math.LN10 * (1 - rSquared + 0.001))
   469→        : 0
   470→
   471→      // Effect size (difference between extreme genotype means)
   472→      const sortedMeans = [...groupMeans].sort((a, b) => a - b)
   473→      const effect = sortedMeans[sortedMeans.length - 1] - sortedMeans[0]
   474→
   475→      return {
   476→        marker: '',
   477→        fStatistic: fStat,
   478→        pValue: Math.max(0, Math.min(1, pValueApprox)),
   479→        rSquared: Math.min(1, rSquared),
   480→        lodScore: Math.min(20, lodScore),
   481→        effect,
   482→        significant: pValueApprox < alpha
   483→      }
   484→    },
   485→    []
   486→  )
   487→
   488→  // Calculate Rogers' genetic distance
   489→  const calculateRogersDistance = useCallback(
   490→    (genos1: string[], genos2: string[]): number => {
   491→      let sumDiff = 0
   492→      let validLoci = 0
   493→
   494→      for (let i = 0; i < genos1.length; i++) {
   495→        const g1 = genos1[i]
   496→        const g2 = genos2[i]
   497→        
   498→        if (g1.length !== 2 || g2.length !== 2) continue
   499→        
   500→        // Count shared alleles
   501→        let sharedAlleles = 0
   502→        if (g1[0] === g2[0] || g1[0] === g2[1]) sharedAlleles++
   503→        if (g1[1] === g2[0] || g1[1] === g2[1]) sharedAlleles++
   504→        
   505→        // Rogers' distance contribution: sqrt(0.5 * (1 - proportionShared))
   506→        const propShared = sharedAlleles / 2
   507→        sumDiff += 0.5 * (1 - propShared)
   508→        validLoci++
   509→      }
   510→
   511→      return validLoci > 0 ? Math.sqrt(sumDiff / validLoci) : 0
   512→    },
   513→    []
   514→  )
   515→
   516→  // Calculate Jaccard similarity coefficient
   517→  const calculateJaccardSimilarity = useCallback(
   518→    (genos1: string[], genos2: string[]): number => {
   519→      let intersection = 0
   520→      let union = 0
   521→
   522→      for (let i = 0; i < genos1.length; i++) {
   523→        const alleles1 = new Set([genos1[i][0], genos1[i][1]])
   524→        const alleles2 = new Set([genos2[i][0], genos2[i][1]])
   525→        
   526→        // Intersection
   527→        let interCount = 0
   528→        for (const a of alleles1) {
   529→          if (alleles2.has(a)) interCount++
   530→        }
   531→        intersection += interCount
   532→        
   533→        // Union
   534→        union += new Set([...alleles1, ...alleles2]).size
   535→      }
   536→
   537→      return union > 0 ? intersection / union : 0
   538→    },
   539→    []
   540→  )
   541→
   542→  // Calculate Nei's genetic distance (1972)
   543→  const calculateNeiDistance = useCallback(
   544→    (genos1: string[], genos2: string[]): number => {
   545→      let sumLog = 0
   546→      let validLoci = 0
   547→
   548→      for (let i = 0; i < genos1.length; i++) {
   549→        const alleles1 = [genos1[i][0], genos1[i][1]]
   550→        const alleles2 = [genos2[i][0], genos2[i][1]]
   551→        
   552→        // Calculate allele frequencies for each population (individual as pop)
   553→        const freq1: Record<string, number> = {}
   554→        const freq2: Record<string, number> = {}
   555→        
   556→        for (const a of alleles1) {
   557→          freq1[a] = (freq1[a] || 0) + 0.5
   558→        }
   559→        for (const a of alleles2) {
   560→          freq2[a] = (freq2[a] || 0) + 0.5
   561→        }
   562→
   563→        // Nei's distance: -ln(sum(sqrt(p_i * q_i)))
   564→        const allAlleles = new Set([...Object.keys(freq1), ...Object.keys(freq2)])
   565→        let identitySum = 0
   566→        
   567→        for (const allele of allAlleles) {
   568→          identitySum += Math.sqrt((freq1[allele] || 0) * (freq2[allele] || 0))
   569→        }
   570→
   571→        if (identitySum > 0) {
   572→          sumLog += Math.log(identitySum)
   573→          validLoci++
   574→        }
   575→      }
   576→
   577→      return validLoci > 0 ? -(sumLog / validLoci) : 0
   578→    },
   579→    []
   580→  )

   582→  // Run complete analysis
   583→  const runAnalysis = useCallback(() => {
   584→    if (!parsedData) {
   585→      setError('Please parse data before running analysis')
   586→      return
   587→    }
   588→
   589→    setIsLoading(true)
   590→    setError(null)
   591→
   592→    try {
   593→      setTimeout(() => {
   594→        try {
   595→          const { samples, markers, genotypes, traitValues } = parsedData
   596→          
   597→          // ========================================
   598→          // 1. MARKER-BASED GENETIC ANALYSIS
   599→          // ========================================
   600→          const markerStats: MarkerStats[] = []
   601→
   602→          for (let j = 0; j < markers.length; j++) {
   603→            const markerGenotypes = genotypes.map(g => g[j])
   604→            const { freq, ho } = calculateAlleleFrequencies(markerGenotypes)
   605→            
   606→            const genotypeCounts: Record<string, number> = {}
   607→            for (const g of markerGenotypes) {
   608→              genotypeCounts[g] = (genotypeCounts[g] || 0) + 1
   609→            }
   610→
   611→            markerStats.push({
   612→              marker: markers[j],
   613→              alleleFreq: freq,
   614→              maf: calculateMAF(freq),
   615→              pic: calculatePIC(freq),
   616→              he: calculateExpectedHeterozygosity(freq),
   617→              ho,
   618→              genotypeCounts
   619→            })
   620→          }
   621→
   622→          // Filter by MAF threshold
   623→          const polymorphicMarkers = markerStats.filter(m => m.maf >= mafThreshold)
   624→
   625→          // ========================================
   626→          // 2. LINKAGE DISEQUILIBRIUM ANALYSIS
   627→          // ========================================
   628→          const ldResults: LDResult[] = []
   629→
   630→          for (let i = 0; i < markers.length; i++) {
   631→            for (let j = i + 1; j < markers.length; j++) {
   632→              const genos1 = genotypes.map(g => g[i])
   633→              const genos2 = genotypes.map(g => g[j])
   634→              
   635→              const ldResult = calculateLD(genos1, genos2, alphaLevel)
   636→              ldResult.marker1 = markers[i]
   637→              ldResult.marker2 = markers[j]
   638→              
   639→              if (ldResult.r2 >= ldCutoff || ldResult.significant) {
   640→                ldResults.push(ldResult)
   641→              }
   642→            }
   643→          }
   644→
   645→          // Sort by r² descending
   646→          ldResults.sort((a, b) => b.r2 - a.r2)
   647→
   648→          // ========================================
   649→          // 3. QTL MAPPING (if trait values available)
   650→          // ========================================
   651→          const qtlResults: QTLResult[] = []
   652→
   653→          if (traitValues && traitValues.length === samples.length) {
   654→            for (let j = 0; j < markers.length; j++) {
   655→              const markerGenotypes = genotypes.map(g => g[j])
   656→              const qtlResult = calculateQTLSMA(markerGenotypes, traitValues, alphaLevel)
   657→              qtlResult.marker = markers[j]
   658→              qtlResults.push(qtlResult)
   659→            }
   660→
   661→            // Sort by R² descending
   662→            qtlResults.sort((a, b) => b.rSquared - a.rSquared)
   663→          }
   664→

   666→          // 4. ASSOCIATION MAPPING METRICS
   667→          // ========================================

   672→              const markerGenotypes = genotypes.map(g => g[j])
   673→              const qtlResult = calculateQTLSMA(markerGenotypes, traitValues, alphaLevel)
   674→              
   675→              // GLM association (similar to SMA but with different interpretation)
   676→              const glmPvalue = qtlResult.pValue
   677→              const glmEffect = qtlResult.effect
   678→              
   679→              // MLM approximation (inflate p-values slightly to account for structure)
   680→              const inflationFactor = 1.2 // Simple approximation
   681→              const mlmPvalue = Math.min(1, glmPvalue * inflationFactor)
   682→              const mlmEffect = glmEffect * 0.9
   683→
   684→              associationResults.push({
   685→                marker: markers[j],
   686→                glmPvalue,
   687→                glmEffect,
   688→                mlmPvalue,
   689→                mlmEffect,
   690→                significant: glmPvalue < alphaLevel
   691→              })
   692→            }
   693→          }
   694→
   695→          // ========================================
   696→          // 5. MARKER-ASSISTED SELECTION (MAS)
   697→          // ========================================
   698→          const masResults: MASResult[] = []
   699→
   700→          if (qtlResults.length > 0) {
   701→            for (const qtl of qtlResults) {
   702→              // Selection efficiency based on R² and significance
   703→              const efficiency = qtl.rSquared * (qtl.significant ? 1 : 0.5)
   704→              
   705→              // Association strength categorization
   706→              let strength: string
   707→              if (qtl.rSquared > 0.3) strength = 'Very Strong'
   708→              else if (qtl.rSquared > 0.2) strength = 'Strong'
   709→              else if (qtl.rSquared > 0.1) strength = 'Moderate'
   710→              else if (qtl.rSquared > 0.05) strength = 'Weak'
   711→              else strength = 'Negligible'
   712→
   713→              // Expected genetic gain with MAS
   714→              // ΔG_MAS = i * h²_MAS * σ_P where h²_MAS is mediated through marker
   715→              const expectedGain = qtl.effect * qtl.rSquared
   716→
   717→              // Cost-benefit assessment
   718→              let costBenefit: string
   719→              if (qtl.significant && qtl.rSquared > 0.15) {
   720→                costBenefit = 'Highly Recommended'
   721→              } else if (qtl.significant && qtl.rSquared > 0.08) {
   722→                costBenefit = 'Recommended'
   723→              } else if (qtl.rSquared > 0.03) {
   724→                costBenefit = 'Optional'
   725→              } else {
   726→                costBenefit = 'Not Recommended'
   727→              }
   728→
   729→              masResults.push({
   730→                marker: qtl.marker,
   731→                selectionEfficiency: efficiency,
   732→                associationStrength: strength,
   733→                expectedGain,
   734→                recommended: qtl.significant && qtl.rSquared > 0.08,
   735→                costBenefit
   736→              })
   737→            }
   738→          }
   739→
   740→          // ========================================
   741→          // 6. GENETIC DISTANCE & SIMILARITY
   742→          // ========================================

   746→            for (let j = i + 1; j < samples.length; j++) {
   747→              const genos1 = genotypes[i]
   748→              const genos2 = genotypes[j]
   749→
   750→              geneticDistances.push({
   751→                sample1: samples[i],
   752→                sample2: samples[j],
   753→                rogersDistance: calculateRogersDistance(genos1, genos2),
   754→                jaccardSimilarity: calculateJaccardSimilarity(genos1, genos2),
   755→                neiDistance: calculateNeiDistance(genos1, genos2)
   756→              })
   757→            }
   758→          }
   759→
   760→          // Sort by Rogers' distance descending
   761→          geneticDistances.sort((a, b) => b.rogersDistance - a.rogersDistance)
   762→
   763→          // ========================================
   764→          // SUMMARY STATISTICS
   765→          // ========================================
   766→          const summary = {
   767→            totalMarkers: markers.length,
   768→            totalSamples: samples.length,
   769→            avgMaf: mean(polymorphicMarkers.map(m => m.maf)),
   770→            avgPic: mean(polymorphicMarkers.map(m => m.pic)),
   771→            avgHe: mean(polymorphicMarkers.map(m => m.he)),
   772→            polymorphicMarkers: polymorphicMarkers.length,
   773→            significantLDPairs: ldResults.filter(l => l.significant).length,
   774→            significantQTLs: qtlResults.filter(q => q.significant).length
   775→          }
   776→

   778→            markerStats,
   779→            ldResults,
   780→            qtlResults,
   781→            associationResults,
   782→            masResults,
   783→            geneticDistances,
   784→            summary
   785→          })
   786→        } catch (err) {
   787→          setError(`Analysis error: ${err instanceof Error ? err.message : 'Unknown error'}`)
   788→        } finally {
   789→          setIsLoading(false)
   790→        }
   791→      }, 100) // Small delay for UI feedback
   792→    } catch (err) {
   793→      setIsLoading(false)
   794→      setError(`Error: ${err instanceof Error ? err.message : 'Unknown error'}`)
   795→    }

   797→    parsedData,
   798→    mafThreshold,
   799→    ldCutoff,

   801→    calculateAlleleFrequencies,
   802→    calculatePIC,
   803→    calculateExpectedHeterozygosity,
   804→    calculateMAF,
   805→    calculateLD,
   806→    calculateQTLSMA,
   807→    calculateRogersDistance,

   810→  ])
   811→
