    42→export interface BioTool {

    44→  name: string
    45→  description: string
    46→  category: ToolCategoryType
    47→  icon: LucideIcon

  1058→// Helper functions
  1059→export function getAllTools(): BioTool[] {
  1060→  return toolCategories.flatMap((category) => category.tools)
  1061→}
  1062→
  1063→export function getToolById(id: string): BioTool | undefined {
  1064→  return getAllTools().find((tool) => tool.id === id)
  1065→}
  1066→
  1067→export function getToolsByCategory(categoryId: ToolCategoryType): BioTool[] {
  1068→  const category = toolCategories.find((cat) => cat.id === categoryId)
  1069→  return category?.tools || []
  1070→}
  1071→
  1072→export function searchTools(query: string): BioTool[] {
  1073→  const lowerQuery = query.toLowerCase()
  1074→  return getAllTools().filter(
  1075→    (tool) =>
  1076→      tool.name.toLowerCase().includes(lowerQuery) ||
  1077→      tool.description.toLowerCase().includes(lowerQuery) ||
  1078→      tool.tags.some((tag) => tag.toLowerCase().includes(lowerQuery))
  1079→  )
  1080→}
