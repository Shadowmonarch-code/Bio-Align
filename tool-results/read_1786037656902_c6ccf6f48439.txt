   113→function useDragDrop(onFiles: (files: File[]) => void) {
   114→  const [isDragging, setIsDragging] = useState(false);
   115→  const dragCounter = useRef(0);
   116→
   117→  const handleDragEnter = useCallback((e: React.DragEvent) => {
   118→    e.preventDefault();
   119→    e.stopPropagation();
   120→    dragCounter.current++;
   121→    if (e.dataTransfer.items && e.dataTransfer.items.length > 0) {
   122→      setIsDragging(true);
   123→    }
   124→  }, []);
   125→
   126→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   127→    e.preventDefault();
   128→    e.stopPropagation();
   129→    dragCounter.current--;
   130→    if (dragCounter.current === 0) {
   131→      setIsDragging(false);
   132→    }
   133→  }, []);
   134→
   135→  const handleDragOver = useCallback((e: React.DragEvent) => {
   136→    e.preventDefault();
   137→    e.stopPropagation();
   138→  }, []);
   139→
   140→  const handleDrop = useCallback(
   141→    (e: React.DragEvent) => {
   142→      e.preventDefault();
   143→      e.stopPropagation();
   144→      setIsDragging(false);
   145→      dragCounter.current = 0;
   146→
   147→      const files = Array.from(e.dataTransfer.files);
   148→      if (files.length > 0) {
   149→        onFiles(files);
   150→      }
   151→    },
   152→    [onFiles]
   153→  );
   154→
   155→  return {
   156→    isDragging,
   157→    handlers: {
   158→      onDragEnter: handleDragEnter,
   159→      onDragLeave: handleDragLeave,
   160→      onDragOver: handleDragOver,
   161→      onDrop: handleDrop,
   162→    },
   163→  };
   164→}

   639→  const [viewMode, setViewMode] = useState<'grid' | 'list'>('list');
   640→  const [previewFile, setPreviewFile] = useState<UploadedFile | null>(null);
   641→  const fileInputRef = useRef<HTMLInputElement>(null);
   642→
   643→  // Handle new files being added
   644→  const handleFiles = useCallback(async (newFiles: File[]) => {
   645→    const uploadedFiles: UploadedFile[] = [];
   646→
   647→    for (const file of newFiles) {
   648→      const validation = validateFile(file);
   649→      const bioType = detectFileType(file.name);
   650→      let preview: string | undefined;
   651→
   652→      // Try to read preview for text-based files
   653→      if (bioType && !bioType.mimeType.startsWith('application/')) {
   654→        try {
   655→          preview = await readPreviewContent(file);
   656→        } catch {
   657→          preview = undefined;
   658→        }
   659→      }
   660→
   661→      uploadedFiles.push({
   662→        id: generateFileId(),
   663→        name: file.name,
   664→        size: file.size,
   665→        type: file.type || bioType?.mimeType || 'application/octet-stream',
   666→        bioType,
   667→        lastModified: file.lastModified,
   668→        status: validation.valid ? 'uploading' : 'error',
   669→        progress: 0,
   670→        error: validation.errors[0],
   671→        preview,
   672→        file,
   673→      });
   674→    }
   675→
   676→    setFiles((prev) => [...prev, ...uploadedFiles]);
   677→
   678→    // Simulate upload progress for each file
   679→    uploadedFiles.forEach((uploadedFile) => {
   680→      if (uploadedFile.status !== 'error') {
   681→        // Start upload simulation
   682→        let progress = 0;
   683→        const interval = setInterval(() => {
   684→          progress += Math.random() * 30 + 10;
   685→          if (progress >= 100) {
   686→            progress = 100;
   687→            clearInterval(interval);
   688→            setFiles((prev) =>
   689→              prev.map((f) =>
   690→                f.id === uploadedFile.id ? { ...f, status: 'ready' as const, progress: 100 } : f
   691→              )
   692→            );
   693→          } else {
   694→            setFiles((prev) =>
   695→              prev.map((f) =>
   696→                f.id === uploadedFile.id ? { ...f, progress: Math.min(progress, 99) } : f
   697→              )
   698→            );
   699→          }
   700→        }, 200);
   701→      }
   702→    });
   703→  }, []);
   704→
   705→  // Remove a single file
   706→  const removeFile = useCallback((id: string) => {
   707→    setFiles((prev) => prev.filter((f) => f.id !== id));
   708→  }, []);
   709→
   710→  // Clear all files
   711→  const clearAll = useCallback(() => {
   712→    setFiles([]);
   713→  }, []);
   714→
   715→  // Open file browser
   716→  const openFileBrowser = () => {
   717→    fileInputRef.current?.click();
   718→  };
   719→
   720→  // Handle file input change
   721→  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   722→    if (e.target.files && e.target.files.length > 0) {
   723→      handleFiles(Array.from(e.target.files));
   724→      e.target.value = '';
   725→    }
   726→  };
   727→
   728→  // Drag and drop hook
   729→  const { isDragging, handlers } = useDragDrop(handleFiles);
   730→
   731→  // Calculate stats
   732→  const totalSize = files.reduce((acc, f) => acc + f.size, 0);
   733→  const readyCount = files.filter((f) => f.status === 'ready').length;
   734→  const errorCount = files.filter((f) => f.status === 'error').length;
   735→
   736→  return (
   737→    <div className="w-full space-y-6">
   738→      {/* Hidden file input */}
   739→      <input
   740→        ref={fileInputRef}
   741→        type="file"
   742→        multiple
   743→        accept=".fasta,.fa,.fna,.faa,.fastq,.fq,.gb,.genbank,.embl,.pdb,.ent,.cif,.mcif,.mol,.mol2,.sdf,.sam,.bam,.cram,.vcf,.bed,.gff,.gff3,.gtf,.csv,.tsv,.txt,.xlsx,.xls,.json,.xml,.zip,.gz,.gzip,.tar,.tar.gz,.bz2,.rar,.7z"
   744→        onChange={handleInputChange}
   745→        className="hidden"
   746→      />
   747→
   748→      {/* Upload Zone */}
   749→      <motion.div
   750→        {...handlers}
   751→        onClick={openFileBrowser}
   752→        whileHover={{ scale: 1.005 }}
   753→        whileTap={{ scale: 0.995 }}
   754→        animate={isDragging ? { scale: 1.02 } : {}}
   755→        className={cn(
   756→          'relative cursor-pointer rounded-xl border-2 border-dashed p-8 md:p-12 transition-all duration-300',
   757→          'hover:border-[#C1121F]/50 hover:bg-[#C1121F]/[0.02]',
   758→          isDragging
   759→            ? 'border-[#C1121F] bg-[#C1121F]/[0.05]'
   760→            : 'border-border hover:border-[#C1121F]/30'
   761→        )}
   762→      >
   763→        {/* Glassmorphism effect overlay */}
   764→        <div
   765→          className={cn(
   766→            'absolute inset-0 rounded-xl opacity-0 transition-opacity duration-300 pointer-events-none',
   767→            'bg-gradient-to-br from-white/[0.08] to-transparent backdrop-blur-[2px]',
   768→            isDragging && 'opacity-100'
   769→          )}
   770→          style={{
   771→            boxShadow: isDragging
   772→              ? 'inset 0 0 30px rgba(193, 18, 31, 0.1)'
   773→              : undefined,
   774→          }}
   775→        />
   776→
   777→        <div className="relative z-10 flex flex-col items-center justify-center text-center">
   778→          {/* Animated upload icon */}
   779→          <motion.div
   780→            animate={
   781→              isDragging
   782→                ? {
   783→                    scale: [1, 1.2, 1],
   784→                    rotate: [0, 5, -5, 0],
   785→                  }
   786→                : {
   787→                    y: [0, -8, 0],
   788→                  }
   789→            }
   790→            transition={
   791→              isDragging
   792→                ? { duration: 0.5, repeat: Infinity }
   793→                : { duration: 2, repeat: Infinity, ease: 'easeInOut' }
   794→            }
   795→            className="mb-4"
   796→          >
   797→            <div
   798→              className={cn(
   799→                'relative p-4 md:p-6 rounded-full transition-colors duration-300',
   800→                isDragging
   801→                  ? 'bg-[#C1121F]/10'
   802→                  : 'bg-muted hover:bg-[#C1121F]/5'
   803→              )}
   804→            >
   805→              <Upload
   806→                className={cn(
   807→                  'size-8 md:size-10 transition-colors duration-300',
   808→                  isDragging ? 'text-[#C1121F]' : 'text-muted-foreground'
   809→                )}
   810→              />
   811→              
   812→              {/* Pulse effect when dragging */}
   813→              {isDragging && (
   814→                <motion.div
   815→                  className="absolute inset-0 rounded-full border-2 border-[#C1121F]"
   816→                  animate={{
   817→                    scale: [1, 1.5],
   818→                    opacity: [0.5, 0],
   819→                  }}
   820→                  transition={{
   821→                    duration: 1,
   822→                    repeat: Infinity,
   823→                  }}
   824→                />
   825→              )}
   826→            </div>
   827→          </motion.div>
   828→
   829→          {/* Text content */}
   830→          <AnimatePresence mode="wait">
   831→            {isDragging ? (
   832→              <motion.div
   833→                key="dragging"
   834→                initial={{ opacity: 0, y: 10 }}
   835→                animate={{ opacity: 1, y: 0 }}
   836→                exit={{ opacity: 0, y: -10 }}
   837→              >
   838→                <h3 className="text-lg font-semibold text-[#C1121F] mb-1">
   839→                  Drop files here
   840→                </h3>
   841→                <p className="text-sm text-muted-foreground">
   842→                  Release to upload your files
   843→                </p>
   844→              </motion.div>
   845→            ) : (
   846→              <motion.div
   847→                key="idle"
   848→                initial={{ opacity: 0, y: 10 }}
   849→                animate={{ opacity: 1, y: 0 }}
   850→                exit={{ opacity: 0, y: -10 }}
   851→              >
   852→                <h3 className="text-lg font-semibold mb-1">
   853→                  Drag & drop bioinformatics files
   854→                </h3>
   855→                <p className="text-sm text-muted-foreground mb-3">
   856→                  or click to browse your computer
   857→                </p>
   858→                <Button
   859→                  type="button"
   860→                  variant="outline"
   861→                  size="sm"
   862→                  className="gap-2"
   863→                  onClick={(e) => {
   864→                    e.stopPropagation();
   865→                    openFileBrowser();
   866→                  }}
   867→                >
   868→                  <Upload className="size-4" />
   869→                  Browse Files
   870→                </Button>
   871→              </motion.div>
   872→            )}
   873→          </AnimatePresence>
   874→
   875→          {/* Quick format badges */}
   876→          {!isDragging && (
   877→            <motion.div
   878→              initial={{ opacity: 0 }}
   879→              animate={{ opacity: 1 }}
   880→              transition={{ delay: 0.2 }}
   881→              className="mt-6 flex flex-wrap justify-center gap-1.5"
   882→            >
   883→              {['FASTA', 'FASTQ', 'PDB', 'VCF', 'BAM', 'CSV'].map((format) => (
   884→                <Badge
   885→                  key={format}
   886→                  variant="outline"
   887→                  className="text-[10px] font-mono px-1.5 py-0 h-5"
   888→                >
   889→                  .{format.toLowerCase()}
   890→                </Badge>
   891→              ))}
   892→              <Badge variant="outline" className="text-[10px] px-1.5 py-0 h-5">
   893→                +{Object.keys(BIO_FILE_TYPES).length - 6} more
   894→              </Badge>
   895→            </motion.div>
   896→          )}
   897→        </div>
   898→      </motion.div>
   899→
   900→      {/* Supported Formats Section */}
   901→      <SupportedFormats />
   902→
   903→      {/* Files Section */}
   904→      {files.length > 0 && (
   905→        <motion.div
   906→          initial={{ opacity: 0, y: 20 }}
   907→          animate={{ opacity: 1, y: 0 }}
   908→          className="space-y-4"
   909→        >
   910→          {/* Toolbar */}
   911→          <div className="flex items-center justify-between">
   912→            <div className="flex items-center gap-4">
   913→              <h3 className="text-lg font-semibold">
   914→                Uploaded Files{' '}
   915→                <span className="text-muted-foreground font-normal">
   916→                  ({files.length})
   917→                </span>
   918→              </h3>
   919→              
   920→              {/* Stats */}
   921→              <div className="hidden sm:flex items-center gap-3 text-sm text-muted-foreground">
   922→                <span className="flex items-center gap-1">
   923→                  <CheckCircle2 className="size-3.5 text-green-500" />
   924→                  {readyCount} ready
   925→                </span>
   926→                {errorCount > 0 && (
   927→                  <span className="flex items-center gap-1">
   928→                    <AlertCircle className="size-3.5 text-red-500" />
   929→                    {errorCount} errors
   930→                  </span>
   931→                )}
   932→                <span>{formatFileSize(totalSize)} total</span>
   933→              </div>
   934→            </div>
   935→
   936→            <div className="flex items-center gap-2">
   937→              {/* View toggle */}
   938→              <div className="hidden sm:flex items-center border rounded-md p-0.5">
   939→                <Button
   940→                  size="icon"
   941→                  variant={viewMode === 'list' ? 'secondary' : 'ghost'}
   942→                  className="size-7"
   943→                  onClick={() => setViewMode('list')}
   944→                >
   945→                  <List className="size-3.5" />
   946→                </Button>
   947→                <Button
   948→                  size="icon"
   949→                  variant={viewMode === 'grid' ? 'secondary' : 'ghost'}
   950→                  className="size-7"
   951→                  onClick={() => setViewMode('grid')}
   952→                >
   953→                  <Grid3X3 className="size-3.5" />
   954→                </Button>
   955→              </div>
   956→
   957→              {/* Clear all button */}
   958→              <Button
   959→                variant="outline"
   960→                size="sm"
   961→                className="gap-1.5 text-destructive hover:text-destructive"
   962→                onClick={clearAll}
   963→              >
   964→                <Trash2 className="size-3.5" />
   965→                Clear All
   966→              </Button>
   967→            </div>
   968→          </div>
   969→
   970→          {/* Mobile stats */}
   971→          <div className="sm:hidden flex items-center gap-3 text-xs text-muted-foreground">
   972→            <span className="flex items-center gap-1">
   973→              <CheckCircle2 className="size-3 text-green-500" />
   974→              {readyCount} ready
   975→            </span>
   976→            {errorCount > 0 && (
   977→              <span className="flex items-center gap-1">
   978→                <AlertCircle className="size-3 text-red-500" />
   979→                {errorCount} errors
   980→              </span>
   981→            )}
   982→            <span>{formatFileSize(totalSize)}</span>
   983→          </div>
   984→
   985→          {/* File list/grid */}
   986→          <AnimatePresence mode="popLayout">
   987→            {viewMode === 'grid' ? (
   988→              <motion.div
   989→                layout
   990→                className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4"
   991→              >
   992→                <AnimatePresence>
   993→                  {files.map((file) => (
   994→                    <FileItem
   995→                      key={file.id}
   996→                      file={file}
   997→                      onRemove={removeFile}
   998→                      onPreview={setPreviewFile}
   999→                      viewMode={viewMode}
  1000→                    />
  1001→                  ))}
  1002→                </AnimatePresence>
  1003→              </motion.div>
  1004→            ) : (
  1005→              <motion.div
  1006→                layout
  1007→                className="space-y-2"
  1008→              >
  1009→                <AnimatePresence>
  1010→                  {files.map((file) => (
  1011→                    <FileItem
  1012→                      key={file.id}
  1013→                      file={file}
  1014→                      onRemove={removeFile}
  1015→                      onPreview={setPreviewFile}
  1016→                      viewMode={viewMode}
  1017→                    />
  1018→                  ))}
  1019→                </AnimatePresence>
  1020→              </motion.div>
  1021→            )}
  1022→          </AnimatePresence>
  1023→        </motion.div>
  1024→      )}
  1025→
  1026→      {/* Empty State */}
  1027→      {files.length === 0 && <EmptyState />}
  1028→
  1029→      {/* Preview Dialog */}
  1030→      <AnimatePresence>
  1031→        {previewFile && (
  1032→          <PreviewDialog file={previewFile} onClose={() => setPreviewFile(null)} />
  1033→        )}
  1034→      </AnimatePresence>
  1035→    </div>
  1036→  );
  1037→}
