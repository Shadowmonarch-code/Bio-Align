   163→   163→
   164→   164→  // Parse genotypic data
   165→   165→  const parseData = useCallback((text: string): boolean => {
   166→   166→    try {
   167→   167→      setError(null)
   168→   168→      const lines = text.trim().split('\n').filter(line => line.trim())
   169→   169→      
   170→   170→      if (lines.length < 3) {
   171→   171→        setError('Data must have at least a header row and 2 sample rows')
   172→   172→        return false
   173→   173→      }
   174→   174→
   175→   175→      const header = lines[0].split(',').map(h => h.trim())
   176→   176→      const samples: string[] = []
   177→   177→      const markers: string[] = header.slice(1, -1) // Exclude Sample and optional Trait column
   178→   178→      const genotypes: string[][] = []
   179→   179→      let traitValues: number[] = []
   180→   180→
   181→   181→      for (let i = 1; i < lines.length; i++) {
   182→   182→        const cols = lines[i].split(',').map(c => c.trim())
   183→   183→        if (cols.length < header.length) continue
   184→   184→        
   185→   185→        samples.push(cols[0])
   186→   186→        const genoRow = cols.slice(1, -1) // Genotypes (exclude sample name and trait)
   187→   187→        genotypes.push(genoRow)
   188→   188→        
   189→   189→        // Parse trait value if present
   190→   190→        if (cols.length > header.length - 1) {
   191→   191→          const traitVal = parseFloat(cols[cols.length - 1])
   192→   192→          if (!isNaN(traitVal)) {
   193→   193→            traitValues.push(traitVal)
   194→   194→          }
   195→   195→        }
   196→   196→      }
   197→   197→
   198→   198→      if (samples.length === 0) {
   199→   199→        setError('No valid sample data found')
   200→   200→        return false
   201→   201→      }
   202→   202→
   203→   203→      setParsedData({
   204→   204→        samples,
   205→   205→        markers,
   206→   206→        genotypes,
   207→   207→        traitValues: traitValues.length === samples.length ? traitValues : undefined
   208→   208→      })
   209→   209→      return true
   210→   210→    } catch (err) {
   211→   211→      setError(`Parse error: ${err instanceof Error ? err.message : 'Unknown error'}`)
   212→   212→      return false
   213→   213→    }
   214→   214→  }, [])
   215→   215→
   216→   216→  // Load sample data
   217→   217→  const loadSampleData = useCallback(() => {

   221→   221→
   222→   222→  // Calculate allele frequencies for a single marker
   223→   223→  const calculateAlleleFrequencies = useCallback(
   224→   224→    (genotypes: string[]): { freq: Record<string, number>; ho: number } => {
   225→   225→      const alleleCounts: Record<string, number> = {}
   226→   226→      let heteroCount = 0
   227→   227→      const n = genotypes.length
   228→   228→
   229→   229→      for (const geno of genotypes) {
   230→   230→        if (geno.length !== 2) continue
   231→   231→        
   232→   232→        const a1 = geno[0]
   233→   233→        const a2 = geno[1]
   234→   234→        
   235→   235→        alleleCounts[a1] = (alleleCounts[a1] || 0) + 1
   236→   236→        alleleCounts[a2] = (alleleCounts[a2] || 0) + 1
   237→   237→        
   238→   238→        if (a1 !== a2) {
   239→   239→          heteroCount++
   240→   240→        }
   241→   241→      }
   242→   242→
   243→   243→      const totalAlleles = 2 * n
   244→   244→      const freq: Record<string, number> = {}
   245→   245→      
   246→   246→      for (const [allele, count] of Object.entries(alleleCounts)) {
   247→   247→        freq[allele] = count / totalAlleles
   248→   248→      }
   249→   249→
   250→   250→      return { freq, ho: heteroCount / n }
   251→   251→    },
   252→   252→    []
   253→   253→  )
   254→   254→
   255→   255→  // Calculate PIC (Polymorphism Information Content) - Botstein et al. (1980)
   256→   256→  const calculatePIC = useCallback((freq: Record<string, number>): number => {
   257→   257→    const alleles = Object.keys(freq)
   258→   258→    let pic = 1
   259→   259→    
   260→   260→    // PIC = 1 - Σp_i² - 2ΣΣ(p_i * p_j)² for i < j
   261→   261→    for (let i = 0; i < alleles.length; i++) {
   262→   262→      const pi = freq[alleles[i]]
   263→   263→      pic -= pi * pi
   264→   264→      
   265→   265→      for (let j = i + 1; j < alleles.length; j++) {
   266→   266→        const pj = freq[alleles[j]]
   267→   267→        pic -= 2 * pi * pi * pj * pj
   268→   268→      }
   269→   269→    }
   270→   270→    
   271→   271→    return Math.max(0, pic)
   272→   272→  }, [])
   273→   273→
   274→   274→  // Calculate expected heterozygosity (gene diversity) - Nei (1973)
   275→   275→  const calculateExpectedHeterozygosity = useCallback((freq: Record<string, number>): number => {
   276→   276→    let he = 1
   277→   277→    for (const p of Object.values(freq)) {
   278→   278→      he -= p * p
   279→   279→    }
   280→   280→    return Math.max(0, he)
   281→   281→  }, [])
   282→   282→
   283→   283→  // Calculate MAF (Minor Allele Frequency)
   284→   284→  const calculateMAF = useCallback((freq: Record<string, number>): number => {
   285→   285→    const frequencies = Object.values(freq)
   286→   286→    return Math.min(...frequencies)
   287→   287→  }, [])
   288→   288→
   289→   289→  // Calculate Linkage Disequilibrium - Hill & Weir (1988)
   290→   290→  const calculateLD = useCallback(
   291→   291→    (
   292→   292→      genotypes1: string[],
   293→   293→      genotypes2: string[],
   294→   294→      alpha: number = 0.05
   295→   295→    ): LDResult => {
   296→   296→      const n = genotypes1.length
   297→   297→      
   298→   298→      // Get alleles for each marker
   299→   299→      const alleles1 = new Set<string>()
   300→   300→      const alleles2 = new Set<string>()
   301→   301→      
   302→   302→      for (const g of genotypes1) {
   303→   303→        alleles1.add(g[0])
   304→   304→        alleles1.add(g[1])
   305→   305→      }
   306→   306→      for (const g of genotypes2) {
   307→   307→        alleles2.add(g[0])
   308→   308→        alleles2.add(g[1])
   309→   309→      }
   310→   310→
   311→   311→      const aArr = Array.from(alleles1).sort()
   312→   312→      const bArr = Array.from(alleles2).sort()
   313→   313→      
   314→   314→      // Use first allele as reference for biallelic case
   315→   315→      const A = aArr[0] || 'A'
   316→   316→      const B = bArr[0] || 'A'
   317→   317→
   318→   318→      // Count haplotype-like frequencies from genotypes
   319→   319→      // For diploid data, we estimate gamete frequencies using EM or simple counting
   320→   320→      let nAB = 0, nAb = 0, naB = 0, nab = 0
   321→   321→      
   322→   322→      for (let i = 0; i < n; i++) {
   323→   323→        const g1 = genotypes1[i]
   324→   324→        const g2 = genotypes2[i]
   325→   325→        
   326→   326→        // Count allele occurrences
   327→   327→        const g1hasA = g1.includes(A)
   328→   328→        const g2hasB = g2.includes(B)
   329→   329→        
   330→   330→        // Simplified: count based on presence/absence
   331→   331→        if (g1hasA && g2hasB) nAB++
   332→   332→        if (g1hasA && !g2hasB) nAb++
   333→   333→        if (!g1hasA && g2hasB) naB++
   334→   334→        if (!g1hasA && !g2hasB) nab++
   335→   335→      }
   336→   336→
   337→   337→      // Allele frequencies
   338→   338→      const pA = (nAB + nAb) / (2 * n)
   339→   339→      const pa = 1 - pA
   340→   340→      const pB = (nAB + naB) / (2 * n)
   341→   341→      const pb = 1 - pB
   342→   342→
   343→   343→      // Expected frequency of AB under independence
   344→   344→      const expectedAB = pA * pB
   345→   345→      const observedAB = nAB / n
   346→   346→      
   347→   347→      // D coefficient
   348→   348→      const D = observedAB - expectedAB
   349→   349→      
   350→   350→      // r² = D² / (pA * pa * pB * pb)
   351→   351→      const denom = pA * pa * pB * pb
   352→   352→      const r2 = denom > 0 ? (D * D) / denom : 0
   353→   353→      
   354→   354→      // D' (standardized disequilibrium coefficient)
   355→   355→      // D' = D / D_max where D_max = min(pA*pb, pa*pB) if D > 0
   356→   356→      //                   D_max = min(pA*pB, pa*pb) if D < 0
   357→   357→      let dMax: number
   358→   358→      if (D > 0) {
   359→   359→        dMax = Math.min(pA * pb, pa * pB)
   360→   360→      } else {
   361→   361→        dMax = Math.min(pA * pB, pa * pb)
   362→   362→      }
   363→   363→      const dPrime = dMax > 0 ? Math.abs(D) / dMax : 0
   364→   364→
   365→   365→      // Chi-square test for LD significance
   366→   366→      // Contingency table: 2x2 (presence/absence of each allele)
   367→   367→      const contingencyTable = [
   368→   368→        [nAB, nAb],
   369→   369→        [naB, nab]
   370→   370→      ]
   371→   371→      
   372→   372→      const chiSqResult = chiSquareTest(contingencyTable)
   373→   373→
   374→   374→      return {
   375→   375→        marker1: '',
   376→   376→        marker2: '',
   377→   377→        r2: Math.min(1, r2),
   378→   378→        dPrime: Math.min(1, dPrime),
   379→   379→        dValue: D,
   380→   380→        chiSquare: chiSqResult.chiSquare,
   381→   381→        pValue: chiSqResult.pValue,
   382→   382→        significant: chiSqResult.pValue < alpha
   383→   383→      }
   384→   384→    },
   385→   385→    []
   386→   386→  )
   387→   387→
   388→   388→  // Single Marker Analysis (SMA) for QTL mapping
   389→   389→  const calculateQTLSMA = useCallback(
   390→   390→    (
   391→   391→      genotypes: string[],
   392→   392→      traitValues: number[],
   393→   393→      alpha: number = 0.05
   394→   394→    ): QTLResult => {
   395→   395→      // Group trait values by genotype
   396→   396→      const groups: Record<string, number[]> = {}
   397→   397→      
   398→   398→      for (let i = 0; i < genotypes.length; i++) {
   399→   399→        const geno = genotypes[i]
   400→   400→        if (!groups[geno]) {
   401→   401→          groups[geno] = []
   402→   402→        }
   403→   403→        groups[geno].push(traitValues[i])
   404→   404→      }
   405→   405→
   406→   406→      const groupKeys = Object.keys(groups)
   407→   407→      const k = groupKeys.length
   408→   408→      const n = traitValues.length
   409→   409→
   410→   410→      if (k < 2) {
   411→   411→        return {
   412→   412→          marker: '',
   413→   413→          fStatistic: 0,
   414→   414→          pValue: 1,
   415→   415→          rSquared: 0,
   416→   416→          lodScore: 0,
   417→   417→          effect: 0,
   418→   418→          significant: false
   419→   419→        }
   420→   420→      }
   421→   421→
   422→   422→      // One-way ANOVA
   423→   423→      const grandMean = mean(traitValues)
   424→   424→      
   425→   425→      // Sum of squares between groups
   426→   426→      let ssb = 0
   427→   427→      const groupMeans: number[] = []
   428→   428→      
   429→   429→      for (const key of groupKeys) {
   430→   430→        const groupData = groups[key]
   431→   431→        const groupMean = mean(groupData)
   432→   432→        groupMeans.push(groupMean)
   433→   433→        ssb += groupData.length * (groupMean - grandMean) ** 2
   434→   434→      }
   435→   435→      
   436→   436→      // Sum of squares within groups (error)
   437→   437→      let ssw = 0
   438→   438→      for (const key of groupKeys) {
   439→   439→        const groupData = groups[key]
   440→   440→        const groupMean = mean(groupData)
   441→   441→        for (const val of groupData) {
   442→   442→          ssw += (val - groupMean) ** 2
   443→   443→        }
   444→   444→      }
   445→   445→
   446→   446→      // Degrees of freedom
   447→   447→      const dfBetween = k - 1
   448→   448→      const dfWithin = n - k
   449→   449→
   450→   450→      // Mean squares
   451→   451→      const msb = ssb / dfBetween
   452→   452→      const msw = ssw / dfWithin
   453→   453→
   454→   454→      // F-statistic
   455→   455→      const fStat = msw > 0 ? msb / msw : 0
   456→   456→
   457→   457→      // R-squared (proportion of variance explained)
   458→   458→      const ssTotal = ssb + ssw
   459→   459→      const rSquared = ssTotal > 0 ? ssb / ssTotal : 0
   460→   460→
   461→   461→      // Approximate p-value using F-distribution
   462→   462→      // Using chi-square approximation for simplicity
   463→   463→      const pValueApprox = 1 - chiSquareCDF(fStat * dfBetween, dfBetween)
   464→   464→
   465→   465→      // LOD score approximation: LOD = (n/2) * log10(1 - R²) for null vs alternative
   466→   466→      // Or LOD ≈ (n * R²) / (2 * ln(10) * (1-R²))
   467→   467→      const lodScore = rSquared < 1 
   468→   468→        ? (n * rSquared) / (2 * Math.LN10 * (1 - rSquared + 0.001))
   469→   469→        : 0
   470→   470→
   471→   471→      // Effect size (difference between extreme genotype means)
   472→   472→      const sortedMeans = [...groupMeans].sort((a, b) => a - b)
   473→   473→      const effect = sortedMeans[sortedMeans.length - 1] - sortedMeans[0]
   474→   474→
   475→   475→      return {
   476→   476→        marker: '',
   477→   477→        fStatistic: fStat,
   478→   478→        pValue: Math.max(0, Math.min(1, pValueApprox)),
   479→   479→        rSquared: Math.min(1, rSquared),
   480→   480→        lodScore: Math.min(20, lodScore),
   481→   481→        effect,
   482→   482→        significant: pValueApprox < alpha
   483→   483→      }
   484→   484→    },
   485→   485→    []
   486→   486→  )
   487→   487→
   488→   488→  // Calculate Rogers' genetic distance
   489→   489→  const calculateRogersDistance = useCallback(
   490→   490→    (genos1: string[], genos2: string[]): number => {
   491→   491→      let sumDiff = 0
   492→   492→      let validLoci = 0
   493→   493→
   494→   494→      for (let i = 0; i < genos1.length; i++) {
   495→   495→        const g1 = genos1[i]
   496→   496→        const g2 = genos2[i]
   497→   497→        
   498→   498→        if (g1.length !== 2 || g2.length !== 2) continue
   499→   499→        
   500→   500→        // Count shared alleles
   501→   501→        let sharedAlleles = 0
   502→   502→        if (g1[0] === g2[0] || g1[0] === g2[1]) sharedAlleles++
   503→   503→        if (g1[1] === g2[0] || g1[1] === g2[1]) sharedAlleles++
   504→   504→        
   505→   505→        // Rogers' distance contribution: sqrt(0.5 * (1 - proportionShared))
   506→   506→        const propShared = sharedAlleles / 2
   507→   507→        sumDiff += 0.5 * (1 - propShared)
   508→   508→        validLoci++
   509→   509→      }
   510→   510→
   511→   511→      return validLoci > 0 ? Math.sqrt(sumDiff / validLoci) : 0
   512→   512→    },
   513→   513→    []
   514→   514→  )
   515→   515→
   516→   516→  // Calculate Jaccard similarity coefficient
   517→   517→  const calculateJaccardSimilarity = useCallback(
   518→   518→    (genos1: string[], genos2: string[]): number => {
   519→   519→      let intersection = 0
   520→   520→      let union = 0
   521→   521→
   522→   522→      for (let i = 0; i < genos1.length; i++) {
   523→   523→        const alleles1 = new Set([genos1[i][0], genos1[i][1]])
   524→   524→        const alleles2 = new Set([genos2[i][0], genos2[i][1]])
   525→   525→        
   526→   526→        // Intersection
   527→   527→        let interCount = 0
   528→   528→        for (const a of alleles1) {
   529→   529→          if (alleles2.has(a)) interCount++
   530→   530→        }
   531→   531→        intersection += interCount
   532→   532→        
   533→   533→        // Union
   534→   534→        union += new Set([...alleles1, ...alleles2]).size
   535→   535→      }
   536→   536→
   537→   537→      return union > 0 ? intersection / union : 0
   538→   538→    },
   539→   539→    []
   540→   540→  )
   541→   541→
   542→   542→  // Calculate Nei's genetic distance (1972)
   543→   543→  const calculateNeiDistance = useCallback(
   544→   544→    (genos1: string[], genos2: string[]): number => {
   545→   545→      let sumLog = 0
   546→   546→      let validLoci = 0
   547→   547→
   548→   548→      for (let i = 0; i < genos1.length; i++) {
   549→   549→        const alleles1 = [genos1[i][0], genos1[i][1]]
   550→   550→        const alleles2 = [genos2[i][0], genos2[i][1]]
   551→   551→        
   552→   552→        // Calculate allele frequencies for each population (individual as pop)
   553→   553→        const freq1: Record<string, number> = {}
   554→   554→        const freq2: Record<string, number> = {}
   555→   555→        
   556→   556→        for (const a of alleles1) {
   557→   557→          freq1[a] = (freq1[a] || 0) + 0.5
   558→   558→        }
   559→   559→        for (const a of alleles2) {
   560→   560→          freq2[a] = (freq2[a] || 0) + 0.5
   561→   561→        }
   562→   562→
   563→   563→        // Nei's distance: -ln(sum(sqrt(p_i * q_i)))
   564→   564→        const allAlleles = new Set([...Object.keys(freq1), ...Object.keys(freq2)])
   565→   565→        let identitySum = 0
   566→   566→        
   567→   567→        for (const allele of allAlleles) {
   568→   568→          identitySum += Math.sqrt((freq1[allele] || 0) * (freq2[allele] || 0))
   569→   569→        }
   570→   570→
   571→   571→        if (identitySum > 0) {
   572→   572→          sumLog += Math.log(identitySum)
   573→   573→          validLoci++
   574→   574→        }
   575→   575→      }
   576→   576→
   577→   577→      return validLoci > 0 ? -(sumLog / validLoci) : 0
   578→   578→    },
   579→   579→    []
   580→   580→  )

   582→   582→  // Run complete analysis
   583→   583→  const runAnalysis = useCallback(() => {
   584→   584→    if (!parsedData) {
   585→   585→      setError('Please parse data before running analysis')
   586→   586→      return
   587→   587→    }
   588→   588→
   589→   589→    setIsLoading(true)
   590→   590→    setError(null)
   591→   591→

   593→   593→      setTimeout(() => {
   594→   594→        try {
   595→   595→          const { samples, markers, genotypes, traitValues } = parsedData
   596→   596→          
   597→   597→          // ========================================
   598→   598→          // 1. MARKER-BASED GENETIC ANALYSIS
   599→   599→          // ========================================
   600→   600→          const markerStats: MarkerStats[] = []
   601→   601→
   602→   602→          for (let j = 0; j < markers.length; j++) {
   603→   603→            const markerGenotypes = genotypes.map(g => g[j])
   604→   604→            const { freq, ho } = calculateAlleleFrequencies(markerGenotypes)
   605→   605→            
   606→   606→            const genotypeCounts: Record<string, number> = {}
   607→   607→            for (const g of markerGenotypes) {
   608→   608→              genotypeCounts[g] = (genotypeCounts[g] || 0) + 1
   609→   609→            }
   610→   610→
   611→   611→            markerStats.push({
   612→   612→              marker: markers[j],
   613→   613→              alleleFreq: freq,
   614→   614→              maf: calculateMAF(freq),
   615→   615→              pic: calculatePIC(freq),
   616→   616→              he: calculateExpectedHeterozygosity(freq),
   617→   617→              ho,
   618→   618→              genotypeCounts
   619→   619→            })
   620→   620→          }
   621→   621→
   622→   622→          // Filter by MAF threshold
   623→   623→          const polymorphicMarkers = markerStats.filter(m => m.maf >= mafThreshold)
   624→   624→
   625→   625→          // ========================================
   626→   626→          // 2. LINKAGE DISEQUILIBRIUM ANALYSIS
   627→   627→          // ========================================
   628→   628→          const ldResults: LDResult[] = []
   629→   629→
   630→   630→          for (let i = 0; i < markers.length; i++) {
   631→   631→            for (let j = i + 1; j < markers.length; j++) {
   632→   632→              const genos1 = genotypes.map(g => g[i])
   633→   633→              const genos2 = genotypes.map(g => g[j])
   634→   634→              
   635→   635→              const ldResult = calculateLD(genos1, genos2, alphaLevel)
   636→   636→              ldResult.marker1 = markers[i]
   637→   637→              ldResult.marker2 = markers[j]
   638→   638→              
   639→   639→              if (ldResult.r2 >= ldCutoff || ldResult.significant) {
   640→   640→                ldResults.push(ldResult)
   641→   641→              }
   642→   642→            }
   643→   643→          }
   644→   644→
   645→   645→          // Sort by r² descending
   646→   646→          ldResults.sort((a, b) => b.r2 - a.r2)
   647→   647→
   648→   648→          // ========================================
   649→   649→          // 3. QTL MAPPING (if trait values available)
   650→   650→          // ========================================
   651→   651→          const qtlResults: QTLResult[] = []
   652→   652→
   653→   653→          if (traitValues && traitValues.length === samples.length) {
   654→   654→            for (let j = 0; j < markers.length; j++) {
   655→   655→              const markerGenotypes = genotypes.map(g => g[j])
   656→   656→              const qtlResult = calculateQTLSMA(markerGenotypes, traitValues, alphaLevel)
   657→   657→              qtlResult.marker = markers[j]
   658→   658→              qtlResults.push(qtlResult)
   659→   659→            }
   660→   660→
   661→   661→            // Sort by R² descending
   662→   662→            qtlResults.sort((a, b) => b.rSquared - a.rSquared)
   663→   663→          }
   664→   664→
   665→   665→          // ========================================
   666→   666→          // 4. ASSOCIATION MAPPING METRICS
   667→   667→          // ========================================
   668→   668→          const associationResults: AssociationResult[] = []
   669→   669→
   670→   670→          if (traitValues && traitValues.length === samples.length) {
   671→   671→            for (let j = 0; j < markers.length; j++) {
   672→   672→              const markerGenotypes = genotypes.map(g => g[j])
   673→   673→              const qtlResult = calculateQTLSMA(markerGenotypes, traitValues, alphaLevel)
   674→   674→              
   675→   675→              // GLM association (similar to SMA but with different interpretation)
   676→   676→              const glmPvalue = qtlResult.pValue
   677→   677→              const glmEffect = qtlResult.effect
   678→   678→              
   679→   679→              // MLM approximation (inflate p-values slightly to account for structure)
   680→   680→              const inflationFactor = 1.2 // Simple approximation
   681→   681→              const mlmPvalue = Math.min(1, glmPvalue * inflationFactor)
   682→   682→              const mlmEffect = glmEffect * 0.9
   683→   683→
   684→   684→              associationResults.push({
   685→   685→                marker: markers[j],
   686→   686→                glmPvalue,
   687→   687→                glmEffect,
   688→   688→                mlmPvalue,
   689→   689→                mlmEffect,
   690→   690→                significant: glmPvalue < alphaLevel
   691→   691→              })
   692→   692→            }
   693→   693→          }
   694→   694→
   695→   695→          // ========================================
   696→   696→          // 5. MARKER-ASSISTED SELECTION (MAS)
   697→   697→          // ========================================
   698→   698→          const masResults: MASResult[] = []
   699→   699→
   700→   700→          if (qtlResults.length > 0) {
   701→   701→            for (const qtl of qtlResults) {
   702→   702→              // Selection efficiency based on R² and significance
   703→   703→              const efficiency = qtl.rSquared * (qtl.significant ? 1 : 0.5)
   704→   704→              
   705→   705→              // Association strength categorization
   706→   706→              let strength: string
   707→   707→              if (qtl.rSquared > 0.3) strength = 'Very Strong'
   708→   708→              else if (qtl.rSquared > 0.2) strength = 'Strong'
   709→   709→              else if (qtl.rSquared > 0.1) strength = 'Moderate'
   710→   710→              else if (qtl.rSquared > 0.05) strength = 'Weak'
   711→   711→              else strength = 'Negligible'
   712→   712→
   713→   713→              // Expected genetic gain with MAS
   714→   714→              // ΔG_MAS = i * h²_MAS * σ_P where h²_MAS is mediated through marker
   715→   715→              const expectedGain = qtl.effect * qtl.rSquared
   716→   716→
   717→   717→              // Cost-benefit assessment
   718→   718→              let costBenefit: string
   719→   719→              if (qtl.significant && qtl.rSquared > 0.15) {
   720→   720→                costBenefit = 'Highly Recommended'
   721→   721→              } else if (qtl.significant && qtl.rSquared > 0.08) {
   722→   722→                costBenefit = 'Recommended'
   723→   723→              } else if (qtl.rSquared > 0.03) {
   724→   724→                costBenefit = 'Optional'
   725→   725→              } else {
   726→   726→                costBenefit = 'Not Recommended'
   727→   727→              }
   728→   728→
   729→   729→              masResults.push({
   730→   730→                marker: qtl.marker,
   731→   731→                selectionEfficiency: efficiency,
   732→   732→                associationStrength: strength,
   733→   733→                expectedGain,
   734→   734→                recommended: qtl.significant && qtl.rSquared > 0.08,
   735→   735→                costBenefit
   736→   736→              })
   737→   737→            }
   738→   738→          }
   739→   739→
   740→   740→          // ========================================
   741→   741→          // 6. GENETIC DISTANCE & SIMILARITY
   742→   742→          // ========================================
   743→   743→          const geneticDistances: GeneticDistanceResult[] = []
   744→   744→
   745→   745→          for (let i = 0; i < samples.length; i++) {
   746→   746→            for (let j = i + 1; j < samples.length; j++) {
   747→   747→              const genos1 = genotypes[i]
   748→   748→              const genos2 = genotypes[j]
   749→   749→
   750→   750→              geneticDistances.push({
   751→   751→                sample1: samples[i],
   752→   752→                sample2: samples[j],
   753→   753→                rogersDistance: calculateRogersDistance(genos1, genos2),
   754→   754→                jaccardSimilarity: calculateJaccardSimilarity(genos1, genos2),
   755→   755→                neiDistance: calculateNeiDistance(genos1, genos2)
   756→   756→              })
   757→   757→            }
   758→   758→          }
   759→   759→
   760→   760→          // Sort by Rogers' distance descending
   761→   761→          geneticDistances.sort((a, b) => b.rogersDistance - a.rogersDistance)
   762→   762→
   763→   763→          // ========================================
   764→   764→          // SUMMARY STATISTICS
   765→   765→          // ========================================
   766→   766→          const summary = {
   767→   767→            totalMarkers: markers.length,
   768→   768→            totalSamples: samples.length,
   769→   769→            avgMaf: mean(polymorphicMarkers.map(m => m.maf)),
   770→   770→            avgPic: mean(polymorphicMarkers.map(m => m.pic)),
   771→   771→            avgHe: mean(polymorphicMarkers.map(m => m.he)),
   772→   772→            polymorphicMarkers: polymorphicMarkers.length,
   773→   773→            significantLDPairs: ldResults.filter(l => l.significant).length,
   774→   774→            significantQTLs: qtlResults.filter(q => q.significant).length
   775→   775→          }
   776→   776→
   777→   777→          setResults({
   778→   778→            markerStats,
   779→   779→            ldResults,
   780→   780→            qtlResults,
   781→   781→            associationResults,
   782→   782→            masResults,
   783→   783→            geneticDistances,
   784→   784→            summary
   785→   785→          })
   786→   786→        } catch (err) {
   787→   787→          setError(`Analysis error: ${err instanceof Error ? err.message : 'Unknown error'}`)
   788→   788→        } finally {
   789→   789→          setIsLoading(false)
   790→   790→        }
   791→   791→      }, 100) // Small delay for UI feedback
   792→   792→    } catch (err) {
   793→   793→      setIsLoading(false)
   794→   794→      setError(`Error: ${err instanceof Error ? err.message : 'Unknown error'}`)
   795→   795→    }
   796→   796→  }, [
   797→   797→    parsedData,
   798→   798→    mafThreshold,
   799→   799→    ldCutoff,
   800→   800→    alphaLevel,
   801→   801→    calculateAlleleFrequencies,
