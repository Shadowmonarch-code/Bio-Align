   131→   131→export default function SequenceAnalysisTool({ selectedTool }: SequenceAnalysisToolProps) {
   132→   132→  // State
   133→   133→  // Map tool IDs from sidebar to internal tool IDs
   134→   134→  const getInitialTool = (): ToolId => {
   135→   135→    if (!selectedTool) return 'pairwise';
   136→   136→    const toolMap: Record<string, ToolId> = {
   137→   137→      'sequence': 'pairwise',
   138→   138→      'blast': 'blast',
   139→   139→      'orf': 'orf',
   140→   140→      'reverse-complement': 'reverse-complement',
   141→   141→      'translate': 'translation',
   142→   142→      'gc-content': 'gc-content',
   143→   143→      'motif-search': 'motif-search',
   144→   144→      'pattern-search': 'pattern-search',
   145→   145→      'restriction-map': 'restriction-map',
   146→   146→    };
   147→   147→    return toolMap[selectedTool] || 'pairwise';
   148→   148→  };
   149→   149→  
   150→   150→  const [activeTool, setActiveTool] = useState<ToolId>(getInitialTool);
   151→   151→  const [sequence1, setSequence1] = useState('');
   152→   152→  const [sequence2, setSequence2] = useState('');
   153→   153→  const [isAnalyzing, setIsAnalyzing] = useState(false);
   154→   154→  const [progress, setProgress] = useState(0);
   155→   155→  const [result, setResult] = useState<AnalysisResult | null>(null);
   156→   156→  const [showAdvanced, setShowAdvanced] = useState(false);
   157→   157→  const [copied, setCopied] = useState(false);
   158→   158→  
   159→   159→  // Tool-specific options state
   160→   160→  const [alignmentOptions, setAlignmentOptions] = useState({
   161→   161→    algorithm: 'global' as 'global' | 'local',
   162→   162→    matchScore: 1,
   163→   163→    mismatchPenalty: -1,
   164→   164→    gapPenalty: -1,
   165→   165→    matrix: 'BLOSUM62',
   166→   166→  });
   167→   167→
   168→   168→  const [blastOptions, setBlastOptions] = useState({
   169→   169→    program: 'blastn',
   170→   170→    database: 'nr',
   171→   171→    eValue: '0.001',
   172→   172→    maxTargets: 10,
   173→   173→    wordSize: 11,
   174→   174→    lowComplexityFilter: true,
   175→   175→  });
   176→   176→
   177→   177→  const [orfOptions, setOrfOptions] = useState({
   178→   178→    geneticCode: 1,
   179→   179→    minLength: 30,
   180→   180→    showAllFrames: true,
   181→   181→    startCodons: ['ATG'],
   182→   182→  });
   183→   183→
   184→   184→  const [translationOptions, setTranslationOptions] = useState({
   185→   185→    frame: 1,
   186→   186→    geneticCode: 1,
   187→   187→  });
   188→   188→
   189→   189→  const [motifPattern, setMotifPattern] = useState('');
   190→   190→  const [patternSearchOptions, setPatternSearchOptions] = useState({
   191→   191→    caseSensitive: false,
   192→   192→    allowMismatch: false,
   193→   193→    maxMismatches: 0,
   194→   194→  });
   195→   195→
   196→   196→  // Get sequence stats for display
   197→   197→  const seq1Stats = useMemo(() => 
   198→   198→    sequence1 ? getSequenceStats(sequence1) : null, [sequence1]
   199→   199→  );
   200→   200→
   201→   201→  const seq2Stats = useMemo(() => 
   202→   202→    sequence2 ? getSequenceStats(sequence2) : null, [sequence2]
   203→   203→  );
   204→   204→
   205→   205→  // Load example sequences
   206→   206→  const loadExampleDNA = useCallback(() => {
   207→   207→    setSequence1(EXAMPLE_SEQUENCES.dna);
   208→   208→  }, []);
   209→   209→
   210→   210→  const loadExampleProtein = useCallback(() => {
   211→   211→    setSequence1(EXAMPLE_SEQUENCES.protein);
   212→   212→  }, []);
   213→   213→
   214→   214→  // Handle file upload
   215→   215→  const handleFileUpload = useCallback((event: React.ChangeEvent<HTMLInputElement>) => {
   216→   216→    const file = event.target.files?.[0];
   217→   217→    if (file) {
   218→   218→      const reader = new FileReader();
   219→   219→      reader.onload = (e) => {
   220→   220→        const content = e.target?.result as string;
   221→   221→        if (content.startsWith('>')) {
   222→   222→          const parsed = parseFASTA(content);
   223→   223→          if (parsed.length > 0) {
   224→   224→            setSequence1(parsed[0].sequence);
   225→   225→            if (parsed.length > 1) {
   226→   226→              setSequence2(parsed[1].sequence);
   227→   227→            }
   228→   228→          }
   229→   229→        } else {
   230→   230→          setSequence1(content);
   231→   231→        }
   232→   232→      };
   233→   233→      reader.readAsText(file);
   234→   234→    }
   235→   235→  }, []);
   236→   236→
   237→   237→  // Run analysis based on active tool
   238→   238→  const runAnalysis = useCallback(async () => {
   239→   239→    setIsAnalyzing(true);
   240→   240→    setProgress(0);
   241→   241→    setResult(null);
   242→   242→
   243→   243→    // Simulate progress
   244→   244→    const progressInterval = setInterval(() => {
   245→   245→      setProgress(prev => Math.min(prev + Math.random() * 15, 90));
   246→   246→    }, 200);
   247→   247→
   248→   248→    // Simulate processing delay
   249→   249→    await new Promise(resolve => setTimeout(resolve, 1000 + Math.random() * 1500));
   250→   250→
   251→   251→    let analysisResult: AnalysisResult;
   252→   252→
   253→   253→    try {
   254→   254→      switch (activeTool) {
   255→   255→        case 'pairwise': {
   256→   256→          const alignFn = alignmentOptions.algorithm === 'global' ? globalAlignment : localAlignment;
   257→   257→          const alignmentResult = alignFn(sequence1, sequence2, {
   258→   258→            matchScore: alignmentOptions.matchScore,
   259→   259→            mismatchPenalty: alignmentOptions.mismatchPenalty,
   260→   260→            gapPenalty: alignmentOptions.gapPenalty,
   261→   261→          });
   262→   262→          analysisResult = { type: 'pairwise', data: alignmentResult, timestamp: new Date() };
   263→   263→          break;
   264→   264→        }
   265→   265→
   266→   266→        case 'blast': {
   267→   267→          const blastResult = mockBLASTSearch(sequence1, {
   268→   268→            database: blastOptions.database,
   269→   269→            maxTargets: blastOptions.maxTargets,
   270→   270→            program: blastOptions.program,
   271→   271→          });
   272→   272→          analysisResult = { type: 'blast', data: blastResult, timestamp: new Date() };
   273→   273→          break;
   274→   274→        }
   275→   275→
   276→   276→        case 'orf': {
   277→   277→          const orfs = findORFs(sequence1, {
   278→   278→            minLength: orfOptions.minLength,
   279→   279→            geneticCode: orfOptions.geneticCode,
   280→   280→            allFrames: orfOptions.showAllFrames,
   281→   281→            startCodons: orfOptions.startCodons,
   282→   282→          });
   283→   283→          analysisResult = { type: 'orf', data: orfs, timestamp: new Date() };
   284→   284→          break;
   285→   285→        }
   286→   286→
   287→   287→        case 'reverse-complement': {
   288→   288→          const rcResult = rc(sequence1);
   289→   289→          analysisResult = { type: 'reverse-complement', data: rcResult, timestamp: new Date() };
   290→   290→          break;
   291→   291→        }
   292→   292→
   293→   293→        case 'translation': {
   294→   294→          const translationResult = translate(
   295→   295→            sequence1,
   296→   296→            translationOptions.frame,
   297→   297→            translationOptions.geneticCode
   298→   298→          );
   299→   299→          analysisResult = { type: 'translation', data: translationResult, timestamp: new Date() };
   300→   300→          break;
   301→   301→        }
   302→   302→
   303→   303→        case 'gc-content': {
   304→   304→          const gcResult = getSequenceStats(sequence1);
   305→   305→          analysisResult = { type: 'gc-content', data: gcResult, timestamp: new Date() };
   306→   306→          break;
   307→   307→        }
   308→   308→
   309→   309→        case 'motif-search':
   310→   310→        case 'pattern-search': {
   311→   311→          const motifResults = searchMotif(sequence1, motifPattern || 'ATG', patternSearchOptions);
   312→   312→          analysisResult = { type: activeTool, data: motifResults, timestamp: new Date() };
   313→   313→          break;
   314→   314→        }
   315→   315→
   316→   316→        case 'restriction-map': {
   317→   317→          const restrictionSites = findRestrictionSites(sequence1);
   318→   318→          analysisResult = { type: 'restriction-map', data: restrictionSites, timestamp: new Date() };
   319→   319→          break;
   320→   320→        }
   321→   321→
   322→   322→        default:
   323→   323→          throw new Error('Unknown tool');
   324→   324→      }
   325→   325→
   326→   326→      clearInterval(progressInterval);
   327→   327→      setProgress(100);
   328→   328→      
   329→   329→      setTimeout(() => {
   330→   330→        setResult(analysisResult);
   331→   331→        setIsAnalyzing(false);
   332→   332→        setProgress(0);
   333→   333→      }, 300);
   334→   334→
   335→   335→    } catch (error) {
   336→   336→      clearInterval(progressInterval);
   337→   337→      setIsAnalyzing(false);
   338→   338→      console.error('Analysis error:', error);
   339→   339→    }
   340→   340→  }, [activeTool, sequence1, sequence2, alignmentOptions, blastOptions, orfOptions, translationOptions, motifPattern, patternSearchOptions]);
   341→   341→
   342→   342→  // Copy to clipboard
   343→   343→  const copyToClipboard = useCallback((text: string) => {
   344→   344→    navigator.clipboard.writeText(text);
   345→   345→    setCopied(true);
   346→   346→    setTimeout(() => setCopied(false), 2000);
   347→   347→  }, []);
   348→   348→
   349→   349→  // Download result
   350→   350→  const downloadResult = useCallback((format: 'json' | 'csv' | 'txt') => {
   351→   351→    if (!result) return;
   352→   352→
   353→   353→    let content = '';
   354→   354→    let filename = '';
   355→   355→    let mimeType = '';
   356→   356→
   357→   357→    switch (format) {
   358→   358→      case 'json':
   359→   359→        content = JSON.stringify(result.data, null, 2);
   360→   360→        filename = `bioalign-${result.type}-result.json`;
   361→   361→        mimeType = 'application/json';
   362→   362→        break;
   363→   363→      case 'csv':
   364→   364→        content = convertToCSV(result.data);
   365→   365→        filename = `bioalign-${result.type}-result.csv`;
   366→   366→        mimeType = 'text/csv';
   367→   367→        break;
   368→   368→      case 'txt':
   369→   369→        content = formatPlainText(result);
   370→   370→        filename = `bioalign-${result.type}-result.txt`;
   371→   371→        mimeType = 'text/plain';
   372→   372→        break;
   373→   373→    }
   374→   374→
   375→   375→    const blob = new Blob([content], { type: mimeType });
   376→   376→    const url = URL.createObjectURL(blob);
   377→   377→    const a = document.createElement('a');
   378→   378→    a.href = url;
   379→   379→    a.download = filename;
   380→   380→    a.click();
   381→   381→    URL.revokeObjectURL(url);
   382→   382→  }, [result]);
   383→   383→
   384→   384→  // Highlight nucleotides in sequence
   385→   385→  const highlightSequence = (seq: string): React.ReactNode[] => {
   386→   386→    return seq.split('').map((char, i) => (
   387→   387→      <span key={i} className={NUCLEOTIDE_COLORS[char.toUpperCase()] || ''}>
   388→   388→        {char}
   389→   389→      </span>
   390→   390→    ));
   391→   391→  };
   392→   392→
   393→   393→  // Render tool-specific options
   394→   394→  const renderToolOptions = () => {
   395→   395→    switch (activeTool) {
   396→   396→      case 'pairwise':
   397→   397→        return (
   398→   398→          <div className="space-y-4">
   399→   399→            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   400→   400→              <div>
   401→   401→                <Label htmlFor="algorithm">Algorithm</Label>

   404→   404→                }>

   407→   407→                    <SelectItem value="global">Global (Needleman-Wunsch)</SelectItem>
   408→   408→                    <SelectItem value="local">Local (Smith-Waterman)</SelectItem>
   409→   409→                  </SelectContent>
   410→   410→                </Select>
   411→   411→              </div>
   412→   412→              <div>
   413→   413→                <Label htmlFor="matrix">Substitution Matrix</Label>
   414→   414→                <Select value={alignmentOptions.matrix} onValueChange={(v) =>
   415→   415→                  setAlignmentOptions(prev => ({ ...prev, matrix: v }))
   416→   416→                }>
   417→   417→                  <SelectTrigger><SelectValue /></SelectTrigger>
   418→   418→                  <SelectContent>
   419→   419→                    {Object.keys(SUBSTITUTION_MATRICES).map(m => (
   420→   420→                      <SelectItem key={m} value={m}>{m}</SelectItem>
   421→   421→                    ))}
   422→   422→                  </SelectContent>
   423→   423→                </Select>
   424→   424→              </div>
   425→   425→            </div>
   426→   426→            
   427→   427→            <Collapsible open={showAdvanced} onOpenChange={setShowAdvanced}>
   428→   428→              <CollapsibleTrigger asChild>
   429→   429→                <Button variant="ghost" size="sm" className="w-full justify-between">
   430→   430→                  <span className="flex items-center gap-2">
   431→   431→                    <Settings2 className="w-4 h-4" />
   432→   432→                    Advanced Scoring Options
   433→   433→                  </span>
   434→   434→                  {showAdvanced ? <ChevronUp className="w-4 h-4" /> : <ChevronDown className="w-4 h-4" />}
   435→   435→                </Button>
   436→   436→              </CollapsibleTrigger>
   437→   437→              <CollapsibleContent className="space-y-4 pt-4">
   438→   438→                <div className="grid grid-cols-3 gap-4">
   439→   439→                  <div>
   440→   440→                    <Label>Match Score</Label>
   441→   441→                    <Input
   442→   442→                      type="number"
   443→   443→                      value={alignmentOptions.matchScore}
   444→   444→                      onChange={(e) => setAlignmentOptions(prev => ({ ...prev, matchScore: Number(e.target.value) }))}
   445→   445→                    />
   446→   446→                  </div>
   447→   447→                  <div>
   448→   448→                    <Label>Mismatch Penalty</Label>
   449→   449→                    <Input
   450→   450→                      type="number"
   451→   451→                      value={alignmentOptions.mismatchPenalty}
   452→   452→                      onChange={(e) => setAlignmentOptions(prev => ({ ...prev, mismatchPenalty: Number(e.target.value) }))}
   453→   453→                    />
   454→   454→                  </div>
   455→   455→                  <div>
   456→   456→                    <Label>Gap Penalty</Label>
   457→   457→                    <Input
   458→   458→                      type="number"
   459→   459→                      value={alignmentOptions.gapPenalty}
   460→   460→                      onChange={(e) => setAlignmentOptions(prev => ({ ...prev, gapPenalty: Number(e.target.value) }))}
   461→   461→                    />
   462→   462→                  </div>
   463→   463→                </div>
   464→   464→              </CollapsibleContent>
   465→   465→            </Collapsible>
   466→   466→
   467→   467→            {/* Second sequence input */}
   468→   468→            <div>
   469→   469→              <Label htmlFor="seq2">Second Sequence</Label>
   470→   470→              <Textarea
   471→   471→                id="seq2"
   472→   472→                placeholder="Enter second sequence..."
   473→   473→                value={sequence2}
   474→   474→                onChange={(e) => setSequence2(e.target.value)}
   475→   475→                rows={4}
   476→   476→                className="font-mono text-sm"
   477→   477→              />
   478→   478→              {seq2Stats && (
   479→   479→                <div className="flex flex-wrap gap-2 mt-2">
   480→   480→                  <Badge variant="secondary">{seq2Stats.length} bp</Badge>
   481→   481→                  <Badge variant="secondary">{seq2Stats.type}</Badge>
   482→   482→                  <Badge variant="secondary">GC: {seq2Stats.gcContent.toFixed(1)}%</Badge>
   483→   483→                </div>
   484→   484→              )}
   485→   485→            </div>
   486→   486→          </div>
   487→   487→        );
   488→   488→
   489→   489→      case 'blast':
   490→   490→        return (
   491→   491→          <div className="space-y-4">
   492→   492→            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">

  1306→  1306→function convertToCSV(data: unknown): string {
  1307→  1307→  if (typeof data === 'string') return data;
  1308→  1308→  if (Array.isArray(data)) {
  1309→  1309→    if (data.length === 0) return '';
  1310→  1310→    const headers = Object.keys(data[0]);
  1311→  1311→    const rows = data.map(item => 
  1312→  1312→      headers.map(h => JSON.stringify((item as Record<string, unknown>)[h] ?? '')).join(',')
  1313→  1313→    );
  1314→  1314→    return [headers.join(','), ...rows].join('\n');
  1315→  1315→  }
  1316→  1316→  if (typeof data === 'object' && data !== null) {
  1317→  1317→    const entries = Object.entries(data as Record<string, unknown>);
  1318→  1318→    return entries.map(([k, v]) => `${k},${JSON.stringify(v)}`).join('\n');
  1319→  1319→  }
  1320→  1320→  return String(data);
  1321→  1321→}
