    96→function SearchDialog({ onNavigate }: SearchDialogProps) {
    97→  const [open, setOpen] = React.useState(false)
    98→  const [searchQuery, setSearchQuery] = React.useState("")
    99→  const inputRef = React.useRef<HTMLInputElement>(null)
   100→
   101→  // Filter tools based on search query
   102→  const filteredTools = React.useMemo(() => {
   103→    if (!searchQuery.trim()) return []
   104→    const query = searchQuery.toLowerCase()
   105→    return searchableTools.filter(tool => 
   106→      tool.name.toLowerCase().includes(query) || 
   107→      tool.category.toLowerCase().includes(query) ||
   108→      tool.id.toLowerCase().includes(query)
   109→    )
   110→  }, [searchQuery])
   111→
   112→  // Filter databases based on search query
   113→  const filteredDatabases = React.useMemo(() => {
   114→    if (!searchQuery.trim()) return []
   115→    const query = searchQuery.toLowerCase()
   116→    return databaseLinks.filter(db => 
   117→      db.name.toLowerCase().includes(query)
   118→    )
   119→  }, [searchQuery])
   120→
   121→  React.useEffect(() => {
   122→    const handleKeyDown = (e: KeyboardEvent) => {
   123→      if ((e.metaKey || e.ctrlKey) && e.key === "k") {
   124→        e.preventDefault()
   125→        setOpen((prev) => !prev)
   126→      }
   127→      // Close on Escape
   128→      if (e.key === "Escape" && open) {
   129→        e.preventDefault()
   130→        setOpen(false)
   131→      }
   132→    }
   133→
   134→    document.addEventListener("keydown", handleKeyDown)
   135→    return () => document.removeEventListener("keydown", handleKeyDown)
   136→  }, [open])
   137→
   138→  // Focus input when dialog opens
   139→  React.useEffect(() => {
   140→    if (open && inputRef.current) {
   141→      setTimeout(() => inputRef.current?.focus(), 100)
   142→    }
   143→    // Reset search when closing
   144→    if (!open) {
   145→      setSearchQuery("")
   146→    }
   147→  }, [open])
   148→
   149→  const handleToolClick = (toolId: string) => {
   150→    setOpen(false)
   151→    if (onNavigate) {
   152→      onNavigate('analysis', toolId)
   153→    }
   154→  }
   155→
   156→  const handleDatabaseClick = (url: string) => {
   157→    window.open(url, '_blank', 'noopener,noreferrer')
   158→    setOpen(false)
   159→  }
   160→
   161→  const hasResults = filteredTools.length > 0 || filteredDatabases.length > 0
   162→  const showInitial = !searchQuery.trim()
   163→
   164→  return (
   165→    <>
   166→      <Tooltip>
   167→        <TooltipTrigger asChild>
   168→          <Button
   169→            variant="outline"
   170→            className="relative h-9 w-full max-w-sm justify-start gap-2 text-sm text-muted-foreground sm:pr-12 cursor-pointer"
   171→            onClick={() => setOpen(true)}
   172→          >
   173→            <Search className="size-4" />
   174→            <span className="hidden lg:inline-flex">Search...</span>
   175→            <span className="inline-flex lg:hidden">Search</span>
   176→            <kbd className="pointer-events-none absolute right-1.5 top-1/2 hidden h-5 -translate-y-1/2 select-none items-center gap-1 rounded border bg-muted px-1.5 font-mono text-[10px] font-medium opacity-100 sm:flex">
   177→              <Command className="size-3" />K
   178→            </kbd>
   179→          </Button>
   180→        </TooltipTrigger>
   181→        <TooltipContent>
   182→          <p>Search <kbd className="rounded border bg-muted px-1 text-xs">⌘K</kbd></p>
   183→        </TooltipContent>
   184→      </Tooltip>
   185→
   186→      {/* Search Overlay */}
   187→      <AnimatePresence>
   188→        {open && (
   189→          <>
   190→            <motion.div
   191→              initial={{ opacity: 0 }}
   192→              animate={{ opacity: 1 }}
   193→              exit={{ opacity: 0 }}
   194→              className="fixed inset-0 z-50 bg-black/40 backdrop-blur-sm"
   195→              onClick={() => setOpen(false)}
   196→            />
   197→            <motion.div
   198→              initial={{ opacity: 0, scale: 0.95, y: -20 }}
   199→              animate={{ opacity: 1, scale: 1, y: 0 }}
   200→              exit={{ opacity: 0, scale: 0.95, y: -20 }}
   201→              transition={{ duration: 0.2, ease: "easeOut" }}
   202→              className="fixed left-1/2 top-[15%] z-50 w-full max-w-lg -translate-x-1/2 overflow-hidden rounded-xl border bg-background shadow-2xl"
   203→              onClick={(e) => e.stopPropagation()}
   204→            >
   205→              {/* Search Input Header */}
   206→              <div className="flex items-center border-b px-4">
   207→                <Search className="size-4 shrink-0 text-muted-foreground" />
   208→                <Input
   209→                  ref={inputRef}
   210→                  value={searchQuery}
   211→                  onChange={(e) => setSearchQuery(e.target.value)}
   212→                  placeholder="Search tools, databases, analysis..."
   213→                  className="border-0 focus-visible:ring-0 focus-visible:ring-offset-0"
   214→                />
   215→                <kbd className="pointer-events-none flex h-5 select-none items-center gap-1 rounded border bg-muted px-1.5 font-mono text-[10px] font-medium text-muted-foreground">
   216→                  ESC
   217→                </kbd>
   218→              </div>
   219→              
   220→              {/* Search Results */}
   221→              <div className="max-h-[400px] overflow-y-auto">
   222→                {showInitial ? (
   223→                  /* Initial State - Quick Actions */
   224→                  <div className="p-4">
   225→                    <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground/60 mb-3">
   226→                      Quick Actions
   227→                    </p>
   228→                    <div className="space-y-1">
   229→                      <button
   230→                        onClick={() => handleToolClick('blast')}
   231→                        className="flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-left text-sm hover:bg-accent transition-colors cursor-pointer"
   232→                      >

   234→                        <span>Run BLAST Search</span>

   236→                      </button>
   237→                      <button
   238→                        onClick={() => handleToolClick('primer')}
   239→                        className="flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-left text-sm hover:bg-accent transition-colors cursor-pointer"
   240→                      >
   241→                        <FolderIcon className="size-4 text-blue-500" />
   242→                        <span>Design Primers</span>
   243→                        <kbd className="ml-auto rounded border bg-muted px-1.5 font-mono text-[10px]">⌘P</kbd>
   244→                      </button>
   245→                      <button
   246→                        onClick={() => onNavigate?.('workspaces')}
   247→                        className="flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-left text-sm hover:bg-accent transition-colors cursor-pointer"
   248→                      >
   249→                        <FolderIcon className="size-4 text-green-500" />
   250→                        <span>Open Workspaces</span>
   251→                        <kbd className="ml-auto rounded border bg-muted px-1.5 font-mono text-[10px]">⌘W</kbd>
   252→                      </button>
   253→                    </div>
   254→
   255→                    <div className="mt-6">
   256→                      <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground/60 mb-3">
   257→                        Popular Databases
   258→                      </p>
   259→                      <div className="grid grid-cols-2 gap-2">
   260→                        {databaseLinks.slice(0, 6).map((db) => (
   261→                          <button
   262→                            key={db.name}
   263→                            onClick={() => handleDatabaseClick(db.url)}
   264→                            className="flex items-center gap-2 rounded-lg px-3 py-2 text-left text-sm hover:bg-accent transition-colors cursor-pointer"
   265→                          >
   266→                            <span className="text-xs">🔗</span>
   267→                            <span className="truncate">{db.name}</span>
   268→                          </button>
   269→                        ))}
   270→                      </div>
   271→                    </div>
   272→                  </div>
   273→                ) : hasResults ? (
   274→                  /* Search Results */
   275→                  <div className="p-2">
   276→                    {filteredTools.length > 0 && (
   277→                      <div className="mb-2">
   278→                        <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground/60 px-3 py-2">
   279→                          Tools ({filteredTools.length})
   280→                        </p>
   281→                        {filteredTools.map((tool) => (
   282→                          <button
   283→                            key={tool.id}
   284→                            onClick={() => handleToolClick(tool.id)}
   285→                            className="flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-left text-sm hover:bg-accent transition-colors cursor-pointer"
   286→                          >
   287→                            <span className="text-base">{tool.icon}</span>
   288→                            <div className="flex-1 min-w-0">
   289→                              <p className="font-medium truncate">{tool.name}</p>
   290→                              <p className="text-xs text-muted-foreground">{tool.category}</p>
   291→                            </div>
   292→                            <ChevronRight className="size-4 text-muted-foreground" />
   293→                          </button>
   294→                        ))}
   295→                      </div>
   296→                    )}
   297→                    {filteredDatabases.length > 0 && (
   298→                      <div className={filteredTools.length > 0 ? "mt-2 pt-2 border-t" : ""}>
   299→                        <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground/60 px-3 py-2">
   300→                          Databases ({filteredDatabases.length})
   301→                        </p>
   302→                        {filteredDatabases.map((db) => (
   303→                          <button
   304→                            key={db.name}
   305→                            onClick={() => handleDatabaseClick(db.url)}
   306→                            className="flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-left text-sm hover:bg-accent transition-colors cursor-pointer"
   307→                          >
   308→                            <span>🔗</span>
   309→                            <span className="flex-1 truncate">{db.name}</span>
   310→                            <span className="text-xs text-muted-foreground">External →</span>
   311→                          </button>
   312→                        ))}
   313→                      </div>
   314→                    )}
   315→                  </div>
   316→                ) : (
   317→                  /* No Results */
   318→                  <div className="p-8 text-center">
   319→                    <p className="text-sm text-muted-foreground">No results found for "{searchQuery}"</p>
   320→                    <p className="text-xs text-muted-foreground mt-1">Try a different search term</p>
   321→                  </div>
   322→                )}
   323→              </div>
   324→            </motion.div>
   325→          </>
   326→        )}
   327→      </AnimatePresence>
   328→    </>
   329→  )

   346→// Theme Toggle Component
   347→function ThemeToggle() {
   348→  const { theme, setTheme } = useTheme()
   349→  const [mounted, setMounted] = React.useState(false)
   350→
   351→  React.useEffect(() => {
   352→    setMounted(true)
   353→  }, [])
   354→
   355→  if (!mounted) {
   356→    return (
   357→      <Button variant="ghost" size="icon" className="size-9">
   358→        <Sun className="size-4" />
   359→      </Button>
   360→    )
   361→  }
   362→
   363→  return (
   364→    <Tooltip>
   365→      <TooltipTrigger asChild>
   366→        <Button
   367→          variant="ghost"
   368→          size="icon"
   369→          className="size-9"
   370→          onClick={() => setTheme(theme === "dark" ? "light" : "dark")}
   371→        >
   372→          <AnimatePresence mode="wait">
   373→            {theme === "dark" ? (
   374→              <motion.div
   375→                key="moon"
   376→                initial={{ rotate: -90, opacity: 0 }}
   377→                animate={{ rotate: 0, opacity: 1 }}
   378→                exit={{ rotate: 90, opacity: 0 }}
   379→                transition={{ duration: 0.2 }}
   380→              >
   381→                <Moon className="size-4" />
   382→              </motion.div>
   383→            ) : (
   384→              <motion.div
   385→                key="sun"
   386→                initial={{ rotate: 90, opacity: 0 }}
   387→                animate={{ rotate: 0, opacity: 1 }}
   388→                exit={{ rotate: -90, opacity: 0 }}
   389→                transition={{ duration: 0.2 }}
   390→              >
   391→                <Sun className="size-4" />
   392→              </motion.div>
   393→            )}
   394→          </AnimatePresence>
   395→          <span className="sr-only">Toggle theme</span>
   396→        </Button>
   397→      </TooltipTrigger>
   398→      <TooltipContent>
   399→        Switch to {theme === "dark" ? "light" : "dark"} mode
   400→      </TooltipContent>
   401→    </Tooltip>
   402→  )
   403→}

   521→        <Separator orientation="vertical" className="mx-1 h-6 hidden sm:block" />
   522→        <UserMenu />
   523→      </div>
   524→    </header>
   525→  )

   528→// Main Dashboard Layout Component
   529→export default function DashboardLayout({
   530→  children,
   531→  title,
   532→  subtitle,
   533→  actions,
   534→  breadcrumbs = [],
   535→  onSidebarNavigate,
   536→}: DashboardLayoutProps) {
   537→  const sidebarState = useSidebarState()
   538→
   539→  // Default breadcrumbs if not provided
   540→  const defaultBreadcrumbs = title
   541→    ? [{ label: "Dashboard", href: "/dashboard" }, { label: title }]
   542→    : []
   543→
   544→  const displayBreadcrumbs = breadcrumbs.length > 0 ? breadcrumbs : defaultBreadcrumbs
   545→
   546→  return (
   547→    <TooltipProvider delayDuration={0}>
   548→      <div className="min-h-screen bg-background">
   549→        {/* Desktop Sidebar */}
   550→        <div className="hidden md:block">
   551→          <DashboardSidebar
   552→            collapsed={sidebarState.collapsed}
   553→            onToggle={sidebarState.toggle}
   554→            onNavigate={onSidebarNavigate}
   555→          />
   556→        </div>
   557→
   558→        {/* Mobile Sidebar */}
   559→        <MobileSidebar
   560→          open={sidebarState.mobileOpen}
   561→          onClose={() => sidebarState.setMobileOpen(false)}
   562→          onNavigate={onSidebarNavigate}
   563→        />
   564→
   565→        {/* Main Content Area */}
   566→        <motion.main
   567→          animate={{
   568→            marginLeft: sidebarState.isMobile ? 0 : sidebarState.collapsed ? 64 : 280,
   569→          }}
   570→          transition={{ type: "spring", stiffness: 300, damping: 30 }}
   571→          className="min-h-screen flex flex-col"
   572→        >
   573→          {/* Header */}
   574→          <Header
   575→            onMobileMenuClick={sidebarState.toggle}
   576→            title={title}
   577→            subtitle={subtitle}
   578→            actions={actions}
   579→            breadcrumbs={displayBreadcrumbs}
   580→            onSearchNavigate={onSidebarNavigate}
   581→          />
   582→
   583→          {/* Page Title & Breadcrumbs Section */}
   584→          {(title || displayBreadcrumbs.length > 0) && (
   585→            <div className="border-b bg-background/50 backdrop-blur-sm px-4 sm:px-6 lg:px-8 py-4">
   586→              {/* Breadcrumbs */}
   587→              {displayBreadcrumbs.length > 0 && (
   588→                <Breadcrumb className="mb-2">
   589→                  <BreadcrumbList>
   590→                    {displayBreadcrumbs.map((crumb, index) => (
   591→                      <React.Fragment key={crumb.label}>
   592→                        <BreadcrumbItem>
   593→                          {index === displayBreadcrumbs.length - 1 || (!crumb.href && !('onClick' in crumb)) ? (
   594→                            <BreadcrumbPage>{crumb.label}</BreadcrumbPage>
   595→                          ) : 'onClick' in crumb && crumb.onClick ? (
   596→                            <button onClick={crumb.onClick} className="flex items-center gap-1.5 hover:text-biored transition-colors">
   597→                              {crumb.label}
   598→                            </button>
   599→                          ) : (
   600→                            <BreadcrumbLink href={crumb.href || '#'}>{crumb.label}</BreadcrumbLink>
   601→                          )}
   602→                        </BreadcrumbItem>
   603→                        {index < displayBreadcrumbs.length - 1 && (
   604→                          <BreadcrumbSeparator />
   605→                        )}
   606→                      </React.Fragment>
   607→                    ))}
   608→                  </BreadcrumbList>
   609→                </Breadcrumb>
   610→              )}
   611→
   612→              {/* Title & Actions Row */}
   613→              <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   614→                <div>
   615→                  {title && (
   616→                    <motion.h1
   617→                      initial={{ opacity: 0, y: -10 }}
   618→                      animate={{ opacity: 1, y: 0 }}
   619→                      className="text-2xl font-bold tracking-tight"
   620→                    >
   621→                      {title}
   622→                    </motion.h1>
   623→                  )}
   624→                  {subtitle && (
   625→                    <motion.p
   626→                      initial={{ opacity: 0 }}
   627→                      animate={{ opacity: 1 }}
   628→                      transition={{ delay: 0.1 }}
   629→                      className="text-sm text-muted-foreground mt-1"
   630→                    >
   631→                      {subtitle}
   632→                    </motion.p>
   633→                  )}
   634→                </div>
   635→                {actions && (
   636→                  <motion.div
   637→                    initial={{ opacity: 0, x: 20 }}
   638→                    animate={{ opacity: 1, x: 0 }}
   639→                    transition={{ delay: 0.15 }}
   640→                    className="flex items-center gap-2"
   641→                  >
   642→                    {actions}
   643→                  </motion.div>
   644→                )}
   645→              </div>
   646→            </div>
   647→          )}
   648→
   649→          {/* Main Content */}
   650→          <ScrollArea className="flex-1">
   651→            <div className="p-4 sm:p-6 lg:p-8">
   652→              <motion.div
   653→                initial={{ opacity: 0, y: 20 }}
   654→                animate={{ opacity: 1, y: 0 }}
   655→                transition={{ duration: 0.3, ease: "easeOut" }}
   656→              >
   657→                {children}
   658→              </motion.div>
   659→            </div>
   660→          </ScrollArea>
   661→        </motion.main>
   662→      </div>
   663→    </TooltipProvider>
   664→  )
   665→}
