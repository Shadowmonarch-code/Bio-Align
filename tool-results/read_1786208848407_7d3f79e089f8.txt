   186→  // Drawing helper functions - defined first to avoid reference errors

   189→    margin: { top: number; right: number; bottom: number; left: number },
   190→    chartWidth: number,
   191→    items: string[],
   192→    colors?: string[],
   193→    legendConfig: { position: string; fontSize: number; fontFamily: string }
   194→  ) => {
   195→    let legendX: number, legendY: number
   196→    
   197→    switch (legendConfig.position) {
   198→      case 'top':
   199→        legendX = margin.left + chartWidth / 2 - (items.length * 60) / 2
   200→        legendY = 10
   201→        break
   202→      case 'bottom':
   203→        legendX = margin.left + chartWidth / 2 - (items.length * 60) / 2
   204→        legendY = margin.top + 350 + 20
   205→        break
   206→      case 'left':
   207→        legendX = 10
   208→        legendY = margin.top + 20

   214→

   221→      ctx.fillRect(legendX, legendY + index * 20, 14, 14)
   222→

   224→      ctx.font = `${legendConfig.fontSize - 1}px ${legendConfig.fontFamily}`

   231→  const drawBarChart = useCallback((
   232→    ctx: CanvasRenderingContext2D, 
   233→    margin: { top: number; right: number; bottom: number; left: number }, 
   234→    chartWidth: number, 
   235→    chartHeight: number,
   236→    cfg: typeof config
   237→  ) => {

   239→    const barWidth = chartWidth / sampleData.labels.length * 0.7
   240→    const gap = chartWidth / sampleData.labels.length * 0.3
   241→
   242→    // Grid lines
   243→    if (cfg.showGrid) {
   244→      ctx.strokeStyle = '#E5E7EB'
   245→      ctx.lineWidth = 0.5
   246→      for (let i = 0; i <= 5; i++) {
   247→        const y = margin.top + (chartHeight / 5) * i
   248→        ctx.beginPath()

   250→        ctx.lineTo(margin.left + chartWidth, y)
   251→        ctx.stroke()

   253→        ctx.fillStyle = '#666666'
   254→        ctx.font = `${cfg.fontSize}px ${cfg.fontFamily}`
   255→        ctx.textAlign = 'right'

   257→        ctx.fillText(value, margin.left - 10, y + 4)
   258→      }
   259→    }
   260→
   261→    // Bars
   262→    sampleData.values.forEach((value, index) => {
   263→      const x = margin.left + (chartWidth / sampleData.labels.length) * index + gap / 2
   264→      const barHeight = (value / maxValue) * chartHeight
   265→      const y = margin.top + chartHeight - barHeight

   267→      ctx.fillStyle = currentColors[index % currentColors.length]
   268→      ctx.fillRect(x, y, barWidth, barHeight)
   269→
   270→      if (cfg.showErrorBars) {
   271→        const errorY = (sampleData.errorValues[index] / maxValue) * chartHeight
   272→        ctx.strokeStyle = '#000000'
   273→        ctx.lineWidth = 1.5
   274→        ctx.beginPath()

   276→        ctx.lineTo(x + barWidth / 2, y + errorY)
   277→        ctx.moveTo(x + barWidth / 2 - 5, y - errorY)
   278→        ctx.lineTo(x + barWidth / 2 + 5, y - errorY)
   279→        ctx.moveTo(x + barWidth / 2 - 5, y + errorY)
   280→        ctx.lineTo(x + barWidth / 2 + 5, y + errorY)
   281→        ctx.stroke()
   282→      }
   283→
   284→      ctx.fillStyle = '#666666'
   285→      ctx.font = `${cfg.fontSize}px ${cfg.fontFamily}`
   286→      ctx.textAlign = 'center'
   287→      ctx.save()
   288→      ctx.translate(x + barWidth / 2, margin.top + chartHeight + 15)
   289→      ctx.rotate(-Math.PI / 6)
   290→      ctx.fillText(sampleData.labels[index], 0, 0)
   291→      ctx.restore()
   292→    })
   293→
   294→    // Axis lines
   295→    ctx.strokeStyle = '#333333'
   296→    ctx.lineWidth = cfg.axisLineWidth
   297→    ctx.beginPath()
   298→    ctx.moveTo(margin.left, margin.top)
   299→    ctx.lineTo(margin.left, margin.top + chartHeight)
   300→    ctx.lineTo(margin.left + chartWidth, margin.top + chartHeight)
   301→    ctx.stroke()
   302→
   303→    if (cfg.showLegend) {
   304→      drawLegend(ctx, margin, chartWidth, ['Treatment A'], undefined, {
   305→        position: cfg.legendPosition,
   306→        fontSize: cfg.fontSize,
   307→        fontFamily: cfg.fontFamily
   308→      })
   309→    }
   310→  }, [sampleData, currentColors, drawLegend])

   312→  const drawLineChart = useCallback((

   449→
   450→    if (cfg.showGrid) {
   451→      ctx.strokeStyle = '#E5E7EB'

   463→      const centerX = margin.left + (chartWidth / boxData.length) * (index + 0.5)

   487→      ctx.strokeRect(centerX - boxWidth / 2, scaleY(box.q3), boxWidth, scaleY(box.q1) - scaleY(box.q3))
   488→
   489→      ctx.strokeStyle = '#000000'
   490→      ctx.lineWidth = 2
   491→      ctx.beginPath()
   492→      ctx.moveTo(centerX - boxWidth / 2, scaleY(box.median))
   493→      ctx.lineTo(centerX + boxWidth / 2, scaleY(box.median))
   494→      ctx.stroke()
   495→
   496→      box.outliers.forEach(outlier => {
   497→        ctx.fillStyle = '#FFFFFF'
   498→        ctx.strokeStyle = currentColors[index % currentColors.length]
   499→        ctx.lineWidth = 1.5
   500→        ctx.beginPath()

   503→        ctx.stroke()
   504→      })
   505→
   506→      ctx.fillStyle = '#666666'
   507→      ctx.font = `${cfg.fontSize}px ${cfg.fontFamily}`

   509→      ctx.save()
   510→      ctx.translate(centerX, margin.top + chartHeight + 15)

   554→    ctx.beginPath()
   555→    ctx.moveTo(margin.left, margin.top)
   556→    ctx.lineTo(margin.left, margin.top + chartHeight)
   557→    ctx.lineTo(margin.left + chartWidth, margin.top + chartHeight)

   587→      ctx.lineWidth = 2

   597→      ctx.textAlign = 'center'
   598→      ctx.textBaseline = 'middle'
   599→      ctx.fillText(`${value}%`, labelX, labelY)
   600→
   601→      startAngle = endAngle
   602→    })
   603→
   604→    const legendItems = ['Category A', 'Category B', 'Category C', 'Category D', 'Category E']

   606→      const legendX = width - 90
   607→      const legendY = 50 + index * 22

   611→
   612→      ctx.fillStyle = '#333333'
   613→      ctx.font = `${cfg.fontSize - 1}px ${cfg.fontFamily}`
   614→      ctx.textAlign = 'left'
   615→      ctx.textBaseline = 'middle'
   616→      ctx.fillText(item, legendX + 16, legendY + 6)
   617→    })
   618→  }, [currentColors])

   653→      case 'pie':

   656→      default:

   661→    ctx.font = `bold ${config.fontSize + 2}px ${config.fontFamily}`
   662→    ctx.textAlign = 'center'
   663→    ctx.fillText(figureTitle || 'Sample Figure Title', width / 2, 25)
