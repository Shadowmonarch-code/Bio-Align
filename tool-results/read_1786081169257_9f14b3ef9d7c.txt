   100→
   101→interface NavbarProps {
   102→  onNavigate?: (view: string) => void;
   103→}
   104→
   105→export default function Navbar({ onNavigate }: NavbarProps) {
   106→  const [isScrolled, setIsScrolled] = React.useState(false);
   107→  const [isMobileOpen, setIsMobileOpen] = React.useState(false);
   108→  const [activeSection, setActiveSection] = React.useState("");
   109→  const [mounted, setMounted] = React.useState(false);
   110→  const [showAuthModal, setShowAuthModal] = React.useState(false);
   111→  const [authMode, setAuthMode] = React.useState<"signin" | "signup">("signin");
   112→  
   113→  // Form state
   114→  const [authEmail, setAuthEmail] = React.useState("");
   115→  const [authPassword, setAuthPassword] = React.useState("");
   116→  const [authName, setAuthName] = React.useState("");
   117→  const [isLoading, setIsLoading] = React.useState(false);
   118→  const [authError, setAuthError] = React.useState("");
   119→  
   120→  const { theme, setTheme } = useTheme();
   121→  const { data: session, status } = useSession();
   122→
   123→  // Handle mount state for theme
   124→  React.useEffect(() => {
   125→    setMounted(true);
   126→  }, []);
   127→
   128→  // Scroll detection for glassmorphism effect
   129→  React.useEffect(() => {
   130→    const handleScroll = () => {
   131→      setIsScrolled(window.scrollY > 20);
   132→      
   133→      // Detect active section
   134→      const sections = navLinks.map((link) => link.href.substring(1));
   135→      for (const section of sections.reverse()) {
   136→        const element = document.getElementById(section);
   137→        if (element) {
   138→          const rect = element.getBoundingClientRect();
   139→          if (rect.top <= 150) {
   140→            setActiveSection(section);
   141→            break;
   142→          }
   143→        }
   144→      }
   145→    };
   146→
   147→    window.addEventListener("scroll", handleScroll, { passive: true });
   148→    return () => window.removeEventListener("scroll", handleScroll);
   149→  }, []);
   150→
   151→  // Toggle theme
   152→  const toggleTheme = () => {
   153→    setTheme(theme === "dark" ? "light" : "dark");
   154→  };
   155→
   156→  // Handle navigation click
   157→  const handleNavClick = (href: string) => {
   158→    setIsMobileOpen(false);
   159→    
   160→    // Check if it's a special navigation action
   161→    if (onNavigate) {
   162→      if (href === '#dashboard' || href === '#get-started') {
   163→        onNavigate('dashboard');
   164→        window.scrollTo({ top: 0 });
   165→        return;
   166→      }
   167→    }
   168→    
   169→    // Regular anchor scrolling
   170→    const elementId = href.substring(1);
   171→    const element = document.getElementById(elementId);
   172→    if (element) {
   173→      element.scrollIntoView({ behavior: "smooth" });
   174→    }
   175→  };
   176→
   177→  // Handle Get Started button click
   178→  const handleGetStarted = () => {
   179→    if (onNavigate) {
   180→      onNavigate('dashboard');
   181→      window.scrollTo({ top: 0 });
   182→    }
   183→  };
   184→
   185→  // Handle Sign In button click
   186→  const handleSignIn = () => {
   187→    setShowAuthModal(true);
   188→    setAuthMode('signin');
   189→    setAuthError("");
   190→  };
   191→
   192→  // Handle Sign Up button click  
   193→  const handleSignUp = () => {
   194→    setShowAuthModal(true);
   195→    setAuthMode('signup');
   196→    setAuthError("");
   197→  };
   198→
   199→  // Handle sign out
   200→  const handleSignOut = async () => {
   201→    await signOut({ callbackUrl: '/' });
   202→  };
   203→
   204→  // Reset form when modal opens
   205→  React.useEffect(() => {
   206→    if (showAuthModal) {
   207→      setAuthEmail("");
   208→      setAuthPassword("");
   209→      setAuthName("");
   210→      setAuthError("");
   211→    }
   212→  }, [showAuthModal, authMode]);
   213→
   214→  // Handle form submission with real authentication
   215→  const handleAuthSubmit = async (e: React.FormEvent) => {
   216→    e.preventDefault();
   217→    setAuthError("");
   218→    setIsLoading(true);
   219→
   220→    try {
   221→      if (authMode === 'signup') {
   222→        // Registration
   223→        const response = await fetch('/api/auth/register', {
   224→          method: 'POST',
   225→          headers: { 'Content-Type': 'application/json' },
   226→          body: JSON.stringify({
   227→            name: authName,
   228→            email: authEmail,
   229→            password: authPassword,
   230→          }),
   231→        });
   232→
   233→        const data = await response.json();
   234→
   235→        if (!response.ok) {
   236→          throw new Error(data.error || 'Registration failed');
   237→        }
   238→
   239→        // After successful registration, sign in
   240→        const result = await signIn('credentials', {
   241→          email: authEmail,
   242→          password: authPassword,
   243→          redirect: false,
   244→        });
   245→
   246→        if (result?.error) {
   247→          throw new Error(result.error);
   248→        }
   249→      } else {
   250→        // Sign in
   251→        const result = await signIn('credentials', {
   252→          email: authEmail,
   253→          password: authPassword,
   254→          redirect: false,
   255→        });
   256→
   257→        if (result?.error) {
   258→          throw new Error('Invalid email or password');
   259→        }
   260→      }
   261→
   262→      // Success - close modal and navigate to dashboard
   263→      setShowAuthModal(false);
   264→      if (onNavigate) {
   265→        onNavigate('dashboard');
   266→        window.scrollTo({ top: 0 });
   267→      }
   268→    } catch (error: any) {
   269→      setAuthError(error.message || 'An error occurred');
   270→    } finally {
   271→      setIsLoading(false);
   272→    }
   273→  };
   274→
   275→  // User initials for avatar
   276→  const getUserInitials = () => {
   277→    if (session?.user?.name) {
   278→      return session.user.name.split(' ').map(n => n[0]).join('').toUpperCase().slice(0, 2);
   279→    }
   280→    if (session?.user?.email) {
   281→      return session.user.email[0].toUpperCase();
   282→    }
   283→    return 'U';
   284→  };
   285→
   286→  return (
   287→    <>
   288→      <motion.header
   289→        initial={{ y: -100, opacity: 0 }}
   290→        animate={{ y: 0, opacity: 1 }}
   291→        transition={{ duration: 0.5 }}
   292→        className={cn(
   293→          "fixed top-0 left-0 right-0 z-50 w-full transition-all duration-300",
   294→          isScrolled
   295→            ? "glass shadow-sm border-b border-border/50"
   296→            : "bg-transparent"
   297→        )}
   298→      >
   299→        <nav
   300→          className={cn(
   301→            "mx-auto flex h-16 max-w-7xl items-center justify-between px-4 sm:px-6 lg:px-8",
   302→            "transition-all duration-300"
   303→          )}
   304→        >
   305→          {/* Logo */}
   306→          <motion.div className="flex items-center gap-2">
   307→            <a
   308→              href="#"
   309→              className="group flex items-center gap-2.5 no-underline"
   310→              onClick={(e) => {
   311→                e.preventDefault();
   312→                window.scrollTo({ top: 0, behavior: "smooth" });
   313→              }}
   314→            >
   315→              <div className="relative flex items-center justify-center">
   316→                <div className="absolute inset-0 rounded-lg bg-biored/20 blur-lg group-hover:bg-biored/30 transition-colors duration-300" />
   317→                <Dna className="relative size-8 text-biored transition-transform duration-300 group-hover:scale-110 group-hover:rotate-12" />
   318→              </div>
   319→              <span className="text-xl font-bold tracking-tight">
   320→                <span className="text-foreground">Bio</span>
   321→                <span className="text-biored">Align</span>
   322→              </span>
   323→            </a>
   324→          </motion.div>
   325→
   326→          {/* Desktop Navigation Links */}
   327→          <div className="hidden lg:flex items-center gap-1">
   328→            {navLinks.map((link) => (
   329→              <a
   330→                key={link.name}
   331→                href={link.href}
   332→                onClick={(e) => {
   333→                  e.preventDefault();
   334→                  handleNavClick(link.href);
   335→                }}
   336→                className={cn(
   337→                  "relative px-3 py-2 text-sm font-medium rounded-md transition-all duration-200 cursor-pointer",
   338→                  "hover:text-foreground hover:bg-accent/10",
   339→                  activeSection === link.href.substring(1)
   340→                    ? "text-biored font-semibold"
   341→                    : "text-muted-foreground"
   342→                )}
   343→              >
   344→                <span className="relative z-10">{link.name}</span>
   345→                {/* Active indicator */}
   346→                {activeSection === link.href.substring(1) && (
   347→                  <motion.div
   348→                    layoutId="activeNav"
   349→                    className="absolute inset-0 rounded-md bg-biored/10 border border-biored/20"
   350→                    transition={{
   351→                      type: "spring",
   352→                      stiffness: 380,
   353→                      damping: 30,
   354→                    }}
   355→                  />
   356→                )}
   357→              </a>
   358→            ))}
   359→          </div>
   360→
   361→          {/* Right Side Actions */}
   362→          <div className="flex items-center gap-2 sm:gap-3">
   363→            {/* Theme Toggle */}
   364→            {mounted && (
   365→              <Button
   366→                variant="ghost"
   367→                size="icon"
   368→                onClick={toggleTheme}
   369→                className="relative size-9 rounded-full overflow-hidden hover:bg-accent/10 cursor-pointer"
   370→                aria-label="Toggle theme"
   371→              >
   372→                <AnimatePresence mode="wait" initial={false}>
   373→                  {theme === "dark" ? (
   374→                    <motion.div
   375→                      key="sun"
   376→                      initial={{ rotate: -90, scale: 0, opacity: 0 }}
   377→                      animate={{ rotate: 0, scale: 1, opacity: 1 }}
   378→                      exit={{ rotate: 90, scale: 0, opacity: 0 }}
   379→                      transition={{ duration: 0.2 }}
   380→                      className="absolute"
   381→                    >
   382→                      <Sun className="size-4 text-yellow-400" />
   383→                    </motion.div>
   384→                  ) : (
   385→                    <motion.div
   386→                      key="moon"
   387→                      initial={{ rotate: 90, scale: 0, opacity: 0 }}
   388→                      animate={{ rotate: 0, scale: 1, opacity: 1 }}
   389→                      exit={{ rotate: -90, scale: 0, opacity: 0 }}
   390→                      transition={{ duration: 0.2 }}
   391→                      className="absolute"
   392→                    >
   393→                      <Moon className="size-4 text-slate-700" />
   394→                    </motion.div>
   395→                  )}
   396→                </AnimatePresence>
   397→              </Button>
   398→            )}
   399→
   400→            {/* Auth Section */}
   401→            {status === "loading" ? (
   402→              <div className="flex items-center gap-2">
   403→                <Loader2 className="size-4 animate-spin text-muted-foreground" />
   404→              </div>
   405→            ) : session?.user ? (
   406→              /* Logged In State */
   407→              <div className="flex items-center gap-2">
   408→                {/* User Menu Button */}
   409→                <Button
   410→                  variant="ghost"
   411→                  size="sm"
   412→                  onClick={() => onNavigate?.("dashboard")}
   413→                  className="hidden sm:flex items-center gap-2 text-muted-foreground hover:text-foreground hover:bg-accent/10 cursor-pointer"
   414→                >
   415→                  <div className="size-7 rounded-full bg-biored/10 flex items-center justify-center">
   416→                    <span className="text-xs font-medium text-biored">{getUserInitials()}</span>
   417→                  </div>
   418→                  <span className="text-sm font-medium">{session.user.name || 'User'}</span>
   419→                </Button>
   420→                
   421→                {/* Sign Out Button */}
   422→                <Button
   423→                  variant="ghost"
   424→                  size="sm"
   425→                  onClick={handleSignOut}
   426→                  className="hidden sm:inline-flex text-muted-foreground hover:text-destructive hover:bg-destructive/10 cursor-pointer gap-2"
   427→                >
   428→                  <LogOut className="size-4" />
   429→                  <span className="hidden md:inline">Sign Out</span>
   430→                </Button>
   431→
   432→                {/* Mobile Sign Out */}
   433→                <Button
   434→                  variant="ghost"
   435→                  size="icon"
   436→                  onClick={handleSignOut}
   437→                  className="sm:hidden size-9 hover:bg-accent/10 cursor-pointer"
   438→                >
   439→                  <LogOut className="size-4" />
   440→                </Button>
   441→              </div>
   442→            ) : (
   443→              /* Logged Out State */
   444→              <>
   445→                {/* Sign In Button - Desktop */}
   446→                <Button
   447→                  variant="ghost"
   448→                  size="sm"
   449→                  onClick={handleSignIn}
   450→                  className="hidden sm:inline-flex text-muted-foreground hover:text-foreground hover:bg-accent/10 cursor-pointer gap-2"
   451→                >
   452→                  <LogIn className="size-4" />
   453→                  Sign In
   454→                </Button>
   455→
   456→                {/* Get Started CTA Button */}
   457→                <Button
   458→                  size="sm"
   459→                  onClick={handleGetStarted}
   460→                  className="hidden sm:inline-flex bg-biored hover:bg-biored-dark text-white shadow-lg shadow-biored/25 hover:shadow-biored/40 transition-all duration-300 group cursor-pointer"
   461→                >
   462→                  Get Started
   463→                  <ArrowRight className="size-4 ml-1 transition-transform duration-200 group-hover:translate-x-0.5" />
   464→                </Button>
   465→              </>
   466→            )}
   467→
   468→            {/* Mobile Menu Button */}
   469→            <Sheet open={isMobileOpen} onOpenChange={setIsMobileOpen}>
   470→              <Button
   471→                variant="ghost"
   472→                size="icon"
   473→                className="lg:hidden size-9 hover:bg-accent/10 cursor-pointer"
   474→                onClick={() => setIsMobileOpen(true)}
   475→              >
   476→                <Menu className="size-5" />
   477→              </Button>
   478→              <SheetContent side="right" className="w-[300px] sm:w-[350px] p-0 bg-background/95 backdrop-blur-xl">
   479→                <div className="flex flex-col h-full">
   480→                  {/* Mobile Header */}
   481→                  <SheetHeader className="p-6 pb-4">
   482→                    <div className="flex items-center justify-between">
   483→                      <SheetTitle className="flex items-center gap-2.5">
   484→                        <Dna className="size-6 text-biored" />
   485→                        <span className="text-xl font-bold">
   486→                          <span className="text-foreground">Bio</span>
   487→                          <span className="text-biored">Align</span>
   488→                        </span>
   489→                      </SheetTitle>
   490→                    </div>
   491→                    <SheetDescription className="sr-only">
   492→                      Navigation menu for BioAlign platform
   493→                    </SheetDescription>
   494→                  </SheetHeader>
   495→
   496→                  <Separator className="bg-border/50" />
   497→
   498→                  {/* Auth Status in Mobile Menu */}
   499→                  <div className="px-4 py-3 border-b border-border/50">
   500→                    {session?.user ? (
   501→                      <div className="flex items-center justify-between">
   502→                        <div className="flex items-center gap-3">
   503→                          <div className="size-9 rounded-full bg-biored/10 flex items-center justify-center">
   504→                            <User className="size-4 text-biored" />
   505→                          </div>
   506→                          <div>
   507→                            <p className="text-sm font-medium">{session.user.name || 'User'}</p>
   508→                            <p className="text-xs text-muted-foreground truncate max-w-[150px]">
   509→                              {session.user.email}
   510→                            </p>
   511→                          </div>
   512→                        </div>
   513→                        <Button
   514→                          variant="ghost"
   515→                          size="sm"
   516→                          onClick={handleSignOut}
   517→                          className="text-destructive hover:text-destructive cursor-pointer"
   518→                        >
   519→                          <LogOut className="size-4 mr-1" />
   520→                          Out
   521→                        </Button>
   522→                      </div>
   523→                    ) : (
   524→                      <div className="flex gap-2">
   525→                        <Button
   526→                          variant="outline"
   527→                          size="sm"
   528→                          className="flex-1 cursor-pointer"
   529→                          onClick={() => {
   530→                            setIsMobileOpen(false);
   531→                            handleSignIn();
   532→                          }}
   533→                        >
   534→                          <LogIn className="size-4 mr-1" />
   535→                          Sign In
   536→                        </Button>
   537→                        <Button
   538→                          size="sm"
   539→                          className="flex-1 bg-biored hover:bg-biored-dark text-white cursor-pointer"
   540→                          onClick={() => {
   541→                            setIsMobileOpen(false);
   542→                            handleGetStarted();
   543→                          }}
   544→                        >
   545→                          Get Started
   546→                        </Button>
   547→                      </div>
   548→                    )}
   549→                  </div>
   550→
   551→                  {/* Mobile Navigation Links */}
   552→                  <div className="flex-1 overflow-y-auto py-4 px-2 scrollbar-hide">
   553→                    <nav className="flex flex-col gap-1">
   554→                      {navLinks.map((link) => {
   555→                        const Icon = link.icon;
   556→                        return (
   557→                          <a
   558→                            key={link.name}
   559→                            href={link.href}
   560→                            onClick={(e) => {
   561→                              e.preventDefault();
   562→                              handleNavClick(link.href);
   563→                            }}
   564→                            className={cn(
   565→                              "flex items-center gap-3 px-4 py-3 rounded-lg transition-all duration-200 no-underline cursor-pointer",
   566→                              "hover:bg-accent/10",
   567→                              activeSection === link.href.substring(1)
   568→                                ? "bg-biored/10 text-biored font-medium"
   569→                                : "text-muted-foreground hover:text-foreground"
   570→                            )}
   571→                          >
   572→                            <Icon className="size-5 shrink-0" />
   573→                            <div className="flex flex-col">
   574→                              <span>{link.name}</span>
   575→                              <span className="text-xs text-muted-foreground/70">
   576→                                {link.description}
   577→                              </span>
   578→                            </div>
   579→                          </a>
   580→                        );
   581→                      })}
   582→                    </nav>
   583→                  </div>
   584→
   585→                  <Separator className="bg-border/50" />
   586→                </div>
   587→              </SheetContent>
   588→            </Sheet>
   589→          </div>
   590→        </nav>
   591→      </motion.header>
   592→
   593→      {/* Spacer to prevent content from hiding behind fixed navbar */}
   594→      <div className="h-16" />
   595→
   596→      {/* Authentication Modal */}
   597→      <Dialog open={showAuthModal} onOpenChange={setShowAuthModal}>
   598→        <DialogContent className="sm:max-w-md p-0 overflow-hidden">
   599→          <div className="relative">
   600→            {/* Header gradient */}
   601→            <div className="bg-gradient-to-br from-biored to-biored-dark p-6 text-white">
   602→              <DialogTitle className="text-xl font-bold text-center">
   603→                {authMode === 'signin' ? 'Welcome Back' : 'Join BioAlign'}
   604→              </DialogTitle>
   605→              <DialogDescription className="text-white/80 text-center mt-1">
   606→                {authMode === 'signin' 
   607→                  ? 'Sign in to access your bioinformatics workspace' 
   608→                  : 'Create your free account to start analyzing'}
   609→              </DialogDescription>
   610→            </div>
   611→
   612→            {/* Error Message */}
   613→            {authError && (
   614→              <div className="mx-6 mt-4 p-3 rounded-lg bg-red-50 dark:bg-red-950/30 border border-red-200 dark:border-red-800">
   615→                <p className="text-sm text-red-600 dark:text-red-400">{authError}</p>
   616→              </div>
   617→            )}
   618→
   619→            {/* Form */}
   620→            <form onSubmit={handleAuthSubmit} className="p-6 space-y-4">
   621→              {authMode === 'signup' && (
   622→                <div className="space-y-2">
   623→                  <Label htmlFor="name">Full Name</Label>
   624→                  <Input 
   625→                    id="name" 
   626→                    placeholder="Dr. Jane Smith" 
   627→                    value={authName}
