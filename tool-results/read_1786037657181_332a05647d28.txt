   519→// ==================== MAIN TOOLS CATALOG COMPONENT ====================
   520→export default function ToolsCatalog() {
   521→  // State
   522→  const [isLoading, setIsLoading] = React.useState(true)
   523→  const [searchQuery, setSearchQuery] = React.useState("")
   524→  const [viewMode, setViewMode] = React.useState<ViewMode>("grid")
   525→  const [expandedCategories, setExpandedCategories] = React.useState<
   526→    Set<string>
   527→  >(new Set())
   528→  const [favorites, setFavorites] = React.useState<Set<string>>(new Set())
   529→  const [activeTab, setActiveTab] = React.useState("all")
   530→  const [statusFilter, setStatusFilter] = React.useState<ToolStatus | "all">(
   531→    "all"
   532→  )
   533→
   534→  // Simulate loading
   535→  React.useEffect(() => {
   536→    const timer = setTimeout(() => setIsLoading(false), 800)
   537→    return () => clearTimeout(timer)
   538→  }, [])
   539→
   540→  // Load favorites from localStorage
   541→  React.useEffect(() => {
   542→    try {
   543→      const saved = localStorage.getItem("bioalign-favorites")
   544→      if (saved) {
   545→        setFavorites(new Set(JSON.parse(saved)))
   546→      }
   547→    } catch (error) {
   548→      console.error("Failed to load favorites:", error)
   549→    }
   550→  }, [])
   551→
   552→  // Save favorites to localStorage
   553→  const saveFavorites = React.useCallback((newFavs: Set<string>) => {
   554→    setFavorites(newFavs)
   555→    try {
   556→      localStorage.setItem(
   557→        "bioalign-favorites",
   558→        JSON.stringify(Array.from(newFavs))
   559→      )
   560→    } catch (error) {
   561→      console.error("Failed to save favorites:", error)
   562→    }
   563→  }, [])
   564→
   565→  // Toggle favorite
   566→  const toggleFavorite = React.useCallback(
   567→    (toolId: string) => {
   568→      const newFavs = new Set(favorites)
   569→      if (newFavs.has(toolId)) {
   570→        newFavs.delete(toolId)
   571→      } else {
   572→        newFavs.add(toolId)
   573→      }
   574→      saveFavorites(newFavs)
   575→    },
   576→    [favorites, saveFavorites]
   577→  )
   578→
   579→  // Toggle category expansion
   580→  const toggleCategory = React.useCallback((categoryId: string) => {
   581→    setExpandedCategories((prev) => {
   582→      const next = new Set(prev)
   583→      if (next.has(categoryId)) {
   584→        next.delete(categoryId)
   585→      } else {
   586→        next.add(categoryId)
   587→      }
   588→      return next
   589→    })
   590→  }, [])
   591→
   592→  // Get filtered tools
   593→  const getFilteredTools = React.useCallback((): BioTool[] => {
   594→    let tools: BioTool[]
   595→
   596→    switch (activeTab) {
   597→      case "recent":
   598→        tools = recentlyUsedTools
   599→        break
   600→      case "favorites":
   601→        tools = getAllTools().filter((t) => favorites.has(t.id))
   602→        break
   603→      case "popular":
   604→        tools = popularTools
   605→        break
   606→      default:
   607→        tools = getAllTools()
   608→    }
   609→
   610→    // Apply search filter
   611→    if (searchQuery.trim()) {
   612→      tools = searchTools(searchQuery.trim())
   613→    }
   614→
   615→    // Apply status filter
   616→    if (statusFilter !== "all") {
   617→      tools = tools.filter((t) => t.status === statusFilter)
   618→    }
   619→
   620→    // Apply tab-specific filtering
   621→    if (activeTab === "favorites") {
   622→      tools = tools.filter((t) => favorites.has(t.id))
   623→    }
   624→
   625→    return tools
   626→  }, [searchQuery, activeTab, statusFilter, favorites])
   627→
   628→  // Get filtered categories
   629→  const getFilteredCategories = React.useCallback((): ToolCategory[] => {
   630→    if (!searchQuery.trim() && statusFilter === "all") {
   631→      return toolCategories
   632→    }
   633→
   634→    return toolCategories
   635→      .map((cat) => ({
   636→        ...cat,
   637→        tools: cat.tools.filter((tool) => {
   638→          const matchesSearch =
   639→            !searchQuery.trim() ||
   640→            tool.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
   641→            tool.description
   642→              .toLowerCase()
   643→              .includes(searchQuery.toLowerCase())
   644→          const matchesStatus =
   645→            statusFilter === "all" || tool.status === statusFilter
   646→          return matchesSearch && matchesStatus
   647→        }),
   648→      }))
   649→      .filter((cat) => cat.tools.length > 0)
