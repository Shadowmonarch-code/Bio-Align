   108→/**
   109→ * Extracts view name and optional toolId from an href
   110→ * Maps URL paths to internal view names for navigation
   111→ */
   112→function getNavigationTarget(href: string): { view: string; toolId?: string } {
   113→  // Tool items - map to analysis view with toolId
   114→  if (href.startsWith("/dashboard/tools/")) {
   115→    const toolId = href.replace("/dashboard/tools/", "")
   116→    return { view: "analysis", toolId }
   117→  }
   118→
   119→  // Plant Breeding section - all map to plant-breeding view
   120→  if (href.startsWith("/plant-breeding")) {
   121→    return { view: "plant-breeding" }
   122→  }
   123→
   124→  // Bioinformatics section - map to analysis view with toolId
   125→  if (href.startsWith("/bioinformatics")) {
   126→    const toolId = href.replace("/bioinformatics/", "")
   127→    return { view: "analysis", toolId }
   128→  }
   129→
   130→  // Thesis Studio section - map to thesis view
   131→  if (href.startsWith("/thesis")) {
   132→    return { view: "thesis" }
   133→  }
   134→
   135→  // Visualization section - could map to a visualization view or tools
   136→  if (href.startsWith("/visualization")) {
   137→    const toolId = href.replace("/visualization/", "")
   138→    return { view: "analysis", toolId: `viz-${toolId}` }
   139→  }
   140→
   141→  // Main navigation items
   142→  switch (href) {
   143→    case "/dashboard":
   144→      return { view: "dashboard" }
   145→    case "/analyze-data":
   146→      return { view: "analyze-data" }
   147→    case "/dashboard/workspaces":
   148→    case "/dashboard/projects":
   149→      return { view: "workspaces" }
   150→    case "/dashboard/databases":
   151→      return { view: "databases" }
   152→    case "/docs":
   153→      return { view: "documentation" }
   154→    case "/dashboard/settings":
   155→      return { view: "settings" }
   156→    case "/dashboard/workflows":
   157→      return { view: "workspaces" }
   158→    default:
   159→      // Fallback: try to extract meaningful view name
   160→      const path = href.replace(/^\//, "")
   161→      // Map common patterns
   162→      if (path.startsWith("plant-breeding")) return { view: "plant-breeding" }
   163→      if (path.startsWith("bioinformatics")) return { view: "analysis", toolId: path.replace("bioinformatics/", "") }
   164→      if (path.startsWith("thesis")) return { view: "thesis" }
   165→      return { view: "dashboard" }
   166→  }
   167→}

   299→// ==================== SIDEBAR CONTENT COMPONENT ====================

   301→interface SidebarContentProps {
   302→  collapsed: boolean
   303→  activeItem: string
   304→  openSections: Record<string, boolean>
   305→  setOpenSections: React.Dispatch<React.SetStateAction<Record<string, boolean>>>
   306→  onNavigate?: (view: string, toolId?: string) => void
   307→}
   308→
   309→function SidebarContent({
   310→  collapsed,
   311→  activeItem,
   312→  openSections,
   313→  setOpenSections,
   314→  onNavigate,
   315→}: SidebarContentProps) {
   316→  const { 
   317→    isAuthenticated,
   318→    userDisplayName, 
   319→    userEmail, 
   320→    userInitials,
   321→  } = useUser()
   322→
   323→  // Handle navigation click
   324→  const handleNavClick = React.useCallback(
   325→    (href: string) => {
   326→      if (onNavigate) {
   327→        const { view, toolId } = getNavigationTarget(href)
   328→        onNavigate(view, toolId)
   329→      }
   330→    },
   331→    [onNavigate]
   332→  )
   333→
   334→  const toggleSection = (sectionId: string) => {
   335→    setOpenSections((prev) => ({
   336→      ...prev,
   337→      [sectionId]: !prev[sectionId],
   338→    }))
   339→  }
   340→
   341→  const renderNavItem = (item: NavItem, isSubItem = false, sectionColor?: string) => {
   342→    const isActive = item.href === activeItem || item.isActive
   343→    
   344→    if (collapsed && !isSubItem) {
   345→      return (
   346→        <Tooltip key={item.title}>
   347→          <TooltipTrigger asChild>
   348→            <Button
   349→              variant="ghost"
   350→              size={isSubItem ? "sm" : "default"}
   351→              className={cn(
   352→                "w-full justify-center relative",
   353→                isSubItem ? "h-8 px-2" : "h-10",
   354→                isActive && sectionColor 
   355→                  ? `bg-[${sectionColor}]/10 hover:bg-[${sectionColor}]/15` 
   356→                  : isActive && "bg-biored/10 text-biored hover:bg-biored/15 hover:text-biored",
   357→                !isActive && "hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   358→              )}
   359→              onClick={() => handleNavClick(item.href)}
   360→            >
   361→              <item.icon className={cn(
   362→                "size-4 shrink-0", 
   363→                isActive && (sectionColor ? `text-[${sectionColor}]` : "text-biored")
   364→              )} />
   365→            </Button>
   366→          </TooltipTrigger>
   367→          <TooltipContent side="right">
   368→            {item.title}
   369→          </TooltipContent>
   370→        </Tooltip>
   371→      )
   372→    }
   373→
   374→    return (
   375→      <Button
   376→        key={item.title}
   377→        variant="ghost"
   378→        size={isSubItem ? "sm" : "default"}
   379→        className={cn(
   380→          "w-full justify-start gap-3 relative",
   381→          isSubItem ? "h-8 pl-7 text-sm" : "h-10 px-3",
   382→          isActive && sectionColor
   383→            ? `bg-[${sectionColor}]/10 text-[${sectionColor}] hover:bg-[${sectionColor}]/15 font-medium`
   384→            : isActive && "bg-biored/10 text-biored hover:bg-biored/15 hover:text-biored font-medium",
   385→          !isActive && "hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   386→        )}
   387→        onClick={() => handleNavClick(item.href)}
   388→      >
   389→        {isActive && (
   390→          <motion.div
   391→            layoutId={`activeIndicator-${item.title}`}
   392→            className="absolute left-0 top-1/2 -translate-y-1/2 w-[3px] h-6 rounded-r-full"
   393→            style={{ backgroundColor: sectionColor || 'var(--biored)' }}
   394→            transition={{ type: "spring", stiffness: 350, damping: 30 }}
   395→          />
   396→        )}
   397→        <item.icon className={cn(
   398→          "size-4 shrink-0", 
   399→          isActive && (sectionColor || "text-biored")
   400→        )} />
   401→        {!collapsed && (
   402→          <motion.span
   403→            initial={{ opacity: 0 }}
   404→            animate={{ opacity: 1 }}
   405→            exit={{ opacity: 0 }}
   406→            transition={{ duration: 0.2 }}
   407→          >
   408→            {item.title}
   409→          </motion.span>
   410→        )}
   411→      </Button>
   412→    )
   413→  }
   414→
   415→  const renderCollapsibleSection = (section: SectionDefinition) => {
   416→    const isOpen = openSections[section.id] ?? false
   417→    const hasNestedGroups = section.nestedGroups && Object.keys(section.nestedGroups).length > 0
   418→
   419→    if (collapsed) {
   420→      return (
   421→        <div key={section.id} className="space-y-1">
   422→          <Tooltip>
   423→            <TooltipTrigger asChild>
   424→              <Button
   425→                variant="ghost"
   426→                size="default"
   427→                className="w-full justify-center h-10 hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   428→                onClick={() => toggleSection(section.id)}
   429→              >
   430→                <span style={{ color: section.color }} className="size-4 flex items-center justify-center">
   431→                  <section.icon className="size-4" />
   432→                </span>
   433→              </Button>
   434→            </TooltipTrigger>
   435→            <TooltipContent side="right">
   436→              <span className="flex items-center gap-2">
   437→                <span>{section.emoji}</span>
   438→                <span>{section.title}</span>
   439→              </span>
   440→            </TooltipContent>
   441→          </Tooltip>
   442→        </div>
   443→      )
   444→    }
   445→
   446→    return (
   447→      <Collapsible key={section.id} open={isOpen} onOpenChange={() => toggleSection(section.id)}>
   448→        <CollapsibleTrigger asChild>
   449→          <Button
   450→            variant="ghost"
   451→            size="default"
   452→            className="w-full justify-between h-10 px-3 hover:bg-sidebar-accent hover:text-sidebar-accent-foreground group"
   453→          >
   454→            <div className="flex items-center gap-3">
   455→              <span style={{ color: section.color }} className="size-4 shrink-0 flex items-center justify-center transition-colors group-hover:opacity-80">
   456→                <section.icon 
   457→                  className="size-4" 
   458→                />
   459→              </span>
   460→              <span className="font-medium">{section.emoji} {section.title}</span>
   461→            </div>
   462→            <motion.div
   463→              animate={{ rotate: isOpen ? 90 : 0 }}
   464→              transition={{ duration: 0.2 }}
   465→            >
   466→              <ChevronRight className="size-4 opacity-50" />
   467→            </motion.div>
   468→          </Button>
   469→        </CollapsibleTrigger>
   470→        <CollapsibleContent>
   471→          <motion.div
   472→            initial={{ height: 0, opacity: 0 }}
   473→            animate={{ height: "auto", opacity: 1 }}
   474→            exit={{ height: 0, opacity: 0 }}
   475→            transition={{ duration: 0.25, ease: "easeInOut" }}
   476→            className="overflow-hidden"
   477→          >
   478→            <div className="mt-1 space-y-0.5 pl-2">
   479→              {section.items.map((item) => {
   480→                // Check if this item has a nested group
   481→                const nestedKey = Object.keys(section.nestedGroups || {}).find(
   482→                  key => item.href.includes(key.toLowerCase().replace(/\s+/g, "-"))
   483→                )
   484→                
   485→                if (nestedKey && section.nestedGroups) {
   486→                  const nestedItems = section.nestedGroups[nestedKey]
   487→                  const isNestedOpen = openSections[`${section.id}-${nestedKey}`] ?? false
   488→

   490→                    <Collapsible key={item.title} open={isNestedOpen} onOpenChange={() =>

   492→                    }>
   493→                      <CollapsibleTrigger asChild>
   494→                        <Button
   495→                          variant="ghost"
   496→                          size="sm"
   497→                          className="w-full justify-between h-8 pl-6 pr-3 text-sm hover:bg-sidebar-accent/50"

   499→                          <div className="flex items-center gap-2">
   500→                            <span style={{ color: section.color }} className="size-3.5 flex items-center justify-center">
   501→                              <item.icon className="size-3.5" />
   502→                            </span>
   503→                            <span>{item.title}</span>
   504→                          </div>
   505→                          <motion.div
   506→                            animate={{ rotate: isNestedOpen ? 90 : 0 }}
   507→                            transition={{ duration: 0.15 }}
   508→                          >
   509→                            <ChevronRight className="size-3 opacity-50" />
   510→                          </motion.div>
   511→                        </Button>
   512→                      </CollapsibleTrigger>
   513→                      <CollapsibleContent>
   514→                        <motion.div
   515→                          initial={{ height: 0, opacity: 0 }}
   516→                          animate={{ height: "auto", opacity: 1 }}
   517→                          exit={{ height: 0, opacity: 0 }}
   518→                          transition={{ duration: 0.2 }}

   524→                            )}
   525→                          </div>
   526→                        </motion.div>
   527→                      </CollapsibleContent>
   528→                    </Collapsible>
   529→                  )
   530→                }
   531→                
   532→                return renderNavItem(item, true, section.color)
   533→              })}
   534→            </div>
   535→          </motion.div>
   536→        </CollapsibleContent>
   537→      </Collapsible>
   538→    )
   539→  }
   540→
   541→  return (
   542→    <div className="flex flex-col h-full">
   543→      {/* Logo Section */}
   544→      <div className={cn(
   545→        "flex items-center h-16 border-b border-sidebar-border px-4",
   546→        collapsed ? "justify-center" : "gap-3"
   547→      )}>
   548→        <div className="flex items-center justify-center w-9 h-9 rounded-lg bg-gradient-to-br from-biored to-biored-dark shadow-lg shadow-biored/25">
   549→          <Dna className="size-5 text-white" />
   550→        </div>
   551→        <AnimatePresence mode="wait">
   552→          {!collapsed && (
   553→            <motion.div
   554→              initial={{ opacity: 0, width: 0 }}
   555→              animate={{ opacity: 1, width: "auto" }}
   556→              exit={{ opacity: 0, width: 0 }}
   557→              transition={{ duration: 0.2, ease: "easeInOut" }}
   558→              className="overflow-hidden whitespace-nowrap"
   559→            >
   560→              <span className="text-lg font-bold tracking-tight gradient-text">BioAlign</span>
   561→            </motion.div>
   562→          )}
   563→        </AnimatePresence>
   564→      </div>
   565→
   566→      {/* Navigation */}
   567→      <ScrollArea className="flex-1 py-4">
   568→        <div className="px-3 space-y-4">
   569→          {/* Main Navigation */}
   570→          <div className="space-y-1">
   571→            {!collapsed && (
   572→              <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground/60 px-3 mb-2">
   573→                Main
   574→              </p>
   575→            )}
   576→            {mainNavItems.map((item) => renderNavItem(item))}
   577→          </div>

   579→          {/* Separator after main */}

   581→
   582→          {/* Collapsible Sections */}
   583→          {collapsibleSections.map((section) => (
   584→            <div key={section.id} className="space-y-1">
   585→              {renderCollapsibleSection(section)}
   586→            </div>
   587→          ))}
   588→
   589→          {/* Separator before other */}
   590→          <Separator className="bg-sidebar-border/50" />

   617→      {/* User Profile Section - Only show when authenticated */}

   631→            {!collapsed && (
   632→              <motion.div
   633→                initial={{ opacity: 0, width: 0 }}
   634→                animate={{ opacity: 1, width: "auto" }}
   635→                exit={{ opacity: 0, width: 0 }}
   636→                transition={{ duration: 0.2, ease: "easeInOut" }}
   637→                className="overflow-hidden whitespace-nowrap"
   638→              >
   639→                <div className="flex flex-col">

   645→              </motion.div>
   646→            )}
   647→          </AnimatePresence>
   648→        </div>
   649→      ) : (

   651→          "flex items-center h-16 px-4 border-t border-sidebar-border justify-center",
   652→        )}>
   653→          <span className="text-xs text-muted-foreground">Not signed in</span>
   654→        </div>
   655→      )}
   656→    </div>
   657→  )
